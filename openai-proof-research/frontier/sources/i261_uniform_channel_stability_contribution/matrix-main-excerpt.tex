% Formula excerpts from a fully read theorem; prose below is reviewer paraphrase.
% Reviewer paraphrase: the same bound applies to every finite matrix size and every measurable Hermitian PSD potential with integrable trace power, for gamma in (1/2,3/2). The constant is sharp, with the displayed one-channel example.
\begin{equation}\label{eq:main-bound}
 \Tr(H_W)_-^\gamma\le C_\gamma\int_\R\tr(W(x)^{\gamma+1/2})\,dx,
 \qquad
 C_\gamma=
 \left(\frac{\gamma-1/2}{\gamma+1/2}\right)^{\gamma-1/2}
 \frac{\Gamma(\gamma+1)}{\sqrt\pi\,\Gamma(\gamma+3/2)}.
\end{equation}

\begin{equation}\label{eq:extremal-potential}
 W(x)=\diag\big((r+1)\sech^2(rx),0,\ldots,0\big).
\end{equation}
