from fractions import Fraction
from itertools import product
import hashlib
import json

class Ring:
    def __init__(self, kind):
        self.kind = kind
        if kind in ('R0', 'R1', 'A'):
            self.basis = tuple(product(range(3), range(4)))
        elif kind == 'M':
            self.basis = tuple(product(range(2), range(3)))
        elif kind == 'N':
            self.basis = ((0,0),(1,0),(0,1),(1,1),(0,2),(0,3))
        elif kind == 'k':
            self.basis = ((0,0),)
        self.allowed = set(self.basis)
    def mon(self, a, b):
        c = 1
        if self.kind == 'R1':
            q, a = divmod(a, 3)
            b += 3*q
            c = (-1)**q
        return ((a,b),c) if (a,b) in self.allowed else (None,0)
    def mul(self, a, b):
        return self.mon(a[0]+b[0],a[1]+b[1])

def rank(columns):
    basis = {}
    for col in columns:
        w = {r:Fraction(c) for r,c in col.items() if c}
        while w:
            r = max(w)
            if r not in basis:
                lead = w[r]
                basis[r] = {s:c/lead for s,c in w.items()}
                break
            lead = w[r]
            for s,c in basis[r].items():
                w[s] = w.get(s,0)-lead*c
                if not w[s]:
                    del w[s]
    return len(basis)

def term(ring,p,j):
    if p < 0:
        return ()
    out = []
    for mask in range(4):
        h = mask.bit_count()
        if p < h or (p-h)%2:
            continue
        for b0 in range((p-h)//2+1):
            b1 = (p-h)//2-b0
            for m in ring.basis:
                if sum(m)+h+3*b0+4*b1 == j:
                    out.append((m,mask,b0,b1))
    return tuple(out)

def diff(ring,forms,p,j,bad=False):
    src, dst = term(ring,p,j), term(ring,p-1,j)
    pos = {v:i for i,v in enumerate(dst)}
    out = []
    # forms[h][i] is a list of (coefficient, exponent) for g_hi.
    for m,mask,b0,b1 in src:
        col = {}
        def put(monom, target_mask, c, t0, t1):
            mm,cc = ring.mul(m,monom)
            if cc:
                r = pos[(mm,target_mask,t0,t1)]
                col[r] = col.get(r,0)+c*cc
                if not col[r]:
                    del col[r]
        for i in range(2):
            sign = (-1)**((mask & ((1<<i)-1)).bit_count())
            if mask & (1<<i):
                put((int(i==0),int(i==1)),mask^(1<<i),sign,b0,b1)
            elif not bad:
                for h,b in enumerate((b0,b1)):
                    if b:
                        for c,monom in forms[h][i]:
                            put(monom,mask|(1<<i),sign*c,b0-int(h==0),b1-int(h==1))
        out.append(col)
    return out

G0 = ((((1,(2,0)),),()),((),((1,(0,3)),)))
G1 = ((((1,(2,0)),),((1,(0,2)),)),((),((1,(0,3)),)))

def table(ring,forms,limit):
    ans={}; compositions=0
    for j in range(4*((limit+1)//2)+8):
        ds={p:diff(ring,forms,p,j) for p in range(1,limit+2)}
        for p in range(2,limit+2):
            for col in ds[p]:
                composition={}
                for k,c in col.items():
                    for r,d in ds[p-1][k].items():
                        composition[r]=composition.get(r,0)+c*d
                assert not any(composition.values()), ('d_squared',ring.kind,p,j)
            compositions+=1
        for p in range(limit+1):
            h=len(term(ring,p,j))-rank(ds[p] if p else [])-rank(ds[p+1])
            assert h>=0
            if h:
                ans[p,j]=h
    return ans,compositions

def expected0(limit):
    ans={}
    for b0 in range(limit//2+1):
        for b1 in range(limit//2+1):
            for ex in (0,1):
                for ey in (0,1):
                    p=2*(b0+b1)+ex+ey; j=3*b0+4*b1+2*ex+3*ey
                    if p<=limit:
                        ans[p,j]=ans.get((p,j),0)+1
    return ans

def expected1(limit):
    return {(p,2*p):1 for p in range(limit+1)}

def bar_h1(base,module):
    # Normalized bar differential, through degree 2, with independent
    # resolution machinery and the same explicit monomial arithmetic.
    plus=[m for m in base.basis if sum(m)>0]
    ans={}
    for j in range(14):
        c0=[m for m in module.basis if sum(m)==j]
        c1=[(m,a) for m in module.basis for a in plus if sum(m)+sum(a)==j]
        c2=[(m,a,b) for m in module.basis for a in plus for b in plus if sum(m)+sum(a)+sum(b)==j]
        p0={m:i for i,m in enumerate(c0)}; p1={v:i for i,v in enumerate(c1)}
        d1=[]; d2=[]
        for m,a in c1:
            mm,c=module.mul(m,a)
            d1.append({p0[mm]:c} if c else {})
        for m,a,b in c2:
            col={}
            mm,c=module.mul(m,a)
            if c:
                col[p1[mm,b]]=c
            ab,c=base.mul(a,b)
            if c:
                ix=p1[m,ab]; col[ix]=col.get(ix,0)-c
            col={r:c for r,c in col.items() if c}
            for r in range(len(c0)):
                assert sum(c*d1[s].get(r,0) for s,c in col.items())==0
            d2.append(col)
        h=len(c1)-rank(d1)-rank(d2)
        if h:
            ans[1,j]=h
    return ans

def annihilator_pair():
    r=Ring('R1')
    # Validate ker(x^2)=image(xy) and ker(xy)=image(x^2).
    def image(mon):
        return [{r.basis.index(mm):c} if c else {} for a in r.basis for mm,c in [r.mul(a,mon)]]
    x2,xy=image((2,0)),image((1,1))
    rx,ry=rank(x2),rank(xy)
    assert rx+ry==len(r.basis)==12
    for a in r.basis:
        aa,c=r.mul(a,(2,0))
        if c:
            bb,d=r.mul(aa,(1,1)); assert not d
    # Nonzero factors are nonzerodivisors in Q[x,y]/(x^3+y^3):
    # neither x nor y divides x^3+y^3, so neither x^2 nor xy shares a factor.
    return {'rank_x2':rx,'rank_xy':ry,'dimension':12}

def main():
    m,n=Ring('M'),Ring('N'); a=Ring('A')
    assert [sum(sum(v)==d for v in m.basis) for d in range(7)]==[1,2,2,1,0,0,0]
    assert [sum(sum(v)==d for v in n.basis) for d in range(7)]==[1,2,2,1,0,0,0]
    # Independent degreewise lex prescription, including ideal closure.
    for d in range(7):
        box=sorted([v for v in a.basis if sum(v)==d],reverse=True)
        keep=sum(sum(v)==d for v in m.basis)
        assert set(box[len(box)-keep:] if keep else [])=={v for v in n.basis if sum(v)==d}
    for u in a.basis:
        if u not in n.allowed:
            for v in a.basis:
                w,c=a.mul(u,v)
                assert not c or w not in n.allowed
    t0,c0=table(m,G0,6); t1,c1=table(m,G1,6); tn,cn=table(n,G0,6)
    assert t0==expected0(6); assert t1==expected1(6)
    for t in (t0,t1):
        assert all(v<=tn.get(k,0) for k,v in t.items())
    for base,forms in ((Ring('R0'),G0),(Ring('R1'),G1)):
        t,_=table(base,forms,4); assert t=={(0,0):1}
    residue,_=table(Ring('k'),G0,4)
    assert sum(residue.values())==sum(p+1 for p in range(5))
    assert bar_h1(Ring('R0'),m)=={k:v for k,v in t0.items() if k[0]==1}
    assert bar_h1(Ring('R1'),m)=={k:v for k,v in t1.items() if k[0]==1}
    assert bar_h1(a,n)=={k:v for k,v in tn.items() if k[0]==1}
    # Failure controls distinguish the very maps at issue.
    bad=diff(m,G1,2,3,bad=True)
    actual=diff(m,G1,2,3)
    assert rank(bad)==0 and rank(actual)==1
    assert t0!=t1 and t0.get((1,3))==1 and t1.get((1,3),0)==0
    assert expected0(6)!=expected1(6)
    pair=annihilator_pair()
    data={name:[[p,j,v] for (p,j),v in sorted(t.items())] for name,t in [('R0_M',t0),('R1_M',t1),('A_N',tn)]}
    print(json.dumps({'status':'PASS_EXACT_DIAGNOSTICS_ONLY_C_UNRESOLVED','homological_limit':6,'fixed_ambient_quotients':1,'two_base_choices':True,'new_C_counterexamples':0,'campaign_count':4,'Tate_compositions_internal_components':c0+c1+cn,'bar_homology_limit':1,'bar_arithmetic_shared':True,'independent_review':False,'exact_annihilator_control':pair,'controls':['omit_even_differential_detected','wrong_base_invariance_detected','free_and_residue_field','bar_Tate_agree_through_1','lex_HF_and_closure'],'tables':data,'data_sha256':hashlib.sha256(json.dumps(data,sort_keys=True).encode()).hexdigest()},sort_keys=True,indent=2))

if __name__=='__main__':
    main()
