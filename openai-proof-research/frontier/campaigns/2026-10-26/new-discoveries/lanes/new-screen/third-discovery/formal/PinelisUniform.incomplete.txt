import Mathlib

open MeasureTheory Set Real

namespace PinelisUniform

noncomputable def radius : ℝ := Real.sqrt 3
noncomputable def height : ℝ := 1 / (2 * radius)
noncomputable def density (x : ℝ) : ℝ :=
  (Set.Icc (-radius) radius).indicator (fun _ => height) x

lemma radius_pos : 0 < radius := by unfold radius; positivity
lemma radius_sq : radius ^ 2 = 3 := Real.sq_sqrt (by norm_num)
lemma height_pos : 0 < height := by unfold height; positivity
lemma density_nonneg (x : ℝ) : 0 ≤ density x := by
  unfold density
  split_ifs <;> positivity
lemma density_measurable : Measurable density :=
  measurable_const.indicator measurableSet_Icc

/-- The exact pointwise log-concavity condition stated below source Conjecture 1.3. -/
def SourceLogConcave (f : ℝ → ℝ) : Prop :=
  ∀ (x y t : ℝ), 0 < t → t < 1 →
    f x ^ (1 - t) * f y ^ t ≤ f ((1 - t) * x + t * y)

lemma density_logConcave : SourceLogConcave density := by
  intro x y t ht0 ht1
  by_cases hx : x ∈ Set.Icc (-radius) radius
  · by_cases hy : y ∈ Set.Icc (-radius) radius
    · have hz : (1 - t) * x + t * y ∈ Set.Icc (-radius) radius := by
        constructor <;> nlinarith [hx.1, hx.2, hy.1, hy.2]
      simp only [density, Set.indicator_of_mem hx, Set.indicator_of_mem hy,
        Set.indicator_of_mem hz]
      rw [← Real.rpow_add height_pos]
      norm_num
    · simp only [density, Set.indicator_of_notMem hy, Real.zero_rpow ht0.ne', mul_zero]
      exact density_nonneg _
  · simp only [density, Set.indicator_of_notMem hx,
      Real.zero_rpow (show 1 - t ≠ 0 by linarith), zero_mul]
    exact density_nonneg _

lemma density_mul_pow_indicator (k : ℕ) :
    (fun x => density x * x ^ k) =
      (Set.Icc (-radius) radius).indicator (fun x => height * x ^ k) := by
  ext x
  by_cases hx : x ∈ Set.Icc (-radius) radius <;> simp [density, hx]

lemma density_mul_pow_integrable (k : ℕ) : Integrable (fun x => density x * x ^ k) := by
  rw [density_mul_pow_indicator]
  apply IntegrableOn.integrable_indicator _ measurableSet_Icc
  exact (continuous_const.mul (continuous_id.pow k)).continuousOn.integrableOn_Icc

lemma density_mul_pow_integral (k : ℕ) :
    (∫ x, density x * x ^ k) =
      height * (radius ^ (k + 1) - (-radius) ^ (k + 1)) / (k + 1) := by
  rw [density_mul_pow_indicator, integral_indicator measurableSet_Icc,
    integral_Icc_eq_integral_Ioc,
    ← intervalIntegral.integral_of_le (show -radius ≤ radius by linarith [radius_pos]),
    intervalIntegral.integral_const_mul, integral_pow]
  ring

lemma density_integrable : Integrable density := by
  simpa using density_mul_pow_integrable 0

lemma density_integral : (∫ x, density x) = 1 := by
  have h := density_mul_pow_integral 0
  simp only [pow_zero, mul_one, Nat.zero_add, pow_one, Nat.cast_one, div_one] at h
  rw [h]
  unfold height
  field_simp

lemma density_mean : (∫ x, density x * x) = 0 := by
  simpa using density_mul_pow_integral 1

lemma density_secondMoment : (∫ x, density x * x ^ 2) = 1 := by
  rw [density_mul_pow_integral]
  norm_num
  unfold height
  have hr := radius_pos.ne'
  field_simp
  nlinarith [radius_sq]

lemma density_at_zero : density 0 = 1 / (2 * Real.sqrt 3) := by
  have h : (0 : ℝ) ∈ Set.Icc (-radius) radius := by
    constructor <;> linarith [radius_pos]
  simp [density, h, height, radius]

lemma density_at_zero_lt_exp : density 0 < Real.exp (-1) := by
  have hr : (3 : ℝ) < 2 * radius := by nlinarith [radius_sq, radius_pos]
  have h1 : height < (1 : ℝ) / 3 := by
    unfold height
    exact one_div_lt_one_div_of_lt (by norm_num) hr
  have h2 : (1 : ℝ) / 3 < Real.exp (-1) := by
    rw [Real.exp_neg]
    exact one_div_lt_one_div_of_lt (Real.exp_pos _) Real.exp_one_lt_three
  have h0 : density 0 = height := by
    have hm : (0 : ℝ) ∈ Set.Icc (-radius) radius := by
      constructor <;> linarith [radius_pos]
    simp [density, hm]
  rw [h0]
  exact h1.trans h2

end PinelisUniform
