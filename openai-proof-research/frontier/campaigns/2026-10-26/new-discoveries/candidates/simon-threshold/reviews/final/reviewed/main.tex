\begin{abstract}
We prove the convexity question posed at the end of Section 3 of Simon's paper on a Mittag--Leffler function of Le Roy type. Rescaling the auxiliary variable expresses the literal threshold as a supremum of positive perspectives of a convex function. We prove finiteness and positivity and identify this supremum with the source's infimum over admissible positive parameters. The distinct complete-monotonicity conjecture remains open here. Independent novelty and eligibility review is pending.
\end{abstract}

Fix $\gamma>1$. Simon \cite[p.~5, Section 3]{Simon} defines, for $0<\alpha<1/\gamma$,
\[
 B_\gamma(\alpha)=\inf\{\beta>0:
 z+\gamma(z^{\beta-\alpha}-z^\beta)\leq1\text{ for every }0<z<1\}.
\]
He states that $\alpha\mapsto B_\gamma(\alpha)/\alpha$ is nondecreasing, and asks whether $\alpha\mapsto B_\gamma(\alpha)$ is convex. We prove the latter assertion.
\begin{theorem}
For every real $\gamma>1$, the literal function $B_\gamma$ is convex on $(0,1/\gamma)$.
\end{theorem}

Put $f(t)=-\log(1-e^{-t})$ for $t>0$. Its derivative
\[
 f'(t)=-\frac{1}{e^t-1}
\]
is strictly increasing, so $f$ is strictly convex. For $s>0$ define
\begin{equation}\label{J}
 J_s(\alpha)=\alpha+\frac{\alpha}{s}
       \bigl(\log\gamma-f(s)+f(s/\alpha)\bigr),\qquad\alpha>0.
\end{equation}

\begin{lemma}\label{representation}
For $\gamma>1$ and $0<\alpha<1/\gamma$,
\[
 B_\gamma(\alpha)=\sup_{s>0}J_s(\alpha),\qquad
 0<\alpha< B_\gamma(\alpha)\leq\alpha+1.
\]
\end{lemma}
\begin{proof}
Set $z=e^{-t}$, $t>0$. Taking logarithms of positive factors shows that the criterion is equivalent to
\[
 \beta\geq \alpha+
 \frac{\log\gamma+\log(1-e^{-\alpha t})-\log(1-e^{-t})}{t}
 \quad(t>0).
\]
Substitute $s=\alpha t$ to obtain exactly $\beta\geq J_s(\alpha)$ for every $s>0$.

For completeness, $\beta=\alpha+1$ is admissible whenever $\gamma\alpha\leq1$: the elementary exponential inequalities give
\[
 \gamma e^{-t}(1-e^{-\alpha t})
 \leq\gamma\alpha te^{-t}\leq te^{-t}\leq1-e^{-t}.
\]
Thus the supremum is finite and at most $\alpha+1$. As $s\to\infty$, $f(s)\to0$. Choose $s>0$ with $f(s)<\log\gamma$; since $f(s/\alpha)>0$, equation \eqref{J} gives $J_s(\alpha)>\alpha>0$. If $M=\sup_{s>0}J_s(\alpha)$, the set in the source's infimum is therefore exactly $[M,\infty)$. This proves the claimed equality and bounds.
\end{proof}

\begin{proof}[Proof of Theorem 1]
For fixed $s>0$, the function $\alpha\mapsto\alpha f(s/\alpha)$ is the classical positive perspective of $f$, and is convex on $(0,\infty)$. Explicitly, for $a,b>0$, $\lambda,\mu\geq0$, $\lambda+\mu=1$, and $c=\lambda a+\mu b>0$, apply convexity of $f$ with weights $\lambda a/c$, $\mu b/c$:
\[
 c f(s/c)\leq\lambda a f(s/a)+\mu b f(s/b).
\]
The remaining terms of \eqref{J} are linear in $\alpha$, so each $J_s$ is convex. The finite pointwise supremum in Lemma \ref{representation} is convex. This proves exactly the source's threshold-convexity assertion.
\end{proof}
