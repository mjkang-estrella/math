\begin{center}
{\Large\bfseries A quartic counterexample to diagonal concavity\\and a real-rootedness criterion}\\[.5em]
Research proof and formalization packet\\
October 26, 2026
\end{center}
\begin{abstract}
We give one explicit quartic construction refuting two conjectures of
Lachaume. The polynomial $P(x)=x^4+8x^3+6x^2+8x+1$ has positive coefficients
and satisfies $4PP''-3(P')^2\le0$ on the entire real line, but has a nonreal
root. The corresponding symmetric polynomial
$f=1+2\sigma_1+\sigma_2+2\sigma_3+\sigma_4$ in four variables has a concave
fourth root on the positive diagonal, while an exact midpoint inequality
shows that $f^{1/4}$ is not concave on the positive orthant.
All inputs and all five coefficients are strictly positive. The complete
statements, including the genuine elementary symmetric functions and the
universal diagonal-concavity assertion, are certified in Lean.
\end{abstract}

\section{The source questions and the construction}
For $x=(x_1,\ldots,x_n)$ write
\[
 \sigma_k(x)=\sum_{1\le i_1<\cdots<i_k\le n}x_{i_1}\cdots x_{i_k},
 \qquad \sigma_0(x)=1.
\]
Let $\Gamma_n=(0,\infty)^n$. Lachaume \cite[Conjecture~1, p.~3]{L17}
asks whether, for $a_0,\ldots,a_p\ge0$, concavity of
$\bigl(\sum_{k=0}^p a_k\sigma_k\bigr)^{1/p}$ on $\Gamma_n$
is equivalent to concavity of its restriction to the positive diagonal.
In the same paper, Conjecture~3 (p.~5) proposes that a polynomial $P$
of degree $n$ with nonnegative coefficients is real-rooted whenever
\begin{equation}\label{criterion}
 PP''+\left(\frac1n-1\right)(P')^2\le0\qquad\text{on all of }\mathbb R.
\end{equation}
The positive-axis restriction of this condition is insufficient for that
conjecture; we verify the displayed condition on the full real line.

\begin{theorem}\label{main}
Set $n=p=4$, $(a_0,\ldots,a_4)=(1,2,1,2,1)$, and
\[
 f=1+2\sigma_1+\sigma_2+2\sigma_3+\sigma_4,
 \qquad P(t)=f(t,t,t,t)=t^4+8t^3+6t^2+8t+1.
\]
Then:
\begin{enumerate}
\item $P$ satisfies \eqref{criterion} on $\mathbb R$ and has a nonreal root.
\item $P^{1/4}$ is concave on $(0,\infty)$.
\item $f^{1/4}$ is positive and twice continuously differentiable on
$\Gamma_4$, but is not concave there.
\end{enumerate}
Thus this single construction refutes both cited conjectures.
\end{theorem}

\section{The global differential condition and a nonreal root}
Differentiating gives
\[
 P'=4t^3+24t^2+12t+8,\qquad P''=12t^2+48t+12.
\]
Expansion yields the exact identity
\begin{equation}\label{square}
 4PP''-3(P')^2=-144(t^2-1)^2.
\end{equation}
This proves \eqref{criterion} for every real $t$, since $n=4$.
All coefficients of $P$ are positive, so $P(t)>0$ for $t>0$.
For the genuine positive fourth root $h(t)=P(t)^{1/4}$,
\[
 h''(t)=\frac{4P(t)P''(t)-3P'(t)^2}{16P(t)^{7/4}}\le0
 \qquad(t>0).
\]
Consequently $h$ is concave on the full positive axis.

To exhibit a nonreal root, put $s=\sqrt3$. The factorization
\begin{equation}\label{factor}
 P(t)=\bigl(t^2+(4-2s)t+1\bigr)
       \bigl(t^2+(4+2s)t+1\bigr)
\end{equation}
follows from $s^2=3$. Since $3/2<s<2$, the number
\[
 z=s-2+i\sqrt{4s-6}
\]
has strictly positive imaginary part. Substitution into the first factor
in \eqref{factor} gives zero: writing $u=s-2$, $v=\sqrt{4s-6}$, the
imaginary part is $v(2u+4-2s)=0$, and the real part is
$u^2-v^2+(4-2s)u+1=0$. Therefore $P(z)=0$, proving the first two assertions
of Theorem~\ref{main}.

\section{A strictly positive midpoint witness}
Fix the first coordinate at $2$. Direct expansion of the elementary
symmetric functions gives
\begin{equation}\label{slice}
 g(t):=f(2,t,t,t)=5+12t+15t^2+4t^3.
\end{equation}
Consider the strictly positive points
\[
 A=(2,1/20,1/20,1/20),\quad
 B=(2,3/20,3/20,3/20),\quad
 C=\frac{A+B}{2}=(2,1/10,1/10,1/10).
\]
Equation~\eqref{slice} gives
\[
 f(A)=\frac{2819}{500},\qquad
 f(C)=\frac{3177}{500},\qquad
 f(B)=\frac{7151}{1000}.
\]
The following are exact rational inequalities:
\[
 \left(\frac{15409}{10000}\right)^4<\frac{2819}{500},\qquad
 \frac{3177}{500}<\left(\frac{15877}{10000}\right)^4,\qquad
 \left(\frac{16352}{10000}\right)^4<\frac{7151}{1000}.
\]
Monotonicity of the fourth power on nonnegative numbers now proves
\[
 f(C)^{1/4}<\frac{15877}{10000}
 <\frac12\left(\frac{15409}{10000}+\frac{16352}{10000}\right)
 <\frac{f(A)^{1/4}+f(B)^{1/4}}2.
\]
The middle rational gap is $7/20000$. This strictly reverses the midpoint
inequality required by concavity. Moreover $f>0$ on $\Gamma_4$ because
all its coefficients and coordinates are positive; hence its fourth
root is smooth there. This completes the proof of Theorem~\ref{main}.

\section{Formal certificate and scope}
The accompanying three Lean modules contain 61 declarations, replayed
from fresh source copies with Lean~4.34.1 and mathlib revision
\texttt{d13f23b7}. The verifier treats warnings as errors and checks the
axioms of every declaration; only
\texttt{propext}, \texttt{Classical.choice}, and \texttt{Quot.sound} occur.
The two source endpoints include the actual polynomial and its derivatives,
a nonreal root of its complex extension, strict positivity and $C^2$
regularity, universal diagonal concavity, and failure of concavity
on the positive orthant. The elementary symmetric functions are literal
subset sums, identified with Mathlib's \texttt{MvPolynomial.esymm}.
No numerical approximation is used in the certificate.

The distinct real-rooted-diagonal sufficient condition is not refuted.
Lachaume's later work \cite[Theorem~2.7(d)]{L18} establishes that condition
using Walsh's theorem. Carptopus \cite{CW} identifies the derivative
convolution in the original Conjectures~4 and~5 with a rescaled Walsh
convolution and explicitly excludes Conjectures~1 and~3 from its scope.
We claim neither a new Walsh theorem nor a resolution of those neighboring
questions. The two failures here form one common construction.
The independent source and prior-art audit is pending; this draft does
not assert worldwide priority or a complete literature census.
