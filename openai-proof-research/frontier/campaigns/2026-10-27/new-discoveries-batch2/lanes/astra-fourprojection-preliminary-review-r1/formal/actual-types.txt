theorem FourProjection.denominator_pos : ∀ (a b : ℝ), a ^ 2 < 1 → b ^ 2 < 1 → 0 < 1 + a * b

theorem FourProjection.sqrt_lower : ∀ (a b : ℝ),
  a ^ 2 < 1 → b ^ 2 < 1 → (1 - (a ^ 2 + b ^ 2)) * (1 + a * b) ≤ √((1 - a ^ 2) * (1 - b ^ 2))

theorem FourProjection.scalar_coefficient : ∀ (a b : ℝ),
  a ^ 2 < 1 → b ^ 2 < 1 → 1 - √((1 - a ^ 2) * (1 - b ^ 2)) / (1 + a * b) ≤ a ^ 2 + b ^ 2

theorem FourProjection.weight_inequality : ∀ (a b u v : ℝ),
  a ^ 2 < 1 →
    b ^ 2 < 1 →
      0 ≤ u →
        0 ≤ v →
          u ^ 2 = (1 + a) * (1 + b) → v ^ 2 = (1 - a) * (1 - b) → (u - v) ^ 2 ≤ (a ^ 2 + b ^ 2) * (u ^ 2 + v ^ 2)

theorem FourProjection.real_pair : ∀ (a b u v x y : ℝ),
  a ^ 2 < 1 →
    b ^ 2 < 1 →
      0 ≤ u →
        0 ≤ v →
          u ^ 2 = (1 + a) * (1 + b) →
            v ^ 2 = (1 - a) * (1 - b) → u * x + v * y = 0 → (x + y) ^ 2 ≤ (a ^ 2 + b ^ 2) * (x ^ 2 + y ^ 2)

theorem FourProjection.complex_pair : ∀ (a b u v : ℝ) (x y : ℂ),
  a ^ 2 < 1 →
    b ^ 2 < 1 →
      0 ≤ u →
        0 ≤ v →
          u ^ 2 = (1 + a) * (1 + b) →
            v ^ 2 = (1 - a) * (1 - b) →
              ↑u * x + ↑v * y = 0 → Complex.normSq (x + y) ≤ (a ^ 2 + b ^ 2) * (Complex.normSq x + Complex.normSq y)

theorem FourProjection.weighted_bound.{u_1, u_2} : ∀ {I : Type u_1} {J : Type u_2} [inst : Fintype I]
  [inst_1 : Fintype J] (a : I → ℝ) (b : J → ℝ) (w : I → J → ℝ),
  (∀ (i : I), ∑ j, w i j ≤ 1) →
    (∀ (j : J), ∑ i, w i j ≤ 1) → ∑ i, ∑ j, (a i ^ 2 + b j ^ 2) * w i j ≤ ∑ i, a i ^ 2 + ∑ j, b j ^ 2

theorem FourProjection.spectral_core.{u_1, u_2} : ∀ {I : Type u_1} {J : Type u_2} [inst : Fintype I]
  [inst_1 : Fintype J] (a : I → ℝ) (b : J → ℝ) (k l : I → J → ℂ) (u v : I → J → ℝ),
  (∀ (i : I), a i ^ 2 < 1) →
    (∀ (j : J), b j ^ 2 < 1) →
      (∀ (i : I) (j : J), 0 ≤ u i j) →
        (∀ (i : I) (j : J), 0 ≤ v i j) →
          (∀ (i : I) (j : J), u i j ^ 2 = (1 + a i) * (1 + b j)) →
            (∀ (i : I) (j : J), v i j ^ 2 = (1 - a i) * (1 - b j)) →
              (∀ (i : I) (j : J), ↑(u i j) * k i j + ↑(v i j) * l i j = 0) →
                (∀ (i : I), ∑ j, Complex.normSq (k i j) ≤ 1) →
                  (∀ (j : J), ∑ i, Complex.normSq (k i j) ≤ 1) →
                    (∀ (i : I), ∑ j, Complex.normSq (l i j) ≤ 1) →
                      (∀ (j : J), ∑ i, Complex.normSq (l i j) ≤ 1) →
                        (∑ i, ∑ j, Complex.normSq (k i j + l i j)) / 4 ≤ (∑ i, a i ^ 2 + ∑ j, b j ^ 2) / 2
