import Source
import Mathlib.RingTheory.PowerSeries.Expand
import Mathlib.Tactic.Ring

set_option autoImplicit false
set_option warningAsError true
namespace InversePeriodDoubling
open PowerSeries

theorem square_eq_expand (P : PowerSeries (ZMod 2)) :
    P ^ 2 = expand 2 (by decide) P := by
  have h : frobenius (ZMod 2) 2 = RingHom.id (ZMod 2) := by
    ext x
    fin_cases x <;> decide
  rw [← map_frobenius_expand 2 (by decide), h, map_id]
  rfl

theorem coeff_square (P : PowerSeries (ZMod 2)) (n : ℕ) :
    (P ^ 2).coeff n = if 2 ∣ n then P.coeff (n / 2) else 0 := by
  rw [square_eq_expand, coeff_expand]

theorem D_quadratic :
    D + X ^ 2 * D + X * D ^ 2 + X ^ 3 * D ^ 2 + X = 0 := by
  ext n
  rcases Nat.even_or_odd n with ⟨k, rfl⟩ | ⟨k, rfl⟩
  · cases k with
    | zero => simp [D, d]
    | succ k =>
      simp only [map_add, coeff_X_pow_mul', coeff_succ_X_mul, coeff_X, map_zero]
      simp only [D, coeff_mk]
      simp only [show k + 1 + (k + 1) = 2 * (k + 1) by omega, d_even]
      split_ifs <;> simp_all [coeff_square, D, coeff_mk, d_even] <;> omega
  · cases k with
    | zero => simp [D, d, coeff_X_pow_mul', coeff_square]; decide
    | succ k =>
      simp only [map_add, coeff_X_pow_mul', coeff_succ_X_mul, coeff_X, map_zero]
      simp only [D, coeff_mk]
      split_ifs <;> simp_all [coeff_square, D, coeff_mk, d_odd] <;> omega

end InversePeriodDoubling
