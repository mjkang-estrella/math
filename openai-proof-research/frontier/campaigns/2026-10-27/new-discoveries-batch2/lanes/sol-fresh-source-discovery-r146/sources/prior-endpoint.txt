/- ## Conjecture 4 -/

namespace Conjecture4

abbrev Ambient (n : ℕ) := EuclideanSpace ℝ (Fin n)

/-- The support function in `Ambient n`, using the same definition as the current planar
Formal Conjectures file. -/
noncomputable def supportFunction {n : ℕ} (K : Set (Ambient n)) (u : Ambient n) : ℝ :=
  ⨆ x : K, inner ℝ (x : Ambient n) u

/-- The Firey `L_p`-sum in `Ambient n`, using the same half-space presentation as the
current planar Formal Conjectures file. -/
noncomputable def lpSum {n : ℕ} (p : ℝ) (K L : Set (Ambient n)) : Set (Ambient n) :=
  {x | ∀ u : Ambient n,
    inner ℝ x u ≤ (supportFunction K u ^ p + supportFunction L u ^ p) ^ (1 / p)}

/-- `K` is invariant under reflection in some centre. -/
def HasCentreOfSymmetry {n : ℕ} (K : Set (Ambient n)) : Prop :=
  ∃ c : Ambient n, ∀ x, x ∈ K ↔ (2 • c - x) ∈ K

/-- Gamma-function extension of the binomial coefficient used in Conjecture 4. -/
noncomputable def generalizedBinomial (x y : ℝ) : ℝ :=
  Real.Gamma (x + 1) / (Real.Gamma (y + 1) * Real.Gamma (x - y + 1))

/-- The conjectured sharp coefficient
`∑ i, choose n i / generalizedBinomial (n / q) (i / q)`. -/
noncomputable def rsConstant (n : ℕ) (q : ℝ) : ℝ :=
  ∑ i ∈ Finset.range (n + 1),
    (Nat.choose n i : ℝ) / generalizedBinomial ((n : ℝ) / q) ((i : ℝ) / q)

/-- The unit cube `[0,1]^n` in the theorem's Euclidean-space model. -/
noncomputable def unitCube (n : ℕ) : Set (Ambient n) :=
  WithLp.ofLp ⁻¹' Set.Icc (0 : Fin n → ℝ) 1

/--
**Conjecture 4 (Fradelizi-Manui-Meyer-Ndiaye, 2026).** For every positive dimension and
every convex body with a centre of symmetry containing the origin, for $p > 1$,
$$
  |K \oplus_p (-K)| \leq
  \left(\sum_{i=0}^n \binom{n}{i}
  \binom{n/q}{i/q}^{-1}\right)|K|,
$$
where $q$ is the Hölder conjugate of $p$.

Equality holds for $K=[0,1]^n$.  Thus the declaration below contains both the general
inequality and the unit-cube equality clause of the paper's Conjecture 4.

The hypothesis `1 ≤ n` records the paper's positive-dimensional convention explicitly.
-/
@[category research solved, AMS 52]
theorem volume_lpSum_le :
    answer(True) ↔
      (∀ (n : ℕ), 1 ≤ n → ∀ (K : Set (Ambient n)),
        Convex ℝ K → IsCompact K → (interior K).Nonempty →
        HasCentreOfSymmetry K → (0 : Ambient n) ∈ K →
        ∀ p q : ℝ, 1 < p → 1 / p + 1 / q = 1 →
          volume (lpSum p K (-K)) ≤ ENNReal.ofReal (rsConstant n q) * volume K) ∧
      (∀ (n : ℕ), 1 ≤ n → ∀ p q : ℝ, 1 < p → 1 / p + 1 / q = 1 →
        volume (lpSum p (unitCube n) (-(unitCube n))) =
          ENNReal.ofReal (rsConstant n q) * volume (unitCube n)) := by
  constructor
  · intro _
    simpa [HasCentreOfSymmetry, lpSum, supportFunction, rsConstant, unitCube,
      generalizedBinomial, LpRogersShephardLean4Web.Conjecture4FullTarget,
      LpRogersShephardLean4Web.Conjecture4Target,
      LpRogersShephardLean4Web.Conjecture4SharpnessTarget,
      LpRogersShephardLean4Web.HasCentreOfSymmetryN,
      LpRogersShephardLean4Web.lpSumN, LpRogersShephardLean4Web.lpSumE,
      LpRogersShephardLean4Web.unitCube,
      LpRogersShephardLean4Web.supportFunctionN,
      LpRogersShephardLean4Web.supportFunctionE,
      LpRogersShephardLean4Web.symmetricRSConstant,
      LpRogersShephardLean4Web.generalizedBinomial] using
        LpRogersShephardLean4Web.conjecture4_full_proved
  · intro _
    trivial

end Conjecture4

open Arxiv.«2607.03582»

/--
**Conjecture 5 (Fradelizi-Manui-Meyer-Ndiaye, 2026).** Among planar convex bodies with a
centre of symmetry containing the origin, for $p > 1$, equality in Corollary 29 holds only for
parallelograms with a vertex at the origin.
-/
@[category research solved, AMS 52]
theorem isParallelogramAtOrigin_of_volume_lpSum_eq :
    answer(True) ↔ ∀ (K : Set ℝ²), Convex ℝ K → IsCompact K → (interior K).Nonempty →
      HasCentreOfSymmetry K → (0 : ℝ²) ∈ K → ∀ p q : ℝ, 1 < p → 1 / p + 1 / q = 1 →
        volume (lpSum p K (-K)) = ENNReal.ofReal (rsConstant q) * volume K →
          IsParallelogramAtOrigin K := by
  constructor
  · intro _
    simpa [Arxiv.«2607.03582».HasCentreOfSymmetry,
      Arxiv.«2607.03582».lpSum, Arxiv.«2607.03582».supportFunction,
      Arxiv.«2607.03582».rsConstant, Arxiv.«2607.03582».IsParallelogramAtOrigin,
      LpRogersShephardLean4Web.Conjecture5Target,
      LpRogersShephardLean4Web.HasCentreOfSymmetry,
      LpRogersShephardLean4Web.lpSum, LpRogersShephardLean4Web.lpSumE,
      LpRogersShephardLean4Web.supportFunction,
      LpRogersShephardLean4Web.supportFunctionE,
      LpRogersShephardLean4Web.rsConstant,
      LpRogersShephardLean4Web.IsParallelogramAtOrigin] using
        LpRogersShephardLean4Web.conjecture5_proved
  · intro _
    trivial
