import FeasibleH
import IncidenceRealH
import FacetDimension
noncomputable section
namespace FullereneGeometry
theorem onH10v45 : pH 45 ∈ facetSection qH 10 := by
 refine ⟨(facet_listed_vertices_H 45).1, ?_⟩
 exact (real_incidence_H 10 45).mp (by decide +kernel)
theorem onH10v46 : pH 46 ∈ facetSection qH 10 := by
 refine ⟨(facet_listed_vertices_H 46).1, ?_⟩
 exact (real_incidence_H 10 46).mp (by decide +kernel)
theorem onH10v49 : pH 49 ∈ facetSection qH 10 := by
 refine ⟨(facet_listed_vertices_H 49).1, ?_⟩
 exact (real_incidence_H 10 49).mp (by decide +kernel)
theorem dimension_H10 : affineDimension (facetSection qH 10) = 2 :=
 section_dimension_eq_of_minor qH 10 (pH 45) (pH 46) (pH 49)
  onH10v45 onH10v46 onH10v49 0 1 (by norm_num [pH])
end FullereneGeometry
