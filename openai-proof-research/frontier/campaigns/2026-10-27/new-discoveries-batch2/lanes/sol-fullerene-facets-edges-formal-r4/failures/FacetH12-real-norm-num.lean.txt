import FeasibleH
import IncidenceRealH
import FacetDimension
noncomputable section
namespace FullereneGeometry
theorem onH12v00 : pH 0 ∈ facetSection qH 12 := by
 refine ⟨(facet_listed_vertices_H 0).1, ?_⟩
 exact (real_incidence_H 12 0).mp (by decide +kernel)
theorem onH12v01 : pH 1 ∈ facetSection qH 12 := by
 refine ⟨(facet_listed_vertices_H 1).1, ?_⟩
 exact (real_incidence_H 12 1).mp (by decide +kernel)
theorem onH12v06 : pH 6 ∈ facetSection qH 12 := by
 refine ⟨(facet_listed_vertices_H 6).1, ?_⟩
 exact (real_incidence_H 12 6).mp (by decide +kernel)
theorem dimension_H12 : affineDimension (facetSection qH 12) = 2 :=
 section_dimension_eq_of_minor qH 12 (pH 0) (pH 1) (pH 6)
  onH12v00 onH12v01 onH12v06 0 1 (by norm_num [pH])
end FullereneGeometry
