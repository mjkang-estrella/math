theorem onH15v03 : pH 3 ∈ facetSection qH 15 := by
 refine ⟨(facet_listed_vertices_H 3).1, ?_⟩
 exact (real_incidence_H 15 3).mp (by decide +kernel)

theorem dimension_H15 : affineDimension (facetSection qH 15) = 2 :=
 section_dimension_eq_of_minor qH 15 (pH 3) (pH 4) (pH 16)
  onH15v03 onH15v04 onH15v16 0 1 (by norm_num [pH])
end FullereneGeometry
