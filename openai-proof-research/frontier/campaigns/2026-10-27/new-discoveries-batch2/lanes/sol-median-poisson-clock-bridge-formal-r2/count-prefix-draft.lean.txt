/-- This is a count-and-prefix probability space, not a clock-path construction. -/
noncomputable def countPrefixLaw (t : ℝ≥0) (k : ℕ) : Measure (ℕ × (Fin k → Vertex)) :=
  (poissonMeasure (10*t)).prod (prefixLaw k)
instance (t : ℝ≥0) (k : ℕ) : IsProbabilityMeasure (countPrefixLaw t k) := by
  unfold countPrefixLaw; infer_instance

theorem count_prefix_mass (t : ℝ≥0) (k : ℕ) (w : Fin k → Vertex) :
    (countPrefixLaw t k).real ({k} ×ˢ {w}) =
      Real.exp (-10*(t:ℝ)) * (t:ℝ)^k / (k.factorial:ℝ) := by
  change ((countPrefixLaw t k) ({k} ×ˢ {w})).toReal = _
  rw [countPrefixLaw, Measure.prod_prod, ENNReal.toReal_mul]
  change (poissonMeasure (10*t)).real {k} * (prefixLaw k).real {w} = _
  rw [poissonMeasure_real_singleton, prefix_mass]
  push_cast
  rw [mul_pow]
  have h : (10:ℝ)^k * (1/10:ℝ)^k = 1 := by rw [← mul_pow]; norm_num
  calc
    Real.exp (-(10*(t:ℝ))) * (10^k * (t:ℝ)^k) / (k.factorial:ℝ) * (1/10:ℝ)^k =
        Real.exp (-10*(t:ℝ)) * (t:ℝ)^k / (k.factorial:ℝ) * (10^k * (1/10:ℝ)^k) := by ring
    _ = _ := by rw [h, mul_one]

end MedianClock
