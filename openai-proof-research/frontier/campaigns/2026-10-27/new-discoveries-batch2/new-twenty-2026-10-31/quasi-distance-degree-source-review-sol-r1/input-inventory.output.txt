"""One explicit public GET plan per invocation; run inside the supplied runner.

The CLI has no batch/list traversal or retry mode. Every redirect is precharged.
This guards this client only, not other tools or arbitrary Python networking.
"""
import contextlib
import datetime
import fcntl
import hashlib
import ipaddress
import json
from pathlib import Path
import re
import sys
import urllib.error
import urllib.parse
import urllib.request

ROOTS = frozenset([
    "https://github.com/openai/math",
    "https://github.com/LeanPool/lean-pool",
    "https://github.com/TauCetiProject/TauCeti",
    "https://github.com/google-deepmind/formal-conjectures",
])
PER_ID_CAP = 12
GLOBAL_CAP = 48
MAX_BYTES = 24 * 1024 * 1024


class PolicyError(RuntimeError):
    pass


def key(url):
    p = urllib.parse.urlsplit(url)
    if p.scheme != "https" or not p.hostname or p.username or p.password:
        raise PolicyError("Only explicit credential-free HTTPS URLs are allowed")
    host = p.hostname.lower()
    if p.port not in (None, 443) or host in ("localhost", "api.github.com") or host.endswith((".local", ".internal", ".localhost")):
        raise PolicyError("Disallowed host/port")
    try:
        if not ipaddress.ip_address(host).is_global:
            raise PolicyError("Nonpublic IP address")
    except ValueError:
        pass
    return host + urllib.parse.unquote(p.path).rstrip("/")


def artifacts(url):
    s = urllib.parse.unquote(url).lower()
    found = set(re.findall(r"(?<!\d)\d{4}\.\d{4,5}(?!\d)", s))
    for doi in re.findall(r"10\.\d{4,9}/[-._;()/:a-z0-9]+", s):
        while any(doi.endswith(t) for t in (".full", ".pdf", ".abstract", ".html", ".xml")):
            doi = doi.rsplit(".", 1)[0]
        found.add(doi)
    return found


class Guard:
    def __init__(self, lane, opener=None):
        self.lane = Path(lane).resolve()
        self.ids = set(map(str, json.loads((self.lane / "reservation.json").read_text())["ids"]))
        self.denied = json.loads((self.lane / "inherited-denials.json").read_text())
        self.opener = opener  # Tests inject a local fake; CLI always uses urllib.
        self.ledger = self.lane / "guard-ledger.jsonl"

    @contextlib.contextmanager
    def locked(self):
        with (self.lane / "guard.lock").open("a") as lock:
            fcntl.flock(lock, fcntl.LOCK_EX)
            records, previous = [], "0" * 64
            if self.ledger.exists():
                for raw in self.ledger.read_bytes().splitlines(keepends=True):
                    record = json.loads(raw)
                    if record.get("seq") != len(records) or record.get("previous") != previous:
                        raise PolicyError("Broken guard ledger chain; no request permitted")
                    records.append(record)
                    previous = hashlib.sha256(raw).hexdigest()
            yield records, previous

    def append(self, records, previous, event):
        record = dict(event, seq=len(records), previous=previous,
                      utc=datetime.datetime.now(datetime.timezone.utc).isoformat())
        raw = (json.dumps(record, sort_keys=True) + "\n").encode()
        with self.ledger.open("ab") as f:
            f.write(raw)
            f.flush()
        return record

    def note(self, event):
        with self.locked() as (records, previous):
            return self.append(records, previous, event)

    def reserve(self, url, routes, plan_sha):
        k = key(url)
        if url in ROOTS or k in {key(x) for x in ROOTS}:
            routes = sorted(self.ids)  # Shared roots can never be undercharged.
        if not routes or not set(routes) <= self.ids or len(routes) != len(set(routes)):
            raise PolicyError("Routes must be exact unique reserved IDs")
        with self.locked() as (records, previous):
            attempts = [r for r in records if r["event"] == "attempt"]
            failures = [r for r in records if r["event"] == "unavailable"]
            denied_keys = set(self.denied["blocked_keys"]) | {r["url_key"] for r in failures}
            denied_artifacts = set(self.denied["blocked_artifacts"])
