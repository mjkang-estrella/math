/-
# Bunkbed E7 for transitive tournaments: base definitions and path lemmas

Literal model, MathDB 382297 (E7 / D7): transitive tournament arcs i<j on `Fin n`;
vertices `Fin n × Bool`; one token per horizontal arc per layer (exists only for i<j)
and ONE bidirectional token per index; outcome = arbitrary Boolean assignment to all
tokens; reachability = reflexive transitive closure of the enabled-arc relation.
-/
import Mathlib.Logic.Relation

/-- Horizontal arc token: ordered pair with `i < j`, carrying the layer. -/
structure HToken (n : Nat) where
  fst : Fin n
  snd : Fin n
  hlt : (fst : Nat) < (snd : Nat)
  layer : Bool

/-- One token per horizontal arc per layer, plus one bidirectional token per index. -/
def Token (n : Nat) := HToken n ⊕ Fin n

/-- Vertices of the two-layer bunkbed. -/
abbrev Vert (n : Nat) := Fin n × Bool

/-- Outcome: a Boolean assignment to every token. -/
def Out (n : Nat) := Token n → Bool

/-- Largest index a token touches (the horizontal target, or the vertical index). -/
def tmax {n : Nat} : Token n → Nat
  | Sum.inl t => (t.snd : Nat)
  | Sum.inr k => (k : Nat)

/-- The horizontal token for arc `(i,l) → (j,l)`, given `i < j`. -/
def htok {n : Nat} (i j : Fin n) (h : (i : Nat) < (j : Nat)) (l : Bool) : Token n :=
  Sum.inl ⟨i, j, h, l⟩

/-- The vertical token of index `i`; ONE token carries BOTH directions. -/
abbrev vtok {n : Nat} (i : Fin n) : Token n := Sum.inr i

@[simp] theorem tmax_htok {n : Nat} (i j : Fin n) (h : (i : Nat) < (j : Nat)) (l : Bool) :
    tmax (htok i j h l) = (j : Nat) := rfl

@[simp] theorem tmax_vtok {n : Nat} (i : Fin n) : tmax (vtok i) = (i : Nat) := rfl

/-- Enabled arc relation, literally: horizontal retention (same layer, `i < j`) or
vertical retention (one shared token enables both directions). -/
def Arc (n : Nat) (ω : Out n) : Vert n → Vert n → Prop := fun x y =>
  (x.2 = y.2 ∧ ∃ h : (x.1 : Nat) < (y.1 : Nat), ω (htok x.1 y.1 h x.2))
  ∨ (x.1 = y.1 ∧ x.2 ≠ y.2 ∧ ω (vtok x.1))

/-- Literal reachability: reflexive transitive closure of the enabled-arc relation. -/
def Reach (n : Nat) (ω : Out n) (x y : Vert n) : Prop :=
  Relation.ReflTransGen (Arc n ω) x y

theorem arc_fst_le {n : Nat} {ω : Out n} {x y : Vert n} (h : Arc n ω x y) :
    (x.1 : Nat) ≤ (y.1 : Nat) := by
  have h' : (x.2 = y.2 ∧ ∃ hh : (x.1 : Nat) < (y.1 : Nat), ω (htok x.1 y.1 hh x.2))
      ∨ (x.1 = y.1 ∧ x.2 ≠ y.2 ∧ ω (vtok x.1)) := h
  rcases h' with ⟨-, h2, -⟩ | ⟨h1, -, -⟩
  · exact Nat.le_of_lt h2
  · exact Nat.le_of_eq (by rw [h1])

/-- A retained arc, given its bit. -/
theorem arc_horiz {n : Nat} {ω : Out n} {i j : Fin n} {l : Bool}
    (hij : (i : Nat) < (j : Nat)) (h : ω (htok i j hij l)) : Arc n ω (i, l) (j, l) :=
  Or.inl ⟨rfl, ⟨hij, h⟩⟩

theorem arc_vert {n : Nat} {ω : Out n} {i : Fin n} {l : Bool} (h : ω (vtok i)) :
    Arc n ω (i, l) (i, !l) := by
  refine Or.inr ⟨rfl, ?_, h⟩
  cases l <;> simp

/-- Along a reachability path the index never decreases. -/
theorem reach_fst_le {n : Nat} {ω : Out n} {x y : Vert n}
    (h : Relation.ReflTransGen (Arc n ω) x y) : (x.1 : Nat) ≤ (y.1 : Nat) := by
  induction h with
  | refl => exact Nat.le_refl _
  | tail _ h2 ih => exact Nat.le_trans ih (arc_fst_le h2)

/-- Single-arc agreement on low endpoints from token agreement. -/
theorem arc_agree {n : Nat} {ω ω' : Out n} {c : Nat}
    (hag : ∀ t : Token n, tmax t ≤ c → ω t = ω' t) {x y : Vert n} (h : Arc n ω x y)
    (hx : (x.1 : Nat) ≤ c) (hy : (y.1 : Nat) ≤ c) : Arc n ω' x y := by
  have h' : (x.2 = y.2 ∧ ∃ hh : (x.1 : Nat) < (y.1 : Nat), ω (htok x.1 y.1 hh x.2))
      ∨ (x.1 = y.1 ∧ x.2 ≠ y.2 ∧ ω (vtok x.1)) := h
  rcases h' with ⟨h1, h2⟩ | ⟨h1, h2, h3⟩
  · rcases h2 with ⟨hlt, hw⟩
    refine Or.inl ⟨h1, ⟨hlt, ?_⟩⟩
    have htm : tmax (htok x.1 y.1 hlt x.2) ≤ c := by rw [tmax_htok]; exact hy
    rw [← hag _ htm]; exact hw
  · refine Or.inr ⟨h1, h2, ?_⟩
    have htm : tmax (vtok x.1) ≤ c := by rw [tmax_vtok]; exact hx
    rw [← hag _ htm]; exact h3

/-- If two outcomes agree on all tokens with `tmax ≤ c`, then reachability from `x`
to any vertex of index `≤ c` agrees: no path may leave the low indices, since along
any path the index is nondecreasing. -/
theorem reach_agree {n : Nat} {ω ω' : Out n} {c : Nat}
    (hag : ∀ t : Token n, tmax t ≤ c → ω t = ω' t) {x y : Vert n} (hy : (y.1 : Nat) ≤ c)
    (h : Relation.ReflTransGen (Arc n ω) x y) :
    Relation.ReflTransGen (Arc n ω') x y := by
  have key : ∀ b : Vert n, Relation.ReflTransGen (Arc n ω) x b → (b.1 : Nat) ≤ c →
      Relation.ReflTransGen (Arc n ω') x b := by
    intro b hb
    induction hb with
    | refl => exact fun _ => Relation.ReflTransGen.refl
    | @tail b' c' h1 h2 ih =>
        intro hcc
        have hbb : (b'.1 : Nat) ≤ c := Nat.le_trans (arc_fst_le h2) hcc
        exact Relation.ReflTransGen.tail (ih hbb) (arc_agree hag h2 hbb hcc)
  exact key y h hy

/-- Reachability from `(u,false)` never goes to an index `< u`. -/
theorem reach_ge {n : Nat} {ω : Out n} {u : Fin n} {x : Vert n}
    (h : Relation.ReflTransGen (Arc n ω) (u, false) x) : (u : Nat) ≤ (x.1 : Nat) :=
  reach_fst_le h

/-- The two entry shapes into a vertex, extracted from `Arc`. -/
theorem arc_cases {n : Nat} {ω : Out n} {x y : Vert n} (h : Arc n ω x y) :
    ((x.2 = y.2 ∧ ∃ hh : (x.1 : Nat) < (y.1 : Nat), ω (htok x.1 y.1 hh x.2) = true)
      ∨ (x.1 = y.1 ∧ x.2 ≠ y.2 ∧ ω (vtok x.1) = true)) := h

theorem arc_horiz' {n : Nat} {ω : Out n} {x y : Vert n}
    (h1 : x.2 = y.2) (hlt : (x.1 : Nat) < (y.1 : Nat)) (hw : ω (htok x.1 y.1 hlt x.2) = true) :
    Arc n ω x y := Or.inl ⟨h1, ⟨hlt, hw⟩⟩

/-- Recursive characterization (the exact new-index closure): for `u < j`, the copy
`(j,l)` is reachable iff some strictly-lower same-layer copy is reachable with its
horizontal bit set, or the vertical partner is reachable with the vertical bit set.
Arcs never decrease the index, so there is no other entry into index `j`. -/
theorem reach_char {n : Nat} {ω : Out n} {u j : Fin n} (huj : (u : Nat) < (j : Nat))
    {l : Bool} :
    Relation.ReflTransGen (Arc n ω) (u, false) (j, l) ↔
      ((∃ i : Fin n, (i : Nat) < (j : Nat) ∧
          Relation.ReflTransGen (Arc n ω) (u, false) (i, l) ∧
          ∃ h : (i : Nat) < (j : Nat), ω (htok i j h l) = true)
        ∨ (Relation.ReflTransGen (Arc n ω) (u, false) (j, !l) ∧ ω (vtok j) = true)) := by
  constructor
  · intro h
    have hne : (j, l) ≠ (u, false) := by
      intro hh
      have h1 : j = u := congrArg Prod.fst hh
      have h2 : (j : Nat) = (u : Nat) := by rw [h1]
      omega
    rcases Relation.ReflTransGen.cases_tail h with heq | ⟨z, h1, h2⟩
    · exact absurd heq hne
    · obtain ⟨z1, z2⟩ := z
      rcases arc_cases h2 with ⟨hz2, hz3⟩ | ⟨hz1, hz2, hz3⟩
      · obtain ⟨hlt, hw⟩ := hz3
        subst hz2
        exact Or.inl ⟨z1, hlt, h1, hlt, hw⟩
      · have hz1' : z1 = j := hz1
        have hz2l : z2 = !l := by cases z2 <;> cases l <;> simp at hz2 ⊢
        rw [hz1', hz2l] at h1
        have hw' : ω (vtok j) = true := by rw [← hz1']; exact hz3
        exact Or.inr ⟨h1, hw'⟩
  · intro h
    rcases h with ⟨i, hik, hreach, hik2, hwt⟩ | ⟨hreach, hwt⟩
    · exact Relation.ReflTransGen.trans hreach
        (Relation.ReflTransGen.single (arc_horiz' rfl hik2 hwt))
    · have hne : (!l : Bool) ≠ l := by cases l <;> simp
      exact Relation.ReflTransGen.trans hreach
        (Relation.ReflTransGen.single
          (show Arc n ω (j, !l) (j, l) from Or.inr ⟨rfl, hne, hwt⟩))

/-
# Bunkbed E7 (r2): prefix state dynamics

Window `Win n j` = tokens with `tmax t ≤ j`; `Asg n j` = Boolean assignments to that
window.  For a source index `u`, `stf n u j w` is the pair of sets of indices `≤ j`
whose layer-0 / layer-1 copies are reachable from `(u,false)` using only the prefix `w`;
it is defined by the exact new-index closure.  The new indicator pair is
`(if z then x||y else x, if z then x||y else y)` with `x,y` the horizontal-entry
disjunctions and `z` the new vertical bit.

Bridge (later lemmas): for any two full outcomes agreeing on the window `j`, literal
`Reach` from `(u,false)` to any index-`≤ j` vertex agrees, and `Reach` into `(j,σ)`
holds iff `j` lies in the corresponding state set.  No counting here.
-/
import BunkbedCount
import Mathlib.Data.Finset.Attach
import Mathlib.Data.Finset.Card

namespace Bun

/-- Tokens of `tmax ≤ j`. -/
def Win (n j : Nat) := {t : Token n // tmax t ≤ j}

/-- Assignments to the `j`-window. -/
def Asg (n j : Nat) := Win n j → Bool

/-- Restrict a `(j+1)`-prefix to the `j`-window. -/
def res (n j : Nat) (w : Asg n (j+1)) : Asg n j :=
  fun t => w ⟨t.1, Nat.le_trans t.2 (Nat.le_succ j)⟩

/-- Horizontal fresh token `(i,m,l)` seen as a window-`m` element (`m < n`, `i < m`). -/
def fresH {n m : Nat} (hm : m < n) (i : Nat) (hic : i < m) (l : Bool) : Win n m :=
  ⟨htok (⟨i, by omega⟩ : Fin n) (⟨m, hm⟩ : Fin n) hic l, by simp [htok, tmax]⟩

/-- Vertical fresh token of index `m` seen as a window-`m` element (`m < n`). -/
def fresV {n m : Nat} (hm : m < n) : Win n m :=
  ⟨Sum.inr (⟨m, hm⟩ : Fin n), by simp [tmax]⟩

/-- Horizontal-entry disjunction into layer `l` at new index `m`, from reachable set `R`. -/
def hor {n m : Nat} (hm : m < n) (R : Finset Nat) (l : Bool) (w : Asg n m) : Bool :=
  decide (((Finset.range m).attach.filter
    fun t : {x // x ∈ Finset.range m} =>
      decide (t.1 ∈ R) && w (fresH hm t.1 (Finset.mem_range.mp t.2) l)).Nonempty)

/-- The exact new-index indicator pair, closed form. -/
def ind (x y z : Bool) : Bool × Bool :=
  (if z then x || y else x, if z then x || y else y)

/-- Pure new-index set update. -/
def stepU (m : Nat) (R S : Finset Nat) (x y z : Bool) : Finset Nat × Finset Nat :=
  let a := (ind x y z).1
  let b := (ind x y z).2
  ((if a then insert m R else R), (if b then insert m S else S))

/-- The start state at index `j` (used at `j = u`): the layer-0 copy is reachable by
reflexivity; the layer-1 copy exactly through the vertical bit. -/
def startST (n j : Nat) (w : Asg n j) : Finset Nat × Finset Nat :=
  (insert j ∅,
    if hj : j < n then (if w (fresV (n := n) (m := j) hj) then insert j ∅ else ∅) else ∅)

/-- Prefix state: sets of indices `≤ j` whose layer-0/1 copies are reachable from
`(u,false)` within the prefix `w`.  Off the strip `u ≤ · ≤ v < n` values are dummies. -/
def stf (n u : Nat) : (j : Nat) → Asg n j → Finset Nat × Finset Nat
  | 0, w => if u = 0 then startST n 0 w else (∅, ∅)
  | j+1, w =>
      if _h1 : j + 1 < u then (∅, ∅)
      else if _h2 : j + 1 = u then startST n (j+1) w
      else
        let p := stf n u j (res n j w)
        if hjn : j + 1 < n then
          stepU (j+1) p.1 p.2
            (hor (m := j+1) hjn p.1 false w)
            (hor (m := j+1) hjn p.2 true w)
            (w (fresV (n := n) (m := j+1) hjn))
        else p

/-- Selector of the layer-`σ` component. -/
def sval (σ : Bool) (p : Finset Nat × Finset Nat) : Finset Nat :=
  if σ then p.2 else p.1

end Bun

import Mathlib.Algebra.BigOperators.Group.Finset.Basic
import Mathlib.Algebra.Order.BigOperators.Group.Finset
import Mathlib.Data.Int.Basic
import Mathlib.Tactic.Linarith

namespace BunDom
open Finset
abbrev State := Nat × Nat
def flip (s : State) : State := (s.2, s.1)
@[simp] theorem flip_flip (s : State) : flip (flip s) = s := by cases s; rfl

theorem sum_flip (S : Finset State) (closed : ∀ s ∈ S, flip s ∈ S)
    (f : State → Int) : ∑ s ∈ S, f (flip s) = ∑ s ∈ S, f s := by
  apply Finset.sum_bij (fun s _ => flip s)
  · exact closed
  · intro a _ b _ h
    simpa using congrArg flip h
  · intro b hb
    exact ⟨flip b, closed b hb, flip_flip b⟩
  · intros; rfl

theorem paired_nonneg (S : Finset State) (closed : ∀ s ∈ S, flip s ∈ S)
    (f : State → Int) (h : ∀ s ∈ S, 0 ≤ f s + f (flip s)) :
    0 ≤ ∑ s ∈ S, f s := by
  have hp := Finset.sum_nonneg h
  rw [Finset.sum_add_distrib, sum_flip S closed f] at hp
  omega

def advance (S : Finset State) (F : State → Int) (w : State → State → Int)
    (q : State) : Int := ∑ s ∈ S, F s * w s q

theorem reflection_step (S : Finset State) (closed : ∀ s ∈ S, flip s ∈ S)
    (F : State → Int) (w : State → State → Int)
    (wn : ∀ s q, 0 ≤ w s q)
    (eqv : ∀ s q, w (flip s) (flip q) = w s q)
    (cross : ∀ s q, s.2 < s.1 → q.1 < q.2 → w s q = 0)
    (dom : ∀ s ∈ S, s.2 < s.1 → F (flip s) ≤ F s)
    (q : State) (hq : q.2 < q.1) :
    advance S F w (flip q) ≤ advance S F w q := by
  have he (s : State) : w (flip s) q = w s (flip q) := by
    simpa using eqv s (flip q)
  have hn : 0 ≤ ∑ s ∈ S, (F s * w s q - F s * w s (flip q)) := by
    apply paired_nonneg S closed
    intro s hs
    rw [eqv s q, he s]
    rcases lt_trichotomy s.2 s.1 with ha | ha | ha
    · have hz := cross s (flip q) ha hq
      have hd := dom s hs ha
      have hw := wn s q
      rw [hz]
      nlinarith
    · have hf : flip s = s := by
        exact Prod.ext ha ha.symm
      rw [hf]
      omega
    · have hz := cross (flip s) (flip q) ha hq
      rw [eqv s q] at hz
      have hd := dom (flip s) (closed s hs) ha
      simp only [flip_flip] at hd
      have hw := wn s (flip q)
      rw [hz]
      nlinarith
  unfold advance
  rw [Finset.sum_sub_distrib] at hn
  exact sub_nonneg.mp hn

theorem reflection_time (S : Finset State) (closed : ∀ s ∈ S, flip s ∈ S)
    (F : Nat → State → Int) (w : Nat → State → State → Int)
    (rec : ∀ t q, F (t+1) q = advance S (F t) (w t) q)
    (wn : ∀ t s q, 0 ≤ w t s q)
    (eqv : ∀ t s q, w t (flip s) (flip q) = w t s q)
    (cross : ∀ t s q, s.2 < s.1 → q.1 < q.2 → w t s q = 0)
    (init : ∀ s ∈ S, s.2 < s.1 → F 0 (flip s) ≤ F 0 s) :
    ∀ t, ∀ s ∈ S, s.2 < s.1 → F t (flip s) ≤ F t s := by
  intro t
  induction t with
  | zero => exact init
  | succ t ih =>
    intro s _ hs
    rw [rec, rec]
    exact reflection_step S closed (F t) (w t) (wn t) (eqv t) (cross t) ih s hs

theorem payoff_nonneg (S : Finset State) (closed : ∀ s ∈ S, flip s ∈ S)
    (F g : State → Int)
    (dom : ∀ s ∈ S, s.2 < s.1 → F (flip s) ≤ F s)
    (anti : ∀ s, g (flip s) = -g s)
    (pos : ∀ s, s.2 < s.1 → 0 ≤ g s) :
    0 ≤ ∑ s ∈ S, F s * g s := by
  apply paired_nonneg S closed
  intro s hs
  rw [anti]
  rcases lt_trichotomy s.2 s.1 with ha | ha | ha
  · have hd := dom s hs ha
    have hg := pos s ha
    nlinarith
  · have hf : flip s = s := by
        exact Prod.ext ha ha.symm
    have hg := anti s
    rw [hf] at hg
    have : g s = 0 := by omega
    simp [this]
  · have hd := dom (flip s) (closed s hs) ha
    simp only [flip_flip] at hd
    have hg := pos (flip s) ha
    rw [anti] at hg
    nlinarith
end BunDom
