# Genus ranges of chord diagrams (329080; Cahoon et al., arXiv:1410.6148, Def. 2.1).
# Ribbon-graph model: backbone = cycle on the 2n endpoints, each endpoint has rotation
# (N,P,C) if the band end is inside and (N,C,P) if outside; faces = cycles of rho o theta,
# genus g = (2 + n - f)/2 (source Lemma 3.1).

def canon(w):
    n=len(w); best=None
    for v in (w,w[::-1]):
        for s in range(n):
            u=v[s:]+v[:s]; m={}; out=[]
            for c in u:
                if c not in m: m[c]=len(m)
                out.append(m[c])
            t=tuple(out)
            if best is None or t<best: best=t
    return best
def prep(w):
    L=len(w); partner=[0]*L; seen={}
    for i,c in enumerate(w):
        if c in seen: partner[i]=seen[c]; partner[seen[c]]=i
        else: seen[c]=i
    theta=[0]*(3*L)
    for i in range(L):
        j=(i+1)%L; theta[3*i]=3*j+1; theta[3*j+1]=3*i
        theta[3*i+2]=3*partner[i]+2
    return theta
def faces(theta,L,mask):
    # mask bit i = 1 -> endpoint i outside
    rho=[0]*(3*L)
    for i in range(L):
        N,P,C=3*i,3*i+1,3*i+2
        if (mask>>i)&1: rho[N]=C;rho[C]=P;rho[P]=N
        else: rho[N]=P;rho[P]=C;rho[C]=N
    seen=bytearray(3*L); f=0
    for h in range(3*L):
        if not seen[h]:
            f+=1; x=h
            while not seen[x]:
                seen[x]=1; x=rho[theta[x]]
    return f
def genus_range(w,stop_max=None):
    L=len(w); n=L//2; theta=prep(w); gs=set()
    # global in/out swap is a mirror: fix bit 0 = 0
    for mask in range(0,1<<L,2):
        f=faces(theta,L,mask); g=(2+n-f)//2; gs.add(g)
        if stop_max is not None and g>stop_max: return gs,False
    return gs,True

def crosses(w,a,b):
    pa=[i for i,c in enumerate(w) if c==a]; pb=[i for i,c in enumerate(w) if c==b]
    return (pa[0]<pb[0]<pa[1])!=(pa[0]<pb[1]<pa[1])
def interlace(w):

def bipartite(n,E):
    adj={i:set() for i in range(n)}
    for a,b in E: adj[a].add(b);adj[b].add(a)
    col={}
    for s in range(n):
        if s in col: continue
        col[s]=0; st=[s]

            v=st.pop()
            for u in adj[v]:
                if u not in col: col[u]=1-col[v]; st.append(u)
                elif col[u]==col[v]: return False
    return True

def dart_data(w,mask):
    L=len(w); partner=[0]*L; seen={}
    for i,c in enumerate(w):
        if c in seen: partner[i]=seen[c]; partner[seen[c]]=i
        else: seen[c]=i
    side=[(mask>>i)&1 for i in range(L)]
    sig=[0]*L
    for i in range(L):
        s=side[i]; step=1 if s==0 else -1; j=(i+step)%L
        while side[j]!=s: j=(j+step)%L
        sig[i]=j
    comp=[-1]*L;f=0

        if comp[i]<0:
            x=i
            while comp[x]<0: comp[x]=f; x=sig[partner[x]]
            f+=1
    extra=(0 not in side)+(1 not in side)
    # dart i represents the boundary segment from endpoint i to sig(i) on side(i).
    # end edge = backbone arc between endpoint L-1 and endpoint 0. Its inner side is the inner boundary
    # segment from the largest inner endpoint l to the smallest inner endpoint a, traversed in the face
    # cycle between darts tau(l) and a; its outer side is the outer segment from the smallest outer
    # endpoint v to the largest outer endpoint u, traversed between tau(v) and u.  So compare comp[a], comp[u].
    inner=[i for i in range(L) if side[i]==0]; outer=[i for i in range(L) if side[i]==1]
    ein=comp[inner[0]] if inner else 'emptyIn'
    eout=comp[outer[-1]] if outer else 'emptyOut'
