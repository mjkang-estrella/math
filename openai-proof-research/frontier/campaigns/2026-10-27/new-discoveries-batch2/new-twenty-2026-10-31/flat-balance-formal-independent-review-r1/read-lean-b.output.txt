475: /-! ## The refutation endpoint
476: 
477: The literal universal statement of the balanced-graph conjecture over all
478: finite simple graphs (quantifying the vertex type over `Type` with a
479: `Fintype` instance, i.e. exactly the finite graphs), and its refutation by
480: the explicit counterexample `K_{3,3}`. -/
481: 
482: /-- The universal balanced-graph conjecture for finite simple graphs, in the
483: exact scope of MathDB 352972. -/
484: def FlatBalancedConjecture : Prop :=
485:   ∀ (V : Type) (hV : Fintype V) (G : SimpleGraph V),
486:     @Flat V hV G → @Balanced V hV G
487: 
488: /-- Existential counterexample: a finite simple graph (`Fin 3 ⊕ Fin 3` with
489: the complete-bipartite adjacency `K33`) that is flat and not balanced. -/
490: theorem flat_not_balanced_exists :
491:     ∃ (V : Type) (_ : Fintype V) (G : SimpleGraph V),
492:       @Flat V _ G ∧ ¬ @Balanced V _ G :=
493:   ⟨Fin 3 ⊕ Fin 3, inferInstance, K33, K33_flat, K33_not_balanced⟩
494: 
495: /-- The literal refutation: the universal balanced-graph conjecture for flat
496: finite simple graphs is false. -/
497: theorem not_flat_balanced_conjecture : ¬ FlatBalancedConjecture := by
498:   intro Hall
499:   obtain ⟨V, hV, G, hflat, hnb⟩ := flat_not_balanced_exists
500:   exact hnb (Hall V hV G hflat)
