  rw [discrepancyNatPos_K33]; omega

number of parameters: 1
fields:
  FlatBalance.SimpleGraph.Adj : V → V → Prop
  FlatBalance.SimpleGraph.adj_symm : ∀ (a b : V), self.Adj a b → self.Adj b a
  FlatBalance.SimpleGraph.adj_irrefl : ∀ (a : V), ¬self.Adj a a
constructor:
  FlatBalance.SimpleGraph.mk {V : Type} (Adj : V → V → Prop) (adj_symm : ∀ (a b : V), Adj a b → Adj b a)
    (adj_irrefl : ∀ (a : V), ¬Adj a a) : FlatBalance.SimpleGraph V
def FlatBalance.IsClique : {V : Type} → FlatBalance.SimpleGraph V → Finset V → Prop :=
fun {V} G s => ∀ a ∈ s, ∀ b ∈ s, a ≠ b → G.Adj a b
def FlatBalance.cliqueFinset : {V : Type} → [Fintype V] → FlatBalance.SimpleGraph V → ℕ → Finset (Finset V) :=
fun {V} [Fintype V] G k => {s ∈ Finset.univ.powerset | s.card = k ∧ FlatBalance.IsClique G s}
def FlatBalance.cliqueCount : {V : Type} → [Fintype V] → FlatBalance.SimpleGraph V → ℕ → ℕ :=
fun {V} [Fintype V] G k => (FlatBalance.cliqueFinset G k).card
def FlatBalance.triangleSet : {V : Type} → [Fintype V] → FlatBalance.SimpleGraph V → V → V → Finset (Finset V) :=
fun {V} [Fintype V] G a b => {s ∈ Finset.univ.powerset | FlatBalance.IsTriangle G s ∧ a ∈ s ∧ b ∈ s}
def FlatBalance.Flat : {V : Type} → [Fintype V] → FlatBalance.SimpleGraph V → Prop :=
fun {V} [Fintype V] G => ∀ (a b : V), G.Adj a b → (FlatBalance.triangleSet G a b).card ≤ 2
def FlatBalance.oddCliqueSum : {V : Type} → [Fintype V] → FlatBalance.SimpleGraph V → ℕ :=
fun {V} [Fintype V] G => ∑ d ∈ Finset.range (Fintype.card V + 1), if d % 2 = 1 then FlatBalance.cliqueCount G d else 0
def FlatBalance.evenCliqueSum : {V : Type} → [Fintype V] → FlatBalance.SimpleGraph V → ℕ :=
fun {V} [Fintype V] G => ∑ d ∈ Finset.range (Fintype.card V + 1), if d % 2 = 0 then FlatBalance.cliqueCount G d else 0
def FlatBalance.evenCliqueSumPos : {V : Type} → [Fintype V] → FlatBalance.SimpleGraph V → ℕ :=
fun {V} [Fintype V] G =>
  ∑ d ∈ Finset.range (Fintype.card V + 1), if d % 2 = 0 ∧ 0 < d then FlatBalance.cliqueCount G d else 0
def FlatBalance.discrepancyNat : {V : Type} → [Fintype V] → FlatBalance.SimpleGraph V → ℕ :=
fun {V} [Fintype V] G => (↑(FlatBalance.oddCliqueSum G) - ↑(FlatBalance.evenCliqueSum G)).natAbs
def FlatBalance.discrepancyNatPos : {V : Type} → [Fintype V] → FlatBalance.SimpleGraph V → ℕ :=
fun {V} [Fintype V] G => (↑(FlatBalance.oddCliqueSum G) - ↑(FlatBalance.evenCliqueSumPos G)).natAbs
def FlatBalance.Balanced : {V : Type} → [Fintype V] → FlatBalance.SimpleGraph V → Prop :=
fun {V} [Fintype V] G => FlatBalance.discrepancyNat G ≤ 1
def FlatBalance.FlatBalancedConjecture : Prop :=
∀ (V : Type) (hV : Fintype V) (G : FlatBalance.SimpleGraph V), FlatBalance.Flat G → FlatBalance.Balanced G
