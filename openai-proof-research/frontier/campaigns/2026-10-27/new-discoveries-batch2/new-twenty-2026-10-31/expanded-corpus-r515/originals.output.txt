        with (self.lane / "guard.lock").open("a") as lock:
            fcntl.flock(lock, fcntl.LOCK_EX)
            records, previous = [], "0" * 64
            if self.ledger.exists():
                for raw in self.ledger.read_bytes().splitlines(keepends=True):
                    record = json.loads(raw)
                    if record.get("seq") != len(records) or record.get("previous") != previous:
                        raise PolicyError("Broken guard ledger chain; no request permitted")
                    records.append(record)
                    previous = hashlib.sha256(raw).hexdigest()
            yield records, previous

    def append(self, records, previous, event):
        record = dict(event, seq=len(records), previous=previous,
                      utc=datetime.datetime.now(datetime.timezone.utc).isoformat())
        raw = (json.dumps(record, sort_keys=True) + "\n").encode()
        with self.ledger.open("ab") as f:
            f.write(raw)
            f.flush()
        return record

    def note(self, event):
        with self.locked() as (records, previous):
            return self.append(records, previous, event)

    def reserve(self, url, routes, plan_sha):
        k = key(url)
        if url in ROOTS or k in {key(x) for x in ROOTS}:
            routes = sorted(self.ids)  # Shared roots can never be undercharged.
        if not routes or not set(routes) <= self.ids or len(routes) != len(set(routes)):
            raise PolicyError("Routes must be exact unique reserved IDs")
        with self.locked() as (records, previous):
            attempts = [r for r in records if r["event"] == "attempt"]
            failures = [r for r in records if r["event"] == "unavailable"]
            denied_keys = set(self.denied["blocked_keys"]) | {r["url_key"] for r in failures}
            denied_artifacts = set(self.denied["blocked_artifacts"])
            for r in failures:
                denied_artifacts.update(r.get("artifacts", []))
            if k in denied_keys or artifacts(url) & denied_artifacts:
                raise PolicyError("Inherited/current unavailable route or artifact: no retry")
            if any(r["url_key"] == k for r in attempts):
                raise PolicyError("Already attempted URL; use saved bytes/outcome, no refetch/query variant")
            if len(attempts) >= GLOBAL_CAP:
                raise PolicyError("Global GET-hop cap48 exhausted")
            for route in routes:
                if sum(route in r["routes"] for r in attempts) >= PER_ID_CAP:
                    raise PolicyError("Per-ID GET-hop cap12 exhausted: " + route)
            return self.append(records, previous, {
                "event": "attempt", "url": url, "url_key": k,
                "routes": routes, "plan_sha256": plan_sha,
                "ordinal": len(attempts) + 1,
            })

    def fetch(self, plan):

        current = plan["url"]
        attempt = self.reserve(current, plan["routes"], plan_sha)
        guard = self

        class Redirect(urllib.request.HTTPRedirectHandler):
            max_redirections = 5
