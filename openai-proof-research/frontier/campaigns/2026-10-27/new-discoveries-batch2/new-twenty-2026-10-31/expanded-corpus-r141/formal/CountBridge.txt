lemma count_sum (s : List G) : s.length = s.count e + s.count a + s.count b + s.count c + s.count d + s.count f := by
 induction s with
 | nil => decide
 | cons x s ih => cases x <;> simp_all [List.count_cons] <;> omega
lemma long_exists (s : List G) (h : s.length = 31) : HasLongOne s 6 := by
 have hsum := count_sum s
 have hc : ∃ x : G, 6 ≤ s.count x := by
  by_contra hn
  have he := hn ∘ (fun h => ⟨e,h⟩)
  have ha := hn ∘ (fun h => ⟨a,h⟩)
  have hb := hn ∘ (fun h => ⟨b,h⟩)
  have hd := hn ∘ (fun h => ⟨d,h⟩)
  have hf := hn ∘ (fun h => ⟨f,h⟩)
  have hh := hn ∘ (fun h => ⟨c,h⟩)
  omega
 obtain ⟨x,hx⟩ := hc
 refine ⟨List.replicate 6 x, List.replicate_sublist_iff.mpr hx, by simp, ?_⟩
 have hp : ∀ y : G, (List.replicate 6 y).prod = 1 := by decide
 exact hp x
lemma E_exists : ∃ n : ℕ, 0 < n ∧ ∀ s : List G, s.length = n → HasLongOne s (Fintype.card G) := by
 refine ⟨31, by omega, ?_⟩
 have hc : Fintype.card G = 6 := by decide
 simpa only [hc] using long_exists
