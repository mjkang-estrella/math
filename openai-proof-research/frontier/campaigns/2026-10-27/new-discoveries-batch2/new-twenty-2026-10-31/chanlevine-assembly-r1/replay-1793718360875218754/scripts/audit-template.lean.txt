import ChanLevineFinal
import Lean.Util.CollectAxioms
open Lean
set_option maxRecDepth 30000
set_option maxHeartbeats 20000000
run_cmd do
  let env ← getEnv
  let locals : Array Name := LOCAL_MODULES
  let mut work : Array Name := #[]
  for (n, _) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? n then
      if locals.contains env.header.moduleNames[idx.toNat]! then
        work := work.push n
        logInfo m!"ROOT {n}"
  let mut seen : NameSet := {}
  while !work.isEmpty do
    let n := work.back!
    work := work.pop
    if seen.contains n then continue
    seen := seen.insert n
    let some ci := env.find? n | throwError "Missing declaration {n}"
    if let .axiomInfo _ := ci then
      logInfo m!"CLOSURE_AXIOM {n}"
      unless n == `propext || n == `Classical.choice || n == `Quot.sound do
        throwError "Forbidden closure axiom {n}"
    let mut deps := ci.type.getUsedConstants
    if let some v := ci.value? (allowOpaque := true) then deps := deps ++ v.getUsedConstants
    if let .inductInfo v := ci then deps := deps ++ v.ctors.toArray
    let impl := Lean.Compiler.getImplementedBy? env n
    let ext := isExtern env n
    let localDecl := match env.getModuleIdxFor? n with
      | some idx => locals.contains env.header.moduleNames[idx.toNat]!
      | none => false
    logInfo m!"CHECK {n} LOCAL {localDecl} UNSAFE {ci.isUnsafe} PARTIAL {ci.isPartial} IMPLEMENTED_BY {impl} EXTERN {ext}"
    if ci.isUnsafe || ci.isPartial || n.toString.endsWith "_unsafe_rec" then
      throwError "Forbidden closure property {n}"
    if localDecl && (impl.isSome || ext) then throwError "Forbidden local implementation {n}"
    for dep in deps do
      logInfo m!"EDGE {n} {dep}"
      work := work.push dep
  logInfo m!"TOTAL_CLOSURE {seen.size}"
  let axs ← collectAxioms `ChanLevine.mathdb_338887
  logInfo m!"ENDPOINT_AXIOMS {axs}"
#print axioms ChanLevine.mathdb_338887
