```lean
/-! Concrete signed permutations; a multiplicative bit represents the sign (-1)^bit. -/
noncomputable section
namespace TypeD
open scoped BigOperators
abbrev Sign := Multiplicative (ZMod 2)
abbrev Signs (n : ℕ) := Fin n → Sign

/-- Coordinate permutations act by transport of the coordinate signs. -/
def permuteSigns (n : ℕ) : Equiv.Perm (Fin n) →* MulAut (Signs n) where
  toFun p :=
    { toFun := fun s i => s (p.symm i)
      invFun := fun s i => s (p i)
      left_inv := by intro s; funext i; simp
      right_inv := by intro s; funext i; simp
      map_mul' := by intro s t; rfl }
  map_one' := by ext s i; rfl
  map_mul' := by intro p q; ext s i; rfl

abbrev SignedPerm (n : ℕ) := SemidirectProduct (Signs n) (Equiv.Perm (Fin n)) (permuteSigns n)

instance (n : ℕ) : Fintype (SignedPerm n) :=
  Fintype.ofEquiv (Signs n × Equiv.Perm (Fin n)) SemidirectProduct.equivProd.symm

/-- Product of the actual coordinate signs, not the sign of the coordinate permutation. -/
def signProduct (n : ℕ) : SignedPerm n →* Sign where
  toFun w := ∏ i, w.left i
  map_one' := by simp
  map_mul' := by
    intro w v
    change (∏ i, w.left i * v.left (w.right.symm i)) = _
    rw [Finset.prod_mul_distrib]
    congr 1
    exact Equiv.prod_comp w.right.symm v.left

abbrev evenSigns (n : ℕ) : Subgroup (SignedPerm n) := (signProduct n).ker
abbrev EvenSignedPerm (n : ℕ) := evenSigns n
instance (n : ℕ) : Fintype (EvenSignedPerm n) := Fintype.ofFinite _

@[simp] theorem even_iff (n : ℕ) (w : SignedPerm n) :
    w ∈ evenSigns n ↔ (∏ i, w.left i) = 1 := Iff.rfl

/-- An ordinary coordinate permutation has no sign changes and is even-signed. -/
def ofPerm (n : ℕ) : Equiv.Perm (Fin n) →* EvenSignedPerm n :=
  (SemidirectProduct.inr : Equiv.Perm (Fin n) →* SignedPerm n).codRestrict
    (evenSigns n) (by intro p; simp [signProduct])

@[simp] theorem ofPerm_left (n : ℕ) (p : Equiv.Perm (Fin n)) :
    (ofPerm n p).val.left = 1 := rfl
@[simp] theorem ofPerm_right (n : ℕ) (p : Equiv.Perm (Fin n)) :
    (ofPerm n p).val.right = p := rfl

def minusSign : Sign := Multiplicative.ofAdd (1 : ZMod 2)
@[simp] theorem minusSign_sq : minusSign * minusSign = 1 := by decide

/-- Flip exactly two distinct coordinates. -/
def pairSigns (n : ℕ) (a b : Fin n) : Signs n :=
  fun i => if i = a ∨ i = b then minusSign else 1

@[simp] theorem prod_pairSigns (n : ℕ) (a b : Fin n) (h : a ≠ b) :
    (∏ i, pairSigns n a b i) = 1 := by
  classical
  have heq : pairSigns n a b =
      (fun i => if i = a then minusSign else 1) *
      (fun i => if i = b then minusSign else 1) := by
    funext i
    by_cases ha : i = a <;> by_cases hb : i = b <;> simp_all [pairSigns]
  change (∏ i, pairSigns n a b i) = 1
  simp only [heq, Pi.mul_apply, Finset.prod_mul_distrib]
  simp

def pairFlip (n : ℕ) (a b : Fin n) (h : a ≠ b) : EvenSignedPerm n :=
  ⟨SemidirectProduct.inl (pairSigns n a b), by simpa [signProduct] using prod_pairSigns n a b h⟩

/-- The 26 adjacent unsigned swaps, followed by the signed swap of the final pair. -/
def simpleD27 (i : Fin 27) : EvenSignedPerm 27 :=
  if h : i.val < 26 then
    ofPerm 27 (Equiv.swap ⟨i.val, by omega⟩ ⟨i.val + 1, by omega⟩)
  else pairFlip 27 25 26 (by decide) * ofPerm 27 (Equiv.swap 25 26)
```

```lean
import Metamatrix
import Mathlib.GroupTheory.GroupAction.Quotient
import Mathlib.Algebra.Group.Action.Prod

noncomputable section
namespace TypeD
open scoped BigOperators
variable {G : Type*} [Group G]

abbrev CosetPair (H K : Subgroup G) := (G ⧸ H) × (G ⧸ K)
abbrev PairOrbits (H K : Subgroup G) := MulAction.orbitRel.Quotient G (CosetPair H K)

/-- The orbit of (H,gK); this uses the actual left cosets and diagonal action. -/
def doubleToPairOrbits (H K : Subgroup G) :
    DoubleCoset.Quotient (H : Set G) (K : Set G) → PairOrbits H K :=
  Quotient.lift (fun g => Quotient.mk'' (((1 : G) : G ⧸ H), (g : G ⧸ K))) (by
    intro x y h
    obtain ⟨a, ha, b, hb, rfl⟩ := DoubleCoset.rel_iff.mp h
    apply Quotient.sound
    change ∃ g : G, g • (((1 : G) : G ⧸ H), ((a * x * b : G) : G ⧸ K)) =
      (((1 : G) : G ⧸ H), (x : G ⧸ K))
    refine ⟨a⁻¹, ?_⟩
    apply Prod.ext
    · change ((a⁻¹ * 1 : G) : G ⧸ H) = ((1 : G) : G ⧸ H)
      rw [mul_one]
      apply QuotientGroup.eq.mpr
      simpa using ha
    · change ((a⁻¹ * (a * x * b) : G) : G ⧸ K) = (x : G ⧸ K)
      apply QuotientGroup.eq.mpr
      simpa [mul_assoc] using K.inv_mem hb)

@[simp] theorem doubleToPairOrbits_mk (H K : Subgroup G) (g : G) :
    doubleToPairOrbits H K (DoubleCoset.mk H K g) =
      Quotient.mk'' (((1 : G) : G ⧸ H), (g : G ⧸ K)) := rfl

theorem doubleToPairOrbits_injective (H K : Subgroup G) :
    Function.Injective (doubleToPairOrbits H K) := by
  intro q r
  induction q using Quotient.inductionOn with
  | _ x =>
    induction r using Quotient.inductionOn with
    | _ y =>
      intro h
      obtain ⟨g, hg⟩ := Quotient.exact h
      have h₁ := congrArg Prod.fst hg
      have h₂ := congrArg Prod.snd hg
      change ((g * 1 : G) : G ⧸ H) = ((1 : G) : G ⧸ H) at h₁
      simp only [mul_one] at h₁
      change ((g * y : G) : G ⧸ K) = (x : G ⧸ K) at h₂
      have hH : g ∈ H := by
        have := QuotientGroup.eq.mp h₁
        simpa using H.inv_mem this
      have hK : (g * y)⁻¹ * x ∈ K := QuotientGroup.eq.mp h₂
      apply DoubleCoset.eq.mpr
      refine ⟨g⁻¹, H.inv_mem hH, ((g * y)⁻¹ * x)⁻¹, K.inv_mem hK, ?_⟩
      simp [mul_assoc]

theorem doubleToPairOrbits_surjective (H K : Subgroup G) :
    Function.Surjective (doubleToPairOrbits H K) := by
  intro q
  induction q using Quotient.inductionOn with
  | _ z =>
    rcases z with ⟨a, b⟩
    induction a using Quotient.inductionOn with
    | _ a =>
      induction b using Quotient.inductionOn with
      | _ b =>
        refine ⟨DoubleCoset.mk H K (a⁻¹ * b), ?_⟩
        apply Quotient.sound
        change ∃ g : G, g • ((a : G ⧸ H), (b : G ⧸ K)) =
          (((1 : G) : G ⧸ H), ((a⁻¹ * b : G) : G ⧸ K))
        refine ⟨a⁻¹, ?_⟩
        change (((a⁻¹ * a : G) : G ⧸ H), ((a⁻¹ * b : G) : G ⧸ K)) = _
        simp

/-- Double cosets are exactly the diagonal orbits of pairs of left cosets. -/
def doubleCosetEquivPairOrbits (H K : Subgroup G) :
    DoubleCoset.Quotient (H : Set G) (K : Set G) ≃ PairOrbits H K :=
  Equiv.ofBijective (doubleToPairOrbits H K)
    ⟨doubleToPairOrbits_injective H K, doubleToPairOrbits_surjective H K⟩

/-- Fixed points in a diagonal product are pairs of fixed points. -/
def fixedPairEquiv (X Y : Type*) [MulAction G X] [MulAction G Y] (g : G) :
    MulAction.fixedBy (X × Y) g ≃ MulAction.fixedBy X g × MulAction.fixedBy Y g where
  toFun z := (⟨z.val.1, congrArg Prod.fst z.property⟩,
    ⟨z.val.2, congrArg Prod.snd z.property⟩)
  invFun z := ⟨(z.1.val, z.2.val), Prod.ext z.1.property z.2.property⟩
  left_inv _ := rfl
  right_inv _ := rfl

/-- Denominator-free orbit Gram formula, using Mathlib's Burnside lemma. -/
theorem doubleCount_mul_card_eq_sum_fixed [Fintype G] (H K : Subgroup G) :
    doubleCount H K * Nat.card G =
      ∑ g : G, Nat.card (MulAction.fixedBy (G ⧸ H) g) *
        Nat.card (MulAction.fixedBy (G ⧸ K) g) := by
  classical
  let : Fintype (CosetPair H K) := Fintype.ofFinite _
  let : Fintype (PairOrbits H K) := Fintype.ofFinite _
  let : ∀ g : G, Fintype (MulAction.fixedBy (CosetPair H K) g) :=
    fun _ => Fintype.ofFinite _
  have hb := MulAction.sum_card_fixedBy_eq_card_orbits_mul_card_group G (CosetPair H K)
  have hc : doubleCount H K = Nat.card (PairOrbits H K) :=
    Nat.card_congr (doubleCosetEquivPairOrbits H K)
  rw [hc, Nat.card_eq_fintype_card, Nat.card_eq_fintype_card, ← hb]
  apply Finset.sum_congr rfl
  intro g _
  rw [← Nat.card_eq_fintype_card, Nat.card_congr (fixedPairEquiv (G ⧸ H) (G ⧸ K) g),
    Nat.card_prod]
```

```lean
/-- Product cancellation for sign transport, valid for every even signed permutation. -/
theorem product_transport (n : ℕ) (w : EvenSignedPerm n) (t : Signs n) :
    (∏ i, w.val.left i * t (w.val.right.symm i) / t i) = 1 := by
  rw [Finset.prod_div_distrib, Finset.prod_mul_distrib,
    Equiv.prod_comp w.val.right.symm t]
  have he : (∏ i, w.val.left i) = 1 := w.property
  rw [he, one_mul, div_self']

/-- The parity obstruction needed in type-D gluing: if all other coordinates have
compatible nonzero signs and the distinguished coordinate is fixed, its sign is positive.
The hypotheses are local sign compatibility, not a cycle-count formula. -/
theorem singleton_sign_positive (n : ℕ) (w : EvenSignedPerm n) (a : Fin n)
    (t : Signs n) (hfix : w.val.right.symm a = a)
    (hcompat : ∀ i, i ≠ a → w.val.left i * t (w.val.right.symm i) = t i) :
    w.val.left a = 1 := by
  classical
  have hprod : (∏ i, w.val.left i * t (w.val.right.symm i) / t i) =
      w.val.left a * t (w.val.right.symm a) / t a := by
    apply Finset.prod_eq_single a
    · intro i _ hia
      rw [hcompat i hia, div_self']
    · simp
  have h := product_transport n w t
  rw [hprod, hfix, mul_div_cancel_right] at h
  exact h

/-- Invariance of a unique zero coordinate forces the coordinate permutation to fix it. -/
theorem unique_zero_fixed (n : ℕ) (p : Equiv.Perm (Fin n)) (a : Fin n)
    (zero : Fin n → Prop) (hzero : ∀ i, zero i ↔ i = a)
    (hinv : ∀ i, zero (p.symm i) ↔ zero i) : p.symm a = a := by
  apply (hzero _).mp
  apply (hinv a).mpr
  exact (hzero a).mpr rfl

/-- The no-exchange statement with the actual unique-zero premise. -/
theorem singleton_zero_no_exchange (n : ℕ) (w : EvenSignedPerm n) (a : Fin n)
    (zero : Fin n → Prop) (t : Signs n)
    (hzero : ∀ i, zero i ↔ i = a)
    (hinv : ∀ i, zero (w.val.right.symm i) ↔ zero i)
    (hcompat : ∀ i, ¬ zero i → w.val.left i * t (w.val.right.symm i) = t i) :
    w.val.left a = 1 := by
  apply singleton_sign_positive n w a t (unique_zero_fixed n w.val.right a zero hzero hinv)
  intro i hia
  apply hcompat i
  simpa [hzero i] using hia

/-- Consequently both sign choices at the singleton extend an invariant sign pattern. -/
theorem singleton_extensions_fixed (n : ℕ) (w : EvenSignedPerm n) (a : Fin n)
    (t : Signs n) (hfix : w.val.right.symm a = a)
    (hcompat : ∀ i, i ≠ a → w.val.left i * t (w.val.right.symm i) = t i)
    (u : Sign) :
    ∀ i, w.val.left i * (Function.update t a u) (w.val.right.symm i) =
      (Function.update t a u) i := by
  classical
  have hpos := singleton_sign_positive n w a t hfix hcompat
  intro i
  by_cases hi : i = a
  · subst i
    simp [hfix, hpos]
  · have hp : w.val.right.symm i ≠ a := by
      intro h
      apply hi
      exact w.val.right.symm.injective (h.trans hfix.symm)
    simpa [Function.update_of_ne hi, Function.update_of_ne hp] using hcompat i hi
```

```lean
import SignedPerm
import Mathlib.Logic.Equiv.Basic
import Mathlib.Data.Fintype.Card
import Mathlib.Data.Multiset.Fintype

/-! Signed permutations act on vectors `Fin n → V` by `(t,π)•f = i ↦ ±f(π⁻¹ i)`. -/
noncomputable section
namespace TypeD
open scoped BigOperators

theorem sign_two (s : Sign) : s = 1 ∨ s = minusSign := by
  revert s; decide

theorem minusSign_ne_one : minusSign ≠ 1 := by decide

section
variable {V : Type*} [InvolutiveNeg V]

/-- The sign bit acts on `V` by negation. -/
def sgnAct (s : Sign) (v : V) : V := if s = 1 then v else -v

@[simp] theorem sgnAct_one (v : V) : sgnAct 1 v = v := by simp [sgnAct]
@[simp] theorem sgnAct_minus (v : V) : sgnAct minusSign v = -v := by
  simp [sgnAct, minusSign_ne_one]

theorem sgnAct_mul (s t : Sign) (v : V) : sgnAct (s * t) v = sgnAct s (sgnAct t v) := by
  rcases sign_two s with rfl | rfl <;> rcases sign_two t with rfl | rfl <;> simp

theorem sgnAct_neg (s : Sign) (v : V) : sgnAct s (-v) = -sgnAct s v := by
  rcases sign_two s with rfl | rfl <;> simp

instance vecAction (n : ℕ) : MulAction (SignedPerm n) (Fin n → V) where
  smul g f := fun i => sgnAct (g.left i) (f (g.right.symm i))
  one_smul f := by
    funext i
    change sgnAct ((1 : Signs n) i) (f ((1 : Equiv.Perm (Fin n)).symm i)) = f i
    rfl
  mul_smul g h f := by
    funext i
    change sgnAct (g.left i * h.left (g.right.symm i))
        (f ((g.right * h.right).symm i)) =
      sgnAct (g.left i) (sgnAct (h.left (g.right.symm i)) (f (h.right.symm (g.right.symm i))))
    rw [sgnAct_mul]
    rfl

theorem smul_apply {n : ℕ} (g : SignedPerm n) (f : Fin n → V) (i : Fin n) :
    (g • f) i = sgnAct (g.left i) (f (g.right.symm i)) := rfl

theorem evenSmul_apply {n : ℕ} (g : EvenSignedPerm n) (f : Fin n → V) (i : Fin n) :
    (g • f) i = sgnAct (g.val.left i) (f (g.val.right.symm i)) := rfl

/-- Completeness: equal multisets of classes give one signed-permutation orbit. -/
theorem exists_smul_eq_of_multiset {n : ℕ} [DecidableEq V] (canon : V → V)
    (hc : ∀ v w, canon v = canon w → w = v ∨ w = -v) (f f' : Fin n → V)
    (h : Multiset.map (fun i => canon (f i)) Finset.univ.val =
      Multiset.map (fun i => canon (f' i)) Finset.univ.val) :
    ∃ b : SignedPerm n, b • f = f' := by
  classical
  have hcnt : ∀ (g : Fin n → V) (c : V), Fintype.card {a // canon (g a) = c} =
      Multiset.count c (Multiset.map (fun i => canon (g i)) Finset.univ.val) := by
    intro g c
    rw [Multiset.count_map, Fintype.card_subtype, Finset.card_def, Finset.filter_val]
    congr 1
    apply Multiset.filter_congr
    intro x _
    exact eq_comm
  have hcard : ∀ c : V, Fintype.card {a // canon (f a) = c} =
      Fintype.card {a // canon (f' a) = c} := by
    intro c
    rw [hcnt, hcnt, h]
  let e : ∀ c, {a // (fun a => canon (f a)) a = c} ≃ {b // (fun b => canon (f' b)) b = c} :=
    fun c => Fintype.equivOfCardEq (hcard c)
  let π : Equiv.Perm (Fin n) := Equiv.ofFiberEquiv e
  have hπ : ∀ a, canon (f' (π a)) = canon (f a) := fun a => Equiv.ofFiberEquiv_map e a
  let t : Signs n := fun i => if f' i = f (π.symm i) then 1 else minusSign
  refine ⟨⟨t, π⟩, ?_⟩
  funext i
  rw [smul_apply]
  change sgnAct (t i) (f (π.symm i)) = f' i
  have hx := hπ (π.symm i)
  simp only [Equiv.apply_symm_apply] at hx
  rcases hc _ _ hx.symm with h1 | h1
  · simp [t, h1]
  · by_cases h2 : f' i = f (π.symm i)
    · simp [t, h2]
    · have ht : t i = minusSign := by simp [t, h2]
      rw [ht, sgnAct_minus, h1]
end
end TypeD
```

```lean
import VecAction
import Generation
import Mathlib.Algebra.BigOperators.Intervals

/-! The chamber vector of type `I` and its elementary properties. -/
noncomputable section
namespace TypeD
open scoped BigOperators
open Finset

/-- Indicator of a missing adjacent generator. -/
def gapInd (I : Finset (Fin 27)) (k : ℕ) : ℕ :=
  if h : k < 27 then (if (⟨k, h⟩ : Fin 27) ∈ I then 0 else 1) else 0

/-- Number of missing adjacent generators among indices `j, …, 24`. -/
def gaps (I : Finset (Fin 27)) (j : ℕ) : ℕ := ∑ k ∈ Ico j 25, gapInd I k

def baseI (I : Finset (Fin 27)) : ℤ := if (25 : Fin 27) ∈ I ∧ (26 : Fin 27) ∈ I then 0 else 1

def lastI (I : Finset (Fin 27)) : ℤ :=
  if (25 : Fin 27) ∈ I then (if (26 : Fin 27) ∈ I then 0 else 1)
  else (if (26 : Fin 27) ∈ I then -1 else 0)

/-- The canonical point of the face of type `I` in the closed fundamental chamber
`x₀ ≥ ⋯ ≥ x₂₅ ≥ |x₂₆|`, with consecutive integer levels. -/
def xI (I : Finset (Fin 27)) (j : Fin 27) : ℤ :=
  if j = 26 then lastI I else baseI I + (gaps I j.val : ℤ)

theorem gaps_succ (I : Finset (Fin 27)) (j : ℕ) (hj : j < 25) :
    gaps I j = gapInd I j + gaps I (j + 1) := by
  unfold gaps
  rw [Finset.sum_eq_sum_Ico_succ_bot hj]

theorem gaps_anti (I : Finset (Fin 27)) {a b : ℕ} (h : a ≤ b) : gaps I b ≤ gaps I a := by
  unfold gaps
  apply Finset.sum_le_sum_of_subset
  intro k hk
  simp only [Finset.mem_Ico] at hk ⊢
  omega

theorem gaps_25 (I : Finset (Fin 27)) : gaps I 25 = 0 := by simp [gaps]

theorem baseI_nonneg (I : Finset (Fin 27)) : 0 ≤ baseI I := by
  unfold baseI; split_ifs <;> norm_num

theorem lastI_abs_le (I : Finset (Fin 27)) : (lastI I).natAbs ≤ (baseI I).natAbs := by
  unfold lastI baseI; split_ifs <;> simp_all

theorem xI_of_ne (I : Finset (Fin 27)) (j : Fin 27) (hj : j ≠ 26) :
    xI I j = baseI I + (gaps I j.val : ℤ) := by simp [xI, hj]

theorem xI_nonneg (I : Finset (Fin 27)) (j : Fin 27) (hj : j ≠ 26) : 0 ≤ xI I j := by
  rw [xI_of_ne I j hj]
  have := baseI_nonneg I
  positivity

theorem xI_25 (I : Finset (Fin 27)) : xI I 25 = baseI I := by
  rw [xI_of_ne I 25 (by decide)]
  simp [gaps_25]

theorem xI_26 (I : Finset (Fin 27)) : xI I 26 = lastI I := by simp [xI]

/-- Absolute values are antitone along the coordinates. -/
theorem xI_abs_anti (I : Finset (Fin 27)) {a b : Fin 27} (hab : a ≤ b) :
    (xI I b).natAbs ≤ (xI I a).natAbs := by
  by_cases hb : b = 26
  · subst hb
    by_cases ha : a = 26
    · subst ha; exact le_rfl
    rw [xI_26, xI_of_ne I a ha]
    have h1 := lastI_abs_le I
    have h2 := baseI_nonneg I
    omega
  · have ha : a ≠ 26 := by
      intro h; subst h; apply hb; exact le_antisymm (by omega) hab
    rw [xI_of_ne I a ha, xI_of_ne I b hb]
    have h1 := gaps_anti I (show a.val ≤ b.val from hab)
    have h2 := baseI_nonneg I
    omega

/-- Adjacent coordinates below the fork. -/
theorem xI_adjacent (I : Finset (Fin 27)) (i : Fin 26) (hi : i.val < 25) :
    xI I i.castSucc = xI I i.succ + (if i.castSucc ∈ I then 0 else 1) := by
  have h1 : i.castSucc ≠ 26 := by intro h; have := congrArg Fin.val h; simp at this; omega
  have h2 : i.succ ≠ 26 := by intro h; have := congrArg Fin.val h; simp at this; omega
  rw [xI_of_ne I _ h1, xI_of_ne I _ h2]
  have hg := gaps_succ I i.val hi
  simp only [Fin.val_succ, Fin.val_castSucc]
  rw [hg]
  unfold gapInd
  have hlt : i.val < 27 := by omega
  simp only [hlt, ↓reduceDIte]
  have hc : (⟨i.val, hlt⟩ : Fin 27) = i.castSucc := rfl
  rw [hc]
  split_ifs <;> push_cast <;> ring

theorem xI_adjacent_abs (I : Finset (Fin 27)) (i : Fin 26) (hi : i.val < 25)
    (h : (xI I i.castSucc).natAbs = (xI I i.succ).natAbs) : i.castSucc ∈ I := by
  have h1 := xI_adjacent I i hi
  have hn : 0 ≤ xI I i.succ := xI_nonneg I _ (by
    intro h; have := congrArg Fin.val h; simp at this; omega)
  by_contra hm
  simp only [hm, ite_false] at h1
  omega

theorem fork_abs (I : Finset (Fin 27)) (h : (xI I 25).natAbs = (xI I 26).natAbs) :
    (25 : Fin 27) ∈ I ∨ (26 : Fin 27) ∈ I := by
  rw [xI_25, xI_26] at h
  by_contra hc
  rw [not_or] at hc
  simp [baseI, lastI, hc.1, hc.2] at h

/-- Zero coordinates of the chamber vector form a terminal block of the D-fork. -/
theorem xI_zero (I : Finset (Fin 27)) (j : Fin 27) (hj : j ≠ 26) (h0 : xI I j = 0) :
    (25 : Fin 27) ∈ I ∧ (26 : Fin 27) ∈ I ∧
      ∀ k : ℕ, ∀ hk : k < 27, j.val ≤ k → k < 25 → (⟨k, hk⟩ : Fin 27) ∈ I := by
  rw [xI_of_ne I j hj] at h0
  have hb := baseI_nonneg I
  have hg : gaps I j.val = 0 := by omega
  have hb0 : baseI I = 0 := by omega
  unfold baseI at hb0
  split_ifs at hb0 with hh
  refine ⟨hh.1, hh.2, ?_⟩
  · intro k hk hk1 hk2
    unfold gaps at hg
    have := (Finset.sum_eq_zero_iff.mp hg) k (by simp; omega)
    unfold gapInd at this
    by_contra hn
    simp [hk, hn] at this
  · omega
```

```lean
import Mathlib.Order.Preorder.Finite

/-! The stabilizer of the chamber vector of type `I` is exactly the parabolic subgroup `W_I`. -/
noncomputable section
namespace TypeD

abbrev PI (I : Finset (Fin 27)) : Subgroup D27 := parabolic simpleD27 I

theorem gen_mem (I : Finset (Fin 27)) {i : Fin 27} (h : i ∈ I) : simpleD27 i ∈ PI I :=
  Subgroup.subset_closure ⟨i, h, rfl⟩

theorem fork_val : (simpleD27 26).val =
    (⟨pairSigns 27 25 26, Equiv.swap 25 26⟩ : SignedPerm 27) := by
  rw [simpleD27_fork]
  apply SemidirectProduct.ext
  · funext k
    simp [pairFlip, ofPerm]
  · simp [pairFlip, ofPerm]

    (⟨1, Equiv.swap i.castSucc i.succ⟩ : SignedPerm 27) := by

  rfl

theorem swap_invariant (x : Fin 27 → ℤ) (a b : Fin 27) (h : x a = x b) (j : Fin 27) :
    x (Equiv.swap a b j) = x j := by
  rw [Equiv.swap_apply_def]
  split_ifs with h1 h2
  · subst h1; exact h.symm
  · subst h2; exact h
  · rfl

theorem perm_smul_eq (x : Fin 27 → ℤ) (a b : Fin 27) (h : x a = x b) :
    (⟨1, Equiv.swap a b⟩ : SignedPerm 27) • x = x := by
  funext j
  rw [smul_apply]
  change sgnAct (1 : Sign) (x ((Equiv.swap a b).symm j)) = x j
  rw [sgnAct_one, Equiv.symm_swap, swap_invariant x a b h]

/-- Every chosen generator fixes the chamber vector. -/
theorem gen_fix (I : Finset (Fin 27)) (i : Fin 27) (hi : i ∈ I) :
    simpleD27 i • xI I = xI I := by
  change (simpleD27 i).val • xI I = xI I
  by_cases h26 : i = 26
  · subst h26
    rw [fork_val]
    have hx : xI I 26 = -xI I 25 := by
      rw [xI_25, xI_26]; unfold lastI baseI; split_ifs <;> simp_all
    funext j
    rw [smul_apply]
    change sgnAct (pairSigns 27 25 26 j) (xI I ((Equiv.swap 25 26).symm j)) = xI I j
    rw [Equiv.symm_swap]
    by_cases j25 : j = 25
    · subst j25
      simp [pairSigns, hx]
    · by_cases j26 : j = 26
      · subst j26
        simp [pairSigns, hx]
      · simp [pairSigns, j25, j26, Equiv.swap_apply_of_ne_of_ne j25 j26]
  · have hlt : i.val < 26 := by
      have := i.isLt
      have : i.val ≠ 26 := fun h => h26 (Fin.ext h)

    by_cases h25 : i'.val < 25
    · rw [xI_adjacent I i' h25]; simp [hi]
    · have hv : i'.val = 25 := by omega

      have e2 : i'.succ = 26 := Fin.ext (by simp [hv])

      rw [e1] at hi
      unfold baseI lastI
      split_ifs <;> simp_all

  rintro _ ⟨i, hi, rfl⟩
  exact gen_fix I i hi

/-! Pure sign elements. -/

    (h0 : xI I j = 0) (h26 : xI I 26 = 0) : pairFlip 27 j 26 hj • xI I = xI I := by

  change sgnAct (pairSigns 27 j 26 i) (xI I i) = xI I i
  by_cases hi : i = j ∨ i = 26
  · rcases hi with rfl | rfl <;> simp [pairSigns, h0, h26]
  · simp [pairSigns, hi]

theorem pairFlip_mem_of_zero (I : Finset (Fin 27)) (j : Fin 27) (hj : j ≠ 26)
    (h0 : xI I j = 0) : pairFlip 27 j 26 hj ∈ PI I := by
  obtain ⟨h25, h26, hall⟩ := xI_zero I j hj h0
  have hfork : pairFlip 27 25 26 (by decide) ∈ PI I := by
    have hm := (PI I).mul_mem (gen_mem I h26) (gen_mem I h25)
    have he : simpleD27 26 * simpleD27 25 = pairFlip 27 25 26 (by decide) := by
      rw [simpleD27_fork]
      have : simpleD27 25 = ofPerm 27 (Equiv.swap 25 26) := by
        have := simpleD27_adjacent (25 : Fin 26)
        exact this
      rw [this, mul_assoc, ← map_mul, Equiv.swap_mul_self, map_one, mul_one]
    rwa [he] at hm

  have key : ∀ m : ℕ, ∀ k : Fin 27, ∀ hk : k ≠ 26, k.val + m = 25 → j.val ≤ k.val →
      pairFlip 27 k 26 hk ∈ PI I := by
    intro m
    induction m with
    | zero =>
      intro k hk hkm _
      have : k = 25 := Fin.ext (by simpa using hkm)
      subst this
      exact hfork
    | succ m ih =>
      intro k hk hkm hjk
      have hk1 : k.val + 1 < 27 := by omega
      let k1 : Fin 27 := ⟨k.val + 1, hk1⟩
      have hk1ne : k1 ≠ 26 := by intro h; have := congrArg Fin.val h; simp [k1] at this; omega

      have hs : i'.succ = k1 := rfl

      have e2 : Equiv.swap k k1 26 = 26 := Equiv.swap_apply_of_ne_of_ne hk.symm hk1ne.symm

  exact key (25 - j.val) j hj (by have : j.val ≠ 26 := fun h => hj (Fin.ext h); omega) le_rfl

theorem sign_part (I : Finset (Fin 27)) : ∀ (m : ℕ) (g : D27), g.val.right = 1 →
    g • xI I = xI I →
    (univ.filter (fun j : Fin 27 => j ≠ 26 ∧ g.val.left j ≠ 1)).card = m → g ∈ PI I := by
  intro m
  induction m with
  | zero =>
    intro g hr _ hc
    have hl : ∀ j : Fin 27, j ≠ 26 → g.val.left j = 1 := by
      intro j hj
      by_contra hne
      have : j ∈ univ.filter (fun j : Fin 27 => j ≠ 26 ∧ g.val.left j ≠ 1) := by simp [hj, hne]
      rw [Finset.card_eq_zero] at hc
      simp [hc] at this
    have hprod : (∏ i, g.val.left i) = 1 := g.property
    have h26 : g.val.left 26 = 1 := by
      rw [← Finset.mul_prod_erase univ _ (mem_univ (26 : Fin 27))] at hprod
      rwa [Finset.prod_eq_one (fun j hj => hl j (Finset.mem_erase.mp hj).1), mul_one] at hprod
    have : g = 1 := by
      apply Subtype.ext
      apply SemidirectProduct.ext
      · funext j
        by_cases hj : j = 26
        · subst hj; exact h26
        · exact hl j hj
      · exact hr
    rw [this]; exact (PI I).one_mem
  | succ m ih =>
    intro g hr hfix hc
    obtain ⟨j, hjmem⟩ : ∃ j, j ∈ univ.filter (fun j : Fin 27 => j ≠ 26 ∧ g.val.left j ≠ 1) := by
      apply Finset.card_pos.mp; omega
    have hj := (Finset.mem_filter.mp hjmem).2
    have hmin : g.val.left j = minusSign := (sign_two _).resolve_left hj.2
    have hx0 : xI I j = 0 := by
      have := congrFun hfix j
      rw [evenSmul_apply, hr, hmin, sgnAct_minus] at this
      have h' : -xI I j = xI I j := this
      omega

      obtain ⟨h25, h26, _⟩ := xI_zero I j hj.1 hx0
      rw [xI_26]; simp [lastI, h25, h26]

    have hf : f ∈ PI I := pairFlip_mem_of_zero I j hj.1 hx0
    have hfx : f • xI I = xI I := pairFlip_smul_of_zero I j hj.1 hx0 hx26

      change g.val.right * (1 : Equiv.Perm (Fin 27)) = 1
      rw [hr, mul_one]
    have hl' : ∀ i, (g * f).val.left i = g.val.left i * pairSigns 27 j 26 i := by
      intro i
      change g.val.left i * (permuteSigns 27 g.val.right (pairSigns 27 j 26)) i = _
      rw [hr, map_one]
      rfl

    have hc' : (univ.filter (fun i : Fin 27 => i ≠ 26 ∧ (g * f).val.left i ≠ 1)).card = m := by
      have heq : univ.filter (fun i : Fin 27 => i ≠ 26 ∧ (g * f).val.left i ≠ 1) =
          (univ.filter (fun i : Fin 27 => i ≠ 26 ∧ g.val.left i ≠ 1)).erase j := by
        ext i
        simp only [mem_filter, mem_univ, true_and, mem_erase, hl']
        by_cases hij : i = j
        · subst hij
          simp [pairSigns, hmin]
        · by_cases hi26 : i = 26
          · simp [hi26]
          · simp [pairSigns, hij, hi26]
      rw [heq, Finset.card_erase_of_mem hjmem, hc]; rfl
    have := ih (g * f) hr' hfix' hc'
    have hg : g = (g * f) * f⁻¹ := by group
    rw [hg]
    exact (PI I).mul_mem this ((PI I).inv_mem hf)

/-! The potential argument. -/

def potential (π : Equiv.Perm (Fin 27)) : ℕ := ∑ j : Fin 27, (j : ℕ) * (π j : ℕ)

theorem potential_le (π : Equiv.Perm (Fin 27)) : potential π ≤ 27 * (26 * 26) := by
  unfold potential
  calc ∑ j : Fin 27, (j : ℕ) * (π j : ℕ) ≤ ∑ _j : Fin 27, 26 * 26 := by
        apply Finset.sum_le_sum
        intro j _
        have h1 : (j : ℕ) ≤ 26 := by omega
        have h2 : (π j : ℕ) ≤ 26 := by omega
        exact Nat.mul_le_mul h1 h2
    _ = 27 * (26 * 26) := by simp

theorem potential_swap (π : Equiv.Perm (Fin 27)) (i : Fin 26) :
    (potential (π * Equiv.swap i.castSucc i.succ) : ℤ) =
      potential π + ((π i.castSucc : ℕ) - (π i.succ : ℕ) : ℤ) := by
  set a := i.castSucc
  set b := i.succ
  have hab : a ≠ b := (Fin.castSucc_lt_succ (i := i)).ne
  have h1 : potential (π * Equiv.swap a b) = ∑ j : Fin 27, (Equiv.swap a b j : ℕ) * (π j : ℕ) := by
    unfold potential
    rw [← Equiv.sum_comp (Equiv.swap a b) (fun j => (Equiv.swap a b j : ℕ) * (π j : ℕ))]
    simp [Equiv.Perm.mul_apply]
  rw [h1, potential]
  push_cast
  rw [← sub_eq_iff_eq_add', ← Finset.sum_sub_distrib]
  rw [Fintype.sum_eq_add a b hab]
  · rw [Equiv.swap_apply_left, Equiv.swap_apply_right]
    have hb : (b : ℕ) = (a : ℕ) + 1 := by simp [a, b]
    rw [hb]; push_cast; ring
  · intro j hj
    rw [Equiv.swap_apply_of_ne_of_ne hj.1 hj.2]; ring

theorem exists_descent (π : Equiv.Perm (Fin 27)) (h : π ≠ 1) :
    ∃ i : Fin 26, π i.succ < π i.castSucc := by
  by_contra hc
  simp only [not_exists, not_lt] at hc
  have hmono : StrictMono π := by
    apply Fin.strictMono_iff_lt_succ.mpr
    intro i
    exact lt_of_le_of_ne (hc i) (fun he => (Fin.castSucc_lt_succ (i := i)).ne (π.injective he))
  apply h
  ext j
  simp [hmono.apply_eq]

theorem abs_invariant (I : Finset (Fin 27)) (g : D27) (hg : g • xI I = xI I) (k : Fin 27) :
    (xI I (g.val.right k)).natAbs = (xI I k).natAbs := by
  have := congrFun hg (g.val.right k)
  rw [evenSmul_apply, Equiv.symm_apply_apply] at this
  rw [← this]
  rcases sign_two (g.val.left (g.val.right k)) with h | h <;> simp [h]

/-- The stabilizer of the chamber vector is contained in the parabolic subgroup. -/
theorem stabilizer_le_PI (I : Finset (Fin 27)) (g : D27) (hg : g • xI I = xI I) : g ∈ PI I := by
  suffices H : ∀ m : ℕ, ∀ g : D27, g • xI I = xI I → 27 * (26 * 26) - potential g.val.right = m →
      g ∈ PI I from H _ g hg rfl
  intro m
  induction m using Nat.strong_induction_on with
  | _ m ih =>
    intro g hg hm
    by_cases hr : g.val.right = 1
    · exact sign_part I _ g hr hg rfl
    obtain ⟨i, hi⟩ := exists_descent _ hr
    set π := g.val.right
    have habs : (xI I i.castSucc).natAbs = (xI I i.succ).natAbs := by
      have h1 := xI_abs_anti I (show i.castSucc ≤ i.succ from (Fin.castSucc_lt_succ (i := i)).le)
      have h2 := xI_abs_anti I hi.le
      rw [abs_invariant I g hg, abs_invariant I g hg] at h2
      omega
    -- a generator in `I` whose permutation part is the adjacent swap
    obtain ⟨s, hsI, hsr, hsx⟩ : ∃ s : D27, s ∈ PI I ∧
        s.val.right = Equiv.swap i.castSucc i.succ ∧ s • xI I = xI I := by
      by_cases h25 : i.val < 25
      · have hmem := xI_adjacent_abs I i h25 habs
        refine ⟨simpleD27 i.castSucc, gen_mem I hmem, by rw [adjacent_val], gen_fix I _ hmem⟩
      · have hv : i.val = 25 := by omega
        have e1 : i.castSucc = 25 := Fin.ext (by simp [hv])
        have e2 : i.succ = 26 := Fin.ext (by simp [hv])
        rw [e1, e2] at habs ⊢
        rcases fork_abs I habs with h | h
        · refine ⟨simpleD27 25, gen_mem I h, ?_, gen_fix I _ h⟩
          have := adjacent_val (25 : Fin 26)
          rw [show (simpleD27 25).val = (simpleD27 (25 : Fin 26).castSucc).val from rfl, this]
          rfl
        · exact ⟨simpleD27 26, gen_mem I h, by rw [fork_val], gen_fix I _ h⟩
    have hgs : (g * s) • xI I = xI I := by rw [mul_smul, hsx, hg]
    have hpot := potential_swap π i
    have hr' : (g * s).val.right = π * Equiv.swap i.castSucc i.succ := by
      change π * s.val.right = _
      rw [hsr]
    have hlt : ((π i.succ : ℕ)) < (π i.castSucc : ℕ) := hi
    have hb1 := potential_le ((g * s).val.right)
    have hb2 := potential_le π
    have hmem := ih (27 * (26 * 26) - potential (g * s).val.right) (by
      rw [hr']; rw [hr'] at hb1; omega) (g * s) hgs rfl
    have hg' : g = (g * s) * s⁻¹ := by group
    rw [hg']
    exact (PI I).mul_mem hmem ((PI I).inv_mem hsI)

theorem stabilizer_eq_PI (I : Finset (Fin 27)) :
    MulAction.stabilizer D27 (xI I) = PI I :=
  le_antisymm (fun g hg => stabilizer_le_PI I g hg) (PI_le_stabilizer I)
```

```lean
theorem natAbs_sgnAct (s : Sign) (v : ℤ) : (sgnAct s v).natAbs = v.natAbs := by
  rcases sign_two s with rfl | rfl <;> simp

theorem multiset_map_perm {α : Type*} (f : Fin n → α) (σ : Equiv.Perm (Fin n)) :
    Multiset.map (fun i => f (σ i)) univ.val = Multiset.map f univ.val := by
  have : Multiset.map (fun i => f (σ i)) univ.val = Multiset.map f (Multiset.map σ univ.val) := by
    rw [Multiset.map_map]; rfl
  rw [this]

  exact h2

  exact eq_comm

/-- Faces of dimension `d` of the type-D arrangement, as integer vectors with consecutive levels.
Glued faces (one zero coordinate) carry the dimension of the two halves. -/
def DFace (d : ℕ) (y : Fin 27 → ℤ) : Prop :=
  (nzLevels y = Icc 1 d ∧ (2 ≤ zeros y ∨ (zeros y = 0 ∧ 2 ≤ ones y))) ∨
  (1 ≤ d ∧ nzLevels y = Icc 1 (d - 1) ∧ zeros y = 1)

theorem DFace_of_absMS {d : ℕ} {y y' : Fin 27 → ℤ} (h : absMS y = absMS y') :
    DFace d y ↔ DFace d y' := by
  simp only [DFace, zeros, ones, nzLevels, h]

theorem DFace_smul {d : ℕ} (b : SignedPerm 27) (y : Fin 27 → ℤ) : DFace d (b • y) ↔ DFace d y :=
  DFace_of_absMS (absMS_smul b y)

  have := congrArg Finset.card h

    (h : absMS y = absMS y') : d = d' := by
  have hz : zeros y = zeros y' := by simp [zeros, h]
  have hl : nzLevels y = nzLevels y' := by simp [nzLevels, h]
  rcases h1 with ⟨a1, b1⟩ | ⟨a1, b1, c1⟩ <;> rcases h2 with ⟨a2, b2⟩ | ⟨a2, b2, c2⟩
  · rw [hl, a2] at a1; exact (Icc_one_inj a1).symm
  · omega
  · omega
  · rw [hl, b2] at b1; have := Icc_one_inj b1; omega
```

```lean
      ((e.symm (g • q)) : X) = g • (e.symm q : X) := by
    intro q
    induction q using QuotientGroup.induction_on with
    | H h =>
      change ((e.symm ((g * h : G) : G ⧸ MulAction.stabilizer G x)) : X) = _
      rw [MulAction.orbitEquivQuotientStabilizer_symm_apply,
        MulAction.orbitEquivQuotientStabilizer_symm_apply, mul_smul]
  apply Nat.card_congr
  exact
    { toFun := fun q => ⟨e.symm q.1, (e.symm q.1).2, by
        rw [← hequi]; congr 2; exact q.2⟩
      invFun := fun y => ⟨e ⟨y.1, y.2.1⟩, by
        change g • e ⟨y.1, y.2.1⟩ = e ⟨y.1, y.2.1⟩
        apply e.symm.injective
        apply Subtype.ext
        rw [hequi, Equiv.symm_apply_apply]
        exact y.2.2⟩
      left_inv := fun q => by
        apply Subtype.ext
        simp
      right_inv := fun y => by
        apply Subtype.ext
        simp }

theorem orbit_finite (x : Fin 27 → ℤ) (P : (Fin 27 → ℤ) → Prop) :
    Finite {y : Fin 27 → ℤ // y ∈ MulAction.orbit D27 x ∧ P y} := by
  have hf : (MulAction.orbit D27 x).Finite := Set.finite_range _
  have : Finite (MulAction.orbit D27 x) := hf.to_subtype
  exact Finite.of_injective (fun y => (⟨y.1, y.2.1⟩ : MulAction.orbit D27 x))
    (fun a b h => Subtype.ext (by simpa using congrArg Subtype.val h))

theorem DFace_orbit (I : Finset (Fin 27)) (y : Fin 27 → ℤ) (hy : y ∈ MulAction.orbit D27 (xI I)) :
    DFace (27 - I.card) y := by
  obtain ⟨h, rfl⟩ := hy
  exact (DFace_smul h.val (xI I)).mpr (xI_DFace I)

/-- The sum over parabolic types of fixed points equals fixed points of the face set. -/
theorem sum_orbit_fixed (p : ℕ) (hp : p ≤ 27) (g : D27) :
    ∑ I ∈ powersetCard p (univ : Finset (Fin 27)),
        Nat.card {y : Fin 27 → ℤ // y ∈ MulAction.orbit D27 (xI I) ∧ g • y = y} =
      Nat.card {y : Fin 27 → ℤ // DFace (27 - p) y ∧ g • y = y} := by
  classical
  have : ∀ I : ↥(powersetCard p (univ : Finset (Fin 27))),
      Finite {y : Fin 27 → ℤ // y ∈ MulAction.orbit D27 (xI I.1) ∧ g • y = y} :=
    fun I => orbit_finite _ _
  rw [← Finset.sum_coe_sort, ← Nat.card_sigma]
  symm
  apply Nat.card_congr
  have hcov : ∀ y : Fin 27 → ℤ, DFace (27 - p) y →
      ∃ I : Finset (Fin 27), I ∈ powersetCard p (univ : Finset (Fin 27)) ∧
        y ∈ MulAction.orbit D27 (xI I) := by
    intro y hy
    obtain ⟨I, hI, h, hh⟩ := DFace_cover (27 - p) (by omega) y hy
    exact ⟨I, by simp [mem_powersetCard]; omega, h, hh⟩
  exact
    { toFun := fun y => ⟨⟨(hcov y.1 y.2.1).choose, (hcov y.1 y.2.1).choose_spec.1⟩,
        ⟨y.1, (hcov y.1 y.2.1).choose_spec.2, y.2.2⟩⟩
      invFun := fun z => ⟨z.2.1, by
        have := DFace_orbit z.1.1 z.2.1 z.2.2.1
        rwa [(mem_powersetCard.mp z.1.2).2] at this, z.2.2.2⟩
      left_inv := fun y => rfl
      right_inv := fun z => by
        have hd : DFace (27 - p) z.2.1 := by
          have := DFace_orbit z.1.1 z.2.1 z.2.2.1
          rwa [(mem_powersetCard.mp z.1.2).2] at this
        apply Sigma.subtype_ext
        · apply Subtype.ext
          change (hcov z.2.1 hd).choose = z.1.1
          obtain ⟨h1, hh1⟩ := (hcov z.2.1 hd).choose_spec.2
          obtain ⟨h2, hh2⟩ := z.2.2.1
          symm
          apply xI_inj z.1.1 _ (h1⁻¹ * h2)
          have e1 : h1 • xI (hcov z.2.1 hd).choose = z.2.1 := hh1
          have e2 : h2 • xI z.1.1 = z.2.1 := hh2
          rw [mul_smul, e2, inv_smul_eq_iff]
          exact e1.symm
        · rfl }
```

```lean
/-- Vectors whose nonzero absolute values are exactly `1, …, k` (type-B faces). -/
def BSet (k : ℕ) (y : Fin 27 → ℤ) : Prop := nzLevels y = Icc 1 k
/-- Those with exactly one zero coordinate. -/
def ZSet (k : ℕ) (y : Fin 27 → ℤ) : Prop := nzLevels y = Icc 1 k ∧ zeros y = 1
/-- No zero, exactly one entry of absolute value one, and that entry equals `u`. -/
def CP (u : ℤ) (k : ℕ) (y : Fin 27 → ℤ) : Prop :=
  nzLevels y = Icc 1 k ∧ zeros y = 0 ∧ ones y = 1 ∧ ∀ i, (y i).natAbs = 1 → y i = u

    (y i).natAbs ≤ k := by
  by_cases h0 : (y i).natAbs = 0
  · omega
  · have : (y i).natAbs ∈ nzLevels y := (mem_nzLevels y _).mpr ⟨h0, i, rfl⟩
    rw [h, mem_Icc] at this

  simp

    ∃ i, (y i).natAbs = 1 := by
  have : 1 ∈ nzLevels y := by rw [h, mem_Icc]; omega
  exact ((mem_nzLevels y 1).mp this).2

  · exact Or.inr (Or.inr ⟨h, Int.sign_eq_neg_one_of_neg h⟩)
  · exact Or.inr (Or.inl ⟨h, by simp [h]⟩)
  · exact Or.inl ⟨h, Int.sign_eq_one_of_pos h⟩

  · simp
  · simp only [sgnAct_minus, lowerV, Int.sign_neg]; ring

    · by_cases h3 : y i = 1
      · simp [lowerV, h3, ← hone i (by omega)]

    · exact absurd h1 this
    · by_cases h3 : y i = -1
      · simp [lowerV, h3, ← hone i (by omega)]
      · have hne : lowerV (y i) ≠ 0 := by simp only [lowerV, h2]; omega
        rw [show (if lowerV (y i) = 0 then u else raiseV (lowerV (y i))) =
          raiseV (lowerV (y i)) from by simp [hne]]
        simp only [raiseV, lowerV, h2]

  exact

        rw [← lower_smul, y.2.2]⟩
      invFun := fun y => ⟨lift y.1, hlift y.1 y.2.1, hlift_fix y.1 y.2.1 y.2.2⟩
```

```lean
import Gluing

/-! The type-D gluing identity for fixed-point counts. -/
noncomputable section
namespace TypeD
open scoped BigOperators
open Finset

theorem fixCard_empty (P : (Fin 27 → ℤ) → Prop) (g : D27) (h : ∀ y, ¬ P y) : fixCard P g = 0 := by
  rw [Nat.card_eq_zero]; left; exact ⟨fun y => h y.1 y.2.1⟩

theorem CP_zero_empty (u : ℤ) (y : Fin 27 → ℤ) : ¬ CP u 0 y := by
  rintro ⟨hl, hz, _, _⟩
  have := (zeros_eq_zero_iff y).mp hz 0
  have := bound_of_levels hl 0
  omega

theorem CP_iff_ZSet_card (u : ℤ) (hu : u = 1 ∨ u = -1) (d : ℕ) (g : D27) :
    fixCard (CP u d) g = fixCard (ZSet (d - 1)) g := by
  rcases Nat.eq_zero_or_pos d with rfl | hd
  · rw [fixCard_empty _ g (CP_zero_empty u), fixCard_empty _ g ZSet_zero_empty]
  · exact card_CP u hu d hd g

theorem one_level_pm (y : Fin 27 → ℤ) (ho : ones y = 1) :
    ∃ i, (y i).natAbs = 1 ∧ ∀ j, (y j).natAbs = 1 → j = i := by
  rw [ones_eq] at ho
  obtain ⟨i, hi⟩ := Finset.card_eq_one.mp ho
  refine ⟨i, ?_, ?_⟩
  · have := congrArg (i ∈ ·) hi; simpa using this
  · intro j hj
    have := congrArg (j ∈ ·) hi; simp at this; exact this.mp hj

theorem gluing (d : ℕ) (g : D27) :
    fixCard (DFace d) g + fixCard (ZSet d) g + fixCard (ZSet (d - 1)) g =
      fixCard (BSet d) g := by
  classical
  let A : (Fin 27 → ℤ) → Prop := fun y =>
    BSet d y ∧ (2 ≤ zeros y ∨ (zeros y = 0 ∧ 2 ≤ ones y))
  have fB : Finite {y : Fin 27 → ℤ // BSet d y ∧ g • y = y} :=
    finite_of_bound d _ (fun y hy => bound_of_levels hy.1)
  have H1 : fixCard (BSet d) g = fixCard A g +
      Nat.card {y : Fin 27 → ℤ // (ZSet d y ∨ CP 1 d y ∨ CP (-1) d y) ∧ g • y = y} := by
    apply card_split
    · intro y
      constructor
      · rintro ⟨hb, hf⟩
        by_cases hz2 : 2 ≤ zeros y
        · exact Or.inl ⟨⟨hb, Or.inl hz2⟩, hf⟩
        by_cases hz1 : zeros y = 1
        · exact Or.inr ⟨Or.inl ⟨hb, hz1⟩, hf⟩
        have hz0 : zeros y = 0 := by omega
        by_cases ho2 : 2 ≤ ones y
        · exact Or.inl ⟨⟨hb, Or.inr ⟨hz0, ho2⟩⟩, hf⟩
```

```lean
/-! Binomial inversion over the integers. -/
namespace TypeD

open Finset

theorem neg_one_pow_sub (N t : ℕ) (h : t ≤ N) :
    ((-1 : ℤ) ^ (N - t)) = (-1) ^ N * (-1) ^ t := by
  have hN : N = (N - t) + t := by omega
  conv_rhs => rw [hN, pow_add]
  rw [mul_assoc, ← pow_add, ← two_mul, pow_mul]
  simp

theorem alt_choose_choose (j k : ℕ) (hjk : j ≤ k) :
    ∑ i ∈ range (k + 1), ((-1 : ℤ) ^ (k - i) * (k.choose i : ℤ) * (i.choose j : ℤ)) =
      if j = k then 1 else 0 := by
  rw [← Finset.sum_range_add_sum_Ico _ (show j ≤ k + 1 by omega)]
  have h1 : ∑ i ∈ range j, ((-1 : ℤ) ^ (k - i) * (k.choose i : ℤ) * (i.choose j : ℤ)) = 0 := by
    apply Finset.sum_eq_zero
    intro i hi
    rw [Finset.mem_range] at hi
    simp [Nat.choose_eq_zero_of_lt hi]
  rw [h1, zero_add, Finset.sum_Ico_eq_sum_range]
  have hN : k + 1 - j = (k - j) + 1 := by omega
  rw [hN]
  have h2 : ∀ t ∈ range (k - j + 1),
      ((-1 : ℤ) ^ (k - (j + t)) * (k.choose (j + t) : ℤ) * ((j + t).choose j : ℤ)) =
        (k.choose j : ℤ) * (-1) ^ (k - j) * ((-1) ^ t * ((k - j).choose t : ℤ)) := by
    intro t ht
    rw [Finset.mem_range] at ht
    have hc := Nat.choose_mul (n := k) (k := j + t) (s := j) (by omega)
    have hc' : ((k.choose (j + t) : ℤ) * ((j + t).choose j : ℤ)) =
        (k.choose j : ℤ) * ((k - j).choose t : ℤ) := by
      have : j + t - j = t := by omega
      rw [this] at hc
      exact_mod_cast hc
    rw [mul_assoc, hc', show k - (j + t) = (k - j) - t by omega, neg_one_pow_sub _ _ (by omega)]
    ring
  rw [Finset.sum_congr rfl h2, ← Finset.mul_sum, Int.alternating_sum_range_choose]
  by_cases h : j = k
  · subst h; simp
  · have : k - j ≠ 0 := by omega
    simp [h, this]

/-- Binomial inversion. -/
theorem binomial_inversion (B W : ℕ → ℤ)
    (h : ∀ m, W m = ∑ k ∈ range (m + 1), (m.choose k : ℤ) * B k) (k : ℕ) :
    B k = ∑ i ∈ range (k + 1), (-1 : ℤ) ^ (k - i) * (k.choose i : ℤ) * W i := by
  have h3 : ∀ i ∈ range (k + 1), (-1 : ℤ) ^ (k - i) * (k.choose i : ℤ) * W i =
      ∑ j ∈ range (k + 1), ((-1 : ℤ) ^ (k - i) * (k.choose i : ℤ) * (i.choose j : ℤ)) * B j := by
    intro i hi
    rw [h i, Finset.mul_sum]
    rw [Finset.mem_range] at hi
    have e : ∀ j, (-1 : ℤ) ^ (k - i) * (k.choose i : ℤ) * ((i.choose j : ℤ) * B j) =
        (-1 : ℤ) ^ (k - i) * (k.choose i : ℤ) * (i.choose j : ℤ) * B j := fun j => by ring
    simp only [e]
    apply Finset.sum_subset
    · intro j hj; simp at hj ⊢; omega
    · intro j _ hj
      simp at hj
      simp [Nat.choose_eq_zero_of_lt (show i < j by omega)]
  rw [Finset.sum_congr rfl h3, Finset.sum_comm]
  rw [Finset.sum_congr rfl (g := fun j => (if j = k then (1 : ℤ) else 0) * B j)]
  · simp
  · intro j hj
    rw [← Finset.sum_mul, alt_choose_choose j k (by simp at hj; omega)]
```

```lean
import Gluing

/-! Relabeling the levels: weak sets `|y i| ≤ m` decompose over exact level sets. -/
noncomputable section
namespace TypeD
open scoped BigOperators
open Finset

/-- Weak type-B vectors: all absolute values at most `m`. -/
def WSet (m : ℕ) (y : Fin 27 → ℤ) : Prop := ∀ i, (y i).natAbs ≤ m
/-- Those with exactly one zero coordinate. -/
def ZWSet (m : ℕ) (y : Fin 27 → ℤ) : Prop := WSet m y ∧ zeros y = 1

/-- An odd relabeling of absolute values. -/
def relab (φ : ℕ → ℕ) (v : ℤ) : ℤ := Int.sign v * (φ v.natAbs : ℤ)

theorem relab_sgnAct (φ : ℕ → ℕ) (s : Sign) (v : ℤ) :
    relab φ (sgnAct s v) = sgnAct s (relab φ v) := by
  rcases sign_two s with rfl | rfl
  · simp
  · simp [relab, Int.sign_neg]

theorem relab_smul (φ : ℕ → ℕ) (g : D27) (y : Fin 27 → ℤ) :
    (fun i => relab φ ((g • y) i)) = g • (fun i => relab φ (y i)) := by
  funext i; simp only [evenSmul_apply, relab_sgnAct]

theorem relab_zero (φ : ℕ → ℕ) : relab φ 0 = 0 := by simp [relab]

theorem relab_abs (φ : ℕ → ℕ) (v : ℤ) (hv : v ≠ 0) : (relab φ v).natAbs = φ v.natAbs := by
  simp [relab, Int.natAbs_mul, Int.natAbs_sign_of_ne_zero hv]

theorem relab_relab (φ ψ : ℕ → ℕ) (v : ℤ) (hv : v ≠ 0) (h1 : 1 ≤ φ v.natAbs)
    (h2 : ψ (φ v.natAbs) = v.natAbs) : relab ψ (relab φ v) = v := by
  have hs : Int.sign (relab φ v) = Int.sign v := by
    unfold relab
    rw [Int.sign_mul, Int.sign_natCast_of_ne_zero (by omega), mul_one, Int.sign_sign]
  have ha : (relab φ v).natAbs = φ v.natAbs := relab_abs φ v hv
  unfold relab at *
  rw [hs, ha, h2, Int.sign_mul_natAbs]

/-- Exact level sets of the same size have equally many fixed vectors. -/
theorem relabel_card (U : Finset ℕ) (hU : ∀ a ∈ U, 1 ≤ a) (Q : ℕ → Prop) (g : D27) :
    Nat.card {y : Fin 27 → ℤ // nzLevels y = U ∧ Q (zeros y) ∧ g • y = y} =
      Nat.card {y : Fin 27 → ℤ // nzLevels y = Icc 1 U.card ∧ Q (zeros y) ∧ g • y = y} := by
  classical
  set k := U.card
  let e : U ≃ Icc 1 k := Fintype.equivOfCardEq (by simp [k])
  let φ : ℕ → ℕ := fun a => if h : a ∈ U then (e ⟨a, h⟩ : ℕ) else 0
  let ψ : ℕ → ℕ := fun b => if h : b ∈ Icc 1 k then (e.symm ⟨b, h⟩ : ℕ) else 0
  have hφ : ∀ a ∈ U, φ a ∈ Icc 1 k ∧ ψ (φ a) = a := by
    intro a ha
    have hm := (e ⟨a, ha⟩).2
    refine ⟨by simp only [φ, ha, ↓reduceDIte]; exact hm, ?_⟩
    simp only [φ, ha, ↓reduceDIte, ψ, hm]
    simp
  have hψ : ∀ b ∈ Icc 1 k, ψ b ∈ U ∧ φ (ψ b) = b := by
    intro b hb
    have hm := (e.symm ⟨b, hb⟩).2
    refine ⟨by simp only [ψ, hb, ↓reduceDIte]; exact hm, ?_⟩
    simp only [ψ, hb, ↓reduceDIte, φ, hm]
    simp
  have hpos : ∀ a ∈ U, 1 ≤ φ a := fun a ha => (mem_Icc.mp (hφ a ha).1).1
  have hposψ : ∀ b ∈ Icc 1 k, 1 ≤ ψ b := fun b hb => hU _ (hψ b hb).1
  -- generic transport between two level sets
  have transport : ∀ (V W : Finset ℕ) (α β : ℕ → ℕ),
      (∀ a ∈ V, α a ∈ W ∧ β (α a) = a ∧ 1 ≤ α a) → (∀ b ∈ W, ∃ a ∈ V, α a = b) →
      (∀ a ∈ V, 1 ≤ a) →
      ∀ y : Fin 27 → ℤ, nzLevels y = V →
        nzLevels (fun i => relab α (y i)) = W ∧ zeros (fun i => relab α (y i)) = zeros y ∧
        (fun i => relab β (relab α (y i))) = y := by
    intro V W α β hα hsurj hV y hy
    have hmem : ∀ i, y i ≠ 0 → (y i).natAbs ∈ V := by
      intro i hi; rw [← hy, mem_nzLevels]; exact ⟨by omega, i, rfl⟩
    have hne : ∀ i, y i ≠ 0 → relab α (y i) ≠ 0 := by
      intro i hi h
      have := relab_abs α _ hi
      rw [h] at this
      have := (hα _ (hmem i hi)).2.2
      simp at *; omega
    refine ⟨?_, ?_, ?_⟩
    · ext b
      rw [mem_nzLevels]
      constructor
      · rintro ⟨hb, i, hi⟩
        by_cases h0 : y i = 0
        · rw [h0, relab_zero] at hi; simp at hi; omega
        · rw [relab_abs α _ h0] at hi; rw [← hi]; exact (hα _ (hmem i h0)).1
      · intro hb
        obtain ⟨a, ha, rfl⟩ := hsurj b hb
        have : a ∈ nzLevels y := by rw [hy]; exact ha
        obtain ⟨ha0, i, hi⟩ := (mem_nzLevels y a).mp this
        refine ⟨by have := (hα a ha).2.2; omega, i, ?_⟩
        have h0 : y i ≠ 0 := by intro h; rw [h] at hi; simp at hi; omega
        rw [relab_abs α _ h0, hi]
    · rw [zeros_eq, zeros_eq]
      apply congrArg Finset.card
      apply Finset.filter_congr
      intro i _
      constructor
      · intro h; by_contra h0; exact hne i h0 h
      · intro h; rw [h, relab_zero]
    · funext i
      by_cases h0 : y i = 0
      · rw [h0, relab_zero, relab_zero]
      · exact relab_relab α β _ h0 (hα _ (hmem i h0)).2.2 (hα _ (hmem i h0)).2.1
  have tU := transport U (Icc 1 k) φ ψ (fun a ha => ⟨(hφ a ha).1, (hφ a ha).2, hpos a ha⟩)
    (fun b hb => ⟨ψ b, (hψ b hb).1, (hψ b hb).2⟩) hU
  have tI := transport (Icc 1 k) U ψ φ (fun b hb => ⟨(hψ b hb).1, (hψ b hb).2, hposψ b hb⟩)
    (fun a ha => ⟨φ a, (hφ a ha).1, (hφ a ha).2⟩) (fun b hb => (mem_Icc.mp hb).1)
  apply Nat.card_congr
  exact
    { toFun := fun y => ⟨fun i => relab φ (y.1 i), (tU y.1 y.2.1).1,
        by rw [(tU y.1 y.2.1).2.1]; exact y.2.2.1,
        by rw [← relab_smul, y.2.2.2]⟩
      invFun := fun y => ⟨fun i => relab ψ (y.1 i), (tI y.1 y.2.1).1,
        by rw [(tI y.1 y.2.1).2.1]; exact y.2.2.1,
        by rw [← relab_smul, y.2.2.2]⟩
      left_inv := fun y => Subtype.ext (tU y.1 y.2.1).2.2
      right_inv := fun y => Subtype.ext (tI y.1 y.2.1).2.2 }

theorem WSet_iff (m : ℕ) (y : Fin 27 → ℤ) : WSet m y ↔ nzLevels y ⊆ Icc 1 m := by
  constructor
  · intro h v hv
    obtain ⟨hv0, i, rfl⟩ := (mem_nzLevels y v).mp hv
    rw [mem_Icc]; exact ⟨by omega, h i⟩
  · intro h i
    by_cases h0 : (y i).natAbs = 0
    · omega
    · have := h ((mem_nzLevels y _).mpr ⟨h0, i, rfl⟩)
      exact (mem_Icc.mp this).2

/-- Weak counts are binomial sums of exact counts. -/
theorem weak_sum (m : ℕ) (Q : ℕ → Prop) (g : D27) :
    Nat.card {y : Fin 27 → ℤ // WSet m y ∧ Q (zeros y) ∧ g • y = y} =
      ∑ k ∈ range (m + 1), m.choose k *
        Nat.card {y : Fin 27 → ℤ // nzLevels y = Icc 1 k ∧ Q (zeros y) ∧ g • y = y} := by
  classical
  have hfin : ∀ U : ↥((Icc 1 m).powerset),
      Finite {y : Fin 27 → ℤ // nzLevels y = U.1 ∧ Q (zeros y) ∧ g • y = y} := by
    intro U
    apply finite_of_bound m
    intro y hy i
    have hsub : nzLevels y ⊆ Icc 1 m := by rw [hy.1]; exact mem_powerset.mp U.2
    exact (WSet_iff m y).mpr hsub i
  have h1 : Nat.card {y : Fin 27 → ℤ // WSet m y ∧ Q (zeros y) ∧ g • y = y} =
      ∑ U ∈ (Icc 1 m).powerset,
        Nat.card {y : Fin 27 → ℤ // nzLevels y = U ∧ Q (zeros y) ∧ g • y = y} := by
    rw [← Finset.sum_coe_sort, ← Nat.card_sigma]
    apply Nat.card_congr
    exact
      { toFun := fun y => ⟨⟨nzLevels y.1, mem_powerset.mpr ((WSet_iff m y.1).mp y.2.1)⟩,
          ⟨y.1, rfl, y.2.2⟩⟩
        invFun := fun z => ⟨z.2.1, (WSet_iff m z.2.1).mpr (by
          rw [z.2.2.1]; exact mem_powerset.mp z.1.2), z.2.2.2⟩
        left_inv := fun y => rfl
        right_inv := fun z => by
          apply Sigma.subtype_ext
          · exact Subtype.ext z.2.2.1
          · rfl }
  rw [h1]
  have h2 : ∀ U ∈ (Icc 1 m).powerset,
      Nat.card {y : Fin 27 → ℤ // nzLevels y = U ∧ Q (zeros y) ∧ g • y = y} =
      Nat.card {y : Fin 27 → ℤ // nzLevels y = Icc 1 U.card ∧ Q (zeros y) ∧ g • y = y} := by
    intro U hU
    apply relabel_card U _ Q g
    intro a ha
    exact (mem_Icc.mp (mem_powerset.mp hU ha)).1
  rw [Finset.sum_congr rfl h2, Finset.sum_powerset_apply_card
    (fun k => Nat.card {y : Fin 27 → ℤ // nzLevels y = Icc 1 k ∧ Q (zeros y) ∧ g • y = y})]
  simp
```

```lean
/-! Canonical representatives of `ℤ × ℤ` modulo `±`, and the finite sets of classes. -/
namespace TypeD

/-- Canonical representative of `{v, -v}`. -/
def canon2 (v : ℤ × ℤ) : ℤ × ℤ := if 0 < v.1 ∨ (v.1 = 0 ∧ 0 ≤ v.2) then v else -v

/-- Sign of a nonzero pair relative to its canonical representative. -/
def sgn2 (v : ℤ × ℤ) : Sign := if 0 < v.1 ∨ (v.1 = 0 ∧ 0 < v.2) then 1 else minusSign

theorem canon2_neg (v : ℤ × ℤ) : canon2 (-v) = canon2 v := by
  obtain ⟨a, b⟩ := v
  simp only [canon2, Prod.neg_mk]
  split_ifs with h1 h2 h2 <;> simp only [neg_neg, Prod.mk.injEq] <;> omega

theorem canon2_cases (v : ℤ × ℤ) : canon2 v = v ∨ canon2 v = -v := by
  unfold canon2; split_ifs
  · exact Or.inl rfl
  · exact Or.inr rfl

theorem canon2_eq (v w : ℤ × ℤ) (h : canon2 v = canon2 w) : w = v ∨ w = -v := by
  rcases canon2_cases v with h1 | h1 <;> rcases canon2_cases w with h2 | h2
  · left; rw [← h2, ← h, h1]
  · right; rw [← neg_neg w, ← h2, ← h, h1]
  · right; rw [← h2, ← h, h1]
  · left; rw [← neg_neg w, ← h2, ← h, h1, neg_neg]

theorem sgnAct_pair (s : Sign) (v : ℤ × ℤ) : sgnAct s v = (sgnAct s v.1, sgnAct s v.2) := by
  rcases sign_two s with rfl | rfl
  · simp
  · simp only [sgnAct_minus]; rfl

theorem canon2_sgnAct (s : Sign) (v : ℤ × ℤ) : canon2 (sgnAct s v) = canon2 v := by
  rcases sign_two s with rfl | rfl
  · simp
  · rw [sgnAct_minus, canon2_neg]

theorem sgn2_neg (v : ℤ × ℤ) (hv : v ≠ 0) : sgn2 (-v) = minusSign * sgn2 v := by
  obtain ⟨a, b⟩ := v
  have hv' : a ≠ 0 ∨ b ≠ 0 := by
    by_contra h; push Not at h; apply hv; rw [h.1, h.2]; rfl
  simp only [sgn2, Prod.neg_mk]
  split_ifs with h1 h2 h2
  · omega
  · simp
  · simp
  · omega

theorem sgn2_sgnAct (s : Sign) (v : ℤ × ℤ) (hv : v ≠ 0) : sgn2 (sgnAct s v) = s * sgn2 v := by
  rcases sign_two s with rfl | rfl
  · simp
  · rw [sgnAct_minus, sgn2_neg v hv]

theorem canon2_fst_eq_zero (v : ℤ × ℤ) : (canon2 v).1 = 0 ↔ v.1 = 0 := by
  rcases canon2_cases v with h | h <;> simp [h]

theorem canon2_snd_eq_zero (v : ℤ × ℤ) : (canon2 v).2 = 0 ↔ v.2 = 0 := by
  rcases canon2_cases v with h | h <;> simp [h]

theorem canon2_eq_zero (v : ℤ × ℤ) : canon2 v = 0 ↔ v = 0 := by
  rcases canon2_cases v with h | h <;> simp [h]

theorem canon2_abs (v : ℤ × ℤ) :
    (canon2 v).1.natAbs = v.1.natAbs ∧ (canon2 v).2.natAbs = v.2.natAbs := by
  rcases canon2_cases v with h | h <;> rw [h] <;> simp

theorem canon2_canon2 (v : ℤ × ℤ) : canon2 (canon2 v) = canon2 v := by
  rcases canon2_cases v with h | h
  · rw [h, h]
  · rw [h, canon2_neg, h]

  obtain ⟨a, b⟩ := v
  simp only [canon2]
  split_ifs with h
  · simpa using h
  · simp only [Prod.neg_mk, Prod.mk.injEq]
    constructor
    · intro h'; exfalso; apply h; omega
    · intro h'; exact absurd h' h

def Cls (i j : ℕ) : Finset (ℤ × ℤ) :=
  ({0} : Finset ℤ) ×ˢ Finset.Icc 0 (j : ℤ) ∪ Finset.Icc 1 (i : ℤ) ×ˢ Finset.Icc (-(j : ℤ)) j

theorem mem_Cls (i j : ℕ) (q : ℤ × ℤ) :
```

```lean
  (msOf x.1.1 x.1.2, parOf x.1.1 x.1.2)

def flipAt (k : Fin 27) : SignedPerm 27 := ⟨fun i => if i = k then minusSign else 1, 1⟩

theorem flipAt_smul (k : Fin 27) (y : Fin 27 → ℤ) (h : y k = 0) : flipAt k • y = y := by
  funext i
  rw [smul_apply]
  change sgnAct (if i = k then minusSign else 1) (y i) = y i
  by_cases hi : i = k
  · subst hi; simp [h]
  · simp [hi]

theorem flipAt_sign (k : Fin 27) : signProduct 27 (flipAt k) = minusSign := by
  change (∏ i : Fin 27, if i = k then minusSign else 1) = minusSign
  simp

theorem flipAt_par (y w : Fin 27 → ℤ) (hz : ¬ zp y w) :
    parOf (flipAt 0 • y) (flipAt 0 • w) = minusSign * parOf y w := by
  classical
  have hz' : ¬ zp (flipAt 0 • y) (flipAt 0 • w) := fun h => hz ((zp_smul _ _ _).mp h)
  unfold parOf
  simp only [hz, hz', ↓reduceIte]
  rw [pair_prod_smul _ _ _ hz, flipAt_sign]

theorem invP_smul (g : D27) (x : PX i j Zc Zc') : invP i j Zc Zc' (g • x) = invP i j Zc Zc' x := by
  unfold invP
  change (msOf (g.val • x.1.1) (g.val • x.1.2), parOf (g • x.1.1) (g • x.1.2)) = _
  rw [msOf_smul, parOf_smul]

theorem invP_complete (x x' : PX i j Zc Zc') (h : invP i j Zc Zc' x = invP i j Zc Zc' x') :
    ∃ g : D27, g • x = x' := by
  classical
  obtain ⟨⟨y, w⟩, hx⟩ := x
  obtain ⟨⟨y', w'⟩, hx'⟩ := x'
  have hm : msOf y w = msOf y' w' := congrArg Prod.fst h
  have hp : parOf y w = parOf y' w' := congrArg Prod.snd h
  obtain ⟨b, hb⟩ := exists_smul_eq_of_multiset canon2 canon2_eq (zipV y w) (zipV y' w') hm
  have hyw : b • y = y' ∧ b • w = w' := by
    rw [← zip_smul] at hb
    exact ⟨funext fun k => congrArg Prod.fst (congrFun hb k),
      funext fun k => congrArg Prod.snd (congrFun hb k)⟩
  by_cases hs : signProduct 27 b = 1
  · refine ⟨⟨b, hs⟩, ?_⟩
    apply Subtype.ext
    exact Prod.ext hyw.1 hyw.2
  have hm1 : signProduct 27 b = minusSign := (sign_two _).resolve_left hs
  by_cases hz : zp y' w'
  · obtain ⟨k, hk1, hk2⟩ := hz
    refine ⟨⟨flipAt k * b, ?_⟩, ?_⟩
    · change signProduct 27 (flipAt k * b) = 1
      rw [map_mul, flipAt_sign, hm1, minusSign_sq]
    · apply Subtype.ext
      apply Prod.ext
      · change (flipAt k * b) • y = y'
        rw [mul_smul, hyw.1, flipAt_smul k y' hk1]
      · change (flipAt k * b) • w = w'
        rw [mul_smul, hyw.2, flipAt_smul k w' hk2]
  · exfalso
    have hz0 : ¬ zp y w := by
      intro h0; apply hz; rw [← hyw.1, ← hyw.2]; exact (zp_smul b y w).mpr h0
    unfold parOf at hp
    simp only [hz, hz0, ↓reduceIte] at hp
    rw [← hyw.1, ← hyw.2, pair_prod_smul _ _ _ hz0, hm1] at hp
    have : (1 : Sign) * ∏ k, sgn2 (y k, w k) = minusSign * ∏ k, sgn2 (y k, w k) := by
      rw [one_mul]; exact hp
    exact minusSign_ne_one (mul_right_cancel this).symm

/-- The condition on multisets coming from the zero counts. -/
def RZ (Zc Zc' : ℕ → Prop) (M : Multiset (ℤ × ℤ)) : Prop :=
  Zc (Multiset.card (M.filter (fun q => q.1 = 0))) ∧
    Zc' (Multiset.card (M.filter (fun q => q.2 = 0)))

theorem invP_props (x : PX i j Zc Zc') :
    Multiset.card (msOf x.1.1 x.1.2) = 27 ∧ (∀ q ∈ msOf x.1.1 x.1.2, q ∈ Cls i j) ∧
      RZ Zc Zc' (msOf x.1.1 x.1.2) := by
  obtain ⟨⟨y, w⟩, ⟨hy1, hy2⟩, ⟨hw1, hw2⟩⟩ := x
  refine ⟨by simp [msOf], ?_, ?_⟩
  · intro q hq
    obtain ⟨k, _, rfl⟩ := Multiset.mem_map.mp hq
    rw [mem_Cls]
    obtain ⟨a1, a2⟩ := canon2_abs (y k, w k)
    exact ⟨canon2_canon2 _, by rw [a1]; exact hy1 k, by rw [a2]; exact hw1 k⟩
  · unfold RZ
    rw [count_fst_zero, count_snd_zero]
    exact ⟨hy2, hw2⟩

theorem realize (M : Multiset (ℤ × ℤ)) (h1 : Multiset.card M = 27) (h2 : ∀ q ∈ M, q ∈ Cls i j)
    (h3 : RZ Zc Zc' M) (s : Sign) (hs : (0 : ℤ × ℤ) ∈ M → s = 1) :
    ∃ x : PX i j Zc Zc', invP i j Zc Zc' x = (M, s) := by
  classical
  obtain ⟨l, rfl⟩ := Quotient.exists_rep M
  have hl : l.length = 27 := by simpa using h1
  let f : Fin 27 → ℤ × ℤ := fun k => l.get (Fin.cast hl.symm k)
  have hfM : Multiset.map f Finset.univ.val = (Quotient.mk _ l : Multiset (ℤ × ℤ)) := by
    rw [Fin.univ_val_map]
    have : List.ofFn f = l := (List.ofFn_congr hl l.get).symm.trans (List.ofFn_get l)
    rw [this]; rfl
  have hmem : ∀ k, f k ∈ Cls i j := by
    intro k
    apply h2
    show f k ∈ (l : Multiset (ℤ × ℤ))
    rw [Multiset.mem_coe]
    exact List.get_mem _ _
  have hcan : ∀ k, canon2 (f k) = f k := fun k => ((mem_Cls i j _).mp (hmem k)).1
  let y : Fin 27 → ℤ := fun k => (f k).1
  let w : Fin 27 → ℤ := fun k => (f k).2
  have hms : msOf y w = Quotient.mk _ l := by
    rw [← hfM]
    unfold msOf
    congr 1
    funext k
    exact hcan k
  have hPy : PP i Zc y := by
    refine ⟨fun k => ((mem_Cls i j _).mp (hmem k)).2.1, ?_⟩
    rw [← count_fst_zero y w, hms]; exact h3.1
  have hPw : PP j Zc' w := by
    refine ⟨fun k => ((mem_Cls i j _).mp (hmem k)).2.2, ?_⟩
    rw [← count_snd_zero y w, hms]; exact h3.2
  let x : PX i j Zc Zc' := ⟨(y, w), hPy, hPw⟩
  by_cases h0 : zp y w
  · refine ⟨x, ?_⟩
    have hs1 : s = 1 := hs (by rw [← hms]; exact (zero_mem_msOf y w).mpr h0)
    show (msOf y w, parOf y w) = _
    rw [hms, hs1]
    unfold parOf
    simp [h0]
  · by_cases hps : parOf y w = s
    · exact ⟨x, by show (msOf y w, parOf y w) = _; rw [hms, hps]⟩
    · let x' : PX i j Zc Zc' := ⟨(flipAt 0 • y, flipAt 0 • w),
        (PP_smul i Zc _ y).mpr hPy, (PP_smul j Zc' _ w).mpr hPw⟩
      refine ⟨x', ?_⟩
      show (msOf (flipAt 0 • y) (flipAt 0 • w), parOf (flipAt 0 • y) (flipAt 0 • w)) = _
      rw [msOf_smul, hms, flipAt_par y w h0]
      congr 1
      rcases sign_two (parOf y w) with h | h <;> rcases sign_two s with h' | h' <;>
        simp_all [minusSign_sq]

/-- The number of orbits of pairs. -/
theorem orbits_eq :
    Nat.card (MulAction.orbitRel.Quotient D27 (PX i j Zc Zc')) =
      Nat.card (MSet (Cls i j) 27 (RZ Zc Zc')) +
        Nat.card (MSet ((Cls i j).erase 0) 27 (RZ Zc Zc')) := by
  classical
  rw [card_orbits_of_invariant (invP i j Zc Zc') (invP_smul i j Zc Zc')
    (invP_complete i j Zc Zc'), ← Nat.card_sum]
  apply Nat.card_congr
  have hprops : ∀ z : Set.range (invP i j Zc Zc'),
      Multiset.card z.1.1 = 27 ∧ (∀ q ∈ z.1.1, q ∈ Cls i j) ∧ RZ Zc Zc' z.1.1 ∧
        ((0 : ℤ × ℤ) ∈ z.1.1 → z.1.2 = 1) := by
    rintro ⟨_, x, rfl⟩
    obtain ⟨a, b, c⟩ := invP_props i j Zc Zc' x
    refine ⟨a, b, c, fun h0 => ?_⟩
    change parOf x.1.1 x.1.2 = 1
    unfold parOf
    simp [(zero_mem_msOf _ _).mp h0]
  exact
    { toFun := fun z =>
        if hs : z.1.2 = 1 then
          Sum.inl ⟨z.1.1, (hprops z).1, (hprops z).2.1, (hprops z).2.2.1⟩
        else
          Sum.inr ⟨z.1.1, (hprops z).1, fun q hq => Finset.mem_erase.mpr
            ⟨fun h0 => hs ((hprops z).2.2.2 (h0 ▸ hq)), (hprops z).2.1 q hq⟩, (hprops z).2.2.1⟩
      invFun := fun s => match s with
        | Sum.inl M => ⟨(M.1, 1), realize i j Zc Zc' M.1 M.2.1 M.2.2.1 M.2.2.2 1 (fun _ => rfl)⟩
        | Sum.inr M => ⟨(M.1, minusSign), realize i j Zc Zc' M.1 M.2.1
            (fun q hq => (Finset.mem_erase.mp (M.2.2.1 q hq)).2) M.2.2.2 minusSign
            (fun h0 => absurd rfl (Finset.mem_erase.mp (M.2.2.1 _ h0)).1)⟩
      left_inv := fun z => by
        by_cases hs : z.1.2 = 1
        · simp only [hs, ↓reduceDIte]
          apply Subtype.ext
          exact Prod.ext rfl hs.symm
        · simp only [hs, ↓reduceDIte]
          apply Subtype.ext
          exact Prod.ext rfl ((sign_two _).resolve_left hs).symm
      right_inv := fun s => by
        rcases s with M | M
        · simp
        · simp [minusSign_ne_one] }

theorem gram_formula :
    ∑ g : D27, fixCard (PP i Zc) g * fixCard (PP j Zc') g =
      (Nat.card (MSet (Cls i j) 27 (RZ Zc Zc')) +
        Nat.card (MSet ((Cls i j).erase 0) 27 (RZ Zc Zc'))) * Nat.card D27 := by
  rw [gram_burnside, orbits_eq]

end
```

```lean
def fmc (K n : ℕ) : ℕ := (K + n - 1).descFactorial n / n.factorial
/-- Binomial coefficient, computed through a descending factorial. -/
def bc (n k : ℕ) : ℕ := n.descFactorial k / k.factorial

def sumTo (f : ℕ → ℤ) : ℕ → ℤ
  | 0 => 0
  | n + 1 => sumTo f n + f n

def sgnp (k : ℕ) : ℤ := if k % 2 = 0 then 1 else -1
/-- Coefficient of the weak character `W_i` in the exact character `B_d`. -/
def al (d i : ℕ) : ℤ := if i ≤ d then sgnp (d - i) * bc d i else 0
/-- Coefficient of `ZW_i` in `D_d = B_d - Z_d - Z_(d-1)`. -/
def be (d i : ℕ) : ℤ := -(al d i + al (d - 1) i)

def gWW (i j : ℕ) : ℤ := fmc (2 * i * j + i + j + 1) 27 + fmc (2 * i * j + i + j) 27
def gZW (i j : ℕ) : ℤ := (j + 1) * fmc (i * (2 * j + 1)) 26 + j * fmc (i * (2 * j + 1)) 26
def gZZ (i j : ℕ) : ℤ := fmc (2 * i * j) 26 + 2 * (i * j * fmc (2 * i * j) 25)

def term (d e i j : ℕ) : ℤ :=
  al d i * al e j * gWW i j + al d i * be e j * gZW j i +
    be d i * al e j * gZW i j + be d i * be e j * gZZ i j

/-- The explicit value of the metamatrix entry with face dimensions `d = 27 - p`, `e = 27 - q`. -/
def Fc (d e : ℕ) : ℤ := sumTo (fun i => sumTo (fun j => term d e i j) (e + 1)) (d + 1)

theorem fmc_eq (K n : ℕ) : fmc K n = Nat.multichoose K n := by
  rw [fmc, Nat.multichoose_eq, Nat.choose_eq_descFactorial_div_factorial]

theorem bc_eq (n k : ℕ) : bc n k = n.choose k := by
  rw [bc, Nat.choose_eq_descFactorial_div_factorial]

theorem sumTo_eq (f : ℕ → ℤ) (n : ℕ) : sumTo f n = ∑ k ∈ range n, f k := by
  induction n with
  | zero => rfl
  | succ n ih => rw [sumTo, ih, Finset.sum_range_succ]

theorem sgnp_eq (k : ℕ) : sgnp k = (-1) ^ k := by
  rw [sgnp, neg_one_pow_eq_pow_mod_two]
  rcases Nat.mod_two_eq_zero_or_one k with h | h <;> simp [h]

theorem al_eq (d i : ℕ) : al d i = (-1) ^ (d - i) * (d.choose i : ℤ) := by
  unfold al
  split_ifs with h
  · rw [sgnp_eq, bc_eq]
  · rw [Nat.choose_eq_zero_of_lt (by omega)]; simp

theorem al_zero_of_lt (d i : ℕ) (h : d < i) : al d i = 0 := by
  simp [al, show ¬ i ≤ d by omega]

theorem be_zero_of_lt (d i : ℕ) (h : d < i) : be d i = 0 := by
  simp [be, al, show ¬ i ≤ d by omega, show ¬ i ≤ d - 1 by omega]

end TypeD
```

```lean
  simp only [Finset.sum_add_distrib, ← Finset.mul_sum]
  rw [gWW_gram, gZW_gram, gZW_gram, gZZ_gram]
  unfold term; ring

  rw [Finset.sum_comm]
  rw [Finset.sum_congr rfl (fun i _ => Finset.sum_comm)]
  simp only [term_gram, ← Finset.sum_mul]
  congr 1
  unfold Fc

  · rw [sumTo_eq]
    apply Finset.sum_eq_zero
    intro j _
    simp [term, al_zero_of_lt d i hi, be_zero_of_lt d i hi]

      (Fc (27 - p.val) (27 - q.val) : ℚ) * (Nat.card D27 : ℚ) := by

  exact mul_right_cancel₀ hpos h4
```

```lean
  rw [← Fin.sum_univ_eq_sum_range (fun k => Eent i.val k * Aent k j.val) 16] at this
  exact this

theorem det_Aint_neg : Aint.det < 0 := by
  have hE : Eint.det = ∏ i, Eint i i :=
    Matrix.det_of_isLowerTriangular Eint (fun i j h => E_lower i j h)
  have hV : Vint.det = ∏ i, Vint i i :=
    Matrix.det_of_isUpperTriangular (fun i j h => V_upper i j h)
  have hEpos : 0 < Eint.det := by
    rw [hE]; exact Finset.prod_pos (fun i _ => E_diag_pos i)
  have hVneg : Vint.det < 0 := by
    rw [hV, Fin.prod_univ_castSucc]
    have h1 : 0 < ∏ i : Fin 15, Vint i.castSucc i.castSucc :=
      Finset.prod_pos (fun i _ => V_diag_pos i)
    have h2 : Vint (Fin.last 15) (Fin.last 15) < 0 := V_last_neg
    exact mul_neg_of_pos_of_neg h1 h2
  have hm := Matrix.det_mul Eint Aint
  rw [EA_eq] at hm
  by_contra hA
  push Not at hA
  have : 0 ≤ Eint.det * Aint.det := mul_nonneg hEpos.le hA
  linarith

/-- **The negative minor.** -/
theorem negativeMinor27 : NegativeMinor27 := by
  unfold NegativeMinor27
  rw [minor_eq]
  have h := RingHom.map_det (Int.castRingHom ℚ) Aint
  rw [RingHom.mapMatrix_apply] at h
  rw [← h, eq_intCast]
  exact_mod_cast det_Aint_neg

/-- **Refutation (E2 endpoint).** The literal metamatrix of the even signed permutation group of
`Fin 27` with its 27 standard generators is not totally positive in the source's strict sense. -/
theorem not_totallyPositive_M27 : ¬ TotallyPositive M27 :=
  negativeMinor_not_totallyPositive negativeMinor27
```
