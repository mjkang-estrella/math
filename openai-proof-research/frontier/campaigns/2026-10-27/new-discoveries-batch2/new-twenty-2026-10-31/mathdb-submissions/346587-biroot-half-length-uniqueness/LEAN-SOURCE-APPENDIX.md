```lean
/-- Includes monoids generated by zero or one element, using the identity. -/
def TwoGenerated (S : Submonoid M) : Prop :=
  ∃ x y : M, S = Submonoid.closure ({x, y} : Set M)

/-- The maximality clause; two-generation of S is a separate hypothesis. -/
def TwoMaximal (S : Submonoid M) : Prop :=
  ∀ T : Submonoid M, TwoGenerated T → S ≤ T → S = T

theorem twoGenerated_map (f : M →* N) {S : Submonoid M}
    (hS : TwoGenerated S) : TwoGenerated (S.map f) := by
  obtain ⟨x, y, rfl⟩ := hS
  refine ⟨f x, f y, ?_⟩
  simp [MonoidHom.map_mclosure]

/-- A proper two-generator extension before decoding would remain proper after
an injective decoding. This is the direction used by the written proof. -/
theorem twoMaximal_reflected (f : M →* N) (hf : Function.Injective f)
    {S : Submonoid M} (hS : TwoMaximal (S.map f)) : TwoMaximal S := by
  intro T hT hST
  have hmap : S.map f ≤ T.map f := by
    intro b hb
    obtain ⟨a, ha, rfl⟩ := hb
    exact ⟨a, hST ha, rfl⟩
  have heq : S.map f = T.map f := hS (T.map f) (twoGenerated_map f hT) hmap
  exact Submonoid.map_injective_of_injective hf heq

/-- Equality of encoded monoids would imply equality of their decoded images. -/
theorem distinct_preimages (f : M →* N) {S T : Submonoid M}
    (hne : S.map f ≠ T.map f) : S ≠ T := by
  intro h
  exact hne (congrArg (fun Q : Submonoid M => Q.map f) h)

/-- Nonidentity indecomposable elements are the basis elements of a word monoid. -/
def AtomIn (S : Submonoid M) (x : M) : Prop :=
  x ∈ S ∧ x ≠ 1 ∧ ∀ a b : M, a ∈ S → b ∈ S → x = a * b → a = 1 ∨ b = 1

/-- Injective decoding preserves and reflects the entire atom set, so a
literal two-element basis is inherited along with maximality. -/

    (S : Submonoid M) (x : M) : AtomIn (S.map f) (f x) ↔ AtomIn S x := by
  constructor
  · intro h
    constructor
    · obtain ⟨a, ha, heq⟩ := h.1
      exact hf heq ▸ ha
    constructor
    · intro hx
      apply h.2.1
      rw [hx, map_one]
    · intro a b ha hb hprod

      rcases h.2.2 (f a) (f b) ⟨a, ha, rfl⟩ ⟨b, hb, rfl⟩ heq with hu | hu

      · exact Or.inr (hf (hu.trans (map_one f).symm))

    refine ⟨⟨x, h.1, rfl⟩, ?_, ?_⟩
    · intro hx
      exact h.2.1 (hf (hx.trans (map_one f).symm))

  | nil => simp
```

```lean
/-! A second scoped checkpoint: incidence coboundary kernels and component counts.
This file does not prove the folding theorem or the bi-root conjecture. -/
namespace BiRoot.Incidence
variable {E V : Type*} (K : Type*) [Field K] (s t : E → V)

def EndpointRelation (u v : V) : Prop := ∃ e : E, s e = u ∧ t e = v

def Component := Quot (EndpointRelation s t)

instance [Finite V] : Finite (Component s t) := inferInstanceAs (Finite (Quot _))

def coboundary : (V → K) →ₗ[K] (E → K) :=
  LinearMap.pi (fun e => (LinearMap.proj (s e) : (V → K) →ₗ[K] K) -
    LinearMap.proj (t e))

@[simp] theorem coboundary_apply (f : V → K) (e : E) :
    coboundary K s t f e = f (s e) - f (t e) := rfl

/-- A function descends to endpoint components exactly when its coboundary vanishes. -/
noncomputable def componentKernelEquiv :
    (Component s t → K) ≃ₗ[K] LinearMap.ker (coboundary K s t) where
  toFun g := ⟨fun v => g (Quot.mk _ v), by
    apply LinearMap.mem_ker.mpr
    funext e
    change g (Quot.mk _ (s e)) - g (Quot.mk _ (t e)) = 0
    have h : (Quot.mk (EndpointRelation s t) (s e)) = Quot.mk _ (t e) :=
      Quot.sound ⟨e, rfl, rfl⟩
    rw [h, sub_self]⟩
  invFun f := Quot.lift (fun v => f.val v) (by
    intro u v h
    obtain ⟨e, rfl, rfl⟩ := h
    have hf := congrFun (LinearMap.mem_ker.mp f.prop) e
    exact sub_eq_zero.mp hf)
  left_inv g := by
    funext q
    refine Quot.inductionOn q ?_
    intro v
    rfl
  right_inv f := by
    apply Subtype.ext
    rfl
  map_add' g h := by
    apply Subtype.ext
    rfl
  map_smul' a g := by
    apply Subtype.ext
    rfl

    Module.finrank K (LinearMap.ker (coboundary K s t)) = Nat.card (Component s t) := by
  let : Fintype (Component s t) := Fintype.ofFinite _
  calc
    Module.finrank K (LinearMap.ker (coboundary K s t)) =
        Module.finrank K (Component s t → K) := (componentKernelEquiv K s t).finrank_eq.symm
    _ = Nat.card (Component s t) := by
      rw [Module.finrank_fintype_fun_eq_card, Nat.card_eq_fintype_card]

    Module.finrank K (LinearMap.range (coboundary K s t)) + Nat.card (Component s t) =
      Fintype.card V := by
  rw [← finrank_coboundary_ker K s t]
  simpa only [Module.finrank_fintype_fun_eq_card] using
    (coboundary K s t).finrank_range_add_finrank_ker

Via the canonical finite-coordinate dual basis this is the usual incidence kernel. -/
def dualCycleSpace := LinearMap.ker (coboundary K s t).dualMap

/-- A subtraction-free cycle-space dimension formula, valid even with isolated
vertices and loops. The endpoint quotient counts undirected components. -/
theorem dualCycleSpace_rank_identity [Fintype E] [Fintype V] :

      Fintype.card E + Nat.card (Component s t) := by
  have hc := coboundary_rank_identity K s t
  have hd := (coboundary K s t).dualMap.finrank_range_add_finrank_ker
  rw [LinearMap.finrank_range_dualMap_eq_finrank_range,
    Subspace.dual_finrank_eq, Module.finrank_fintype_fun_eq_card] at hd
  change Module.finrank K (LinearMap.range (coboundary K s t)) +
    Module.finrank K (dualCycleSpace K s t) = Fintype.card E at hd
  calc
    Module.finrank K (dualCycleSpace K s t) + Fintype.card V =
        Module.finrank K (dualCycleSpace K s t) +
          (Module.finrank K (LinearMap.range (coboundary K s t)) +

              Module.finrank K (dualCycleSpace K s t) + n) hc.symm
    _ = (Module.finrank K (LinearMap.range (coboundary K s t)) +
          Module.finrank K (dualCycleSpace K s t)) + Nat.card (Component s t) := by ac_rfl
    _ = Fintype.card E + Nat.card (Component s t) := by rw [hd]
```

```lean
variable {K E A : Type*} [Field K] [Fintype E] [Fintype A] [DecidableEq A]
variable (Z : Submodule K (E → K)) (χ : E → A)

def colorSpace (a : A) : Submodule K (E → K) where
  carrier := {z | z ∈ Z ∧ ∀ e, χ e ≠ a → z e = 0}
  zero_mem' := ⟨Z.zero_mem, fun _ _ => rfl⟩
  add_mem' := by
    intro x y hx hy
    refine ⟨Z.add_mem hx.1 hy.1, ?_⟩
    intro e he
    change x e + y e = 0
    rw [hx.2 e he, hy.2 e he, zero_add]
  smul_mem' := by
    intro c x hx
    refine ⟨Z.smul_mem c hx.1, ?_⟩
    intro e he
    change c * x e = 0
    rw [hx.2 e he, mul_zero]

def sumColors : (∀ a, colorSpace Z χ a) →ₗ[K] Z where
  toFun f := ⟨∑ a, (f a).val, Z.sum_mem (fun a _ => (f a).prop.1)⟩
  map_add' := by
    intro f g
    apply Subtype.ext
    change (∑ a, ((f + g) a).val) = (∑ a, (f a).val) + (∑ a, (g a).val)
    simp only [Pi.add_apply, Submodule.coe_add, Finset.sum_add_distrib]
  map_smul' := by
    intro c f
    apply Subtype.ext
    change (∑ a, ((c • f) a).val) = c • (∑ a, (f a).val)
    simp only [Pi.smul_apply, Submodule.coe_smul, Finset.smul_sum]

    by_contra hn
    apply hz.2.1
    funext e

  have hy : y ∈ Z := color_restriction_mem Z χ hdim z hz.1 (χ e₀)

    intro h
    have hc := congrFun h e₀
    simp only [y, ite_eq_left rfl, Pi.zero_apply] at hc
    exact he₀ hc

    apply hye

  intro e hze
  by_contra hne
  exact hfull e hze (by simp only [y, ite_eq_right hne])
```

```lean
theorem pow_prefix_of_period (w : List α) (d n : ℕ)
    (hlen : w.length = n * d) (hper : List.HasPeriod w d) :
    FreeMonoid.ofList w = (FreeMonoid.ofList (w.take d)) ^ n := by

    subst w
    rfl
  | succ n ih =>
    by_cases hn : n = 0

      have hwlen : w.length = d := by simpa using hlen
      have ht : w.take d = w := List.take_of_length_le hwlen.le
      simp only [Nat.zero_add, ht, pow_one]

        rw [List.length_drop, hlen, Nat.succ_mul]
        exact Nat.add_sub_cancel_right (n * d) d
      have hdropge : d ≤ (w.drop d).length := by
        rw [hdrop]

          simpa only [List.append_nil, List.take_append_drop] using hper
        exact hcat.factor

        obtain ⟨tail, htail⟩ := hper.drop_prefix d
        have h := congrArg (List.take d) htail

          rw [← FreeMonoid.ofList_append, List.take_append_drop]
        _ = FreeMonoid.ofList (w.take d) * (FreeMonoid.ofList (w.take d)) ^ n := by
          rw [ih (w.drop d) hdrop hpdrop, ht]
        _ = (FreeMonoid.ofList (w.take d)) ^ (n + 1) := (pow_succ' _ _).symm

theorem period_of_two_splits (w x y : List α) (hxy : w = x ++ y)
    (hyx : w = y ++ x) : List.HasPeriod w x.length := by
  unfold List.HasPeriod
  have ht : w.take x.length = x := by rw [hxy]; simp
  rw [ht]
  refine ⟨x, ?_⟩
  calc
    w ++ x = (x ++ y) ++ x := by rw [hxy]
    _ = x ++ (y ++ x) := List.append_assoc _ _ _
    _ = x ++ w := by rw [← hyx]

theorem primitive_no_two_positive_splits (w : FreeMonoid α) (hw : PrimitiveWord w)
    (x y : List α) (hxy : w.toList = x ++ y) (hyx : w.toList = y ++ x)
    (hx : 0 < x.length) (hxshort : x.length < w.length) : False := by
  let d := x.length.gcd y.length
  have hlen : x.length + y.length = w.length := by
    simpa only [List.length_append, FreeMonoid.length] using congrArg List.length hxy.symm
  have hp := period_of_two_splits w.toList x y hxy hyx
  have hq := period_of_two_splits w.toList y x hyx hxy
  have hperiod : List.HasPeriod w.toList d := hp.gcd hq (by
    rw [hlen]
    exact Nat.sub_le _ _)
  have hdvd : d ∣ w.length := by
    rw [← hlen]
    exact Nat.dvd_add (Nat.gcd_dvd_left _ _) (Nat.gcd_dvd_right _ _)
  obtain ⟨n, hn⟩ := hdvd
  have hnlen : w.toList.length = n * d := by
    simpa only [FreeMonoid.length, Nat.mul_comm] using hn
  have hpow : w = (FreeMonoid.ofList (w.toList.take d)) ^ n :=
    pow_prefix_of_period w.toList d n hnlen hperiod
  have hn1 : n = 1 := hw.2 _ n hpow
  have hwd : w.length = d := by simpa only [hn1, Nat.mul_one] using hn
  have hdle : d ≤ x.length := Nat.le_of_dvd hx (Nat.gcd_dvd_left _ _)
  exact (Nat.ne_of_lt (lt_of_le_of_lt hdle hxshort)) hwd.symm

theorem primitive_not_fixed_rotation (w : FreeMonoid α) (hw : PrimitiveWord w)
    (k : ℕ) (hk : 0 < k) (hkshort : k < w.length) :
    w.toList.rotate k ≠ w.toList := by
  intro hrot
  have hxy : w.toList = w.toList.take k ++ w.toList.drop k :=
    (List.take_append_drop k w.toList).symm
  have hyx : w.toList = w.toList.drop k ++ w.toList.take k := by
    rw [List.rotate_eq_drop_append_take hkshort.le] at hrot
    exact hrot.symm
  have htake : (w.toList.take k).length = k := by
    rw [List.length_take]
    change min k w.length = k
    exact Nat.min_eq_left hkshort.le
  have hxp : 0 < (w.toList.take k).length := by rw [htake]; exact hk
  have hxlt : (w.toList.take k).length < w.length := by rw [htake]; exact hkshort
  exact primitive_no_two_positive_splits w hw _ _ hxy hyx hxp hxlt
```

```lean
def readFrom (step : Q → α → Option Q) (q : Q) : List α → Option Q
  | [] => some q
  | a :: w => (step q a).bind (fun r => readFrom step r w)

theorem readFrom_append (step : Q → α → Option Q) (q : Q) (u v : List α) :
    readFrom step q (u ++ v) = (readFrom step q u).bind (fun r => readFrom step r v) := by
  induction u generalizing q with
  | nil => rfl
  | cons a u ih =>
    cases hs : step q a with
    | none => simp only [List.cons_append, readFrom, hs, Option.bind_none]
    | some r => simpa only [List.cons_append, readFrom, hs, Option.bind_some] using ih r

/-- Every positive closed word can be split into genuine first-return words. -/
theorem exists_firstReturn_decomposition (step : Q → α → Option Q) (root : Q)
    (w : List α) (hw : readFrom step root w = some root) :
    ∃ chunks : List (List α), chunks.flatten = w ∧
      ∀ z ∈ chunks, FirstReturnWord (readFrom step root) root z := by
  classical
  generalize hn : w.length = n
  induction n using Nat.strong_induction_on generalizing w with
  | h n ih =>
    by_cases hnil : w = []
    · subst w
      exact ⟨[], rfl, by simp only [List.not_mem_nil, false_implies, implies_true]⟩
    · have hex : ∃ k : ℕ, 0 < k ∧ k ≤ w.length ∧
          readFrom step root (w.take k) = some root := by
        refine ⟨w.length, List.length_pos_iff.mpr hnil, le_rfl, ?_⟩
        simpa only [List.take_length] using hw
      let k := Nat.find hex
      have hk := Nat.find_spec hex
      change 0 < k ∧ k ≤ w.length ∧ readFrom step root (w.take k) = some root at hk
      let u := w.take k
      let v := w.drop k
      have hlen : u.length = k := by
        simp only [u, List.length_take, Nat.min_eq_left hk.2.1]
      have hune : u ≠ [] := by
        intro he
        have hzero := congrArg List.length he
        rw [hlen] at hzero
        exact (Nat.ne_of_gt hk.1) hzero
      have hfirst : FirstReturnWord (readFrom step root) root u := by
        refine ⟨hune, hk.2.2, ?_⟩
        intro x y hxy hx hy hxret
        have hypos : 0 < y.length := List.length_pos_iff.mpr hy
        have hxpos : 0 < x.length := List.length_pos_iff.mpr hx

          rw [← List.length_append, ← hxy, hlen]
        have hxlt : x.length < k := by omega
        have hwxy : w = x ++ (y ++ v) := by
          calc
            w = u ++ v := (List.take_append_drop k w).symm
            _ = (x ++ y) ++ v := by rw [hxy]
            _ = x ++ (y ++ v) := List.append_assoc _ _ _
        have hxtake : w.take x.length = x := by rw [hwxy]; simp
        have hcandidate : 0 < x.length ∧ x.length ≤ w.length ∧
            readFrom step root (w.take x.length) = some root := by
          refine ⟨hxpos, by omega, ?_⟩
          rw [hxtake]
          exact hxret

flattening the two piece lists gives the concatenated positive word. -/
theorem append_firstReturn_decompositions (step : Q → α → Option Q) (root : Q)
```

```lean
import TargetDefinitions

/-! Universe-polymorphic first-return induction. The geometric step is an explicit
hypothesis. This module does not assert the source conjecture unconditionally. -/
namespace BiRoot.Descent
universe u

local instance {α : Type u} [DecidableEq α] : DecidableEq (FreeMonoid α) :=
  inferInstanceAs (DecidableEq (List α))

structure Problem (α : Type u) [DecidableEq α] where
  ρ : α → ℕ
  positive : ∀ a, 0 < ρ a
  w : FreeMonoid α
  primitive : PrimitiveWord w
  X : Finset (FreeMonoid α)
  U : Finset (FreeMonoid α)
  primitiveX : PrimitivePair X
  primitiveU : PrimitivePair U
  distinct : X ≠ U
  memX : w ∈ generatedMonoid X
  memU : w ∈ generatedMonoid U

def cost {α : Type u} [DecidableEq α] (p : Problem α) : ℕ :=
  (∑ x ∈ p.X, weightedLength p.ρ x) + (∑ x ∈ p.U, weightedLength p.ρ x)

def Bound {α : Type u} [DecidableEq α] (p : Problem α) : Prop :=
  weightedLength p.ρ p.w + 3 ≤ cost p

/-- A concrete change of alphabet, with all unproved geometric input exposed.
Primitive words and pairs need not be assumed over S: decoding reflects them. -/
structure ReturnReduction {α : Type u} [DecidableEq α] (p : Problem α) where
  S : Set (List α)
  finite : S.Finite
  prefixFree : InformationTheory.IsPrefixFree S
  nonemptyLetters : [] ∉ S
  w' : FreeMonoid S
  X' : Finset (FreeMonoid S)
  U' : Finset (FreeMonoid S)
  decodeW : decode S w' = p.w
  decodeX : X'.image (decode S) = p.X
  decodeU : U'.image (decode S) = p.U
  memX : w' ∈ generatedMonoid X'
  memU : w' ∈ generatedMonoid U'
  shorter : w'.length < p.w.length

namespace ReturnReduction
variable {α : Type u} [DecidableEq α] {p : Problem α}

noncomputable def encoded (r : ReturnReduction p) : Problem r.S := by
  classical
  exact
    { ρ := fun b => weightedLength p.ρ (FreeMonoid.ofList b.val)
      positive := returnWeight_pos r.S r.nonemptyLetters p.ρ p.positive
      w := r.w'
      primitive := primitiveWord_reflected (decode r.S) (r.decodeW.symm ▸ p.primitive)
      X := r.X'
      U := r.U'
      primitiveX := primitivePair_reflected (decode r.S)
        (decode_injective r.S r.prefixFree r.nonemptyLetters) r.X'
        (r.decodeX.symm ▸ p.primitiveX)
      primitiveU := primitivePair_reflected (decode r.S)
        (decode_injective r.S r.prefixFree r.nonemptyLetters) r.U'
        (r.decodeU.symm ▸ p.primitiveU)
      distinct := by
        intro h
        apply p.distinct
        rw [← r.decodeX, ← r.decodeU, h]
      memX := r.memX
      memU := r.memU }

theorem weighted_word (r : ReturnReduction p) :
    weightedLength r.encoded.ρ r.encoded.w = weightedLength p.ρ p.w := by
  change weightedLength (fun b : r.S =>
    weightedLength p.ρ (FreeMonoid.ofList b.val)) r.w' = _
  rw [← weightedLength_decode, r.decodeW]

theorem cost_eq (r : ReturnReduction p) : cost r.encoded = cost p := by
  classical
  have hi := decode_injective r.S r.prefixFree r.nonemptyLetters
  have hsum (Z : Finset (FreeMonoid r.S)) :
      (∑ z ∈ Z.image (decode r.S), weightedLength p.ρ z) =
        ∑ z ∈ Z, weightedLength r.encoded.ρ z := by
    rw [Finset.sum_image]
    · apply Finset.sum_congr rfl
      intro z _
      exact weightedLength_decode r.S p.ρ z
    · intro a _ b _ hab
      exact hi hab
  unfold cost
  change _ = (∑ x ∈ p.X, weightedLength p.ρ x) +
    (∑ x ∈ p.U, weightedLength p.ρ x)
  rw [← r.decodeX, ← r.decodeU, hsum, hsum]
  rfl

theorem transfer (r : ReturnReduction p) (hb : Bound r.encoded) : Bound p := by
  unfold Bound at hb ⊢
  rw [r.weighted_word, r.cost_eq] at hb
  exact hb
end ReturnReduction

/-- This is the remaining geometric dichotomy, not an assumption hidden in a
new constant. Every use of conditional induction takes a proof of Step. -/
def Step : Prop :=
  ∀ (α : Type u) [DecidableEq α] (p : Problem α),
    Bound p ∨ Nonempty (ReturnReduction p)

/-- The induction hypothesis quantifies over types, not only over words in the
original alphabet. The return alphabet is a subtype of List α in the same universe. -/
theorem bound_of_step (step : Step.{u}) :
    ∀ (α : Type u) [DecidableEq α] (p : Problem α), Bound p := by
  classical
  have main : ∀ n : ℕ, ∀ (α : Type u) [DecidableEq α] (p : Problem α),
      p.w.length = n → Bound p := by
    intro n
    induction n using Nat.strong_induction_on with
    | h n ih =>
      intro α _ p hn
      have hcase : Bound p ∨ Nonempty (ReturnReduction p) := step α p
      rcases hcase with hb | hr
      · exact hb
      · obtain ⟨r⟩ := hr
        apply r.transfer
        exact ih r.w'.length (by simpa only [hn] using r.shorter)
          r.S r.encoded rfl
  intro α _ p
  exact main p.w.length α p rfl

/-- A conditional literal endpoint. It has an explicit Step argument. -/
theorem half_length_of_step (step : Step.{u}) (α : Type u) [DecidableEq α] :
    HalfLengthUniquenessStatement α := by
  apply weighted_implies_half
  intro ρ hρ w hw X U hX hU hne hx hu
  let p : Problem α := ⟨ρ, hρ, w, hw, X, U, hX, hU, hne, hx, hu⟩
```

```lean
/-- Injective changes of vertex coordinates do not change an incidence kernel. -/
theorem cycleSpace_embedding (s t : I → V) (f : V → W)
    (hf : Function.Injective f) :
    Incidence.cycleSpace K (f ∘ s) (f ∘ t) = Incidence.cycleSpace K s t := by
  classical
  have heval (z : I → K) (v : V) :
      Incidence.boundary K (f ∘ s) (f ∘ t) z (f v) =
        Incidence.boundary K s t z v := by
    simp only [Incidence.boundary, LinearMap.coe_mk, AddHom.coe_mk,
      Function.comp_apply, hf.eq_iff]
  ext z
  constructor
  · intro hz
    apply LinearMap.mem_ker.mpr
    funext v
    exact (heval z v).symm.trans (congrFun (LinearMap.mem_ker.mp hz) (f v))
  · intro hz
    apply LinearMap.mem_ker.mpr
    funext w
    by_cases hw : ∃ v, f v = w
    · obtain ⟨v, rfl⟩ := hw
      rw [heval]
      exact congrFun (LinearMap.mem_ker.mp hz) v

        (if f (t i) = w then 1 else 0))) = 0

      intro i _

      have ht : f (t i) ≠ w := fun h => hw ⟨t i, h⟩
```

```lean
namespace BiRoot.Gluing

/-- Restricting positional edges of an incidence forest leaves a forest. -/
theorem forest_restrict (s t : I → V) (P : I → Prop) [DecidablePred P]
    (hz : Incidence.cycleSpace K s t = ⊥) :
    Incidence.cycleSpace K (fun i : {i // P i} => s i.val)
      (fun i : {i // P i} => t i.val) = ⊥ := by
  classical
  apply (Incidence.cycleSpace K (fun i : {i // P i} => s i.val)
    (fun i : {i // P i} => t i.val)).eq_bot_iff.mpr
  intro z hz'
  let z' : I → K := fun i => if h : P i then z ⟨i, h⟩ else 0
  have heval (v : V) : Incidence.boundary K s t z' v =
      Incidence.boundary K (fun i : {i // P i} => s i.val)
        (fun i : {i // P i} => t i.val) z v := by
    change (∑ i, z' i * ((if s i = v then 1 else 0) -
      (if t i = v then 1 else 0))) = ∑ i : {i // P i}, _
    calc
      _ = ∑ i ∈ Finset.univ.filter P, z' i *
          ((if s i = v then 1 else 0) - (if t i = v then 1 else 0)) := by
        rw [Finset.sum_filter]
        apply Finset.sum_congr rfl
        intro i _
        by_cases hi : P i <;> simp [z', hi]
      _ = _ := by
        rw [Finset.sum_subtype (p := P) (Finset.univ.filter P) (by simp)]
        apply Finset.sum_congr rfl
        intro i _
        simp [z', i.prop]
  have hm : z' ∈ Incidence.cycleSpace K s t := by
    apply LinearMap.mem_ker.mpr
    funext v
    rw [heval]
    exact congrFun (LinearMap.mem_ker.mp hz') v
  rw [hz, Submodule.mem_bot] at hm
  change z = 0
  funext i
  have h := congrFun hm i.val
  simpa only [z', dite_eq_left i.prop, Pi.zero_apply] using h

variable {E A : Type*} [Fintype E] [Fintype A] [DecidableEq A]

omit [Fintype A] in
/-- For an edge-coordinate forest, each label uses at most as many positions
as original edge coordinates of that label. -/
theorem color_card_le (es et : I → E) (lab : E → A) (χ : I → A)
    (hcs : ∀ i, lab (es i) = χ i) (hct : ∀ i, lab (et i) = χ i)
    (hz : Incidence.cycleSpace K es et = ⊥) (a : A) :
    Fintype.card {i // χ i = a} ≤ Fintype.card {e // lab e = a} := by
  classical
  let es' (i : {i // χ i = a}) : {e // lab e = a} :=
    ⟨es i.val, (hcs i.val).trans i.prop⟩
  let et' (i : {i // χ i = a}) : {e // lab e = a} :=
    ⟨et i.val, (hct i.val).trans i.prop⟩
  have hf := forest_restrict K es et (fun i => χ i = a) hz
  have he := ColorBridge.cycleSpace_embedding K es' et' Subtype.val Subtype.val_injective
  have hz' : Incidence.cycleSpace K es' et' = ⊥ := by
    rw [← he]
    exact hf
  have hi : Function.Injective (Incidence.boundary K es' et') :=
    LinearMap.ker_eq_bot.mp hz'
  have h := LinearMap.finrank_le_finrank_of_injective hi
  simpa only [Module.finrank_fintype_fun_eq_card] using h

/-- Sum a weight over a finite set by its label fibers. -/
theorem sum_by_color (lab : E → A) (ρ : A → ℕ) :
    (∑ e, ρ (lab e)) = ∑ a, ρ a * Fintype.card {e // lab e = a} := by
  classical
  calc
    (∑ e, ρ (lab e)) = ∑ e, ∑ a, if lab e = a then ρ a else 0 := by simp
    _ = ∑ a, ∑ e, if lab e = a then ρ a else 0 := Finset.sum_comm
    _ = _ := by
      apply Finset.sum_congr rfl
      intro a _
      rw [← Finset.sum_filter]
      have hc : (Finset.univ.filter (fun e => lab e = a)).card =
          Fintype.card {e // lab e = a} := by
        exact (Fintype.card_subtype _).symm
      simp only [Finset.sum_const, smul_eq_mul, hc, mul_comm]

/-- Abstract weighted forest deficit from the total cardinality deficit. -/
theorem weighted_deficit (es et : I → E) (lab : E → A) (χ : I → A)
    (hcs : ∀ i, lab (es i) = χ i) (hct : ∀ i, lab (et i) = χ i)
    (hz : Incidence.cycleSpace K es et = ⊥)
    (hthree : Fintype.card I + 3 ≤ Fintype.card E)
    (ρ : A → ℕ) (hρ : ∀ a, 0 < ρ a) :
    (∑ i, ρ (χ i)) + 3 ≤ ∑ e, ρ (lab e) := by
  have hle (a : A) := color_card_le K es et lab χ hcs hct hz a
  have hsub : (∑ i, (ρ (χ i) - 1)) ≤ ∑ e, (ρ (lab e) - 1) := by
    rw [sum_by_color χ (fun a => ρ a - 1), sum_by_color lab (fun a => ρ a - 1)]
    apply Finset.sum_le_sum
    intro a _
    exact Nat.mul_le_mul_left _ (hle a)
  have hi : (∑ i, ρ (χ i)) = Fintype.card I + ∑ i, (ρ (χ i) - 1) := by
    calc
      _ = ∑ i, ((ρ (χ i) - 1) + 1) := by
        apply Finset.sum_congr rfl
        intro i _
        exact (Nat.sub_add_cancel (hρ (χ i))).symm
      _ = _ := by simp [Finset.sum_add_distrib, add_comm]
  have he : (∑ e, ρ (lab e)) = Fintype.card E + ∑ e, (ρ (lab e) - 1) := by
    calc
      _ = ∑ e, ((ρ (lab e) - 1) + 1) := by
        apply Finset.sum_congr rfl
        intro e _
        exact (Nat.sub_add_cancel (hρ (lab e))).symm
      _ = _ := by simp [Finset.sum_add_distrib, add_comm]
  omega
end Weighted
```

```lean
/-- Encode a given decomposition into an alphabet containing all its chunks. -/
def encodeChunks (S : Set (List α)) (cs : List (List α))
    (hcs : ∀ c ∈ cs, c ∈ S) : FreeMonoid S :=
  FreeMonoid.ofList (cs.attach.map (fun c => (⟨c.val, hcs c.val c.prop⟩ : S)))

theorem decode_encodeChunks (S : Set (List α)) (cs : List (List α))
    (hcs : ∀ c ∈ cs, c ∈ S) :
    (decode S (encodeChunks S cs hcs)).toList = cs.flatten := by
  rw [decode_toList]
  simp [encodeChunks, List.map_map]

theorem encodeChunks_length (S : Set (List α)) (cs : List (List α))
    (hcs : ∀ c ∈ cs, c ∈ S) :
    (encodeChunks S cs hcs).length = cs.length := by
  simp [encodeChunks, FreeMonoid.length]

/-- A finite family, rather than all return words of the graph, suffices. -/
theorem finite_return_family (δ : Q → α → Option Q) (root : Q)
    (Z : List (FreeMonoid α))
    (hZ : ∀ z ∈ Z, readFrom δ root z.toList = some root) :
    ∃ (S : Set (List α)), S.Finite ∧
      (∀ c ∈ S, FirstReturnWord (readFrom δ root) root c) ∧
      ∀ z ∈ Z, ∃ z' : FreeMonoid S, decode S z' = z := by
  classical
  have hex (z : {z // z ∈ Z}) := exists_firstReturn_decomposition δ root z.val.toList
    (hZ z.val z.prop)
  let cs (z : {z // z ∈ Z}) := Classical.choose (hex z)
  have hcs (z : {z // z ∈ Z}) : (cs z).flatten = z.val.toList ∧
      ∀ c ∈ cs z, FirstReturnWord (readFrom δ root) root c :=
    Classical.choose_spec (hex z)
  let L := Z.attach.flatMap cs
  let S : Set (List α) := {c | c ∈ L}
  have hmem (z : {z // z ∈ Z}) (c : List α) (hc : c ∈ cs z) : c ∈ S := by
    exact List.mem_flatMap.mpr ⟨z, List.mem_attach _ _, hc⟩
  refine ⟨S, L.finite_toSet, ?_, ?_⟩
  · intro c hc
    obtain ⟨z, _, hcz⟩ := List.mem_flatMap.mp hc
    exact (hcs z).2 c hcz
  · intro z hz
    let zc : {z // z ∈ Z} := ⟨z, hz⟩
    refine ⟨encodeChunks S (cs zc) (hmem zc), ?_⟩
    apply FreeMonoid.toList.injective
    rw [decode_encodeChunks, (hcs zc).1]

/-- Prefix freeness and injectivity for the actually constructed finite code. -/
theorem finite_return_code (δ : Q → α → Option Q) (root : Q)
    (Z : List (FreeMonoid α))
    (hZ : ∀ z ∈ Z, readFrom δ root z.toList = some root) :
    ∃ (S : Set (List α)), S.Finite ∧
      InformationTheory.IsPrefixFree S ∧ [] ∉ S ∧ Function.Injective (decode S) ∧
      ∀ z ∈ Z, ∃ z' : FreeMonoid S, decode S z' = z := by
  obtain ⟨S, hfin, hret, henc⟩ := finite_return_family δ root Z hZ
  have hpf : InformationTheory.IsPrefixFree S := by
    intro x hx y hy hp
    exact firstReturn_prefixFree (readFrom δ root) root x
      (hret x hx) y (hret y hy) hp
  have hε : [] ∉ S := by
    intro h
    exact (hret [] h).1 rfl
  exact ⟨S, hfin, hpf, hε, decode_injective S hpf hε, henc⟩

/-- The chosen finite code itself, not an unconnected chunk count, has a shorter
preimage of any loop with a non-root prefix. -/
theorem finite_return_shorter (δ : Q → α → Option Q) (root : Q)
    (w : FreeMonoid α) (hw : readFrom δ root w.toList = some root)
    (hvisit : ∃ i, i ≤ w.length ∧ readFrom δ root (w.toList.take i) ≠ some root) :
    ∃ (S : Set (List α)), S.Finite ∧ ∃ w' : FreeMonoid S,
      decode S w' = w ∧ w'.length < w.length ∧
        InformationTheory.IsPrefixFree S ∧ [] ∉ S := by
  obtain ⟨cs, hflat, hret⟩ := exists_firstReturn_decomposition δ root w.toList hw
  let S : Set (List α) := {c | c ∈ cs}
  have hpf : InformationTheory.IsPrefixFree S := by
    intro x hx y hy hp
    exact firstReturn_prefixFree (readFrom δ root) root x (hret x hx) y (hret y hy) hp
  have hε : [] ∉ S := by
    intro h
    exact (hret [] h).1 rfl
  refine ⟨S, cs.finite_toSet, encodeChunks S cs (fun _ h => h), ?_, ?_, hpf, hε⟩
  · apply FreeMonoid.toList.injective
    rw [decode_encodeChunks, hflat]
  · rw [encodeChunks_length]
    exact firstReturn_shortens_of_nonroot_prefix δ root w.toList cs hflat hret hvisit
```

```lean
    (w' : FreeMonoid S) (h : decode S w' ∈ generatedMonoid (X'.image (decode S))) :

same universe; its finiteness is unnecessary for this algebraic construction. -/
theorem returnReduction_of_reader (p : Problem α)
    (δ : Q → α → Option Q) (root : Q)
    (hw : readFrom δ root p.w.toList = some root)
    (hX : ∀ x ∈ p.X, readFrom δ root x.toList = some root)
    (hU : ∀ x ∈ p.U, readFrom δ root x.toList = some root)
    (hvisit : ∃ i, i ≤ p.w.length ∧
      readFrom δ root (p.w.toList.take i) ≠ some root) :
    Nonempty (ReturnReduction p) := by
  classical
  let Z := p.w :: (p.X.toList ++ p.U.toList)
  have hZ : ∀ z ∈ Z, readFrom δ root z.toList = some root := by
    intro z hz
    rcases List.mem_cons.mp hz with rfl | hz
    · exact hw
    · rcases List.mem_append.mp hz with hx | hu
      · exact hX z (Finset.mem_toList.mp hx)
      · exact hU z (Finset.mem_toList.mp hu)
  obtain ⟨S, hfin, hret, henc⟩ := finite_return_family δ root Z hZ
  have hpf : InformationTheory.IsPrefixFree S := by
    intro x hx y hy hp
    exact firstReturn_prefixFree (readFrom δ root) root x (hret x hx) y (hret y hy) hp
  have hε : [] ∉ S := fun h => (hret [] h).1 rfl
  have hf : Function.Injective (decode S) := decode_injective S hpf hε
  let enc (z : FreeMonoid α) : FreeMonoid S :=
    if hz : z ∈ Z then Classical.choose (henc z hz) else 1
  have hdecode (z : FreeMonoid α) (hz : z ∈ Z) : decode S (enc z) = z := by
    simp only [enc, dite_eq_left hz]
    exact Classical.choose_spec (henc z hz)
  have hzw : p.w ∈ Z := List.mem_cons_self
  have hzX (z : FreeMonoid α) (hz : z ∈ p.X) : z ∈ Z := by simp [Z, hz]
  have hzU (z : FreeMonoid α) (hz : z ∈ p.U) : z ∈ Z := by simp [Z, hz]
  let X' := p.X.image enc
  let U' := p.U.image enc
  have hdx : X'.image (decode S) = p.X := by
    rw [Finset.image_image]
    convert p.X.image_id using 1
    apply Finset.image_congr
    intro z hz
    exact hdecode z (hzX z hz)
  have hdu : U'.image (decode S) = p.U := by
    rw [Finset.image_image]
    convert p.U.image_id using 1
    apply Finset.image_congr
    intro z hz
    exact hdecode z (hzU z hz)
  have hdw : decode S (enc p.w) = p.w := hdecode p.w hzw
  refine ⟨⟨S, hfin, hpf, hε, enc p.w, X', U', hdw, hdx, hdu, ?_, ?_, ?_⟩⟩
  · apply mem_generated_of_decode S hf
    rw [hdw, hdx]
    exact p.memX
  · apply mem_generated_of_decode S hf
    rw [hdw, hdu]
    exact p.memU
  · have hs := decode_shorter_of_nonroot δ root S hret (enc p.w)
      (by simpa only [hdw] using hvisit)
    simpa only [hdw] using hs

changes. The direct branch still must provide the weighted deficit. -/

      (∀ x ∈ p.X, readFrom δ root x.toList = some root) ∧
      (∀ x ∈ p.U, readFrom δ root x.toList = some root) ∧
      ∃ i, i ≤ p.w.length ∧ readFrom δ root (p.w.toList.take i) ≠ some root

  · obtain ⟨Q, δ, root, hw, hX, hU, hvisit⟩ := hr
```

```lean
/-- A cyclic monoid has at most its generating element as an atom. This uses
no cancellation or torsion assumptions. -/
theorem cyclic_atom_eq {M : Type*} [Monoid M] (r x : M)
    (hx : AtomIn (Submonoid.closure ({r} : Set M)) x) : x = r := by
  obtain ⟨n, rfl⟩ := Submonoid.mem_closure_singleton.mp hx.1
  cases n with
  | zero => exact (hx.2.1 (pow_zero r)).elim
  | succ n =>
    have hr : r ∈ Submonoid.closure ({r} : Set M) := Submonoid.subset_closure (by simp)
    rcases hx.2.2 (r ^ n) r (Submonoid.pow_mem _ hr n) hr (pow_succ r n) with hn | he
    · rw [pow_succ, hn, one_mul]
    · exact (hx.2.1 (by simp [he])).elim

    (hx : generatedMonoid X ≤ T) (hu : generatedMonoid U ≤ T) : X = U := by
  have heX := hX.2.1 T hT hx
  have heU := hU.2.1 T hT hu

  rw [hX.2.2 z, hU.2.2 z, heX, heU]

    ¬ generatedMonoid X ≤ Submonoid.closure ({r} : Set (FreeMonoid α)) := by
  intro hle
  have ht : TwoGenerated (Submonoid.closure ({r} : Set (FreeMonoid α))) :=
    ⟨r, r, by simp⟩

    (X : Finset (FreeMonoid α)) (hX : PrimitivePair X) :
    2 ≤ ∑ x ∈ X, weightedLength ρ x := by
  calc
    2 = ∑ _x ∈ X, (1 : ℕ) := by simp [hX.1]
    _ ≤ ∑ x ∈ X, weightedLength ρ x := by
      apply Finset.sum_le_sum
      intro x hx
```

```lean
/-! One universal deterministic/codeterministic quotient of a positive labelled
multigraph. The rank nonincrease theorem is not asserted here. -/
namespace BiRoot.Folding
universe u

structure Graph (A : Type u) where
  V : Type u
  E : Type u
  root : V
  src : E → V
  dst : E → V
  lab : E → A

namespace Graph
variable {A : Type u} (G : Graph A)

/-- A positive directed path records actual edges, with its literal word. -/
inductive Path : G.V → List A → G.V → Prop
  | nil (v : G.V) : Path v [] v
  | cons (e : G.E) {w : List A} {v : G.V} :
      Path (G.dst e) w v → Path (G.src e) (G.lab e :: w) v

def StronglyConnected : Prop := ∀ u v : G.V, ∃ w, G.Path u w v

def rank [Fintype G.V] [Fintype G.E] : ℤ :=
  (Fintype.card G.E : ℤ) - Fintype.card G.V + 1

def edgeWeight [Fintype G.E] (ρ : A → ℕ) : ℕ := ∑ e, ρ (G.lab e)

/-- Least vertex equivalence closed under both fold rules. -/
inductive FoldEq : G.V → G.V → Prop
  | refl (v) : FoldEq v v
  | symm {u v} : FoldEq u v → FoldEq v u
  | trans {u v w} : FoldEq u v → FoldEq v w → FoldEq u w
  | forward (e f : G.E) : G.lab e = G.lab f →
      FoldEq (G.src e) (G.src f) → FoldEq (G.dst e) (G.dst f)
  | backward (e f : G.E) : G.lab e = G.lab f →
      FoldEq (G.dst e) (G.dst f) → FoldEq (G.src e) (G.src f)

def vertexSetoid : Setoid G.V where
  r := G.FoldEq
  iseqv := ⟨FoldEq.refl, FoldEq.symm, FoldEq.trans⟩

def edgeSetoid : Setoid G.E where
  r e f := G.lab e = G.lab f ∧ G.FoldEq (G.src e) (G.src f)
  iseqv := ⟨fun _ => ⟨rfl, FoldEq.refl _⟩,
    fun h => ⟨h.1.symm, FoldEq.symm h.2⟩,
    fun h k => ⟨h.1.trans k.1, FoldEq.trans h.2 k.2⟩⟩

def FV := Quotient G.vertexSetoid
def FE := Quotient G.edgeSetoid

instance [Finite G.V] : Finite G.FV := inferInstanceAs (Finite (Quotient _))
instance [Finite G.E] : Finite G.FE := inferInstanceAs (Finite (Quotient _))

noncomputable instance [Finite G.V] : Fintype G.FV := Fintype.ofFinite _
noncomputable instance [Finite G.E] : Fintype G.FE := Fintype.ofFinite _

def qV : G.V → G.FV := Quotient.mk _
def qE : G.E → G.FE := Quotient.mk _

def fsrc : G.FE → G.FV := Quotient.lift (fun e => G.qV (G.src e))
  (fun _ _ h => Quotient.sound h.2)

def fdst : G.FE → G.FV := Quotient.lift (fun e => G.qV (G.dst e))
  (fun e f h => Quotient.sound (FoldEq.forward e f h.1 h.2))

def flab : G.FE → A := Quotient.lift G.lab (fun _ _ h => h.1)

/-- Both uniqueness directions hold on the single quotient, without choosing
any order of local folds. -/
theorem target_unique (e f : G.FE) (hs : G.fsrc e = G.fsrc f)
    (hl : G.flab e = G.flab f) : G.fdst e = G.fdst f := by
  induction e using Quotient.inductionOn with
  | _ e =>
    induction f using Quotient.inductionOn with
    | _ f =>
      apply Quotient.sound
      exact FoldEq.forward e f hl (Quotient.exact hs)

theorem source_unique (e f : G.FE) (ht : G.fdst e = G.fdst f)
    (hl : G.flab e = G.flab f) : G.fsrc e = G.fsrc f := by
  induction e using Quotient.inductionOn with
  | _ e =>
    induction f using Quotient.inductionOn with
    | _ f =>
      apply Quotient.sound
      exact FoldEq.backward e f hl (Quotient.exact ht)

/-- Literal partial transition function of the folded quotient. -/
noncomputable def step (v : G.FV) (a : A) : Option G.FV := by
  classical
  exact if h : ∃ e : G.FE, G.fsrc e = v ∧ G.flab e = a then
    some (G.fdst (Classical.choose h)) else none

theorem step_edge (e : G.FE) : G.step (G.fsrc e) (G.flab e) = some (G.fdst e) := by
  classical
  have h : ∃ f : G.FE, G.fsrc f = G.fsrc e ∧ G.flab f = G.flab e := ⟨e, rfl, rfl⟩
  simp only [step, dite_eq_left h]
  congr 1
  exact G.target_unique _ _ (Classical.choose_spec h).1 (Classical.choose_spec h).2

theorem step_spec (v : G.FV) (a : A) (r : G.FV) (h : G.step v a = some r) :
    ∃ e : G.FE, G.fsrc e = v ∧ G.flab e = a ∧ G.fdst e = r := by
  classical
  by_cases hex : ∃ e : G.FE, G.fsrc e = v ∧ G.flab e = a
  · refine ⟨Classical.choose hex, (Classical.choose_spec hex).1,
      (Classical.choose_spec hex).2, ?_⟩
    simpa only [step, dite_eq_left hex, Option.some.injEq] using h
  · simp only [step, dite_eq_right hex, reduceCtorEq] at h

/-- Codeterminism of the actual partial reader. -/
theorem step_codeterministic : Gluing.Codeterministic G.step := by
  intro a v u r hv hu
  obtain ⟨e, he, hla, hte⟩ := G.step_spec v a r hv
  obtain ⟨f, hf, hlb, htf⟩ := G.step_spec u a r hu
  rw [← he, ← hf]
  exact G.source_unique e f (hte.trans htf.symm) (hla.trans hlb.symm)

/-- Every original path is a path in the deterministic partial reader. -/
theorem path_preserved {v u : G.V} {w : List A} (h : G.Path v w u) :
    readFrom G.step (G.qV v) w = some (G.qV u) := by
  induction h with
  | nil v => rfl
  | cons e h ih =>
    change (G.step (G.qV (G.src e)) (G.lab e)).bind _ = _
    have he := G.step_edge (G.qE e)
    change G.step (G.qV (G.src e)) (G.lab e) = some (G.qV (G.dst e)) at he
    rw [he, Option.bind_some]
    exact ih

/-- Strong connectivity is preserved, stated in the partial-reader language. -/
theorem strong_connectivity_preserved (hc : G.StronglyConnected) :
    ∀ v u : G.FV, ∃ w, readFrom G.step v w = some u := by
  intro v u
  induction v using Quotient.inductionOn with
  | _ v =>
    induction u using Quotient.inductionOn with
    | _ u =>
      obtain ⟨w, hw⟩ := hc v u
      exact ⟨w, G.path_preserved hw⟩

/-- Universal minimality of the quotient identifications: any map respecting
both fold rules identifies FoldEq-related vertices. -/
theorem foldEq_minimal {W : Type*} (f : G.V → W)
    (hf : ∀ e g, G.lab e = G.lab g → f (G.src e) = f (G.src g) →
      f (G.dst e) = f (G.dst g))
    (hb : ∀ e g, G.lab e = G.lab g → f (G.dst e) = f (G.dst g) →
      f (G.src e) = f (G.src g))
    {v u : G.V} (h : G.FoldEq v u) : f v = f u := by
  induction h with
  | refl v => rfl
  | symm _ ih => exact ih.symm
  | trans _ _ ih₁ ih₂ => exact ih₁.trans ih₂
  | forward e g hl _ ih => exact hf e g hl ih
  | backward e g hl _ ih => exact hb e g hl ih
```

```lean
/-! Rank monotonicity of the least fold quotient, via an injective map between
quotients of potential spaces. This avoids an auxiliary sequence of local folds. -/
namespace BiRoot.Folding
open scoped BigOperators
universe u

section Constants
variable (K : Type*) [Field K] {V : Type*} (R : Setoid V)

def constantSpace : Submodule K (V → K) where
  carrier := {f | ∀ v w, R v w → f v = f w}
  zero_mem' := fun _ _ _ => rfl
  add_mem' := by
    intro f g hf hg v w h
    change f v + g v = f w + g w
    rw [hf v w h, hg v w h]
  smul_mem' := by
    intro c f hf v w h
    change c * f v = c * f w
    rw [hf v w h]

noncomputable def constantEquiv : (Quotient R → K) ≃ₗ[K] constantSpace K R where
  toFun f := ⟨fun v => f (Quotient.mk R v),
    fun _ _ h => congrArg f (Quotient.sound h)⟩
  invFun f := Quotient.lift f.val (fun v w h => f.prop v w h)
  left_inv f := by
    funext q
    exact Quotient.inductionOn q (fun _ => rfl)
  right_inv f := by
    apply Subtype.ext
    rfl
  map_add' _ _ := rfl
  map_smul' _ _ := rfl

theorem constant_finrank [Finite V] :
    Module.finrank K (constantSpace K R) = Nat.card (Quotient R) := by
  let : Fintype (Quotient R) := Fintype.ofFinite _
  rw [← (constantEquiv K R).finrank_eq, Module.finrank_fintype_fun_eq_card,
    Nat.card_eq_fintype_card]
end Constants

namespace Graph
variable {A : Type u} (G : Graph A) (K : Type*) [Field K]

/-- Folded-vertex constant potentials have folded-edge constant gradients. -/
theorem constant_coboundary :
    constantSpace K G.vertexSetoid ≤
      (constantSpace K G.edgeSetoid).comap (Incidence.coboundary K G.src G.dst) := by
  intro f hf e g h
  change f (G.src e) - f (G.dst e) = f (G.src g) - f (G.dst g)
  rw [hf _ _ h.2, hf _ _ (FoldEq.forward e g h.1 h.2)]

/-- Conversely, a gradient constant on fold-edge classes forces its potential
constant on fold-vertex classes. The backward rule uses the very relation it
generates only as an edge-class membership proof, not as an induction premise. -/
theorem constant_of_coboundary (f : G.V → K)
    (hf : Incidence.coboundary K G.src G.dst f ∈ constantSpace K G.edgeSetoid) :
    f ∈ constantSpace K G.vertexSetoid := by
  intro v w h
  induction h with
  | refl _ => rfl
  | symm _ ih => exact ih.symm
  | trans _ _ ih₁ ih₂ => exact ih₁.trans ih₂
  | forward e g hl hsrc ih =>
    have h := hf e g ⟨hl, hsrc⟩
    change f (G.src e) - f (G.dst e) = f (G.src g) - f (G.dst g) at h
    rw [ih] at h
    exact sub_right_inj.mp h
  | backward e g hl hdst ih =>
    have h := hf e g ⟨hl, FoldEq.backward e g hl hdst⟩
    change f (G.src e) - f (G.dst e) = f (G.src g) - f (G.dst g) at h
    rw [ih] at h
    exact sub_left_inj.mp h

noncomputable def quotientCoboundary :
    (G.V → K) ⧸ constantSpace K G.vertexSetoid →ₗ[K]
      (G.E → K) ⧸ constantSpace K G.edgeSetoid :=
  Submodule.mapQ _ _ (Incidence.coboundary K G.src G.dst) (G.constant_coboundary K)

theorem quotientCoboundary_injective : Function.Injective (G.quotientCoboundary K) := by
  apply LinearMap.ker_eq_bot.mp
  apply (LinearMap.ker (G.quotientCoboundary K)).eq_bot_iff.mpr
  intro q hq
  obtain ⟨f, rfl⟩ := (constantSpace K G.vertexSetoid).mkQ_surjective q
  apply (Submodule.Quotient.mk_eq_zero _).mpr
  apply G.constant_of_coboundary K f
  have hzero := LinearMap.mem_ker.mp hq
  change (Submodule.Quotient.mk (Incidence.coboundary K G.src G.dst f) :
    (G.E → K) ⧸ constantSpace K G.edgeSetoid) = 0 at hzero
  exact (Submodule.Quotient.mk_eq_zero _).mp hzero

/-- Rank of the single least fold quotient does not increase. -/
theorem folded_rank_le [Fintype G.V] [Fintype G.E] :
    (Fintype.card G.FE : ℤ) - Fintype.card G.FV + 1 ≤ G.rank := by
  have hdim := LinearMap.finrank_le_finrank_of_injective (G.quotientCoboundary_injective ℚ)
  have hv := Submodule.finrank_quotient_add_finrank (constantSpace ℚ G.vertexSetoid)
  have he := Submodule.finrank_quotient_add_finrank (constantSpace ℚ G.edgeSetoid)
  rw [constant_finrank, Module.finrank_fintype_fun_eq_card] at hv he
  change Module.finrank ℚ ((G.V → ℚ) ⧸ constantSpace ℚ G.vertexSetoid) +
    Nat.card G.FV = Fintype.card G.V at hv
  change Module.finrank ℚ ((G.E → ℚ) ⧸ constantSpace ℚ G.edgeSetoid) +
    Nat.card G.FE = Fintype.card G.E at he
  rw [Nat.card_eq_fintype_card] at hv he
  unfold rank
  omega
```

```lean
/-- A functional directed graph admits at most one outgoing label per state. -/
def UniqueOutgoingLabel (δ : Q → A → Option Q) : Prop :=
  ∀ q a b r s, δ q a = some r → δ q b = some s → a = b

/-- Any two readable words in a one-outgoing-label reader are prefix comparable. -/
theorem readable_prefix_comparable (δ : Q → A → Option Q)
    (hδ : UniqueOutgoingLabel δ) (q : Q) (x y : List A)
    (hx : ∃ r, readFrom δ q x = some r) (hy : ∃ r, readFrom δ q y = some r) :
    x <+: y ∨ y <+: x := by
  induction x generalizing q y with
  | nil => exact Or.inl List.nil_prefix
  | cons a x ih =>
    cases y with
    | nil => exact Or.inr List.nil_prefix
    | cons b y =>
      obtain ⟨r, hr⟩ := hx
      obtain ⟨s, hs⟩ := hy
      cases ha : δ q a with
      | none => simp only [readFrom, ha, Option.bind_none, reduceCtorEq] at hr
      | some q' =>
        cases hb : δ q b with
        | none => simp only [readFrom, hb, Option.bind_none, reduceCtorEq] at hs
        | some q'' =>
          have hab := hδ q a b q' q'' ha hb
          subst b
          have he : q'' = q' := Option.some.inj (hb.symm.trans ha)
          subst q''
          have hxt : readFrom δ q' x = some r := by simpa only [readFrom, ha, Option.bind_some] using hr
          have hyt : readFrom δ q' y = some s := by simpa only [readFrom, ha, Option.bind_some] using hs
          rcases ih q' y ⟨r, hxt⟩ ⟨s, hyt⟩ with hp | hp
          · exact Or.inl ((List.prefix_cons_inj a).mpr hp)
          · exact Or.inr ((List.prefix_cons_inj a).mpr hp)

/-- All genuine first returns in a functional graph are the same word. -/
theorem unique_firstReturn (δ : Q → A → Option Q) (hδ : UniqueOutgoingLabel δ)
    (root : Q) (x y : List A)
    (hx : FirstReturnWord (readFrom δ root) root x)
    (hy : FirstReturnWord (readFrom δ root) root y) : x = y := by
  have hc := readable_prefix_comparable δ hδ root x y ⟨root, hx.2.1⟩ ⟨root, hy.2.1⟩
  rcases hc with hp | hp
  · exact firstReturn_prefixFree (readFrom δ root) root x hx y hy hp
  · exact (firstReturn_prefixFree (readFrom δ root) root y hy x hx hp).symm

/-- A nonempty loop in a functional reader supplies a generator for every loop. -/
theorem functional_loops_cyclic (δ : Q → A → Option Q)
    (hδ : UniqueOutgoingLabel δ) (root : Q)
    (z : FreeMonoid A) (hz : z ≠ 1)
    (hclosed : readFrom δ root z.toList = some root) :
    ∃ r : FreeMonoid A, r ≠ 1 ∧
      ∀ w : FreeMonoid A, readFrom δ root w.toList = some root → ∃ n : ℕ, w = r ^ n := by
  obtain ⟨cs, hflat, hret⟩ := exists_firstReturn_decomposition δ root z.toList hclosed
  have hcs : cs ≠ [] := by
    intro he
    have hz' : z.toList = [] := by simpa only [he, List.flatten_nil] using hflat.symm
    exact hz (FreeMonoid.toList.injective hz')
  obtain ⟨r, hr⟩ := List.exists_mem_of_ne_nil cs hcs
  have hrret := hret r hr
  refine ⟨FreeMonoid.ofList r, ?_, ?_⟩
  · intro he
    exact hrret.1 (congrArg FreeMonoid.toList he)
  · intro w hw
    obtain ⟨ds, hdflat, hdret⟩ := exists_firstReturn_decomposition δ root w.toList hw
    have hall : ∀ d ∈ ds, d = r := fun d hd => unique_firstReturn δ hδ root d r
      (hdret d hd) hrret
    have haux : ∀ ls : List (List A), (∀ d ∈ ls, d = r) →
        FreeMonoid.ofList ls.flatten = FreeMonoid.ofList r ^ ls.length := by
      intro ls
      induction ls with
      | nil => intro _; rfl
      | cons d ls ih =>
        intro hall
        have he : d = r := hall d List.mem_cons_self
        have ht : ∀ c ∈ ls, c = r := fun c hc => hall c (List.mem_cons_of_mem d hc)
        change FreeMonoid.ofList (d ++ ls.flatten) = _
        rw [FreeMonoid.ofList_append, ih ht, he, List.length_cons, pow_succ']
    have hp := haux ds hall
    exact ⟨ds.length, by simpa only [hdflat, FreeMonoid.ofList_toList] using hp⟩
```

```lean
    have hpow : w = y ^ l.length := hp.symm.trans (List.prod_eq_pow_length l y hall)
    have hn1 := hw.2 y l.length hpow
    have hwy : w = y := by simpa only [hn1, pow_one] using hpow
    have hyX : y ∈ X := Finset.mem_of_mem_erase (by rw [hy]; exact Finset.mem_singleton_self _)
    exact hnot (hwy ▸ hyX)
  obtain ⟨a, b, hab⟩ := List.append_of_mem hxl
  refine ⟨a.prod, b.prod, ?_, ?_⟩
  · apply Submonoid.list_prod_mem
    intro y hy
    apply Submonoid.subset_closure
    apply hl y
    rw [hab]
    exact List.mem_append_left _ hy
  · rw [← hp, hab, List.prod_append, List.prod_cons]
    exact (mul_assoc _ _ _).symm

/-- A vertex reached inside a used generator is reached by a prefix of w. -/
theorem prefix_coverage [DecidableEq A] (δ : Q → A → Option Q) (root : Q)
    (X : Finset (FreeMonoid A)) (hX : PrimitivePair X)
    (w : FreeMonoid A) (hw : PrimitiveWord w) (hmem : w ∈ generatedMonoid X)
    (hnot : w ∉ X) (hloops : ∀ x ∈ X, readFrom δ root x.toList = some root)
    (x : FreeMonoid A) (hx : x ∈ X) (i : ℕ) (hi : i ≤ x.length) (v : Q)
    (hv : readFrom δ root (x.toList.take i) = some v) :
    ∃ j, j ≤ w.length ∧ readFrom δ root (w.toList.take j) = some v := by
  obtain ⟨a, b, ha, he⟩ := generator_occurs X hX w hw hmem hnot x hx
  have hra := generatedMonoid_closed δ root X hloops ha
  change readFrom δ root a.toList = some root at hra
  refine ⟨a.length + i, ?_, ?_⟩
  · rw [he, FreeMonoid.length_mul, FreeMonoid.length_mul]
    omega
  · have ht : w.toList.take (a.length + i) = a.toList ++ x.toList.take i := by
      rw [he]
      change ((a.toList ++ x.toList) ++ b.toList).take (a.toList.length + i) = _
      rw [List.append_assoc, List.take_append, Nat.add_sub_cancel_left,
        List.take_of_length_le (by omega), List.take_append_of_le_length hi]
    rw [ht, readFrom_append, hra, Option.bind_some]
    exact hv
```

```lean
      · simpa using ha

  have hcard : (S.erase a).card = 1 := by
    rw [Finset.card_erase_of_mem ha]
```

```lean
    cases h : η v a with
    | none => rfl
    | some r => simpa only [Option.map_some, Option.bind_some] using ih r

theorem sumReader_codeterministic (hd : Gluing.Codeterministic δ)
    (he : Gluing.Codeterministic η) : Gluing.Codeterministic (sumReader δ η) := by
  intro a v w r hv hw
  cases v with
  | inl v =>
    cases w with
    | inl w =>
      cases hvr : δ v a with
      | none => simp only [sumReader, hvr, Option.map_none, reduceCtorEq] at hv
      | some q =>
        have hr : r = Sum.inl q := by simpa only [sumReader, hvr, Option.map_some, Option.some.injEq] using hv.symm
        subst r
        have hwr : δ w a = some q := by simpa only [sumReader, Option.map_eq_some_iff,
          Sum.inl.injEq, exists_eq_right] using hw
        exact congrArg Sum.inl (hd a v w q hvr hwr)
    | inr w =>
      cases hvr : δ v a <;> cases hwr : η w a <;>
        simp only [sumReader, hvr, hwr, Option.map_some, Option.map_none,
          Option.some.injEq, reduceCtorEq] at hv hw
      cases hv.trans hw.symm
  | inr v =>
    cases w with
    | inl w =>
      cases hvr : η v a <;> cases hwr : δ w a <;>
        simp only [sumReader, hvr, hwr, Option.map_some, Option.map_none,
          Option.some.injEq, reduceCtorEq] at hv hw
      cases hv.trans hw.symm
    | inr w =>
      cases hvr : η v a with
      | none => simp only [sumReader, hvr, Option.map_none, reduceCtorEq] at hv
      | some q =>
        have hr : r = Sum.inr q := by simpa only [sumReader, hvr, Option.map_some, Option.some.injEq] using hv.symm
        subst r
        have hwr : η w a = some q := by simpa only [sumReader, Option.map_eq_some_iff,
          Sum.inr.injEq, exists_eq_right] using hw
        exact congrArg Sum.inr (he a v w q hvr hwr)

/-- Defined transitions split exactly over the two vertex summands. -/
def sumTransitionEquiv :
    Gluing.TransitionEdge (sumReader δ η) ≃ Gluing.TransitionEdge δ ⊕ Gluing.TransitionEdge η where
  toFun e := match e with
    | ⟨(Sum.inl v, a), h⟩ => Sum.inl ⟨(v, a), by simpa only [sumReader, Option.isSome_map] using h⟩
    | ⟨(Sum.inr v, a), h⟩ => Sum.inr ⟨(v, a), by simpa only [sumReader, Option.isSome_map] using h⟩
  invFun e := match e with
    | Sum.inl ⟨(v, a), h⟩ => ⟨(Sum.inl v, a), by simpa only [sumReader, Option.isSome_map] using h⟩
    | Sum.inr ⟨(v, a), h⟩ => ⟨(Sum.inr v, a), by simpa only [sumReader, Option.isSome_map] using h⟩
  left_inv e := by rcases e with ⟨⟨v, a⟩, h⟩; cases v <;> rfl
  right_inv e := by rcases e with e | e <;> rcases e with ⟨⟨v, a⟩, h⟩ <;> rfl

theorem sumReader_weight [Fintype A] [Fintype V] [Fintype W] (ρ : A → ℕ) :
    (∑ e : Gluing.TransitionEdge (sumReader δ η), ρ e.val.2) =
      (∑ e : Gluing.TransitionEdge δ, ρ e.val.2) +
      (∑ e : Gluing.TransitionEdge η, ρ e.val.2) := by
  classical
  rw [← Equiv.sum_comp (sumTransitionEquiv δ η).symm (fun e => ρ e.val.2)]
  rw [Fintype.sum_sum_type]
  rfl

#print axioms sumTransitionEquiv
```

```lean
import GeneratorCoverage
import SumReaders

namespace BiRoot.Descent
universe u
variable {A : Type u} [Fintype A] [DecidableEq A]

structure PairAutomaton (X : Finset (FreeMonoid A)) where
  Q : Type u
  [vertices : Fintype Q]
  δ : Q → A → Option Q
  root : Q
  codeterministic : Gluing.Codeterministic δ
  count : Fintype.card (Gluing.TransitionEdge δ) + 1 = Fintype.card Q + 2
  loops : ∀ x ∈ X, readFrom δ root x.toList = some root
  connected : ∀ v q, ∃ l, readFrom δ v l = some q
  weight : ∀ ρ : A → ℕ, (∑ e : Gluing.TransitionEdge δ, ρ e.val.2) ≤ ∑ x ∈ X, weightedLength ρ x
  coverage : ∀ v, ∃ x ∈ X, ∃ i, i ≤ x.length ∧ readFrom δ root (x.toList.take i) = some v

attribute [instance] PairAutomaton.vertices

noncomputable def pairAutomaton (X : Finset (FreeMonoid A)) (hX : PrimitivePair X) :
    PairAutomaton X := by
  classical
  let G := Flower.graph X
  have hloops : ∀ x ∈ X, readFrom G.step (G.qV G.root) x.toList = some (G.qV G.root) :=
    fun x hx => G.path_preserved (Flower.loop X ⟨x, hx⟩)
  have hr := G.folded_pair_rank_two (Flower.connected X) (Flower.rank_two X hX) X hX hloops
  refine { Q := G.FV, root := G.qV G.root, δ := G.step
           codeterministic := G.step_codeterministic
           count := ?_, loops := hloops
           connected := G.strong_connectivity_preserved (Flower.connected X)
           weight := ?_, coverage := ?_ }
  · have he := Fintype.card_congr G.transitionEquiv
    change (Fintype.card G.FE : ℤ) - Fintype.card G.FV + 1 = 2 at hr
    omega
  · intro ρ
    have hs : (∑ e : Gluing.TransitionEdge G.step, ρ e.val.2) = G.folded.edgeWeight ρ := by
      exact (Equiv.sum_comp G.transitionEquiv (fun e => ρ e.val.2)).symm
    rw [hs]
    exact (G.folded_weight_le ρ).trans_eq (Flower.weight X ρ)
  · intro q
    induction q using Quotient.inductionOn with
    | _ v =>
      have hne : X.Nonempty := Finset.card_pos.mp (by rw [hX.1]; decide)
      obtain ⟨x, i, hi, hp⟩ := Flower.vertex_prefix X v hne
      exact ⟨x.val, x.prop, i, hi, G.path_preserved hp⟩

theorem PairAutomaton.common_coverage {X : Finset (FreeMonoid A)} (F : PairAutomaton X)
    (hX : PrimitivePair X) (w : FreeMonoid A) (hw : PrimitiveWord w)
    (hmem : w ∈ generatedMonoid X) (hnot : w ∉ X) :
    ∀ v, ∃ i, i ≤ w.length ∧ readFrom F.δ F.root (w.toList.take i) = some v := by
  intro v
  obtain ⟨x, hx, i, hi, hv⟩ := F.coverage v
  exact prefix_coverage F.δ F.root X hX w hw hmem hnot F.loops x hx i hi v hv

end BiRoot.Descent
```

```lean
import PairAutomaton
import GeometryAssembly

set_option linter.unnecessarySimpa false

namespace BiRoot.Descent
universe u
variable {A : Type u} [Fintype A] [DecidableEq A]

theorem geometricInput_exists (p : Problem A) (hx : p.w ∉ p.X) (hu : p.w ∉ p.U) :
    Nonempty (GeometricInput p) := by
  classical
  let : NeZero p.w.length := ⟨by
    intro h
    have he : p.w.toList = [] := List.length_eq_zero_iff.mp h
    exact p.primitive.1 (FreeMonoid.toList.injective he)⟩
  let F := pairAutomaton p.X p.primitiveX
  let H := pairAutomaton p.U p.primitiveU
  have hwF : readFrom F.δ F.root p.w.toList = some F.root :=
    generatedMonoid_closed F.δ F.root p.X F.loops p.memX
  have hwH : readFrom H.δ H.root p.w.toList = some H.root :=
    generatedMonoid_closed H.δ H.root p.U H.loops p.memU
  obtain ⟨s, hs0, hs, _⟩ := cyclic_trajectory F.δ F.root p.w hwF
  obtain ⟨t, ht0, ht, _⟩ := cyclic_trajectory H.δ H.root p.w hwH
  let δ := sumReader F.δ H.δ
  let sl : Fin p.w.length → F.Q ⊕ H.Q := fun i => Sum.inl (s i)
  let tr : Fin p.w.length → F.Q ⊕ H.Q := fun i => Sum.inr (t i)
  have hsl : sl 0 = Sum.inl F.root := congrArg Sum.inl hs0
  have htr : tr 0 = Sum.inr H.root := congrArg Sum.inr ht0
  refine ⟨{ V := F.Q ⊕ H.Q, δ := δ, s := sl, t := tr
            hs := ?_, ht := ?_
            codeterministic := sumReader_codeterministic F.δ H.δ F.codeterministic H.codeterministic
            separated := fun _ h => by cases h
            count := ?_, weight := ?_
            loopsX := ?_, loopsU := ?_, loopW := ?_
            reach := ?_, coverage := ?_ }⟩
  · intro i
    change (F.δ (s i) (p.w.toList.get i)).map Sum.inl = _
    rw [hs i]
    rfl
  · intro i
    change (H.δ (t i) (p.w.toList.get i)).map Sum.inr = _
    rw [ht i]
    rfl
  · have he := Fintype.card_congr (sumTransitionEquiv F.δ H.δ)
    rw [Fintype.card_sum] at he
    change Fintype.card (Gluing.TransitionEdge δ) =
      Fintype.card (Gluing.TransitionEdge F.δ) + Fintype.card (Gluing.TransitionEdge H.δ) at he
    have hF := F.count
    have hH := H.count
    change Fintype.card (Gluing.TransitionEdge δ) + 2 = Fintype.card (F.Q ⊕ H.Q) + 4
    rw [Fintype.card_sum]
    omega
  · change (∑ e : Gluing.TransitionEdge (sumReader F.δ H.δ), p.ρ e.val.2) ≤ cost p
    rw [sumReader_weight]
    exact Nat.add_le_add (F.weight p.ρ) (H.weight p.ρ)
  · intro x hx
    rw [hsl]
    simpa only [δ, read_sum_left, F.loops x hx, Option.map_some]
  · intro x hx
    rw [htr]
    simpa only [δ, read_sum_right, H.loops x hx, Option.map_some]
  · rw [hsl]
    simpa only [δ, read_sum_left, hwF, Option.map_some]
  · intro v
    cases v with
    | inl v =>
      obtain ⟨l, hl⟩ := F.connected v F.root
      obtain ⟨m, hm⟩ := F.connected F.root v
      refine ⟨Sum.inl F.root, Or.inl hsl.symm, ⟨l, ?_⟩, ⟨m, ?_⟩⟩
      · simpa only [δ, read_sum_left, hl, Option.map_some]
      · simpa only [δ, read_sum_left, hm, Option.map_some]
    | inr v =>
      obtain ⟨l, hl⟩ := H.connected v H.root
      obtain ⟨m, hm⟩ := H.connected H.root v
      refine ⟨Sum.inr H.root, Or.inr htr.symm, ⟨l, ?_⟩, ⟨m, ?_⟩⟩
      · simpa only [δ, read_sum_right, hl, Option.map_some]
      · simpa only [δ, read_sum_right, hm, Option.map_some]
  · intro v
    cases v with
    | inl v =>
      obtain ⟨i, hi, hv⟩ := F.common_coverage p.primitiveX p.w p.primitive p.memX hx v
      refine ⟨Sum.inl F.root, Or.inl hsl.symm, i, hi, ?_⟩
      simpa only [δ, read_sum_left, hv, Option.map_some]
    | inr v =>
      obtain ⟨i, hi, hv⟩ := H.common_coverage p.primitiveU p.w p.primitive p.memU hu v
      refine ⟨Sum.inr H.root, Or.inr htr.symm, i, hi, ?_⟩
      simpa only [δ, read_sum_right, hv, Option.map_some]

/-- The formerly unproved geometry contract, discharged for every finite alphabet. -/
theorem nontrivialGeometry : NontrivialGeometry.{u} := by
  intro A _ _ p hx hu
  obtain ⟨c⟩ := geometricInput_exists p hx hu
  exact c.certificates
```

```lean
/-- Positive integer weighted bound, including the sharp uniform deficit three. -/
theorem weighted_primitive_intersection_add_three
    {A : Type u} [Fintype A] [DecidableEq A]
    (ρ : A → ℕ) (hρ : ∀ a, 0 < ρ a)
    (w : FreeMonoid A) (hw : PrimitiveWord w) (X U : Finset (FreeMonoid A))
    (hX : PrimitivePair X) (hU : PrimitivePair U) (hne : X ≠ U)
    (hxw : w ∈ generatedMonoid X) (huw : w ∈ generatedMonoid U) :
    weightedLength ρ w + 3 ≤
      (∑ x ∈ X, weightedLength ρ x) + (∑ u ∈ U, weightedLength ρ u) := by
  let p : Descent.Problem A := ⟨ρ, hρ, w, hw, X, U, hX, hU, hne, hxw, huw⟩
  exact Descent.bound_of_finiteStep
    (Descent.finiteStep_of_geometry (Descent.geometry_of_nontrivial Descent.nontrivialGeometry)) A p

/-- The literal half-length statement of the source, with no geometry premise. -/
theorem half_length_biroot_unique {A : Type u} [Fintype A] [DecidableEq A]
    (w : FreeMonoid A) (hw : PrimitiveWord w) (X U : Finset (FreeMonoid A))
    (hX : PrimitivePair X) (hU : PrimitivePair U)
    (hxw : w ∈ generatedMonoid X) (huw : w ∈ generatedMonoid U)

  Descent.half_length_of_nontrivial_geometry Descent.nontrivialGeometry A
```
