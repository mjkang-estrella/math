import Mathlib.Algebra.Group.Submonoid.Basic

/-!
# Definitions of Kashiwabara (EJC 16 (2009) #R55, arXiv:0809.3682), transcribed literally

* `K V`: the set `K = (V choose 2)` of unordered pairs of distinct vertices.
* `IsEqualPartition`, `EqPart V`: equal partitions, a partition of `V` into two sets of the
  same size, and the set `C = C(V)` of them.
* `Bc c`: the complete bipartite graph whose partite sets are the two blocks of `c`.
* `chi c q`: the vector `χ_{q,c} ∈ ℕ^{K ∪ C}` (here `ℕ^{K ⊕ C}`).
* `PM V`: `PM(V) = {χ_{q,c} | q is a perfect matching of B_c for some c ∈ C}`.
* `N M`: `N(M)`, all nonnegative integral combinations of `M` (the additive submonoid
  generated by `M`); `Nbar M`: `N̄(M) = {v | k v ∈ N(M) for some k ≥ 1}`.
* `chiE G`: the characteristic vector `χ_E` of the edges of `G` on `K`.
* `BFactorizable G` (Definition 4): every `v ∈ N̄(PM(V))` with `v|_K = χ_E` lies in `N(PM(V))`.
* `prism k hk`: the `k`-gonal prism on `Fin k × Bool`.
* `HAP V r`, `InP G HA x`: an HAP set for `r` days and the polytope `P(G, HA)` (Conditions 1–4).
-/

namespace Prism317750

variable (V : Type*) [Fintype V] [DecidableEq V]

/-- `K = (V choose 2)`: unordered pairs of distinct vertices. -/
abbrev K := {e : Sym2 V // ¬ e.IsDiag}

/-- An equal partition of `V`: a partition `{H, A}` of `V` into two sets of the same size. -/
def IsEqualPartition (c : Finset (Finset V)) : Prop :=
  ∃ H A : Finset V, Disjoint H A ∧ H ∪ A = Finset.univ ∧ H.card = A.card ∧ c = {H, A}

/-- `C = C(V)`: the set of equal partitions of `V`. -/
abbrev EqPart := {c : Finset (Finset V) // IsEqualPartition V c}

variable {V}

/-- `B_c`: the complete bipartite graph whose partite sets are the two blocks of `c`. -/
def Bc (c : EqPart V) : SimpleGraph V where
  Adj a b := ∃ X ∈ c.1, ∃ Y ∈ c.1, X ≠ Y ∧ a ∈ X ∧ b ∈ Y
  symm := ⟨fun _ _ ⟨X, hX, Y, hY, hXY, ha, hb⟩ => ⟨Y, hY, X, hX, hXY.symm, hb, ha⟩⟩
  loopless := by
    refine ⟨fun a ⟨X, hX, Y, hY, hXY, ha, hb⟩ => ?_⟩
    obtain ⟨H, A, hd, -, -, hc⟩ := c.2
    rw [hc, Finset.mem_insert, Finset.mem_singleton] at hX hY
    rcases hX with rfl | rfl <;> rcases hY with rfl | rfl
    · exact hXY rfl
    · exact Finset.disjoint_left.mp hd ha hb
    · exact Finset.disjoint_left.mp hd hb ha
    · exact hXY rfl

open scoped Classical in
/-- The vector `χ_{q,c} ∈ ℕ^{K ⊕ C}`: `1` on the edges of `q`, `1` at `c`, `0` elsewhere. -/
noncomputable def chi (c : EqPart V) (q : (Bc c).Subgraph) : K V ⊕ EqPart V → ℕ
  | Sum.inl e => if e.1 ∈ q.edgeSet then 1 else 0
  | Sum.inr c' => if c' = c then 1 else 0

variable (V) in
/-- `PM(V) = {χ_{q,c} | q is a perfect matching of B_c for some c ∈ C}`. -/
def PM : Set (K V ⊕ EqPart V → ℕ) :=
  {v | ∃ (c : EqPart V) (q : (Bc c).Subgraph), q.IsPerfectMatching ∧ v = chi c q}

/-- `N(M)`: the set of all nonnegative integral combinations of elements of `M`. -/
def N (M : Set (K V ⊕ EqPart V → ℕ)) : AddSubmonoid (K V ⊕ EqPart V → ℕ) :=
  AddSubmonoid.closure M

/-- `N̄(M) = {v ∈ ℕ^{K ∪ C} | k v ∈ N(M) for some k ≥ 1}`. -/
def Nbar (M : Set (K V ⊕ EqPart V → ℕ)) : Set (K V ⊕ EqPart V → ℕ) :=
  {v | ∃ m : ℕ, 1 ≤ m ∧ m • v ∈ N M}

open scoped Classical in
/-- `χ_E`: the characteristic function of the edge set of `G` on `K`. -/
noncomputable def chiE (G : SimpleGraph V) : K V → ℕ :=
  fun e => if e.1 ∈ G.edgeSet then 1 else 0

/-- Definition 4: `G` is B-factorizable if every `v ∈ N̄(PM(V))` with `v|_K = χ_E`
satisfies `v ∈ N(PM(V))`. -/
def BFactorizable (G : SimpleGraph V) : Prop :=
  ∀ v ∈ Nbar (PM V), (fun e => v (Sum.inl e)) = chiE G → v ∈ N (PM V)

/-- The `k`-gonal prism `C_k □ K_2` on `Fin k × Bool`: two `k`-cycles `(i, b) ~ (i + 1, b)`
(indices mod `k`) and the matching edges `(i, false) ~ (i, true)`. -/
def prism (k : ℕ) (hk : 3 ≤ k) : SimpleGraph (Fin k × Bool) where
  Adj x y := (x.2 = y.2 ∧ (y.1.val = (x.1.val + 1) % k ∨ x.1.val = (y.1.val + 1) % k)) ∨
    (x.1 = y.1 ∧ x.2 ≠ y.2)
  symm := by
    refine ⟨fun x y h => ?_⟩
    rcases h with ⟨h1, h2 | h2⟩ | ⟨h1, h2⟩
    · exact Or.inl ⟨h1.symm, Or.inr h2⟩
    · exact Or.inl ⟨h1.symm, Or.inl h2⟩
    · exact Or.inr ⟨h1.symm, h2.symm⟩
  loopless := by
    refine ⟨fun x h => ?_⟩
    have key : x.1.val ≠ (x.1.val + 1) % k := by
      intro h'
      have hx := x.1.isLt
      rcases Nat.lt_or_ge (x.1.val + 1) k with h1 | h1
      · rw [Nat.mod_eq_of_lt h1] at h'; omega
      · have : x.1.val + 1 = k := by omega
        rw [this, Nat.mod_self] at h'; omega
    rcases h with ⟨_, h | h⟩ | ⟨_, h⟩
    · exact key h
    · exact key h
    · exact h rfl

/-- An HAP set for `r` days: `d ↦ {H(d), A(d)} ∈ C` (days `0, …, r-1`). -/
structure HAP (r : ℕ) where
  H : Fin r → Finset V
  A : Fin r → Finset V
  eq : ∀ d, IsEqualPartition V {H d, A d}

/-- The pair `{a, b} ∈ K` for `a ≠ b`. -/
def pair (a b : V) (h : a ≠ b) : K V := ⟨s(a, b), Sym2.mk_isDiag_iff.not.mpr h⟩

/-- The polytope `P(G, HA)`: `x` satisfies Conditions 1–4 of Section 3.1. -/
def InP (G : SimpleGraph V) {r : ℕ} (HA : HAP (V := V) r) (x : K V → Fin r → ℝ) : Prop :=
  -- Condition 1
  (∀ e : K V, e.1 ∈ G.edgeSet → ∑ d, x e d = 1) ∧
  (∀ e : K V, e.1 ∉ G.edgeSet → ∀ d, x e d = 0) ∧
  -- Condition 2
  (∀ (d : Fin r) (a : V), ∑ b : {b : V // b ≠ a}, x (pair a b.1 b.2.symm) d = 1) ∧
  -- Condition 3
  (∀ (d : Fin r) (e : K V), e.1 ∈ G.edgeSet → 0 ≤ x e d ∧ x e d ≤ 1) ∧
  -- Condition 4
  (∀ (d : Fin r) (a b : V) (h : a ≠ b), s(a, b) ∈ G.edgeSet →
    ((a ∈ HA.H d ∧ b ∈ HA.H d) ∨ (a ∈ HA.A d ∧ b ∈ HA.A d)) → x (pair a b h) d = 0)

/-- An integral point. -/
def IsIntegral {r : ℕ} (x : K V → Fin r → ℝ) : Prop := ∀ e d, ∃ z : ℤ, x e d = z

/-- The Corollary 9 property for `r` days: for any HAP set, `P(G, HA)` contains an integral
point whenever it is non-empty. -/
def PolytopeIntegral (G : SimpleGraph V) (r : ℕ) : Prop :=
  ∀ HA : HAP (V := V) r, (∃ x, InP G HA x) → ∃ x, InP G HA x ∧ IsIntegral x

/-- Conjecture 17 of the source: the graph of every `k`-gonal prism is B-factorizable. -/
def Conjecture17 : Prop := ∀ (k : ℕ) (hk : 3 ≤ k), BFactorizable (prism k hk)

end Prism317750

instance (hk : 3 ≤ k) : DecidableRel (prism k hk).Adj := fun x y => by

/-- The vertex `(n mod k, b)`. -/
def vtx (n : ℕ) (b : Bool) : Fin k × Bool := (⟨n % k, Nat.mod_lt _ (by omega)⟩, b)

lemma vtx_eq_iff {n m : ℕ} {b c : Bool} : vtx hk n b = vtx hk m c ↔ n % k = m % k ∧ b = c := by
  simp [vtx, Prod.ext_iff, Fin.ext_iff]

lemma vtx_add (n : ℕ) (b : Bool) : vtx hk (n + k) b = vtx hk n b := by
  rw [vtx_eq_iff]; simp

lemma vtx_iter (n t : ℕ) (b : Bool) : vtx hk (n + t * k) b = vtx hk n b := by
  rw [vtx_eq_iff]; simp

lemma vtx_surj (a : Fin k × Bool) : ∃ m b, a = vtx hk (m + 1) b := by
  refine ⟨a.1.val + k - 1, a.2, ?_⟩
  have e : a.1.val + k - 1 + 1 = a.1.val + k := by omega
  simp only [vtx, e, Nat.add_mod_right, Nat.mod_eq_of_lt a.1.isLt]

include hk in
lemma mod_two_ne (m : ℕ) : (m + 2) % k ≠ m % k := by
  intro h
  have h' : m + 2 ≡ m + 0 [MOD k] := h
  have h2 := Nat.ModEq.add_left_cancel' m h'
  unfold Nat.ModEq at h2
  rw [Nat.mod_eq_of_lt (by omega : 2 < k), Nat.zero_mod] at h2
  omega

lemma adj_succ (n : ℕ) (b : Bool) : (prism k hk).Adj (vtx hk n b) (vtx hk (n + 1) b) := by
  left
  refine ⟨rfl, Or.inl ?_⟩
  simp [vtx, Nat.add_mod]

lemma adj_rung (n : ℕ) : (prism k hk).Adj (vtx hk n false) (vtx hk n true) := by
  right; exact ⟨rfl, by simp [vtx]⟩

/-- The three neighbours of `vtx (m+1) b`. -/
lemma adj_iff (m : ℕ) (b : Bool) (a : Fin k × Bool) :
    (prism k hk).Adj (vtx hk (m + 1) b) a ↔
      a = vtx hk (m + 2) b ∨ a = vtx hk m b ∨ a = vtx hk (m + 1) (!b) := by
  obtain ⟨⟨j, hj⟩, c⟩ := a
  have e1 : ((m + 1) % k + 1) % k = (m + 2) % k := by
    rw [Nat.mod_add_mod]
  have e2 : (m + 1) % k = (j + 1) % k ↔ j = m % k := by
    constructor
    · intro h
      have h' : m + 1 ≡ j + 1 [MOD k] := h
      have := Nat.ModEq.add_right_cancel' 1 h'
      unfold Nat.ModEq at this
      rw [Nat.mod_eq_of_lt hj] at this; exact this.symm
    · intro h
      have h' : m ≡ j [MOD k] := by unfold Nat.ModEq; rw [Nat.mod_eq_of_lt hj, h]
      exact Nat.ModEq.add_right 1 h'
  simp only [prism, vtx, Prod.mk.injEq, Fin.mk.injEq, e1, e2]
  constructor
  · rintro (⟨rfl, h | h⟩ | ⟨h1, h2⟩)
    · left; exact ⟨h, rfl⟩
    · right; left; exact ⟨h, rfl⟩
    · right; right; refine ⟨h1.symm, ?_⟩; revert h2; cases b <;> cases c <;> simp
  · rintro (⟨h, rfl⟩ | ⟨h, rfl⟩ | ⟨h1, h2⟩)
    · left; exact ⟨rfl, Or.inl h⟩
    · left; exact ⟨rfl, Or.inr h⟩
    · right; refine ⟨h1.symm, ?_⟩; rw [h2]; cases b <;> simp

lemma nbr_ne1 (m : ℕ) (b : Bool) : vtx hk (m + 2) b ≠ vtx hk m b := by
  rw [Ne, vtx_eq_iff]; exact fun h => mod_two_ne hk m h.1

lemma nbr_ne2 (m : ℕ) (b : Bool) : vtx hk (m + 2) b ≠ vtx hk (m + 1) (!b) := by
  rw [Ne, vtx_eq_iff]; intro h; cases b <;> simp at h

lemma nbr_ne3 (m : ℕ) (b : Bool) : vtx hk m b ≠ vtx hk (m + 1) (!b) := by
  rw [Ne, vtx_eq_iff]; intro h; cases b <;> simp at h

lemma neighborFinset_eq (m : ℕ) (b : Bool) :
    (prism k hk).neighborFinset (vtx hk (m + 1) b) =
      {vtx hk (m + 2) b, vtx hk m b, vtx hk (m + 1) (!b)} := by
  ext a
  rw [SimpleGraph.mem_neighborFinset, adj_iff]
  simp

/-- The prism is 3-regular. -/
theorem prism_degree (a : Fin k × Bool) : (prism k hk).degree a = 3 := by
  obtain ⟨m, b, rfl⟩ := vtx_surj hk a
  rw [← SimpleGraph.card_neighborFinset_eq_degree, neighborFinset_eq]
  rw [Finset.card_insert_of_notMem, Finset.card_insert_of_notMem, Finset.card_singleton]
  · simpa using nbr_ne3 hk m b
  · simp only [Finset.mem_insert, Finset.mem_singleton, not_or]

theorem prism_card_edges : (prism k hk).edgeFinset.card = 3 * k := by
  have h := SimpleGraph.sum_degrees_eq_twice_card_edges (prism k hk)
  simp only [prism_degree, Finset.sum_const, Finset.card_univ, Fintype.card_prod, Fintype.card_fin,
    Fintype.card_bool, smul_eq_mul] at h
  omega

/-- A function supported on the neighbours of `a`: its sum over `b ≠ a` is the sum over the
three neighbours. -/
lemma sum_ne_nbr {M : Type*} [AddCommMonoid M] (m : ℕ) (b : Bool)
    (g : Fin k × Bool → M) (hg : ∀ c, ¬ (prism k hk).Adj (vtx hk (m + 1) b) c → g c = 0) :
    ∑ c : {c : Fin k × Bool // c ≠ vtx hk (m + 1) b}, g c.1 =
      g (vtx hk (m + 2) b) + g (vtx hk m b) + g (vtx hk (m + 1) (!b)) := by
  rw [← Finset.sum_subtype (Finset.univ.filter (· ≠ vtx hk (m + 1) b)) (by simp) g]
  rw [← Finset.sum_subset (s₁ := (prism k hk).neighborFinset (vtx hk (m + 1) b))]
  · rw [neighborFinset_eq, Finset.sum_insert, Finset.sum_insert, Finset.sum_singleton, add_assoc]
    · simpa using nbr_ne3 hk m b
    · simp only [Finset.mem_insert, Finset.mem_singleton, not_or]
      exact ⟨nbr_ne1 hk m b, nbr_ne2 hk m b⟩
  · intro c hc
    rw [SimpleGraph.mem_neighborFinset] at hc
    simp only [Finset.mem_filter, Finset.mem_univ, true_and]
    exact fun h => (prism k hk).loopless.irrefl _ (h ▸ hc)
  · intro c _ hc
    rw [SimpleGraph.mem_neighborFinset] at hc
    exact hg c hc

import Mathlib.Algebra.Order.Field.Basic
import Mathlib.Algebra.BigOperators.Fin
import Mathlib.Algebra.Group.Nat.Even

import Mathlib.Tactic.Linarith

/-!
# The abstract ladder problem

The prism `C_k □ K_2` with vertices `u_n, w_n` (`n ∈ ℕ`, read modulo `k`), edges
`a_n = u_n u_{n+1}`, `b_n = w_n w_{n+1}`, `r_n = u_n w_n`, a fractional point `x` and the
side labels of the three days are encoded as `k`-periodic sequences indexed by `ℕ`.
This file contains the encoding, the third-colour function, lateral alternation
(Step 1 of the proof), the cut-parity fact (P2), the u↔w swap and the two colouring
constructors (type E and type S).
-/

namespace Prism317750

/-- The third colour: for `a ≠ b`, the unique colour different from both. -/
def third (a b : Fin 3) : Fin 3 :=
  if a ≠ 0 ∧ b ≠ 0 then 0 else if a ≠ 1 ∧ b ≠ 1 then 1 else 2

lemma third_ne_left {a b : Fin 3} (h : a ≠ b) : third a b ≠ a := by
  revert a b; decide

lemma third_ne_right {a b : Fin 3} (h : a ≠ b) : third a b ≠ b := by
  revert a b; decide

lemma third_comm (a b : Fin 3) : third a b = third b a := by
  revert a b; decide

lemma eq_third {a b c : Fin 3} (hab : a ≠ b) (hca : c ≠ a) (hcb : c ≠ b) : c = third a b := by
  revert a b c; decide

/-- The abstract ladder problem: side labels `P n b d` of vertex `(n, b)` on day `d`
(`b = false` is the `u`-cycle) and a fractional point given by the values on the
edges `a_n`, `b_n`, `r_n`. -/
structure Ladder (k : ℕ) where
  P : ℕ → Bool → Fin 3 → Bool
  xa : ℕ → Fin 3 → ℝ
  xb : ℕ → Fin 3 → ℝ
  xr : ℕ → Fin 3 → ℝ
  P_per : ∀ n b d, P (n + k) b d = P n b d
  xa_per : ∀ n d, xa (n + k) d = xa n d
  xb_per : ∀ n d, xb (n + k) d = xb n d
  xa_nn : ∀ n d, 0 ≤ xa n d
  xb_nn : ∀ n d, 0 ≤ xb n d
  xr_nn : ∀ n d, 0 ≤ xr n d
  xa_sum : ∀ n, xa n 0 + xa n 1 + xa n 2 = 1
  xb_sum : ∀ n, xb n 0 + xb n 1 + xb n 2 = 1
  xr_sum : ∀ n, xr n 0 + xr n 1 + xr n 2 = 1
  va : ∀ n d, xa n d + xr (n + 1) d + xa (n + 1) d = 1
  vb : ∀ n d, xb n d + xr (n + 1) d + xb (n + 1) d = 1
  sa : ∀ n d, xa n d ≠ 0 → P n false d ≠ P (n + 1) false d
  sb : ∀ n d, xb n d ≠ 0 → P n true d ≠ P (n + 1) true d
  sr : ∀ n d, xr n d ≠ 0 → P n false d ≠ P n true d

namespace Ladder

variable {k : ℕ} (L : Ladder k)

/-- Colour `c` is allowed on `a_n`. -/
def okA (n : ℕ) (c : Fin 3) : Prop := L.P n false c ≠ L.P (n + 1) false c
/-- Colour `c` is allowed on `b_n`. -/
def okB (n : ℕ) (c : Fin 3) : Prop := L.P n true c ≠ L.P (n + 1) true c
/-- Colour `c` is allowed on the rung `r_n`. -/
def okR (n : ℕ) (c : Fin 3) : Prop := L.P n false c ≠ L.P n true c
/-- Colour `c` is allowed on both `a_n` and `b_n` (the set `L_n` of the proof). -/
def inL (n : ℕ) (c : Fin 3) : Prop := L.okA n c ∧ L.okB n c
/-- An arc of the layered digraph `Γ` from `(n, c)` to `(n+1, c')`. -/
def arc (n : ℕ) (c c' : Fin 3) : Prop :=
  L.inL n c ∧ L.inL (n + 1) c' ∧ c ≠ c' ∧ L.okR (n + 1) (third c c')

lemma per_iter {α : Type*} {f : ℕ → α} (hf : ∀ n, f (n + k) = f n) (n t : ℕ) :
    f (n + t * k) = f n := by
  induction t with
  | zero => simp
  | succ t ih => rw [Nat.succ_mul, ← Nat.add_assoc, hf, ih]

lemma P_iter (n t : ℕ) (b : Bool) (d : Fin 3) : L.P (n + t * k) b d = L.P n b d :=
  per_iter (f := fun n => L.P n b d) (fun n => L.P_per n b d) n t

lemma okR_per (n : ℕ) (c : Fin 3) : L.okR (n + k) c ↔ L.okR n c := by
  simp [okR, L.P_per]

lemma okR_iter (n t : ℕ) (c : Fin 3) : L.okR (n + t * k) c ↔ L.okR n c := by
  simp [okR, L.P_iter]

lemma arc_per (n : ℕ) (c c' : Fin 3) : L.arc (n + k) c c' ↔ L.arc n c c' := by
  have h1 : n + k + 1 = n + 1 + k := by omega
  have h2 : n + 1 + k + 1 = n + 1 + 1 + k := by omega
  simp only [arc, inL, okA, okB, okR, h1, h2, L.P_per]

lemma okA_of_pos {n : ℕ} {c : Fin 3} (h : 0 < L.xa n c) : L.okA n c := L.sa n c h.ne'
lemma okB_of_pos {n : ℕ} {c : Fin 3} (h : 0 < L.xb n c) : L.okB n c := L.sb n c h.ne'
lemma okR_of_pos {n : ℕ} {c : Fin 3} (h : 0 < L.xr n c) : L.okR n c := L.sr n c h.ne'

lemma xr_zero {n : ℕ} {c : Fin 3} (h : ¬ L.okR n c) : L.xr n c = 0 := by
  by_contra h'; exact h (L.sr n c h')

/-- (P2): a colour allowed on both `a_n` and `b_n` has the same rung bit on `r_n` and `r_{n+1}`. -/
lemma p2 {n : ℕ} {c : Fin 3} (h : L.inL n c) : L.okR n c ↔ L.okR (n + 1) c := by
  obtain ⟨h1, h2⟩ := h
  unfold okA at h1; unfold okB at h2; unfold okR
  revert h1 h2
  cases L.P n false c <;> cases L.P (n + 1) false c <;> cases L.P n true c <;>
    cases L.P (n + 1) true c <;> simp

/-- Lateral alternation (Step 1). -/
lemma alt (n : ℕ) (d : Fin 3) :
    L.xa n d - L.xb n d = if Even n then L.xa 0 d - L.xb 0 d else -(L.xa 0 d - L.xb 0 d) := by
  induction n with
  | zero => simp
  | succ n ih =>
    have h1 := L.va n d
    have h2 := L.vb n d
    by_cases he : Even n
    · have : ¬ Even (n + 1) := by rw [Nat.even_add_one]; exact not_not.mpr he
      rw [ite_eq_left he] at ih; rw [ite_eq_right this]; linarith
    · have : Even (n + 1) := by rw [Nat.even_add_one]; exact he

/-- A proper 3-edge-colouring of the ladder compatible with the labels. -/
structure Coloring {k : ℕ} (L : Ladder k) where
  ca : ℕ → Fin 3
  cb : ℕ → Fin 3
  cr : ℕ → Fin 3
  ca_per : ∀ n, ca (n + k) = ca n
  cb_per : ∀ n, cb (n + k) = cb n
  cr_per : ∀ n, cr (n + k) = cr n
  oka : ∀ n, L.okA n (ca n)
  okb : ∀ n, L.okB n (cb n)
  okr : ∀ n, L.okR n (cr n)
  pa : ∀ n, ca n ≠ ca (n + 1) ∧ ca n ≠ cr (n + 1) ∧ ca (n + 1) ≠ cr (n + 1)
  pb : ∀ n, cb n ≠ cb (n + 1) ∧ cb n ≠ cr (n + 1) ∧ cb (n + 1) ≠ cr (n + 1)

  cb_per := C.ca_per
  cr_per := C.cr_per
  oka := C.okb
  okb := C.oka
  okr n := (C.okr n).symm
  pa := C.pb
  pb := C.pa

    (h2 : ∀ n, ¬ Even n → 0 < L.xa n q ∧ 0 < L.xb n p)
    (hr : ∀ n, L.okR n (third p q)) : Nonempty (Coloring L) := by
  have hpar : ∀ n, Even (n + k) ↔ Even n := fun n => by
    rw [Nat.even_add]; exact ⟨fun h => h.mpr hk, fun h => ⟨fun _ => hk, fun _ => h⟩⟩
  have hs : ∀ n, Even (n + 1) ↔ ¬ Even n := fun n => Nat.even_add_one
  have t1 := third_ne_left hpq
  have t2 := third_ne_right hpq
  refine ⟨{
    ca := fun n => if Even n then p else q
    cb := fun n => if Even n then q else p

    ca_per := fun n => by simp only [hpar]
    cb_per := fun n => by simp only [hpar]
    cr_per := fun n => rfl
    oka := fun n => ?_
    okb := fun n => ?_
    okr := hr
    pa := fun n => ?_
    pb := fun n => ?_ }⟩
  · by_cases h : Even n
    · simp only [ite_eq_left h]; exact L.okA_of_pos (h1 n h).1
    · simp only [ite_eq_right h]; exact L.okA_of_pos (h2 n h).1
  · by_cases h : Even n
    · simp only [ite_eq_left h]; exact L.okB_of_pos (h1 n h).2
    · simp only [ite_eq_right h]; exact L.okB_of_pos (h2 n h).2
  · by_cases h : Even n
    · have h' : ¬ Even (n + 1) := by rw [hs]; exact not_not.mpr h
      simp only [ite_eq_left h, ite_eq_right h']
      exact ⟨hpq, t1.symm, t2.symm⟩

      simp only [ite_eq_right h, ite_eq_left h']

  · by_cases h : Even n

      simp only [ite_eq_left h, ite_eq_right h']
      exact ⟨hpq.symm, t2.symm, t1.symm⟩
    · have h' : Even (n + 1) := by rw [hs]; exact h
      simp only [ite_eq_right h, ite_eq_left h']
      exact ⟨hpq, t1.symm, t2.symm⟩

/-- `p` is a path of `Γ` on the layers `0, …, m`. -/
def IsPath (p : ℕ → Fin 3) (m : ℕ) : Prop := ∀ i, i < m → L.arc i (p i) (p (i + 1))

/-- One turn: a path from colour `c` at layer `0` to colour `c'` at layer `k`. -/
def T (c c' : Fin 3) : Prop := ∃ p, L.IsPath p k ∧ p 0 = c ∧ p k = c'

lemma arc_iter (n t : ℕ) (c c' : Fin 3) : L.arc (n + t * k) c c' ↔ L.arc n c c' := by
  induction t with
  | zero => simp
  | succ t ih => rw [Nat.succ_mul, ← Nat.add_assoc, L.arc_per, ih]

lemma splice {p q : ℕ → Fin 3} (hp : L.IsPath p k) (hq : L.IsPath q k) {n : ℕ} (hn : n < k)
    (h : L.arc n (p n) (q (n + 1))) : L.T (p 0) (q k) := by
  refine ⟨fun i => if i ≤ n then p i else q i, ?_, by simp, ?_⟩
  · intro i hi
    rcases lt_trichotomy i n with h1 | h1 | h1
    · have h2 : i + 1 ≤ n := h1
      simp only [ite_eq_left h1.le, ite_eq_left h2]
      exact hp i hi
    · subst h1
      simp only [ite_eq_left (le_refl i), ite_eq_right (Nat.not_succ_le_self i)]
      exact h
    · have h2 : ¬ i ≤ n := not_le.mpr h1
      have h3 : ¬ i + 1 ≤ n := by omega
      simp only [ite_eq_right h2, ite_eq_right h3]
      exact hq i hi
  · simp only [ite_eq_right (not_le.mpr hn)]

lemma splice_eq {p q : ℕ → Fin 3} (hp : L.IsPath p k) (hq : L.IsPath q k) {n : ℕ} (hn : n ≤ k)
    (h : p n = q n) : L.T (p 0) (q k) := by
  rcases hn.lt_or_eq with hn | hn
  · exact L.splice hp hq hn (by rw [h]; exact hq n hn)
  · subst hn; exact ⟨p, hp, rfl, h⟩

  refine L.typeE hk (fun n => p (n % k)) (fun n => by simp) (fun n => ?_)
  obtain ⟨r, t, hr, rfl⟩ : ∃ r t, r < k ∧ n = r + t * k :=
    ⟨n % k, n / k, Nat.mod_lt _ hk, (Nat.mod_add_div' n k).symm⟩

    (t01 : L.T c0 c1) (t12 : L.T c1 c2) (t20 : L.T c2 c0) : False := by

  obtain ⟨p1, hp1, e10, e1k⟩ := id t12
  obtain ⟨p2, hp2, e20, e2k⟩ := id t20
  have b01 : p0 1 ≠ p1 1 := fun h => by
    have := L.splice_eq hp0 hp1 hk h; rw [e00, e1k] at this; exact hno2 _ _ this t20
  have b12 : p1 1 ≠ p2 1 := fun h => by
    have := L.splice_eq hp1 hp2 hk h; rw [e10, e2k] at this; exact hno2 _ _ this t01
  have b02 : p0 1 ≠ p2 1 := fun h => by
    have := L.splice_eq hp0 hp2 hk h; rw [e00, e2k] at this; exact hno1 _ this
  have a0 := (hp0 0 hk).2.2.1
  have a1 := (hp1 0 hk).2.2.1
  have a2 := (hp2 0 hk).2.2.1
  have cover : ∀ y, L.okR 1 y := by
    intro y
    have h01' : p0 0 ≠ p1 0 := by rw [e00, e10]; exact h01
    have h02' : p0 0 ≠ p2 0 := by rw [e00, e20]; exact h02
    have h12' : p1 0 ≠ p2 0 := by rw [e10, e20]; exact h12
    rcases fin3_cover h01' h02' h12' b01 b02 b12 a0 a1 a2 y with h | h | h <;> rw [h]
    · exact (hp0 0 hk).2.2.2
    · exact (hp1 0 hk).2.2.2
    · exact (hp2 0 hk).2.2.2
  by_cases hb : p1 1 = p0 0
  · have hne : p0 0 ≠ p2 1 := fun h => b12 (hb.trans h)
    have := L.splice hp0 hp2 hk ⟨(hp0 0 hk).1, (hp2 0 hk).2.1, hne, cover _⟩
    rw [e00, e2k] at this; exact hno1 _ this
  · have := L.splice hp0 hp1 hk ⟨(hp0 0 hk).1, (hp1 0 hk).2.1, Ne.symm hb, cover _⟩
    rw [e00, e1k] at this; exact hno2 _ _ this t20

/-- (2a): the successor lemma. -/
lemma succ_arc (hE : ∀ n d, L.xa n d = L.xb n d) (n : ℕ) (c : Fin 3) (hc : 0 < L.xa n c) :
    ∃ c', 0 < L.xa (n + 1) c' ∧ L.arc n c c' := by
  have key : ∀ c : Fin 3, ∃ d e : Fin 3, c ≠ d ∧ c ≠ e ∧ d ≠ e ∧ third c d = e ∧
      third c e = d := by decide
  obtain ⟨d, e, hcd, hce, hde, hte, htd⟩ := key c
  have mk : ∀ c', c ≠ c' → 0 < L.xa (n + 1) c' → 0 < L.xr (n + 1) (third c c') →
      L.arc n c c' := fun c' hne h1 h2 =>
    ⟨⟨L.okA_of_pos hc, L.okB_of_pos (by rw [← hE]; exact hc)⟩,
     ⟨L.okA_of_pos h1, L.okB_of_pos (by rw [← hE]; exact h1)⟩, hne, L.okR_of_pos h2⟩
  by_contra hcon
  push Not at hcon
  have hd : L.xa (n + 1) d ≤ 0 ∨ L.xr (n + 1) e ≤ 0 := by
    by_contra h; push Not at h
    exact hcon d h.1 (mk d hcd h.1 (by rw [hte]; exact h.2))
  have he : L.xa (n + 1) e ≤ 0 ∨ L.xr (n + 1) d ≤ 0 := by
    by_contra h; push Not at h
    exact hcon e h.1 (mk e hce h.1 (by rw [htd]; exact h.2))
  have S1 := sum3 (L.xa n) (L.xa_sum n) hcd hce hde
  have S2 := sum3 (L.xa (n + 1)) (L.xa_sum (n + 1)) hcd hce hde
  have S3 := sum3 (L.xr (n + 1)) (L.xr_sum (n + 1)) hcd hce hde
  have Vc := L.va n c
  have Vd := L.va n d
  have Ve := L.va n e
  have := L.xa_nn n d; have := L.xa_nn n e
  have := L.xa_nn (n + 1) c; have := L.xa_nn (n + 1) d; have := L.xa_nn (n + 1) e
  have := L.xr_nn (n + 1) c; have := L.xr_nn (n + 1) d; have := L.xr_nn (n + 1) e
  rcases hd with hd | hd <;> rcases he with he | he <;> linarith

lemma reach (hE : ∀ n d, L.xa n d = L.xb n d) (c : Fin 3) (hc : 0 < L.xa 0 c) :
    ∀ m, ∃ p, L.IsPath p m ∧ p 0 = c ∧ 0 < L.xa m (p m) := by
  intro m
  induction m with
  | zero => exact ⟨fun _ => c, fun i hi => absurd hi (Nat.not_lt_zero _), rfl, hc⟩
  | succ m ih =>
    obtain ⟨p, hp, hp0, hpm⟩ := ih
    obtain ⟨c', hc', harc⟩ := L.succ_arc hE m (p m) hpm
    refine ⟨fun i => if i ≤ m then p i else c', ?_, by simp [hp0], ?_⟩
    · intro i hi
      rcases (Nat.lt_succ_iff.mp hi).lt_or_eq with h1 | h1
      · have h2 : i + 1 ≤ m := h1
        simp only [ite_eq_left h1.le, ite_eq_left h2]
        exact hp i h1
      · subst h1
        simp only [ite_eq_left (le_refl i), ite_eq_right (Nat.not_succ_le_self i)]
        exact harc
    · simp only [ite_eq_right (Nat.not_succ_le_self m)]; exact hc'

/-- Step 2: if `x_a = x_b` there is a type-E colouring. -/
theorem existsE (hk : 0 < k) (hE : ∀ n d, L.xa n d = L.xb n d) : Nonempty (Coloring L) := by
  by_cases hloop : ∃ c, L.T c c
  · obtain ⟨c, hc⟩ := hloop; exact L.closed hk hc
  push Not at hloop
  have hno2 : ∀ c c', L.T c c' → ¬ L.T c' c := fun c c' h1 h2 => L.no_two hk hloop h1 h2
  have tot : ∀ c, 0 < L.xa 0 c → ∃ c', 0 < L.xa 0 c' ∧ L.T c c' := by
    intro c hc
    obtain ⟨p, hp, hp0, hpk⟩ := L.reach hE c hc k
    refine ⟨p k, ?_, p, hp, hp0, rfl⟩
    have := L.xa_per 0 (p k); rw [Nat.zero_add] at this; rwa [← this]
  obtain ⟨c0, hc0⟩ : ∃ c, 0 < L.xa 0 c := by
    by_contra h; push Not at h
    have := L.xa_sum 0; linarith [h 0, h 1, h 2]
  obtain ⟨c1, hc1, t01⟩ := tot c0 hc0
  obtain ⟨c2, hc2, t12⟩ := tot c1 hc1
  obtain ⟨c3, _, t23⟩ := tot c2 hc2
  have n01 : c0 ≠ c1 := fun h => hloop c0 (h ▸ t01)
  have n12 : c1 ≠ c2 := fun h => hloop c1 (h ▸ t12)
  have n23 : c2 ≠ c3 := fun h => hloop c2 (h ▸ t23)
  have n02 : c0 ≠ c2 := fun h => hno2 c0 c1 t01 (h ▸ t12)
  have n13 : c1 ≠ c3 := fun h => hno2 c1 c2 t12 (h ▸ t23)
  have e30 : c3 = c0 := fin3_last n01 n02.symm n12.symm n13.symm n23.symm
  exact (L.no_three hk hloop hno2 n01 n02 n12 t01 t12 (e30 ▸ t23)).elim

end Ladder

lemma sum3s (f : Fin 3 → ℝ) (s : ℝ) (hf : f 0 + f 1 + f 2 = s) {c d e : Fin 3} (hcd : c ≠ d)

/-- The bounds (F) for even layers. -/
lemma F_even {n : ℕ} (h : Even n) (c : Fin 3) :
    L.xa 0 c - L.xb 0 c ≤ L.xa n c ∧ -(L.xa 0 c - L.xb 0 c) ≤ L.xb n c := by
  have := L.alt n c
  rw [ite_eq_left h] at this
  constructor <;> linarith [L.xa_nn n c, L.xb_nn n c]

/-- The bounds (F) for odd layers. -/
lemma F_odd {n : ℕ} (h : ¬ Even n) (c : Fin 3) :
    -(L.xa 0 c - L.xb 0 c) ≤ L.xa n c ∧ L.xa 0 c - L.xb 0 c ≤ L.xb n c := by
  have := L.alt n c
  rw [ite_eq_right h] at this
  constructor <;> linarith [L.xa_nn n c, L.xb_nn n c]

/-- Propagation of a rung exclusion around the prism, via (P2). -/
lemma propagate (hk : 0 < k) (c : Fin 3) (H : ∀ m, ¬ L.okR (m + 1) c → L.inL (m + 1) c)
    {j : ℕ} (hj : ¬ L.okR j c) : ∀ n, ¬ L.okR n c := by
  have hj' : ¬ L.okR (j + k) c := by rwa [L.okR_per]
  have fwd : ∀ t, ¬ L.okR (j + k + t) c := by
    intro t
    induction t with
    | zero => simpa using hj'
    | succ t ih =>
      have e : j + k + t = (j + k + t - 1) + 1 := by omega
      have hL := H (j + k + t - 1) (by rw [← e]; exact ih)
      rw [← e] at hL
      rw [← Nat.add_assoc]
      rwa [← L.p2 hL]
  intro n
  have hM : j + k ≤ (j + 1) * k := by nlinarith
  have := fwd (n + (j + 1) * k - (j + k))
  rw [show j + k + (n + (j + 1) * k - (j + k)) = n + (j + 1) * k by omega, L.okR_iter] at this

lemma step3 (hk : 0 < k) (hev : Even k) {α β : Fin 3} (hab : α ≠ β)
    (hα : 0 < L.xa 0 α - L.xb 0 α) (hβ : L.xa 0 β - L.xb 0 β < 0)
    (hγ : 0 ≤ L.xa 0 (third α β) - L.xb 0 (third α β)) : Nonempty (Coloring L) := by
  set γ := third α β with hγdef
  have hga : γ ≠ α := third_ne_left hab
  have hgb : γ ≠ β := third_ne_right hab
  have Sa : ∀ n, L.xa n α + L.xa n β + L.xa n γ = 1 := fun n =>
    sum3s _ _ (L.xa_sum n) hab hga.symm hgb.symm

    refine L.typeS hev α γ hga.symm (fun n hn => ?_) (fun n hn => ?_) (fun n => e ▸ hβR n)

    · exact ⟨hpos n, by linarith [(L.F_odd hn α).2]⟩

/-- The abstract ladder theorem: every ladder problem has a colouring. -/
theorem coloring (hk : 0 < k) : Nonempty (Coloring L) := by
  by_cases hE : ∀ d, L.xa 0 d = L.xb 0 d
  · refine L.existsE hk (fun n d => ?_)
    have := L.alt n d
    rw [hE d] at this
    split_ifs at this <;> linarith
  have hev : Even k := by
    by_contra h; exact hE (L.delta_zero_of_odd h)
  push Not at hE
  obtain ⟨d, hd⟩ := hE
  have hsum : (L.xa 0 0 - L.xb 0 0) + (L.xa 0 1 - L.xb 0 1) + (L.xa 0 2 - L.xb 0 2) = 0 := by
    linarith [L.xa_sum 0, L.xb_sum 0]
  obtain ⟨e, f, hde, hdf, hef⟩ := others d
  have hs := sum3s (fun c => L.xa 0 c - L.xb 0 c) 0 hsum hde hdf hef
  -- a positive and a negative coordinate
  obtain ⟨α, hα⟩ : ∃ α, 0 < L.xa 0 α - L.xb 0 α := by
    by_contra h; push Not at h
    have := h d; have := h e; have := h f
    exact hd (by linarith)
  obtain ⟨β, hβ⟩ : ∃ β, L.xa 0 β - L.xb 0 β < 0 := by
    by_contra h; push Not at h
    have := h d; have := h e; have := h f
    exact hd (by linarith)
  have hab : α ≠ β := fun h => by rw [h] at hα; linarith
  rcases le_or_gt 0 (L.xa 0 (third α β) - L.xb 0 (third α β)) with hγ | hγ
  · exact L.step3 hk hev hab hα hβ hγ
  · have h := L.swap.step3 hk hev (α := β) (β := α) hab.symm
      (by show 0 < L.xb 0 β - L.xa 0 β; linarith)
      (by show L.xb 0 α - L.xa 0 α < 0; linarith)
      (by show 0 ≤ L.xb 0 (third β α) - L.xa 0 (third β α); rw [third_comm]; linarith)
    exact ⟨L.unswap h.some⟩

end Ladder

/-!
# The polytope theorem for the prism (the Corollary 9 form)

A point of `P(prism, HA)` gives an abstract ladder problem; its colouring gives an integral
point of `P(prism, HA)`.
-/

set_option linter.unusedSectionVars false

section Generic

variable {V : Type*} [Fintype V] [DecidableEq V]

/-- In an equal partition `{H, A}`, `A` is the complement of `H`. -/
lemma eqpart_compl {H A : Finset V} (h : IsEqualPartition V {H, A}) (a : V) :
    a ∈ A ↔ a ∉ H := by
  obtain ⟨H', A', hd, hu, -, he⟩ := h
  have hall : ∀ x, x ∈ H' ∨ x ∈ A' := fun x => by
    rw [← Finset.mem_union, hu]; exact Finset.mem_univ x
  have hdis : ∀ x, x ∈ H' → x ∉ A' := fun x hx => Finset.disjoint_left.mp hd hx
  have hH : H ∈ ({H', A'} : Finset (Finset V)) := he ▸ by simp
  have hA : A ∈ ({H', A'} : Finset (Finset V)) := he ▸ by simp
  have hH' : H' ∈ ({H, A} : Finset (Finset V)) := he.symm ▸ by simp
  have hA' : A' ∈ ({H, A} : Finset (Finset V)) := he.symm ▸ by simp
  simp only [Finset.mem_insert, Finset.mem_singleton] at hH hA hH' hA'
  rcases hH with rfl | rfl <;> rcases hA with rfl | rfl
  · rcases hA' with h | h <;> subst h <;>
      exact absurd (hall a) (by intro h; rcases h with h | h <;> exact hdis a h h)
  · exact ⟨fun h1 h2 => hdis a h2 h1, fun h1 => (hall a).resolve_left h1⟩
  · exact ⟨fun h1 h2 => hdis a h1 h2, fun h1 => (hall a).resolve_right h1⟩
  · rcases hH' with h | h <;> subst h <;>
      exact absurd (hall a) (by intro h; rcases h with h | h <;> exact hdis a h h)

/-- Same side on a day iff the `H`-labels agree. -/
lemma sameSide_iff {H A : Finset V} (h : IsEqualPartition V {H, A}) (a b : V) :
    ((a ∈ H ∧ b ∈ H) ∨ (a ∈ A ∧ b ∈ A)) ↔ decide (a ∈ H) = decide (b ∈ H) := by
  rw [eqpart_compl h a, eqpart_compl h b]
  by_cases ha : a ∈ H <;> by_cases hb : b ∈ H <;> simp [ha, hb]

/-- `x` read on ordered pairs of vertices. -/
noncomputable def xe {r : ℕ} (x : K V → Fin r → ℝ) (a b : V) (d : Fin r) : ℝ :=
  if h : a ≠ b then x (pair a b h) d else 0

lemma pair_swap (a b : V) (h : a ≠ b) : pair a b h = pair b a h.symm :=
  Subtype.ext Sym2.eq_swap

lemma xe_of_ne {r : ℕ} (x : K V → Fin r → ℝ) {a b : V} (h : a ≠ b) (d : Fin r) :
    xe x a b d = x (pair a b h) d := by
  simp [xe, h]

lemma xe_symm {r : ℕ} (x : K V → Fin r → ℝ) (a b : V) (d : Fin r) :
    xe x a b d = xe x b a d := by
  by_cases h : a = b
  · subst h; rfl
  · rw [xe_of_ne x h, xe_of_ne x (Ne.symm h), pair_swap]

variable {G : SimpleGraph V} {r : ℕ} {HA : HAP (V := V) r} {x : K V → Fin r → ℝ}

lemma InP.xe_zero (hx : InP G HA x) {a b : V} (h : ¬ G.Adj a b) (d : Fin r) : xe x a b d = 0 := by
  by_cases hab : a = b
  · simp [xe, hab]
  · rw [xe_of_ne x hab]; exact hx.2.1 (pair a b hab) h d

lemma InP.xe_nn (hx : InP G HA x) {a b : V} (h : G.Adj a b) (d : Fin r) : 0 ≤ xe x a b d := by
  rw [xe_of_ne x (G.ne_of_adj h)]; exact (hx.2.2.2.1 d (pair a b (G.ne_of_adj h)) h).1

lemma InP.xe_sum (hx : InP G HA x) {a b : V} (h : G.Adj a b) : ∑ d, xe x a b d = 1 := by
  simp only [xe_of_ne x (G.ne_of_adj h)]; exact hx.1 (pair a b (G.ne_of_adj h)) h

lemma InP.xe_vert (hx : InP G HA x) (a : V) (d : Fin r) :
    ∑ b : {b : V // b ≠ a}, xe x a b.1 d = 1 := by
  rw [← hx.2.2.1 d a]
  refine Finset.sum_congr rfl (fun b _ => ?_)
  rw [xe_of_ne x b.2.symm]

lemma InP.xe_side (hx : InP G HA x) {a b : V} (h : G.Adj a b) (d : Fin r)

  rw [xe_of_ne x (G.ne_of_adj h)]
  exact hx.2.2.2.2 d a b _ h ((sameSide_iff (HA.eq d) a b).mpr hs)

lemma three_ind (u v w d : Fin 3) (huv : u ≠ v) (huw : u ≠ w) (hvw : v ≠ w) :
    ((if u = d then 1 else 0) + (if v = d then 1 else 0) + (if w = d then 1 else 0) : ℝ) = 1 := by
  have : ∀ u v w d : Fin 3, u ≠ v → u ≠ w → v ≠ w →
      (u = d ∧ v ≠ d ∧ w ≠ d) ∨ (u ≠ d ∧ v = d ∧ w ≠ d) ∨ (u ≠ d ∧ v ≠ d ∧ w = d) := by
    decide
  rcases this u v w d huv huw hvw with ⟨h1, h2, h3⟩ | ⟨h1, h2, h3⟩ | ⟨h1, h2, h3⟩ <;>
    simp [h1, h2, h3]

  conv_rhs => rw [← Nat.mod_add_div' n k]
  exact (Ladder.per_iter hf _ _).symm

/-- The ladder problem of a point of `P(prism, HA)`. -/
noncomputable def toLadder (HA : HAP (V := Fin k × Bool) 3) (x : K (Fin k × Bool) → Fin 3 → ℝ)
    (hx : InP (prism k hk) HA x) : Ladder k where
  P n b d := decide (vtx hk n b ∈ HA.H d)
  xa n d := xe x (vtx hk n false) (vtx hk (n + 1) false) d
  xb n d := xe x (vtx hk n true) (vtx hk (n + 1) true) d
  xr n d := xe x (vtx hk n false) (vtx hk n true) d
  P_per n b d := by simp only [vtx_add]
  xa_per n d := by simp only [show n + k + 1 = n + 1 + k by omega, vtx_add]
  xb_per n d := by simp only [show n + k + 1 = n + 1 + k by omega, vtx_add]
  xa_nn n d := hx.xe_nn (adj_succ hk n false) d
  xb_nn n d := hx.xe_nn (adj_succ hk n true) d
  xr_nn n d := hx.xe_nn (adj_rung hk n) d
  xa_sum n := by rw [← Fin.sum_univ_three]; exact hx.xe_sum (adj_succ hk n false)
  xb_sum n := by rw [← Fin.sum_univ_three]; exact hx.xe_sum (adj_succ hk n true)
  xr_sum n := by rw [← Fin.sum_univ_three]; exact hx.xe_sum (adj_rung hk n)
  va n d := by
    have h := hx.xe_vert (vtx hk (n + 1) false) d
    rw [sum_ne_nbr hk n false _ (fun c hc => hx.xe_zero hc d)] at h
    rw [xe_symm x (vtx hk (n + 1) false) (vtx hk n false)] at h
    have e : (!false) = true := rfl
    rw [e] at h
    linarith
  vb n d := by
    have h := hx.xe_vert (vtx hk (n + 1) true) d
    rw [sum_ne_nbr hk n true _ (fun c hc => hx.xe_zero hc d)] at h
    rw [xe_symm x (vtx hk (n + 1) true) (vtx hk n true)] at h
    have e : (!true) = false := rfl
    rw [e, xe_symm x (vtx hk (n + 1) true) (vtx hk (n + 1) false)] at h
    linarith
  sa n d h := fun hs => h (hx.xe_side (adj_succ hk n false) d hs)
  sb n d h := fun hs => h (hx.xe_side (adj_succ hk n true) d hs)
  sr n d h := fun hs => h (hx.xe_side (adj_rung hk n) d hs)

/-- The colour of the side edges of a colouring. -/
def sideC {L : Ladder k} (C : Coloring L) (b : Bool) (n : ℕ) : Fin 3 :=
  if b then C.cb n else C.ca n

/-- The colour of an edge `{a, b}` of the prism. -/
def colOf {L : Ladder k} (C : Coloring L) (a b : Fin k × Bool) : Fin 3 :=
  if a.2 = b.2 then
    (if b.1.val = (a.1.val + 1) % k then sideC C a.2 a.1.val else sideC C a.2 b.1.val)
  else C.cr a.1.val

lemma not_both (i j : Fin k) (h1 : j.val = (i.val + 1) % k) (h2 : i.val = (j.val + 1) % k) :
    False := by
  apply mod_two_ne hk i.val
  rw [Nat.mod_eq_of_lt i.isLt]
  have : (i.val + 2) % k = ((i.val + 1) % k + 1) % k := by rw [Nat.mod_add_mod]
  rw [this, ← h1, ← h2]

lemma colOf_symm {L : Ladder k} (C : Coloring L) {a b : Fin k × Bool}
    (h : (prism k hk).Adj a b) : colOf C a b = colOf C b a := by
  obtain ⟨⟨i, hi⟩, s⟩ := a
  obtain ⟨⟨j, hj⟩, t⟩ := b
  rcases h with ⟨h1, h2 | h2⟩ | ⟨h1, h2⟩
  · simp only at h1 h2; subst h1
    have h3 : ¬ i = (j + 1) % k := fun h3 => not_both hk ⟨i, hi⟩ ⟨j, hj⟩ h2 h3
    simp only [colOf, ite_true]
    rw [ite_eq_left h2, ite_eq_right h3]
  · simp only at h1 h2; subst h1
    have h3 : ¬ j = (i + 1) % k := fun h3 => not_both hk ⟨i, hi⟩ ⟨j, hj⟩ h3 h2
    simp only [colOf, ite_true]
    rw [ite_eq_left h2, ite_eq_right h3]
  · simp only [Fin.mk.injEq] at h1 h2; subst h1
    simp [colOf, h2, Ne.symm h2]

/-- The integral point of a colouring. -/
def xOf {L : Ladder k} (C : Coloring L) (e : K (Fin k × Bool)) (d : Fin 3) : ℝ :=
  Sym2.lift ⟨fun a b => if (prism k hk).Adj a b ∧ colOf C a b = d then 1 else 0, fun a b => by
    by_cases h : (prism k hk).Adj a b
    · dsimp only; rw [colOf_symm hk C h]; simp [h, h.symm]
    · have h' : ¬ (prism k hk).Adj b a := fun h' => h h'.symm
      simp [h, h']⟩ e.1

lemma xOf_pair {L : Ladder k} (C : Coloring L) (a b : Fin k × Bool) (h : a ≠ b) (d : Fin 3) :
    xOf hk C (pair a b h) d = if (prism k hk).Adj a b ∧ colOf C a b = d then 1 else 0 := rfl

lemma colOf_succ {L : Ladder k} (C : Coloring L) (m : ℕ) (b : Bool) :
    colOf C (vtx hk (m + 1) b) (vtx hk (m + 2) b) = sideC C b (m + 1) := by
  have e : (m + 2) % k = ((m + 1) % k + 1) % k := by rw [Nat.mod_add_mod]
  simp only [colOf, vtx, e, ite_true]
  cases b <;> simp [sideC, per_mod C.ca_per, per_mod C.cb_per]

lemma colOf_pred {L : Ladder k} (C : Coloring L) (m : ℕ) (b : Bool) :
    colOf C (vtx hk (m + 1) b) (vtx hk m b) = sideC C b m := by
  have e : ¬ m % k = ((m + 1) % k + 1) % k := by
    rw [Nat.mod_add_mod]; exact fun h => mod_two_ne hk m h.symm
  simp only [colOf, vtx, e, ite_true, ite_false]
  cases b <;> simp [sideC, per_mod C.ca_per, per_mod C.cb_per]

lemma colOf_rung {L : Ladder k} (C : Coloring L) (m : ℕ) (b : Bool) :
    colOf C (vtx hk (m + 1) b) (vtx hk (m + 1) (!b)) = C.cr (m + 1) := by
  cases b <;> simp [colOf, vtx, per_mod C.cr_per]

/-- The polytope theorem: `P(prism, HA)` contains an integral point whenever it is non-empty
(the Corollary 9 form of Conjecture 17). -/
theorem prism_polytope : PolytopeIntegral (prism k hk) 3 := by
  rintro HA ⟨x, hx⟩
  set L := toLadder hk HA x hx
  obtain ⟨C⟩ := L.coloring (by omega)
  refine ⟨xOf hk C, ⟨?_, ?_, ?_, ?_, ?_⟩, ?_⟩
  · rintro ⟨e, he⟩ hE
    induction e using Sym2.ind with
    | h a b =>
      have hab : a ≠ b := fun h => he (Sym2.mk_isDiag_iff.mpr h)
      have hadj : (prism k hk).Adj a b := hE
      show ∑ d, xOf hk C (pair a b hab) d = 1
      simp only [xOf_pair, hadj, true_and]
      rw [Finset.sum_ite_eq]; simp
  · rintro ⟨e, he⟩ hE d
    induction e using Sym2.ind with
    | h a b =>
      have hab : a ≠ b := fun h => he (Sym2.mk_isDiag_iff.mpr h)
      have hadj : ¬ (prism k hk).Adj a b := hE
      show xOf hk C (pair a b hab) d = 0
      simp [xOf_pair, hadj]
  · intro d a
    obtain ⟨m, b, rfl⟩ := vtx_surj hk a
    have := sum_ne_nbr hk m b (fun c => if (prism k hk).Adj (vtx hk (m + 1) b) c ∧
      colOf C (vtx hk (m + 1) b) c = d then (1 : ℝ) else 0) (fun c hc => by simp [hc])
    rw [show (∑ c : {c // c ≠ vtx hk (m + 1) b}, xOf hk C (pair (vtx hk (m + 1) b) c.1 c.2.symm) d)
        = ∑ c : {c // c ≠ vtx hk (m + 1) b}, (if (prism k hk).Adj (vtx hk (m + 1) b) c.1 ∧
          colOf C (vtx hk (m + 1) b) c.1 = d then (1 : ℝ) else 0) from rfl, this]
    have a1 : (prism k hk).Adj (vtx hk (m + 1) b) (vtx hk (m + 2) b) := by
      rw [adj_iff]; left; rfl
    have a2 : (prism k hk).Adj (vtx hk (m + 1) b) (vtx hk m b) := by
      rw [adj_iff]; right; left; rfl
    have a3 : (prism k hk).Adj (vtx hk (m + 1) b) (vtx hk (m + 1) (!b)) := by
      rw [adj_iff]; right; right; rfl
    simp only [a1, a2, a3, true_and, colOf_succ, colOf_pred, colOf_rung]
    cases b
    · obtain ⟨p1, p2, p3⟩ := C.pa m
      have := three_ind (C.ca (m + 1)) (C.ca m) (C.cr (m + 1)) d (Ne.symm p1) p3 p2
      simpa [sideC] using this
    · obtain ⟨p1, p2, p3⟩ := C.pb m
      have := three_ind (C.cb (m + 1)) (C.cb m) (C.cr (m + 1)) d (Ne.symm p1) p3 p2
      simpa [sideC] using this
  · rintro d ⟨e, he⟩ -
    induction e using Sym2.ind with
    | h a b =>
      have hab : a ≠ b := fun h => he (Sym2.mk_isDiag_iff.mpr h)
      show 0 ≤ xOf hk C (pair a b hab) d ∧ xOf hk C (pair a b hab) d ≤ 1
      rw [xOf_pair]; split_ifs <;> norm_num
  · intro d a b hab hE hs
    rw [xOf_pair]
    have hadj : (prism k hk).Adj a b := hE
    rw [sameSide_iff (HA.eq d)] at hs
    simp only [hadj, true_and, ite_eq_right_iff, one_ne_zero, imp_false]
    intro hc
    obtain ⟨m, b0, rfl⟩ := vtx_surj hk a
    rcases (adj_iff hk m b0 b).mp hadj with rfl | rfl | rfl
    · rw [colOf_succ] at hc
      cases b0
      · have hc' : C.ca (m + 1) = d := hc
        subst hc'; exact C.oka (m + 1) hs
      · have hc' : C.cb (m + 1) = d := hc
        subst hc'; exact C.okb (m + 1) hs
    · rw [colOf_pred] at hc
      cases b0
      · have hc' : C.ca m = d := hc
        subst hc'; exact C.oka m hs.symm
      · have hc' : C.cb m = d := hc
        subst hc'; exact C.okb m hs.symm
    · rw [colOf_rung] at hc
      subst hc
      cases b0
      · exact C.okr (m + 1) hs
      · exact C.okr (m + 1) hs.symm
  · rintro ⟨e, he⟩ d
    induction e using Sym2.ind with
    | h a b =>
      have hab : a ≠ b := fun h => he (Sym2.mk_isDiag_iff.mpr h)
      show ∃ z : ℤ, xOf hk C (pair a b hab) d = z
      rw [xOf_pair]; split_ifs
      · exact ⟨1, by simp⟩
      · exact ⟨0, by simp⟩

  let cd : Fin r → EqPart V := fun d => ⟨{HA.H d, HA.A d}, HA.eq d⟩
  have hy01 : ∀ e d, y e d = 0 ∨ y e d = 1 := by
    intro e d
    by_cases he : e.1 ∈ G.edgeSet
    · obtain ⟨n, hn⟩ := hyI e d
      obtain ⟨h0, h1⟩ := hyP.2.2.2.1 d e he
      rw [hn] at h0 h1 ⊢

/-- The bridge: the Corollary 9 property for `r` days implies Definition 4. -/
theorem bfact_of_polytope (G : SimpleGraph V) (r : ℕ) (hne : Nonempty V)
    (hreg : ∀ a, ∑ b : {b : V // b ≠ a}, chiE G (pair a b.1 b.2.symm) = r)
    (hP : PolytopeIntegral G r) : BFactorizable G := by
  intro v ⟨m, hm1, hmv⟩ hvE
  have hvE' : ∀ e, v (Sum.inl e) = chiE G e := fun e => congrFun hvE e
  obtain ⟨z, Z1, Z2, Z3⟩ := zrep_of_mem hmv
  simp only [Pi.smul_apply, smul_eq_mul] at Z1 Z2
  have hm0 : (0 : ℝ) < m := by exact_mod_cast hm1
  -- the HA part has total weight `r`
  have hsumC : ∑ c, v (Sum.inr c) = r := by
    obtain ⟨a₀⟩ := hne
    have h1 : m * ∑ c, v (Sum.inr c) = m * r := by
      calc m * ∑ c, v (Sum.inr c) = ∑ c, m * v (Sum.inr c) := Finset.mul_sum _ _ _
        _ = ∑ c, ∑ b : {b : V // b ≠ a₀}, z c (pair a₀ b.1 b.2.symm) := by simp only [Z2]
        _ = ∑ b : {b : V // b ≠ a₀}, ∑ c, z c (pair a₀ b.1 b.2.symm) := Finset.sum_comm
        _ = ∑ b : {b : V // b ≠ a₀}, m * chiE G (pair a₀ b.1 b.2.symm) := by simp only [Z1, hvE']
        _ = m * r := by rw [← Finset.mul_sum, hreg a₀]
    exact Nat.eq_of_mul_eq_mul_left (by omega) h1
  -- `z c = 0` when `v c = 0`
  have hz0 : ∀ c e, v (Sum.inr c) = 0 → z c e = 0 := by
    intro c e hc
    obtain ⟨a, b, h, rfl⟩ := K_pair e
    have := Z2 c a
    rw [hc, Nat.mul_zero, Finset.sum_eq_zero_iff] at this
    exact this ⟨b, Ne.symm h⟩ (Finset.mem_univ _)
  obtain ⟨f, hfR, hfN⟩ := days (fun c => v (Sum.inr c)) hsumC
  have hvpos : ∀ d, 0 < v (Sum.inr (f d)) := by
    intro d
    have := hfN (fun c => if c = f d then 1 else 0)
    simp only [mul_ite, mul_one, mul_zero, Finset.sum_ite_eq', Finset.mem_univ, ite_true] at this
    rw [← this]
    exact Finset.sum_pos' (fun _ _ => Nat.zero_le _) ⟨d, Finset.mem_univ _, by simp⟩
  -- the HAP set
  have hpart : ∀ d, ∃ H A : Finset V, Disjoint H A ∧ H ∪ A = Finset.univ ∧ H.card = A.card ∧
      (f d).1 = {H, A} := fun d => (f d).2
  choose H A hd hu hcard hc using hpart
  let HA : HAP (V := V) r := ⟨H, A, fun d => ⟨H d, A d, hd d, hu d, hcard d, rfl⟩⟩
  -- the point of `P(G, HA)`
  let x : K V → Fin r → ℝ := fun e d => (z (f d) e : ℝ) * (1 / (m * v (Sum.inr (f d))))
  have hx1 : ∀ e, ∑ d, x e d = v (Sum.inl e) := by
    intro e
    rw [hfR (fun c => (z c e : ℝ) * (1 / (m * v (Sum.inr c))))]
    have : ∀ c, (v (Sum.inr c) : ℝ) * ((z c e : ℝ) * (1 / (m * v (Sum.inr c)))) =
        (z c e : ℝ) * (1 / m) := by
      intro c
      by_cases hc : v (Sum.inr c) = 0
      · simp [hc, hz0 c e hc]
      · have : (v (Sum.inr c) : ℝ) ≠ 0 := by exact_mod_cast hc
        calc (v (Sum.inr c) : ℝ) * ((z c e : ℝ) * (1 / (m * v (Sum.inr c))))
            = (z c e : ℝ) * (1 / m) * ((v (Sum.inr c) : ℝ) / v (Sum.inr c)) := by ring
          _ = (z c e : ℝ) * (1 / m) := by rw [div_self this, mul_one]
    simp only [this, ← Finset.sum_mul]
    rw [← Nat.cast_sum, Z1 e]
    push_cast
    field_simp
  have hxnn : ∀ e d, 0 ≤ x e d := fun e d => by
    apply mul_nonneg (Nat.cast_nonneg _); positivity
  have hxP : InP G HA x := by
    refine ⟨fun e he => ?_, fun e he d => ?_, fun d a => ?_, fun d e he => ?_,
      fun d a b h hE hs => ?_⟩
    · rw [hx1, hvE']; simp [chiE, he]
    · have h0 : v (Sum.inl e) = 0 := by rw [hvE']; simp [chiE, he]
      have := Z1 e
      rw [h0, Nat.mul_zero, Finset.sum_eq_zero_iff] at this
      simp [x, this (f d) (Finset.mem_univ _)]
    · simp only [x, ← Finset.sum_mul]
      rw [← Nat.cast_sum, Z2]
      have : (v (Sum.inr (f d)) : ℝ) ≠ 0 := by exact_mod_cast (hvpos d).ne'
      push_cast
      field_simp
    · refine ⟨hxnn e d, ?_⟩
      have h1 := hx1 e
      rw [hvE'] at h1
      simp only [chiE, he, ite_true, Nat.cast_one] at h1
      rw [← h1]
      exact Finset.single_le_sum (fun i _ => hxnn e i) (Finset.mem_univ d)
    · have hn : ¬ (Bc (f d)).Adj a b := by
        rw [Bc_adj_iff (hd d) (hu d) (hc d)]; exact not_not.mpr hs
      have : z (f d) (pair a b h) = 0 := by
        by_contra h'; exact hn (Z3 _ a b h h')
      simp [x, this]
  -- an integral point
  obtain ⟨y, hyP, hyI⟩ := hP HA ⟨x, hxP⟩
  obtain ⟨w, hwN, hwK, hwC⟩ := lemma8_fwd G HA y hyP hyI hd hu
  have hfd : ∀ d, (⟨{HA.H d, HA.A d}, HA.eq d⟩ : EqPart V) = f d :=
    fun d => Subtype.ext (hc d).symm
  have hw : w = v := by
    funext i
    rcases i with e | c
    · rw [hwK, hvE']
    · rw [hwC]
      simp only [hfd]
      rw [hfN (fun c' => if c = c' then 1 else 0)]
      simp
  rw [← hw]; exact hwN
