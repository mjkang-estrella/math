A.3. Numerical reproduction
   The following Python script reproduces the numerical eigenvalue and signed distance reported in Theorem 5.

import numpy as np, mpmath as mp
n = 7
A = np.zeros((n, n))                                             # vertices 0..5 = cycle, 6 = twin of 0
for i in range(5): A[i, i+1] = 1
A[5, 0] = A[5, 6] = 1; A[6, 1] = 1
L = (np.diag(A.sum(1)) - A) / n
print(np.linalg.eigvals(L))

mp.mp.dps = 50
roots, err = mp.polyroots([1, -7, 20, -30, 25, -11], error=True,
                          maxsteps=200, extraprec=200)
y = [r for r in roots if 0 < mp.re(r) < 1 and mp.im(r) > 0][0]
lam = y / 7
mu = mp.exp(-2j*mp.pi/7); l1 = (1-mu)/7; l2 = l1 + (1-mu**2)/7
e = l2 - l1; nr = mp.mpc(-e.imag, e.real)
print(lam, err, mp.re((lam-l1)*mp.conj(nr))/abs(nr))   # > 0: outside S_7
