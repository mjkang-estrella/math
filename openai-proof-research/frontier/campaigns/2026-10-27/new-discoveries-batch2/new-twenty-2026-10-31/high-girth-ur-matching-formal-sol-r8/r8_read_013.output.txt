module
public import URMatchingSurvivorDegree
public import URMatchingSurvivingColours
@[expose] public section
namespace URMatching
open scoped Classical
def LeftSurvivor {N : ℕ} {h : ℕ} {σ : PermutationTuples N 100} (v : Survivor h σ) : Prop :=
  ∃ x, v.val = Sum.inl x

theorem survivor_bipartite {N h : ℕ} (σ : PermutationTuples N 100)
    (u v : Survivor h σ) (huv : (survivorGraph h σ).Adj u v) :
    (LeftSurvivor u ∧ ¬ LeftSurvivor v) ∨ (¬ LeftSurvivor u ∧ LeftSurvivor v) := by
  obtain ⟨u,hu⟩ := u
  obtain ⟨v,hv⟩ := v
  cases u <;> cases v <;> simp_all [LeftSurvivor, survivorGraph, inducedGraph, permutationGraph, permutationAdj]

def SurvivorNoHole {N : ℕ} (h a : ℕ) (σ : PermutationTuples N 100) : Prop :=
  ∀ A B : Finset (Fin N), A.card = a → B.card = a →
    (∀ x ∈ A, Sum.inl x ∉ removedVertices N h σ) →
    (∀ y ∈ B, Sum.inr y ∉ removedVertices N h σ) →
    ∃ u v : Survivor h σ, (∃ x ∈ A, u.val = Sum.inl x) ∧
      (∃ y ∈ B, v.val = Sum.inr y) ∧ (survivorGraph h σ).Adj u v

theorem induced_no_hole {N h a : ℕ} (σ : PermutationTuples N 100)
    (hh : ∀ A B : Finset (Fin N), A.card = a → B.card = a → ∃ c, ∃ x ∈ A, σ c x ∈ B) :
    SurvivorNoHole h a σ := by
  intro A B hA hB hleft hright
  obtain ⟨c,x,hx,hy⟩ := hh A B hA hB
  exact ⟨⟨Sum.inl x,hleft x hx⟩, ⟨Sum.inr (σ c x),hright _ hy⟩,
    ⟨x,hx,rfl⟩,⟨σ c x,hy,rfl⟩,c,rfl⟩

theorem exists_induced_survivor (h : ℕ) : ∃ a : ℕ, 0 < a ∧ h ≤ a ∧
    ∃ σ : PermutationTuples (4*a) 100,
      100*(removedVertices (4*a) h σ).card < 4*a ∧
      DegreeBound (survivorGraph h σ) 100 ∧ SurvivorNoHole h a σ := by
  obtain ⟨a,ha,hh,σ,hs,hn⟩ := exists_sparse_deletion_tuple h
  exact ⟨a,ha,hh,σ,hs,survivor_degree_bound h σ,induced_no_hole σ hn⟩
end URMatching

module
public import URMatchingBase
@[expose] public section
namespace URMatching
universe v
variable {V : Type v}
/-- The bipartition is an actual proper two-coloring of every graph edge. -/
def AcrossPartition (G : SimpleGraph V) (P : V → Prop) : Prop :=
  ∀ x y, G.Adj x y → (P x ↔ ¬ P y)

theorem partition_orients_edge (G : SimpleGraph V) (P : V → Prop)
    (bip : AcrossPartition G P) (a : Sym2 V) (he : a ∈ G.edgeSet) :
    ∃ x y, P x ∧ ¬ P y ∧ a = s(x,y) := by
  classical
  revert he
  refine Sym2.inductionOn a (fun x y hxy => ?_)
  have hadj : G.Adj x y := hxy
  by_cases hp : P x
  · exact ⟨x, y, hp, (bip x y hadj).mp hp, rfl⟩
  · have hpy : P y := by
      by_contra hny
      exact hp ((bip x y hadj).mpr hny)
    exact ⟨y, x, hpy, hp, by simp [Sym2.eq_swap]⟩
end URMatching

namespace URMatching
universe v
variable {V : Type v} [DecidableEq V]

/-- The directed-cycle/topological obstruction for ANY literal bipartite UR
matching, including partial matchings, odd size, and uncovered ambient vertices. -/
theorem ur_partition_empty_rectangle (G : SimpleGraph V) (P : V → Prop)
    (bip : AcrossPartition G P) (M : Finset (Sym2 V)) (ur : IsUR G M) :
    ∃ A B : Finset V, A.card = M.card / 2 ∧ B.card = M.card / 2 ∧
      (∀ x ∈ A, P x) ∧ (∀ y ∈ B, ¬ P y) ∧
      Disjoint A B ∧ ∀ x ∈ A, ∀ y ∈ B, ¬ G.Adj x y := by
  classical
  apply ur_oriented_empty_rectangle_sides G M P ur
  intro a ha
  exact partition_orients_edge G P bip a (ur.1.1 a ha)
end URMatching

FILE src/URMatchingFinite.lean
module
public import URMatchingBase
@[expose] public section
namespace URMatching
universe u
variable {V : Type u} (G : SimpleGraph V)

@[simp] theorem empty_isMatching : IsMatching G ∅ := by
  simp [IsMatching]

@[simp] theorem covered_empty : Covered (∅ : Finset (Sym2 V)) = ∅ := by
  ext v
  simp [Covered]

theorem eq_empty_of_covered_empty (M : Finset (Sym2 V)) (h : Covered M = ∅) : M = ∅ := by
  apply Finset.eq_empty_iff_forall_notMem.mpr
  intro e he
  obtain ⟨⟨x, y⟩, rfl⟩ := Sym2.mk_surjective e
  have hv : x ∈ Covered M := ⟨s(x, y), he, Sym2.mem_mk_left x y⟩
  rw [h] at hv
  exact hv

@[simp] theorem empty_isUR : IsUR G ∅ := by
  refine ⟨empty_isMatching G, ?_⟩
  intro M _ hc
  exact eq_empty_of_covered_empty M (hc.trans covered_empty)

variable [Fintype V]

theorem card_le_matchingNumber {M : Finset (Sym2 V)} (h : IsMatching G M) :
    M.card ≤ matchingNumber G := by
  classical
  exact Finset.le_sup (f := Finset.card) (Finset.mem_filter.mpr ⟨Finset.mem_univ _, h⟩)

theorem card_le_urMatchingNumber {M : Finset (Sym2 V)} (h : IsUR G M) :
    M.card ≤ urMatchingNumber G := by
  classical
  exact Finset.le_sup (f := Finset.card) (Finset.mem_filter.mpr ⟨Finset.mem_univ _, h⟩)

theorem exists_maximum_matching : ∃ M : Finset (Sym2 V),
    IsMatching G M ∧ M.card = matchingNumber G := by
  classical
  let S := (Finset.univ : Finset (Finset (Sym2 V))).filter (IsMatching G)
  have hs : S.Nonempty := ⟨∅, Finset.mem_filter.mpr ⟨Finset.mem_univ _, empty_isMatching G⟩⟩
  obtain ⟨M, hm, heq⟩ := Finset.exists_mem_eq_sup S hs Finset.card
  exact ⟨M, (Finset.mem_filter.mp hm).2, heq.symm⟩

theorem exists_maximum_urMatching : ∃ M : Finset (Sym2 V),
    IsUR G M ∧ M.card = urMatchingNumber G := by
  classical
  let S := (Finset.univ : Finset (Finset (Sym2 V))).filter (IsUR G)
  have hs : S.Nonempty := ⟨∅, Finset.mem_filter.mpr ⟨Finset.mem_univ _, empty_isUR G⟩⟩
  obtain ⟨M, hm, heq⟩ := Finset.exists_mem_eq_sup S hs Finset.card
  exact ⟨M, (Finset.mem_filter.mp hm).2, heq.symm⟩

theorem urMatchingNumber_le_matchingNumber : urMatchingNumber G ≤ matchingNumber G := by
  obtain ⟨M, hm, heq⟩ := exists_maximum_urMatching G
  rw [← heq]
  exact card_le_matchingNumber G hm.1

/-- This exact universal characterization verifies that the supremum ranges over all matchings. -/
theorem matchingNumber_le_iff (k : ℕ) :
    matchingNumber G ≤ k ↔ ∀ M, IsMatching G M → M.card ≤ k := by
  classical
  constructor
  · intro h M hm; exact (card_le_matchingNumber G hm).trans h
  · intro h
    exact Finset.sup_le fun M hm => h M (Finset.mem_filter.mp hm).2

theorem urMatchingNumber_le_iff (k : ℕ) :
    urMatchingNumber G ≤ k ↔ ∀ M, IsUR G M → M.card ≤ k := by
  classical
  constructor
  · intro h M hm; exact (card_le_urMatchingNumber G hm).trans h
  · intro h
    exact Finset.sup_le fun M hm => h M (Finset.mem_filter.mp hm).2
end URMatching

module
public import URMatchingURSides
@[expose] public section
namespace URMatching
universe v
variable {V : Type v} [DecidableEq V]

/-- Arbitrary literal edge-set matching, with each edge oriented across a fixed
vertex partition. The indexing and all endpoint disjointness are proved. -/
theorem ur_oriented_empty_rectangle_sides (G : SimpleGraph V) (M : Finset (Sym2 V))
    (P : V → Prop) (ur : IsUR G M)
    (orient : ∀ a ∈ M, ∃ x y, P x ∧ ¬ P y ∧ a = s(x,y)) :
    ∃ A B : Finset V, A.card = M.card / 2 ∧ B.card = M.card / 2 ∧
      (∀ x ∈ A, P x) ∧ (∀ y ∈ B, ¬ P y) ∧
      Disjoint A B ∧ ∀ x ∈ A, ∀ y ∈ B, ¬ G.Adj x y := by
  classical
  have ho : ∀ a : M, ∃ x y, P x ∧ ¬ P y ∧ a.val = s(x,y) :=
    fun a => orient a.val a.property
  choose x y hx hy he using ho
  let f : Sum M M → V := Sum.elim x y
  have hf : Function.Injective f := by
    intro a b hab
    cases a with
    | inl i =>
      cases b with
      | inl j =>
        have hi : x i ∈ i.val := by rw [he i]; simp
        have hj : x i ∈ j.val := by rw [he j]; simp [show x i = x j from hab]
        have hij := ur.1.2 i.val i.property j.val j.property (x i) hi hj
        exact congrArg Sum.inl (Subtype.ext hij)
      | inr j =>
        change x i = y j at hab
        exact False.elim (hy j (hab ▸ hx i))
    | inr i =>
      cases b with
      | inl j =>
        change y i = x j at hab
        exact False.elim (hy i (hab.symm ▸ hx j))
      | inr j =>
        have hi : y i ∈ i.val := by rw [he i]; simp
        have hj : y i ∈ j.val := by rw [he j]; simp [show y i = y j from hab]
        have hij := ur.1.2 i.val i.property j.val j.property (y i) hi hj
        exact congrArg Sum.inr (Subtype.ext hij)
  let e : Sum M M ↪ V := ⟨f, hf⟩
  have hedges : indexedEdges e (Equiv.refl M) = M := by
    ext a
    constructor
    · intro ha
      obtain ⟨i, _, rfl⟩ := Finset.mem_image.mp ha
      change s(x i, y i) ∈ M
      rw [← he i]
      exact i.property
    · intro ha
      exact Finset.mem_image.mpr ⟨⟨a,ha⟩, Finset.mem_univ _, (he ⟨a,ha⟩).symm⟩
  have hu : IsUR G (indexedEdges e (Equiv.refl M)) := hedges.symm ▸ ur
  obtain ⟨A, B, hA, hB, hleft, hright, hd, hempty⟩ := ur_indexed_empty_rectangle_sides G e hu
  refine ⟨A, B, by simpa using hA, by simpa using hB, ?_, ?_, hd, hempty⟩
  · intro w hw
    obtain ⟨i, rfl⟩ := hleft w hw
    exact hx i
  · intro w hw
    obtain ⟨i, rfl⟩ := hright w hw
    exact hy i
end URMatching
