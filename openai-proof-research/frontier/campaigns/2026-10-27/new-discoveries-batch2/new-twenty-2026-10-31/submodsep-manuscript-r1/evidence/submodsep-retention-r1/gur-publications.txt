const cards = [...document.querySelectorAll('.pub-card')];
const sections = [...document.querySelectorAll('.year-section')];
const topicButtons = [...document.querySelectorAll('.topic-filter')];
const q = document.querySelector('#q');
const year = document.querySelector('#year');
const type = document.querySelector('#type');
const count = document.querySelector('#count');
const empty = document.querySelector('#empty');
const bibModal = document.querySelector('#bibModal');
const bibText = document.querySelector('#bibText');
const copyBib = document.querySelector('#copyBib');
const closeBib = document.querySelector('#closeBib');
const abstractModal = document.querySelector('#abstractModal');
const abstractText = document.querySelector('#abstractText');
const abstractSource = document.querySelector('#abstractSource');
const closeAbstract = document.querySelector('#closeAbstract');
let activeTopic = '';

function applyFilters() {
  const query = q.value.trim().toLowerCase();
  const wantedYear = year.value;
  const wantedType = type.value;
  const wantedTopic = activeTopic;
  let visible = 0;
  cards.forEach(card => {
    const ok = (!query || card.dataset.search.includes(query)) &&
      (!wantedYear || card.dataset.year === wantedYear) &&
      (!wantedType || card.dataset.type === wantedType) &&
      (!wantedTopic || (card.dataset.topics || '').split(' ').includes(wantedTopic));
    card.hidden = !ok;
    if (ok) visible += 1;
  });
  sections.forEach(section => {
    const sectionCards = [...section.querySelectorAll('.pub-card')];
    section.hidden = sectionCards.every(card => card.hidden);
  });
  count.textContent = visible;
  empty.style.display = visible ? 'none' : 'block';
}
[q, year, type].forEach(control => control.addEventListener('input', applyFilters));
topicButtons.forEach(button => button.addEventListener('click', () => {
  activeTopic = button.dataset.topic || '';
  topicButtons.forEach(candidate => candidate.classList.toggle('is-active', candidate === button));
  applyFilters();
}));
applyFilters();

function openBibtex(text) {
  bibText.value = text;
  bibModal.hidden = false;
  requestAnimationFrame(() => {
    bibText.focus();
    bibText.select();
  });
}

function closeBibtex() {
  bibModal.hidden = true;
  bibText.value = '';
  copyBib.textContent = 'Copy';
}

function openAbstract(text, source) {
  abstractText.textContent = text;
  abstractSource.textContent = source ? `Source: ${source}` : '';
  abstractModal.hidden = false;
  closeAbstract.focus();
}

function closeAbstractModal() {
  abstractModal.hidden = true;
  abstractText.textContent = '';
  abstractSource.textContent = '';
}

document.addEventListener('click', event => {
  const button = event.target.closest('.bib-link');
  if (button) {
    openBibtex(button.dataset.bibtex || '');
    return;
  }
  const abstractButton = event.target.closest('.abstract-link');
  if (abstractButton) {
    openAbstract(abstractButton.dataset.abstract || '', abstractButton.dataset.source || '');
    return;
  }
  if (event.target === bibModal) closeBibtex();
  if (event.target === abstractModal) closeAbstractModal();
});

copyBib.addEventListener('click', async () => {
  bibText.select();
  try {
    await navigator.clipboard.writeText(bibText.value);
    copyBib.textContent = 'Copied';
  } catch {
    document.execCommand('copy');
    copyBib.textContent = 'Copied';
  }
});
closeBib.addEventListener('click', closeBibtex);
closeAbstract.addEventListener('click', closeAbstractModal);
document.addEventListener('keydown', event => {
  if (event.key === 'Escape' && !bibModal.hidden) closeBibtex();
  if (event.key === 'Escape' && !abstractModal.hidden) closeAbstractModal();
});
