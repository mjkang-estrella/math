public import URMatchingPrescriptions
@[expose] public section
namespace URMatching
open scoped Classical
/-- Ordered cycle descriptions with two distinct vertex lists and arbitrary colours.
Index k encodes half-length k+1; at k=0 valid descriptions require distinct colours. -/
abbrev CycleDescription (N k : ℕ) :=
  (Fin (k+1) ↪ Fin N) × (Fin (k+1) ↪ Fin N) ×
    (Fin (k+1) → Fin 100) × (Fin (k+1) → Fin 100)

def ValidDescription {N k : ℕ} (w : CycleDescription N k) : Prop :=
  k = 0 → w.2.2.1 0 ≠ w.2.2.2 0

/-- The two edges at each Y vertex, with cyclically consecutive X vertices. -/
def DescriptionOccurs {N k : ℕ} (w : CycleDescription N k)
    (σ : PermutationTuples N 100) : Prop :=
  ∀ i : Fin (k+1), σ (w.2.2.1 i) (w.1 i) = w.2.1 i ∧
    σ (w.2.2.2 i) (w.1 (i+1)) = w.2.1 i

noncomputable def descriptionEncoding (N k : ℕ) : CycleDescription N k →
    (Fin (k+1) → Fin N) × (Fin (k+1) → Fin N) ×
      (Fin (k+1) → Fin 100) × (Fin (k+1) → Fin 100) :=
  fun w => (w.1, w.2.1, w.2.2.1, w.2.2.2)

theorem descriptionEncoding_injective (N k : ℕ) :
    Function.Injective (descriptionEncoding N k) := by
  intro a b h
  obtain ⟨a₁,a₂,a₃,a₄⟩ := a
  obtain ⟨b₁,b₂,b₃,b₄⟩ := b
  simp only [descriptionEncoding, Prod.mk.injEq] at h
  obtain ⟨h₁,h₂,h₃,h₄⟩ := h
  have e₁ : a₁ = b₁ := DFunLike.coe_injective h₁
  have e₂ : a₂ = b₂ := DFunLike.coe_injective h₂
  exact e₁ ▸ e₂ ▸ h₃ ▸ h₄ ▸ rfl

/-- Ordered witness enumeration bound. The validity restriction only reduces it. -/
theorem description_count_bound (N k : ℕ) :
    Fintype.card {w : CycleDescription N k // ValidDescription w} ≤
      N ^ (k+1) * (N ^ (k+1) * (100 ^ (k+1) * 100 ^ (k+1))) := by
  classical
  calc
    _ ≤ Fintype.card (CycleDescription N k) := Fintype.card_subtype_le _
    _ ≤ Fintype.card ((Fin (k+1) → Fin N) × (Fin (k+1) → Fin N) ×
        (Fin (k+1) → Fin 100) × (Fin (k+1) → Fin 100)) :=
      Fintype.card_le_of_injective _ (descriptionEncoding_injective N k)
    _ = _ := by simp
end URMatching

module
public import URMatchingRemovedVertices
public import Mathlib.Combinatorics.SimpleGraph.Basic
@[expose] public section
namespace URMatching
open scoped Classical
def permutationAdj {N : ℕ} (σ : PermutationTuples N 100) : BipartiteVertex N → BipartiteVertex N → Prop
  | .inl x, .inr y => ∃ c, σ c x = y
  | .inr y, .inl x => ∃ c, σ c x = y
  | _, _ => False

def permutationGraph {N : ℕ} (σ : PermutationTuples N 100) : SimpleGraph (BipartiteVertex N) where
  Adj := permutationAdj σ
  symm := ⟨by intro u v; cases u <;> cases v <;> exact fun h => h⟩
  loopless := ⟨by intro u; cases u <;> exact fun h => h⟩

abbrev Survivor {N : ℕ} (h : ℕ) (σ : PermutationTuples N 100) :=
  {v : BipartiteVertex N // v ∉ removedVertices N h σ}

def inducedGraph {V : Type*} (G : SimpleGraph V) (P : Set V) : SimpleGraph P where
  Adj u v := G.Adj u.val v.val
  symm := ⟨fun _ _ h => G.adj_symm h⟩
  loopless := ⟨fun u => G.loopless.irrefl u.val⟩

def survivorGraph {N : ℕ} (h : ℕ) (σ : PermutationTuples N 100) : SimpleGraph (Survivor h σ) :=
  inducedGraph (permutationGraph σ) {v | v ∉ removedVertices N h σ}
end URMatching

module
public import URMatchingJointTuple
@[expose] public section
namespace URMatching
open scoped Classical
abbrev BipartiteVertex (N : ℕ) := Sum (Fin N) (Fin N)
noncomputable def witnessVertices {N k : ℕ} (w : CycleDescription N k) : Finset (BipartiteVertex N) :=
  Finset.univ.image (fun i => Sum.inl (w.1 i)) ∪ Finset.univ.image (fun i => Sum.inr (w.2.1 i))

theorem witness_vertices_bound {N k : ℕ} (w : CycleDescription N k) :
    (witnessVertices w).card ≤ 2*(k+1) := by
  have hx := Finset.card_image_le (s := (Finset.univ : Finset (Fin (k+1)))) (f := fun i => (Sum.inl (w.1 i) : BipartiteVertex N))
  have hy := Finset.card_image_le (s := (Finset.univ : Finset (Fin (k+1)))) (f := fun i => (Sum.inr (w.2.1 i) : BipartiteVertex N))
  have hu := Finset.card_union_le (Finset.univ.image (fun i => (Sum.inl (w.1 i) : BipartiteVertex N)))
    (Finset.univ.image (fun i => (Sum.inr (w.2.1 i) : BipartiteVertex N)))
  simp only [Finset.card_univ, Fintype.card_fin] at hx hy
  unfold witnessVertices
  omega

noncomputable def occurringWitnesses (N k : ℕ) (σ : PermutationTuples N 100) : Finset (ValidWitness N k) :=
  Finset.univ.filter fun w => DescriptionOccurs w.val σ
noncomputable def removedAt (N k : ℕ) (σ : PermutationTuples N 100) : Finset (BipartiteVertex N) :=
  (occurringWitnesses N k σ).biUnion fun w => witnessVertices w.val
noncomputable def removedVertices (N h : ℕ) (σ : PermutationTuples N 100) : Finset (BipartiteVertex N) :=
  (Finset.range h).biUnion fun k => removedAt N k σ

theorem removedAt_bound (N k : ℕ) (σ : PermutationTuples N 100) :
    (removedAt N k σ).card ≤ 2*(k+1)*witnessCount N k σ := by
  classical
  have he : (occurringWitnesses N k σ).card = witnessCount N k σ := by
    simp only [occurringWitnesses, witnessCount, Fintype.card_subtype]
  have hu := scaled_union_card (occurringWitnesses N k σ)
    (fun w => witnessVertices w.val) 1 (2*(k+1)) (fun w _ => by simpa using witness_vertices_bound w.val)
  simpa only [removedAt, Nat.one_mul, he, Nat.mul_comm] using hu

theorem removed_vertices_bound (N h : ℕ) (σ : PermutationTuples N 100) :
    (removedVertices N h σ).card ≤ witnessWeight N h σ := by
  classical
  exact Finset.card_biUnion_le.trans
    (nat_sum_mono (Finset.range h) _ _ (fun k _ => removedAt_bound N k σ))

theorem witness_vertex_removed {N h k : ℕ} (σ : PermutationTuples N 100)
    (w : ValidWitness N k) (hk : k < h) (hw : DescriptionOccurs w.val σ)
    (v : BipartiteVertex N) (hv : v ∈ witnessVertices w.val) :
    v ∈ removedVertices N h σ := by
  classical
  exact Finset.mem_biUnion.mpr ⟨k, Finset.mem_range.mpr hk,
    Finset.mem_biUnion.mpr ⟨w, Finset.mem_filter.mpr ⟨Finset.mem_univ _,hw⟩,hv⟩⟩
end URMatching
