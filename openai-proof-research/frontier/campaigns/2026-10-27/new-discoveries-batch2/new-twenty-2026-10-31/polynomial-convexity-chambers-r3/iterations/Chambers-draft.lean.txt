/-- Ties concern coordinate labels, irrespective of edge coincidences. -/
theorem volume_coordinate_ne {d : ℕ} (i j : Fin d) (hij : i ≠ j) :
    ∀ᵐ t : Fin d → ℝ ∂volume, t i ≠ t j := by
  cases d with
  | zero => exact Fin.elim0 i
  | succ d =>
    obtain ⟨k, hk⟩ := i.exists_succAbove_eq (Ne.symm hij)
    subst j
    have h : ∀ᵐ p : ℝ × (Fin d → ℝ) ∂(volume.prod volume), p.1 ≠ p.2 k := by
      apply (ae_prod_iff_ae_ae (measurableSet_eq_fun measurable_fst
        ((measurable_pi_apply k).comp measurable_snd)).compl).mpr
      exact Filter.Eventually.of_forall fun x => by
        rw [volume_pi]
        exact (Measure.ae_eval_ne (fun _ : Fin d => (volume : Measure ℝ)) k x).mono
          (fun _ ht => Ne.symm ht)
    have hp := (volume_preserving_piFinSuccAbove (fun _ : Fin (d+1) => ℝ) i).quasiMeasurePreserving.ae h
    simpa using hp

theorem volume_coordinates_injective (d : ℕ) :
    ∀ᵐ t : Fin d → ℝ ∂volume, Function.Injective t := by
  have h : ∀ᵐ t : Fin d → ℝ ∂volume, ∀ i j, i ≠ j → t i ≠ t j := by
    simp only [ae_all_iff]
    exact fun i j hij => volume_coordinate_ne i j hij
  exact h.mono fun _ ht i j heq => by
    by_contra hij
    exact ht i j hij heq

/-- Strict descending labelled chamber inside the positive orthant. -/
def chamber {d : ℕ} (σ : Equiv.Perm (Fin d)) : Set (Fin d → ℝ) :=
  {t | (∀ i, 0 < t i) ∧ StrictAnti (t ∘ σ)}

theorem chamber_measurable {d : ℕ} (σ : Equiv.Perm (Fin d)) :
    MeasurableSet (chamber σ) := by
  have heq : chamber σ =
      (⋂ i, {t : Fin d → ℝ | 0 < t i}) ∩
      (⋂ i, ⋂ j, ⋂ (_ : i < j), {t : Fin d → ℝ | t (σ j) < t (σ i)}) := by
    ext t
    simp [chamber, StrictAnti]
  rw [heq]
  exact (MeasurableSet.iInter fun i => measurableSet_lt measurable_const (measurable_pi_apply i)).inter
    (MeasurableSet.iInter fun i => MeasurableSet.iInter fun j => MeasurableSet.iInter fun _ =>
      measurableSet_lt (measurable_pi_apply (σ j)) (measurable_pi_apply (σ i)))

theorem chamber_unique {d : ℕ} {t : Fin d → ℝ} {σ τ : Equiv.Perm (Fin d)}
    (hσ : t ∈ chamber σ) (hτ : t ∈ chamber τ) : σ = τ := by
  have ht : Function.Injective t := by
    have h : Function.Injective (t ∘ σ) := hσ.2.injective
    exact (h.comp σ.symm.injective : Function.Injective ((t ∘ σ) ∘ σ.symm))
  have heq := Tuple.unique_antitone hσ.2.antitone hτ.2.antitone
  ext i
  exact ht (congrFun heq i)

theorem chamber_cover {d : ℕ} {t : Fin d → ℝ} (ht : Function.Injective t)
    (hpos : ∀ i, 0 < t i) : ∃ σ, t ∈ chamber σ := by
  let σ := Tuple.sort (fun i => OrderDual.toDual (t i))
  have ha : Antitone (t ∘ σ) := Tuple.monotone_sort (fun i => OrderDual.toDual (t i))
  exact ⟨σ, hpos, ha.strictAnti_of_injective (ht.comp σ.injective)⟩

end
end PolynomialConvexity
