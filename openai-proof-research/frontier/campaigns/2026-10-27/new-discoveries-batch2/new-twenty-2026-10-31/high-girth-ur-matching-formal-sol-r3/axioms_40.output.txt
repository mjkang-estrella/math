@[expose] def URMatching.IsMatching.{u} : {V : Type u} → SimpleGraph V → Finset (Sym2 V) → Prop :=
fun {V} G M => (∀ e ∈ M, e ∈ G.edgeSet) ∧ ∀ e ∈ M, ∀ f ∈ M, ∀ v ∈ e, v ∈ f → e = f
@[expose] def URMatching.IsUR.{u} : {V : Type u} → SimpleGraph V → Finset (Sym2 V) → Prop :=
fun {V} G M =>
  URMatching.IsMatching G M ∧
    ∀ (M' : Finset (Sym2 V)), URMatching.IsMatching G M' → URMatching.Covered M' = URMatching.Covered M → M' = M
