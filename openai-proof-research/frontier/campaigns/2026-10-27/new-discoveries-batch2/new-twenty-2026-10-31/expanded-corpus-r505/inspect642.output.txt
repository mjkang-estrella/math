Let $f(n)$ be the maximal number of edges in a graph on $n$ vertices such that all cycles have more vertices than chords. Is it true that $f(n)\ll n$?
     

     
 #642 :  [CES96]  [Er97d,p.84] 
     

 
     

 graph theory  | 

 cycles 

     

   

  

A problem of Hamburger and Szegedy. A chord is an edge between two vertices of the cycle which are not consecutive in the cycle.  Chen, Erdős, and Staton  [CES96]  proved $f(n) \ll n^{3/2}$. Draganić, Methuku, Munhá Correia, and Sudakov  [DMMS24]  have improved this to\[f(n) \ll n(\log n)^8.\]

document.addEventListener("DOMContentLoaded", function () {
  function escapeHtml(str) {
    return String(str).replace(/[&<>"']/g, m => ({
      "&": "&",
      " ": ">",
      '"': """,
      "'": "'"
    })[m]);
  }

  function normalizeHref(href) {
    href = href.trim();
    if (!/^https?:\/\//i.test(href) && !href.startsWith("/")) {
      href = "https://" + href;
    }

    return href;
  }

  let mathJaxQueue = Promise.resolve();

  function clearMathJax(element) {
    if (window.MathJax && MathJax.typesetClear) {
      MathJax.typesetClear([element]);
    }
  }

  function typesetMath(element) {
    if (!window.MathJax) return;

    if (MathJax.typesetPromise) {
      mathJaxQueue = mathJaxQueue
        .then(() => {
          if (MathJax.startup && MathJax.startup.promise) {
            return MathJax.startup.promise;
          }
        })
        .then(() => MathJax.typesetPromise([element]))
        .catch(err => {
          console.error("MathJax typeset failed:", err);
        });

      return;
    }

    if (MathJax.Hub) {
      MathJax.Hub.Queue(["Typeset", MathJax.Hub, element]);
    }
  }

  function renderPreview(textarea, preview, label) {
    let raw = textarea.value;
    const anchors = [];
    let idx = 0;

    raw = raw.replace(
      / ([\s\S]*?) /gi,
      (match, qHref, labelText) => {
        const href = qHref.slice(1, -1);
        const token = `__A_TAG_${idx++}__`;

        anchors.push({
          token,
          href: normalizeHref(href),
          label: labelText
        });

        return token;
      }
    );

    let safe = escapeHtml(raw);

    anchors.forEach(a => {
      const html = ` ${escapeHtml(a.label)} `;
      safe = safe.replace(a.token, html);
    });

    safe = safe.replace(/\{PROBLEM=(.+?)\}/g, (_, p1) =>
      ` [${escapeHtml(p1)}] `
    );

    safe = safe.replace(/\n/g, " ");
    clearMathJax(preview);

      label.classList.add("hidden");
      return;
    }

    preview.innerHTML = safe;
    preview.classList.remove("hidden");

    typesetMath(preview);
  }
