\begin{abstract}
Let $L(x)=\sum_{r=1}^m x_rX_r$, where the $X_r$ are real symmetric
$m\times m$ matrices and $\sum_rX_r^2\preceq KI$ for a fixed finite $K$.
For the Gaussian inverse Mills ratio $q$, we prove that the mean squared
row curl of $q(L)$ controls both the lack of full symmetry of the
coefficient tensor and the sum of all squared coefficient commutators.
After division by $m$, the estimate has a modulus independent of the
dimension, bounded by $C_K/\log(e/t)$ when the normalized curl energy
$t$ is small. The proof compares divided differences at $x$ and $-x$
and controls a cutoff on all tensor indices by Gaussian operator
moments. We also construct a fully symmetric tensor with a controlled
operator-energy bound and controlled algebraic residual.
\end{abstract}

\section{Introduction}
Let $\gamma_m$ be standard Gaussian measure on $\R^m$, and write
$\varphi$ and $\Phi$ for the one-dimensional Gaussian density and
distribution function. Put
\[
 q(a)=\frac{\varphi(a)}{\Phi(-a)}.
\]
For a real symmetric matrix $A$, the matrix $q(A)$ is defined by spectral
calculus; $Dq(A)[H]$ denotes its Fr\'echet derivative in direction $H$.
Consider a linear symmetric matrix field
\begin{equation}\label{eq:field}
 L(x)=\sum_{r=1}^m x_rX_r,\qquad
 B=\sum_{r=1}^m X_r^2\preceq KI,\qquad
 \X_{ijr}=(X_r)_{ij}.
\end{equation}
The tensor $\X$ is symmetric in its first two indices. Let $\Sym$
be the orthogonal projection onto fully symmetric tensors of order
three. All matrix and tensor norms below are unnormalized Frobenius
norms, unless an operator norm is indicated. Define
\begin{align}
 t_q(X)&=\frac1m\E_{\gamma_m}\sum_{i=1}^m\sum_{r<s}
 \left|(Dq(L)[X_r])_{is}-(Dq(L)[X_s])_{ir}\right|^2,\label{eq:curl}\\
 \mathcal E(X)&=\|\X-\Sym\X\|^2+
                 \sum_{r,s=1}^m\|[X_r,X_s]\|^2.\label{eq:residual}
\end{align}
Thus $mt_q(X)$ is the squared $L^2$ norm of the row curl of $q(L)$.
The second sum in \eqref{eq:residual} includes every ordered pair.

\begin{theorem}\label{thm:main}
For each $K\ge0$ there is a nondecreasing function
$\chi_K:[0,\infty)\to[0,\infty)$, independent of $m$, with
$\chi_K(0)=0$ and $\lim_{t\downarrow0}\chi_K(t)=0$, such that
\begin{equation}\label{eq:main}
 \frac{\mathcal E(X)}m\le\chi_K(t_q(X))
\end{equation}
for every family \eqref{eq:field}. There is a finite constant $C_K$
depending only on $K$ such that
\begin{equation}\label{eq:rate}
 \chi_K(t)\le\frac{C_K}{\log(e/t)}\qquad(0<t\le1/2).
\end{equation}
An explicit admissible modulus is given in \eqref{eq:modulus}.
\end{theorem}

No normalization $B\approx I$ or full coefficient symmetry is assumed.
The fixed bound on $B$ is a hypothesis throughout. The theorem estimates
the defining algebraic residual; it does not estimate distance to a
simultaneously diagonalizable family with constants uniform in $m$.

The exact compatibility problem occurs in the entropy manuscript
\cite[section ``From vanishing slack to exponential products'']{Entropy}.
There, if $B=I$ and $q(L)$ is a Jacobian, its first two Taylor terms
at the origin imply full coefficient symmetry and commutation. The
present theorem gives a quantitative implication from an integrated
curl bound, with constants independent of dimension. A pointwise
Taylor argument does not give that estimate without additional control
of the field near the origin. None of the entropy inequalities asserted
in \cite{Entropy} is an assumption of Theorem~\ref{thm:main}.

Spectral averaging of commutators appears in the linear equality
argument of \cite{Mahler}, and fourth-moment spectral truncation appears
in \cite[section 10]{Tracial}. The comparison at the two reflected
points is the step that controls coefficient symmetry before it is
known. Results on almost commuting matrices, such as
\cite{FK}, start from commutator bounds; their family estimates retain
dependence on the number of matrices. The algebraic characterization
of symmetric orthogonally decomposable tensors in \cite{BDHR} concerns
the exact zero set. We do not use either result as a stability theorem
for the residual \eqref{eq:residual}.

\section{Scalar estimates and reflected divided differences}
\begin{lemma}\label{lem:scalar}
The inverse Mills ratio satisfies $0<q'<1$ and $q''>0$ on $\R$.
For $R\ge1$, set
\[
 a_R=\min_{[-R,R]}q',\qquad h_R=\min_{[-R,R]}q''.
\]
Then
\begin{equation}\label{eq:scalarbounds}
 a_R\ge R\varphi(R),\qquad
 h_R\ge\frac{R}{2\pi}\exp(-R^2-4R-20).
\end{equation}
\end{lemma}
\begin{proof}
Let $Y_a$ have the law of $G-a$ conditional on $G>a$, with $G$ standard
Gaussian, and put $d(a)=\E Y_a=q(a)-a>0$. Integration by parts gives
\[
 q'=qd,\qquad \Var(Y_a)=1-qd,\qquad
 q''=q\{d^2-\Var(Y_a)\}.
\]
The first two identities give $0<q'<1$. For $u\ge0$ write
$S_a(u)=\Phi(-a-u)/\Phi(-a)$. Since $q$ is strictly increasing,
\[
 S_a(u+v)=S_a(u)S_a(v)\exp\left(-\int_0^v
               [q(a+u+s)-q(a+s)]\,ds\right)
 <S_a(u)S_a(v)
\]
when $u,v>0$. The survival integral formulas for the first two moments
therefore give
\begin{equation}\label{eq:qsecond}
 q''(a)=2q(a)\int_0^\infty\int_0^\infty
       [S_a(u)S_a(v)-S_a(u+v)]\,du\,dv>0.
\end{equation}
Thus $q'$ is increasing and
$a_R=q'(-R)=q(-R)(q(-R)+R)\ge R\varphi(R)$.
For $a\in[-R,R]$ and $u,v\in[1,2]$, all arguments in the preceding
integral lie in $[-R,R+4]$. The bound $q'<1$ and $q(0)<1$ imply
$q\le R+5$ there. Hence
\[
 S_a(u)S_a(v)\ge e^{-4(R+5)},\qquad
 \int_0^v[q(a+u+s)-q(a+s)]\,ds\ge q'(-R).
\]
Since $0<q'(-R)<1$, we have $1-e^{-q'(-R)}\ge q'(-R)/2$.
Restricting \eqref{eq:qsecond} to the unit square $[1,2]^2$, and using
$q(a)\ge\varphi(R)$, proves the second bound.
\end{proof}

For a differentiable scalar function define the divided difference
$f[a,b]=(f(a)-f(b))/(a-b)$ when $a\ne b$, and $f[a,a]=f'(a)$.
In an orthonormal eigenbasis of $A$,
\begin{equation}\label{eq:frechet}
 (Dq(A)[H])_{ij}=q[\lambda_i,\lambda_j]H_{ij}.
\end{equation}
This formula follows first for polynomials by differentiating their
powers and then for smooth functions by approximation in $C^1$ on a
compact spectral interval. In particular $Dq(A)$ is a contraction in
Frobenius norm. It follows that \eqref{eq:curl} is finite.

\begin{lemma}\label{lem:pair}
Suppose $|\lambda_i|,|\lambda_s|,|\lambda_r|\le R$ and $A,C\in\R$.
Set
\[
 e_+=q[\lambda_i,\lambda_s]A-q[\lambda_i,\lambda_r]C,
 \qquad
 e_-=q[-\lambda_i,-\lambda_s]A-q[-\lambda_i,-\lambda_r]C.
\]
Then
\begin{align}
 |A-C|^2&\le 2a_R^{-2}(e_+^2+e_-^2),\label{eq:pair1}\\
 |(\lambda_i+\lambda_s)A-(\lambda_i+\lambda_r)C|^2
       &\le M_R(e_+^2+e_-^2),\label{eq:pair2}\\
 M_R&=16R^2a_R^{-2}+16h_R^{-2}(1+2a_R^{-2}).\label{eq:M}
\end{align}
\end{lemma}
\begin{proof}
Abbreviate the four divided differences by $\alpha,\beta,\gamma,\eta$
in the order in which they occur. They lie in $[a_R,1]$. Also
\[
 \partial_b q[a,b]=\int_0^1 u q''((1-u)a+ub)\,du\ge h_R/2.
\]
Consequently $\alpha-\beta$ and $\gamma-\eta$ have opposite signs, and
$|\alpha-\beta|\ge(h_R/2)|\lambda_s-\lambda_r|$.
Writing $d=A-C$, the equations are
$e_+=\alpha d+(\alpha-\beta)C$ and
$e_-=\gamma d+(\gamma-\eta)C$.
If $\lambda_s\ne\lambda_r$, multiply these by
$|\gamma-\eta|$ and $|\alpha-\beta|$ and add. The terms in $C$ cancel,
and the coefficient of $d$ is at least $a_R$ times the sum of the
weights. This proves \eqref{eq:pair1}. If the eigenvalues coincide,
$e_+=\alpha d$ gives the same bound directly. Moreover,
\[
 |(\lambda_s-\lambda_r)C|^2
 \le8h_R^{-2}(1+2a_R^{-2})(e_+^2+e_-^2).
\]
Apply the squared triangle inequality to
$(\lambda_i+\lambda_s)d+(\lambda_s-\lambda_r)C$
to obtain \eqref{eq:pair2}.
\end{proof}

At a fixed $x$, diagonalize $L(x)$ by $U$ and rotate all three tensor
indices:
\[
 \widetilde\X_{isr}=\sum_{a,b,c}U_{ai}U_{bs}U_{cr}\X_{abc},\qquad
 \widetilde X_r=U^T\left(\sum_c U_{cr}X_c\right)U.
\]
By \eqref{eq:frechet}, the correspondingly rotated curl tensor has
entries $e_+$ in Lemma~\ref{lem:pair}, with
$A=\widetilde\X_{isr}$ and $C=\widetilde\X_{irs}$.
The same basis diagonalizes $L(-x)=-L(x)$ and gives $e_-$.
These are the derivatives of $q(L)$ evaluated at $-x$, so no extra
sign from differentiating a reflected composition occurs. We rotate
already differentiated tensors and never differentiate an eigenbasis.

\section{Gaussian moments and the spectral cutoff}
Let $L_j$ denote the action of $L(x)$ on the $j$th factor of
$(\R^m)^{\otimes3}$.
\begin{lemma}\label{lem:moments}
Under \eqref{eq:field},
\begin{equation}\label{eq:moments}
 \E L^2=B\preceq KI,\qquad \E L^4\preceq3K^2I.
\end{equation}
For any deterministic tensor $V$ and any $j,a\in\{1,2,3\}$,
\begin{equation}\label{eq:slot}
 \E\|L_jV\|^2\le K\|V\|^2,\qquad
 \E\|L_jL_aV\|^2\le3K^2\|V\|^2.
\end{equation}
\end{lemma}
\begin{proof}
Gaussian orthogonality proves the second-moment identity. Wick's
formula gives
\[
 \E L^4=B^2+\sum_rX_rBX_r+\sum_{r,s}X_rX_sX_rX_s.
\]
The first two terms are bounded above by $K^2I$. The final term is
self-adjoint and has operator norm at most $K^2$. Indeed, the
row-column Cauchy--Schwarz inequality applied to $A_{rs}=B_{rs}=X_rX_s$
gives this bound, since both
$\sum A_{rs}A_{rs}^*$ and $\sum B_{rs}^*B_{rs}$ are at most $K^2I$.
This proves \eqref{eq:moments}. The first assertion of \eqref{eq:slot}
follows by tracing against the fixed marginal of $V$ in slot $j$.
For the second, the case $j=a$ uses the fourth moment. If $j\ne a$,
the slot operators commute and
$L_j^2L_a^2\preceq(L_j^4+L_a^4)/2$; take expectations.
\end{proof}

Let $J$ interchange slots two and three, and set
\[
 S=(I-J)\X,\qquad Z(x)=(I-J)(L_1+L_2)\X.
\]
The entries of $Z$ form the ordered row curl of $L^2$. For $R\ge1$
put $P=\mathbf1_{[-R,R]}(L)$ and $\Pi=P^{\otimes3}$. The projection
$\Pi$ commutes with $J$ and with each $L_j$, and
\[
 I-\Pi\preceq R^{-2}\sum_{j=1}^3L_j^2.
\]
Lemma~\ref{lem:moments} therefore gives
\begin{align}
 \E\|(I-\Pi)S\|^2&\le12KR^{-2}\|\X\|^2,\label{eq:tailS}\\
 \E\|(I-\Pi)Z\|^2
 &\le8R^{-2}\sum_{j=1}^3\sum_{a=1}^2\E\|L_jL_a\X\|^2
 \le144K^2R^{-2}\|\X\|^2.\label{eq:tailZ}
\end{align}
In the second estimate we move $I-J$ through $\Pi$, bound its norm by
$2$, and apply the squared triangle inequality to $L_1+L_2$.
The deterministic tensor in every use of \eqref{eq:slot} is $\X$,
not $Z(x)$. The estimate applies to the third slot without a bound on
its flattening Gram matrix.

On the range of $\Pi$, Lemma~\ref{lem:pair} applies to each ordered
triple. Each ordered curl norm is twice the sum over $r<s$.
Reflection preserves Gaussian measure, so the integrated sum of
$e_+^2+e_-^2$ is at most $4mt_q(X)$. Orthogonal splitting and
$\|\X\|^2=\tr B\le Km$ yield
\begin{align}
 \frac{\|S\|^2}{m}&\le8a_R^{-2}t_q(X)+\frac{12K^2}{R^2},\label{eq:Sbound}\\
 \frac{\E\|Z\|^2}{m}&\le4M_Rt_q(X)+\frac{144K^3}{R^2}.\label{eq:Zbound}
\end{align}
The cutoff depends on $x$, but it is not differentiated. In particular,
these estimates do not require the event $\|L(x)\|_{\rm op}\le R$ to
have large probability.

\section{Extraction of the algebraic residual}
\begin{lemma}\label{lem:algebra}
For the tensors above,
\begin{equation}\label{eq:symmetry}
 \|\X-\Sym\X\|^2=\tfrac13\|S\|^2,
 \qquad
 \sum_{r,s}\|[X_r,X_s]\|^2\le2\E\|Z\|^2+18K\|S\|^2.
\end{equation}
\end{lemma}
\begin{proof}
Since $\X$ is symmetric in slots one and two, its full symmetrization
is the average of three distinct slot permutations. These have equal
norms and equal pairwise inner products. Expanding the projection
error and $\|\X-J\X\|^2$ proves the first identity.
For the second, put $S_{rs}=X_re_s-X_se_r$. Direct expansion gives
\begin{equation}\label{eq:polynomial}
 Z_{\cdot rs}(x)=[X_r,X_s]x+
 \sum_t x_t\{X_rS_{ts}-X_sS_{tr}+X_tS_{rs}\}.
\end{equation}
Each of the three error arrays has squared coefficient norm at most
$K\|S\|^2$. For example,
\[
 \sum_{r,s,t}|X_rS_{ts}|^2
 =\sum_{s,t}\langle S_{ts},BS_{ts}\rangle\le K\|S\|^2.
\]
The square of their sum is at most $9K\|S\|^2$. Gaussian coefficient
orthogonality and $\|A-B\|^2\le2\|A\|^2+2\|B\|^2$ now prove the bound.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Combine \eqref{eq:Sbound}, \eqref{eq:Zbound} and
\eqref{eq:symmetry}. With
\[
 F_K(R)=8(1/3+18K)a_R^{-2}+8M_R,\qquad D_K=4K^2+504K^3,
\]
we obtain $\mathcal E(X)/m\le F_K(R)t_q(X)+D_K/R^2$ for every $R\ge1$.
Also $\|\X-\Sym\X\|^2\le Km$ and
\[
 \sum_{r,s}\|[X_r,X_s]\|^2
 \le4\sum_{r,s}\|X_rX_s\|^2=4\tr B^2\le4K^2m.
\]
Consequently an admissible choice is
\begin{equation}\label{eq:modulus}
 \chi_K(t)=\min\left\{K+4K^2,
       \inf_{R\ge1}\left(F_K(R)t+\frac{D_K}{R^2}\right)\right\}.
\end{equation}
It is nondecreasing. Given a tolerance, first fix $R$ to make
$D_K/R^2$ small and then let $t\downarrow0$. This proves the claimed
limit and $\chi_K(0)=0$, uniformly in dimension.

For the rate, \eqref{eq:scalarbounds} implies
$a_R^{-2}\le2\pi e^{R^2}$ and $h_R^{-2}\le4\pi^2e^{50R^2}$ for $R\ge1$.
Thus $F_K(R)\le C(1+K)e^{52R^2}$ for an absolute constant $C$.
For $0<t\le e^{-104}$ take $R^2=\log(1/t)/104$. The two terms in
\eqref{eq:modulus} are bounded by
$C(1+K)\sqrt t+104D_K/\log(1/t)$, which is at most
$C_K/\log(e/t)$. On $[e^{-104},1/2]$ use $K+4K^2$ and enlarge $C_K$.
If $K=0$, every matrix vanishes and the assertion is immediate.
\end{proof}

\begin{corollary}\label{cor:general}
Replace $q$ by any $f\in C^2(\R)$ satisfying $0<f'\le1$ and $f''>0$.
For each fixed $K$, the resulting curl energy controls
$\mathcal E(X)/m$ by a modulus tending to zero, independent of $m$.
The formula \eqref{eq:modulus} applies with the compact minima of $f'$
and $f''$ in place of $a_R,h_R$.
\end{corollary}
\begin{proof}
The reflected divided-difference argument and the Gaussian moment
estimates use only these compact lower bounds. The preceding proof up
to the logarithmic-rate calculation therefore applies verbatim.
\end{proof}
The rate in Theorem~\ref{thm:main} uses the specific bounds for $q$.
Strict convexity cannot be discarded in Corollary~\ref{cor:general}.
For example, take the fully symmetric tensor in dimension two with
$\X_{111}=1/\sqrt2$ and
$\X_{122}=\X_{212}=\X_{221}=-1/\sqrt2$, all other entries zero.
Then $B=I$ and $\sum_{r,s}\|[X_r,X_s]\|^2=4$, whereas the curl of
$f(L)$ is zero for any affine $f$.

\section{Symmetric repair with bounded energy}
Full symmetrization alone need not preserve the bound on $B$.
The next proposition controls that loss by projecting all slots.
\begin{proposition}\label{prop:repair}
Assume $K>0$, put $T=\Sym\X$ and $e^2=\|T-\X\|^2$.
There exists a fully symmetric tensor $W$, with symmetric slices $W_r$,
such that $B_W=\sum_rW_r^2\preceq4KI$ and
\begin{align}
 \|W-\X\|^2&\le26e^2,\label{eq:repair1}\\
 \|B_W-B\|^2&\le234Ke^2,\label{eq:repair2}\\
 \sum_{r,s}\|[W_r,W_s]-[X_r,X_s]\|^2&\le936Ke^2.\label{eq:repair3}
\end{align}
In particular, if $\beta=\|B-I\|^2/m$ and $\eta=\mathcal E(X)/m$, then
\begin{equation}\label{eq:repairedresidual}
 \frac{\|B_W-I\|^2+\sum_{r,s}\|[W_r,W_s]\|^2}{m}
 \le2\beta+(2+2340K)\eta.
\end{equation}
\end{proposition}
\begin{proof}
Flatten the first slot of $\X$ as a map
$M_X:\R^m\otimes\R^m\to\R^m$. Symmetry of the matrix slices gives
$M_XM_X^*=B$. Let $P=\mathbf1_{[0,4K]}(B_T)$,
$k=\operatorname{rank}(I-P)$ and $a=\|(I-P)_1T\|$.
Spectral calculus and the triangle inequality give
\[
 2\sqrt{Kk}\le a\le e+\sqrt{Kk}.
\]
Thus $a\le2e$. Put $W=P^{\otimes3}T$. Its full symmetry and the
commuting projection inequality give
$\|T-W\|^2\le3a^2\le12e^2$, hence \eqref{eq:repair1}.
The first flattening is $PM_T(P\otimes P)$, so
$B_W\preceq PB_TP\preceq4KI$.
The rectangular product estimate
\[
 \|MM^*-NN^*\|_F\le(\|M\|_{\rm op}+\|N\|_{\rm op})\|M-N\|_F
\]
gives \eqref{eq:repair2}, since $9K\cdot26=234K$.
For any symmetric families $D,A$,
\[
 \left(\sum_{r,s}\|[D_r,A_s]\|^2\right)^{1/2}
 \le2\|B_A\|_{\rm op}^{1/2}\|D\|.
\]
This follows by bounding the two product orders separately and summing
against $\sum_sA_s^2$. Apply it to
$[W_r-X_r,X_s]+[W_r,W_s-X_s]$.
The resulting norm is at most $6\sqrt K\|W-\X\|$, proving
\eqref{eq:repair3}. Squared triangle inequalities for normalization
and commutators give \eqref{eq:repairedresidual}.
\end{proof}

For instance, $X_1=I_m$ and $X_r=0$ for $r>1$ have $B=I_m$,
but their full symmetrization satisfies
\[
 B_T=\diag((m+8)/9,2/9,\ldots,2/9).
\]
Thus the projection in Proposition~\ref{prop:repair} can be necessary.
The proposition starts from an already bounded $B$; it gives no
uniform cap for an arbitrary uncapped family.

Theorem~\ref{thm:main} and Proposition~\ref{prop:repair} also give a
direct interface for applications. If an argument supplies
$t_q(X)\le Au$ and $\|B-I\|^2/m\le Du$, with $A,D,K$ independent of
$m$, then the repaired tensor satisfies
\[
 \frac{\|B_W-I\|^2+\sum_{r,s}\|[W_r,W_s]\|^2}{m}
 \le2Du+(2+2340K)\chi_K(Au).
\]
Obtaining a global orthogonal decomposition from this residual is a
further stability problem. No such assertion, or unrestricted
stability theorem for entropy, is claimed here.
