\begin{abstract}
Assume the isoperimetric profile and equality classification for the unit
cubic three-torus stated below. We determine the leading perimeter cost
of nesting a fixed-volume ball inside a slightly thinner optimal tube
or slab, allowing both interfaces to deform. For ball--tube contact the
cost is $K(R)\eps^2$, with an explicit positive coefficient. For
ball--slab contact it is $2\pi\eps^2/|\log\eps|$. Both limits hold for
the minimum over all nested finite-perimeter pairs, without symmetry or
regularity assumptions on competitors. A common two-volume correction
and a perimeter lattice comparison give scalar quasiminimality of exact
minimizers. Improved convergence then reduces the lower bounds to a
fixed-circle trace problem and a shrinking-annulus capacity estimate.
\end{abstract}
\maketitle

\section{The two nesting problems}
We work on $\T^3=\R^3/\Z^3$ with its unit cubic metric. Volume is
Lebesgue measure and $P$ is De Giorgi perimeter. All inclusions between
measurable sets are understood almost everywhere. The scalar input is
the following profile and classification from \cite{OpenAI}.

\begin{assumption}\label{ass:profile}
For $0<V<1$, put $v=\min(V,1-V)$. The minimum perimeter at volume $V$ is
\[
 I(V)=\min\{(36\pi)^{1/3}v^{2/3},\ 2\sqrt{\pi v},\ 2\}.
\]
For $0<V<1/2$, the minimizers are, up to isometry and null sets,
balls when $V<4\pi/81$, circular tubes about shortest closed geodesics
when $4\pi/81<V<1/\pi$, and coordinate slabs when $1/\pi<V<1/2$.
At the two transition volumes precisely the adjacent types occur;
at $V=1/2$ the minimizers are coordinate slabs. Above half volume,
take complements of the corresponding minimizers.
\end{assumption}
Only strict branches and their limiting volumes occur below.
This assumption is the exact scalar input; neither the scalar
isoperimetric problem nor its classification is reproved here.

For $2/9<R<1/\pi$, define
\begin{equation}\label{eq:D}
 D_R(\eps)=\inf\left\{P(E)+P(F)-4\pi R^2-2\pi(R-\eps):
 \begin{array}{l}E\subset F,\\|E|=4\pi R^3/3,\\|F|=\pi(R-\eps)^2\end{array}\right\}.
\end{equation}
For $1/(2\pi)<R<1/3$, define
\begin{equation}\label{eq:L}
 L_R(\eps)=\inf\left\{P(E)+P(F)-4\pi R^2-2:
 \begin{array}{l}E\subset F,\\|E|=4\pi R^3/3,\\|F|=2(R-\eps)\end{array}\right\}.
\end{equation}
The infima run over finite-perimeter pairs.

\begin{theorem}\label{thm:tube}
Assume Assumption~\ref{ass:profile}. For each fixed $2/9<R<1/\pi$,
\[
 \lim_{\eps\downarrow0}\frac{D_R(\eps)}{\eps^2}
 =K(R):=\frac{4\pi R}{R+4G_R},\qquad
 G_R=R^2-\frac R2\cot\frac1{2R}>0.
\]
\end{theorem}
\begin{theorem}\label{thm:slab}
Assume Assumption~\ref{ass:profile}. For each fixed $1/(2\pi)<R<1/3$,
\[
 \lim_{\eps\downarrow0}\frac{|\log\eps|L_R(\eps)}{\eps^2}=2\pi.
\]
\end{theorem}
The radius remains fixed in both limits. No endpoint-uniform estimate
is asserted. Smooth nested pairs attain each coefficient
asymptotically, whereas the lower bounds apply to every admissible
sequence of finite-perimeter pairs.

The independent scalar minimizers touch along a circle in the tube
problem and at two points in the slab problem. A point on a
two-dimensional interface has zero $H^1$ capacity, which accounts for
the different scales. Logarithmic response in droplet contact is
classical, both for rigid plates \cite{LGL,Lv} and deformable droplet
and bubble contacts \cite{MW,Ginot}. Alama et al.\
\cite[Theorem 4.5]{ABLW} obtain a logarithmic second-order selection
term for optimal shells in a nonlocal problem. Here we determine the
sharp nonlinear minimum over all finite-perimeter nested pairs, with
both interfaces free. The quadratic forms, Green functions, and planar capacity
calculation used below are classical tools.

\section{A common regularity reduction}\label{sec:regularity}
Let $B_R$ be a radius-$R$ ball, $C_R$ the radius-$R$ tube about the
$z$-axis, and $S_R=\{-R<z<R\}$ the coordinate slab.
In either geometry there is a patch of $\partial B_R$ strictly inside
the outer set and a patch of the outer boundary disjoint from the ball.
These free patches permit simultaneous volume correction before any
regularity of the approximating pair has been established.

\begin{lemma}\label{lem:flow}
Let $(E_0,F_0)$ be either $(B_R,C_R)$ or $(B_R,S_R)$ in the ranges above.
For nested pairs $(A,B)$ sufficiently close to $(E_0,F_0)$ in
$L^1\times L^1$, with $P(A)+P(B)\le M$, two sufficiently small
volume errors of total magnitude $q$ can be corrected by one smooth
ambient diffeomorphism. The corrected pair remains nested and its
summed perimeter is at most $P(A)+P(B)+C_Mq$.
\end{lemma}
\begin{proof}
Choose smooth vector fields $X_1,X_2$ supported in the two free patches,
with positive outward flux through the respective boundary. Their
fluxes through the other limiting set vanish. For the composition
$\Phi_s$ of their flows, the two-volume map is
\[
 V_{A,B}(s)=\left(\int_A\det D\Phi_s,\int_B\det D\Phi_s\right).
\]
At $(E_0,F_0)$ and $s=0$ its derivative is diagonal with positive
entries. The map has uniform $C^2$ bounds in $s$ for all measurable
pairs. Its derivatives vary continuously in $L^1\times L^1$ because
the integrands are smooth and bounded. The inverse-function theorem,
or its contraction-map proof with the fixed diagonal derivative,
therefore gives a correcting $s$ with $|s|\le Cq$ uniformly nearby.
Applying the same diffeomorphism preserves inclusion. The perimeter
change-of-variables formula and $\|\operatorname{cof}D\Phi_s-I\|_\infty
\le C|s|$ give $P(\Phi_s H)\le(1+C|s|)P(H)$ for either set.
\end{proof}

\begin{lemma}\label{lem:quasi}
Suppose exact minimizing pairs $(E_j,F_j)$ for either problem converge
in $L^1\times L^1$ to its limiting pair, with uniformly bounded
perimeters, and put $M=\sup_j(P(E_j)+P(F_j))<\infty$.
For all sufficiently large $j$, both components are scalar
$\omega$-minimizers with one constant $\omega$ independent of $j$.
More precisely,
$P(E_j)\le P(H)+\omega\rho^3$ whenever
$E_j\symdiff H\Subset B_\rho(x)$, and likewise for $F_j$.
\end{lemma}
\begin{proof}
The perimeter lattice inequality states
\[
 P(A\cap B)+P(A\cup B)\le P(A)+P(B).
\]
It follows by strict BV approximation and the gradient identity for
minimum and maximum of two smooth functions. Given a competitor $H$
for $E_j$, use the nested pair $(H\cap F_j,H\cup F_j)$. Pointwise
checking the three states of $E_j\subset F_j$ gives
\[
 |(H\cap F_j)\symdiff E_j|+|(H\cup F_j)\symdiff F_j|
 =|H\symdiff E_j|.
\]
If $P(H)\ge P(E_j)$ the conclusion is immediate. Otherwise the new
summed perimeter is uniformly bounded. When $\rho<\rho_0$ for a
fixed sufficiently small $\rho_0$, its $L^1$ change and its two volume
errors are small enough for Lemma~\ref{lem:flow}. Correct the volumes
and use paired minimality to obtain
\[
 P(E_j)+P(F_j)\le P(H)+P(F_j)+C|H\symdiff E_j|.
\]
The small-ball volume bound proves the assertion for $\rho<\rho_0$.
For larger balls choose also $\omega\ge M/\rho_0^3$ and use the
global perimeter bound. For an outer competitor replace the pair by
$(E_j\cap H,E_j\cup H)$; the same identity and cancellation apply.
\end{proof}

We use the improved-convergence theorem stated in
Acerbi--Fusco--Morini \cite[Definition 2.5 and Theorem 4.2]{AFM}.
A sequence of scalar $\omega$-minimizers on the flat torus, with
bounded perimeter and $L^1$ limit a $C^2$ set, eventually has a global
normal-graph boundary over that limit; the graph heights tend to zero
in $C^{1,\alpha}$ for every $\alpha<1/2$.
AFM state this classical regularity result without proof and refer to
earlier regularity theory. Lemma~\ref{lem:quasi} verifies its hypotheses
separately for the two components. Contact between the limiting
interfaces causes no extra hypothesis in this scalar application.
The theorem also applies to the disconnected slab boundary.

We use the open representatives bounded by these graphs, with their
occupied side fixed by $L^1$ convergence. Inclusion almost everywhere
then implies $E_j\subset\overline{F_j}$ pointwise and the same
containment for the inner boundary. This regularity statement concerns
exact minimizers only. Every competitor has energy at least the exact
minimum, which will extend the lower bounds to the original class.

\section{Quadratic expansions and centering}\label{sec:expansions}
For a radial graph of height $\psi$ over the sphere $\partial B_R$,
exact volume gives
\begin{equation}\label{eq:sphere-volume}
 \int_{\partial B_R}\left(\psi+\frac{\psi^2}{R}
 +\frac{\psi^3}{3R^2}\right)=0.
\end{equation}
The area density is
$(1+\psi/R)^2\sqrt{1+|\nabla\psi|^2/(1+\psi/R)^2}$.
Consequently, for $C^1$-small graphs,
\begin{align}
 P(E)-4\pi R^2&=\tfrac12 Q_B(\psi)+o(1)\|\psi\|_{H^1}^2,
 \label{eq:sphere-area}\\
 Q_B(\psi)&=\int_{\partial B_R}(|\nabla\psi|^2-2\psi^2/R^2).
 \nonumber
\end{align}
The error follows by bounding the cubic volume term by
$C\|\psi\|_\infty\int\psi^2$ and the square-root remainder by
$C\|\nabla\psi\|_\infty^2\int|\nabla\psi|^2$.
Thus it is controlled by the actual $H^1$ energy, even under
concentration of normalized gradients.

For a tube graph of radial height $\eta$ over $Z_r=\partial C_r$,
\begin{align}
 &\int_{Z_r}(\eta+\eta^2/(2r))=0,\label{eq:tube-volume}\\
 &P(F)-2\pi r=\tfrac12Q_{C,r}(\eta)+o(1)\|\eta\|_{H^1}^2,
 \quad Q_{C,r}(\eta)=\int_{Z_r}(|\nabla\eta|^2-\eta^2/r^2).
 \label{eq:tube-area}
\end{align}
Indeed its area density is
$(1+\eta/r)\sqrt{1+\eta_z^2+\eta_\theta^2/(r+\eta)^2}$.
For a slab, let $V^\pm$ be the outward heights of its two faces.
Its volume condition is $\int V^++\int V^-=0$ and its deficit is
\begin{equation}\label{eq:slab-area}
 P(F)-2=\tfrac12\sum_\pm\int_{\T^2}|\nabla V^\pm|^2
 +o(1)\sum_\pm\|V^\pm\|_{H^1}^2.
\end{equation}

Let $\nu=(x-c)/R$ be the unit normal of a reference sphere centered at
$c$, let $\nu_i$ be its coordinate components, and set
$\nu_{\rm tr}=(\nu_1,\nu_2)$. These are the normal components of
translations. Choose the spherical reference center so that
$\int\psi\nu_i=0$ for its three translation modes. For a tube,
choose its transverse reference center so that
$\int\eta\cos\theta=\int\eta\sin\theta=0$.
The derivative of each moment map under a reference translation is
minus the Gram matrix of the corresponding normal components. It is
invertible at the limiting graph. The finite-dimensional implicit
function theorem gives these centers for every sufficiently small
$C^1$ graph. This changes representations, not the actual sets.
For a slab, translate its reference center to make both face means zero;
the means sum to zero and change by opposite constants.

On the sphere the Jacobi eigenvalues are $[l(l+1)-2]/R^2$.
After removing the constant and translation modes, there is an $H^1$
spectral gap. On the tube they are
$(m^2-1)/r^2+4\pi^2k^2$.
After removing its constant and two translation modes, the same is true
when $r>1/(2\pi)$. Equations~\eqref{eq:sphere-volume} and
\eqref{eq:tube-volume} show that the omitted means are
$O(\|\psi\|_2^2)$ and $O(\|\eta\|_2^2)$; their squared contributions
are absorbable. Thus the corresponding scalar deficits bound the
squared $H^1$ norms from below. For the mean-zero slab faces, ordinary
Poincar\'e inequalities give the same conclusion.

\section{Ball--tube contact}
\subsection{Compactness and the equatorial constraint}
Put $r=R-\eps$. The ellipsoid with semiaxes $r,r,R^3/r^2$ has the
inner volume and lies in $C_r$. Its longest semiaxis tends to $R<1/2$,
so it is embedded. Its perimeter differs from the sphere's by
$O_R(\eps^2)$, since the first variation at fixed volume vanishes.
Therefore $0\le D_R(\eps)\le C_R\eps^2$.
The direct method on the compact torus gives exact minimizers:
perimeter is lower semicontinuous, and volumes and inclusion pass to
$L^1$ limits.

Along any sequence realizing a lower limit, the two scalar deficits
tend to zero. Assumption~\ref{ass:profile} identifies every limiting
pair as a radius-$R$ ball nested in a radius-$R$ shortest tube. The
transverse projection of the ball is a radius-$R$ disk inside the
tube's disk of the same area, so their centers coincide. A common
translation and signed coordinate permutation normalize the limit to
$(B_R,C_R)$. No arbitrary rotation of the torus is used.
Apply Section~\ref{sec:regularity} and center as in
Section~\ref{sec:expansions}. Write the sphere center relative to the
tube axis as $(h,0)$, after a common translation. Initially $h=o(1)$.
The energy bound and spectral gaps give
\begin{equation}\label{eq:scale-tube}
 \|\psi\|_{H^1}+\|\eta\|_{H^1}=O(\eps).
\end{equation}
Identify all tube graphs with $Z_R$. Along a subsequence,
$u_j=\psi_j/\eps_j$ and $v_j=\eta_j/\eps_j$ converge weakly in $H^1$
and strongly in $L^2$ to mean-zero $u,v$. Their equatorial traces
converge strongly in $L^2$: the trace map takes $H^1$ to $H^{1/2}$
on the circle, which embeds compactly into $L^2$.
The negative terms pass by strong convergence, hence
\begin{equation}\label{eq:weak-energy}
 \liminf_j\frac{D_R(\eps_j)}{\eps_j^2}
 \ge\tfrac12\{Q_B(u)+Q_{C,R}(v)\}.
\end{equation}

At the inner equator its transverse position is
$y_j(\theta)=h_j+(R+\psi_j(\theta,0))e_\theta$.
Let $\phi_j(\theta)=\arg y_j(\theta)$.
These circle maps tend to the identity in $C^1$ and have uniformly
bounded Jacobians and inverse Jacobians. Actual inclusion and a dot
product with $e_\theta$ give exactly
\begin{equation}\label{eq:trace-exact}
 \eps_j+h_j\cdot e_\theta+\psi_j(\theta,0)
 \le\eta_j(\phi_j(\theta),0).
\end{equation}
The two trace norms are $O(\eps_j)$ by \eqref{eq:scale-tube}.
On the half-circle where $h_j\cdot e_\theta\ge0$,
\eqref{eq:trace-exact} gives
$h_j\cdot e_\theta\le|\eta_j(\phi_j(\theta),0)|+|\psi_j(\theta,0)|$.
Taking $L^2$ norms yields $c|h_j|=O(\eps_j)$. Pass to $h_j/\eps_j\to d$.
Composition by $\phi_j$ preserves the strong trace limit: approximate
the limiting trace by smooth functions and use the uniform composition
bounds. Dividing \eqref{eq:trace-exact} by $\eps_j$ yields
\[
 v(\theta,0)-u(\theta,0)\ge1+d\cdot e_\theta.
\]
Replace $u$ by $\widetilde u=u+d\cdot\nu_{\rm tr}$.
The added translation Jacobi field has zero mean and zero bilinear
Jacobi pairing with every $H^1$ function. Its energy is zero, so
\[
 Q_B(\widetilde u)=Q_B(u),\qquad v-\widetilde u\ge1
 \quad\text{on the equator}.
\]

\subsection{The quadratic minimum}
On the zero-mean spaces the two forms are nonnegative.
Common angular averaging preserves the trace inequality and removes
only nonnegative Fourier blocks. We may therefore minimize among
axial functions $u(t)$, $v(z)$, where $t=\cos\varphi$ on the sphere.
For prescribed $A=u(0)$, the mean-zero function
\[
 u_A(t)=A(1-2|t|)
\]
satisfies $Q_B(u_A)/2=4\pi A^2$.
For a mean-zero function $w$ of trace zero, integration by parts on
the hemispheres gives $Q_B(u_A,w)=0$, since its Jacobi image is the
same constant on both hemispheres. Nonnegativity then proves this is
the minimum at trace $A$.

On mean-zero period-one functions, $L=-d^2/dz^2-R^{-2}$ is positive.
Its periodic Green function with zero mean is
\[
 g_R(z)=R^2-\frac{R\cos((|z|-1/2)/R)}{2\sin(1/(2R))},
 \qquad -1/2\le z\le1/2.
\]
It has derivative jump $-1$ at zero, matching periodic derivatives,
and $Lg_R=\delta_0-1$. Thus $G_R=g_R(0)>0$, either by energy
positivity or directly because $\cot(1/(2R))<0$ in our range.
Testing against a mean-zero function and applying energy
Cauchy--Schwarz gives, at trace $D=v(0)$,
\[
 \tfrac12Q_{C,R}(v)\ge\frac{\pi R D^2}{G_R},
\]
with equality at $v=Dg_R/G_R$. The remaining scalar minimization is
\[
 \min_{D-A\ge1}\left(4\pi A^2+\frac{\pi R D^2}{G_R}\right)
 =\frac{4\pi R}{R+4G_R}=K(R).
\]
The minimizing traces are $A=-R/(R+4G_R)$ and
$D=4G_R/(R+4G_R)$. Together with \eqref{eq:weak-energy}, this proves
the lower bound in Theorem~\ref{thm:tube}.

\subsection{Recovery}
Smooth the two axial minimizing functions, preserving evenness,
periodicity, and smoothness at the sphere poles. Subtract their means.
They converge uniformly and in $H^1$. Rescale a fixed approximation
to give trace gap $1+\gamma$, with $\gamma>0$.
For each fixed smooth pair $U,V$, exact volume equations give constant
corrections $s,t=O(\eps^2)$ to the radii
$R+\eps U+s$ and $R-\eps+\eps V+t$.
At sphere height parameter $\tau=z/R$, the tube containment margin is
\[
 R(1-\sqrt{1-\tau^2})
 +\eps[-1+V(R\tau)-U(\tau)\sqrt{1-\tau^2}]+O(\eps^2).
\]
The bracket is positive in a fixed strip about the equator; outside
that strip the zeroth-order term is positive. Hence the entire inner
boundary lies in the tube. Every tube slice is a centered disk;
maximizing transverse distance on each compact inner slice proves
containment of the whole region. The expansions of
Section~\ref{sec:expansions} give the required energy. Take first
$\eps\downarrow0$, then improve the smooth approximation, then let
$\gamma\downarrow0$. This proves the matching upper bound.

\section{Ball--slab contact}
\subsection{A logarithmic recovery sequence}\label{sec:slab-upper}
Let $\ell=|\log\eps|$, fix small $\delta,\eta>0$, and put
$a=\sqrt\eps\,\ell$. In each polar chart choose a radial cutoff
$\chi$ equal to one up to radius $a$, zero beyond $\delta$, and
logarithmic in between. Smooth the two corners on comparable annuli.
Direct polar integration gives
\begin{align*}
 \int|\nabla\chi|^2&=\frac{2\pi+o(1)}{\log(\delta/a)},&
 \int\chi^2&=O_\delta(\ell^{-2})+O(a^2),\\
 \int\chi&=O_\delta(\ell^{-1})+O(a^2),&
 \|\nabla\chi\|_\infty&=O((a\ell)^{-1}).
\end{align*}
Smoothing changes the first integral by $O(\ell^{-2})$.
The spherical metric in horizontal polar coordinates differs from the
plane by $O_R(\delta^2)$.

Move the sphere inward by $t\chi$ at both poles, with
$t=(1+\eta)\eps/2$, and each slab face outward by $t\chi$.
Smooth bumps on fixed free patches restore the two volumes exactly.
Their coefficients are $O_\delta(\eps/\ell)$ by the nonzero first
volume derivatives, so their squared-energy cost is
$O_\delta(\eps^2/\ell^2)$. All graphs are $C^1$-small.
Within horizontal radius $a/2$, converting the sphere's normal
height to vertical height changes $t$ by a factor $1+o(1)$.
Thus the available margin is $(1+\eta+o(1))\eps$ against the imposed
thinning $\eps$. Outside that core, the original quadratic separation
is at least $c_Ra^2\gg\eps$ until leaving the polar charts; elsewhere
it is bounded below away from zero. This proves containment of the
inner boundary between the two faces. Extrema of each nonempty compact
vertical slice lie on that boundary, so the whole region is contained,
without any connected-slice assumption.

The spherical potential, correction bumps, and area remainders are
$o(\eps^2/\ell)$. Each of four patches contributes one half of
$t^2$ times its Dirichlet energy. Since
$\log(\delta/a)=\ell/2-\log\ell+\log\delta$,
\[
 \limsup_{\eps\downarrow0}\frac{\ell L_R(\eps)}{\eps^2}
 \le2\pi(1+\eta)^2(1+O_R(\delta^2)).
\]
Let $\delta,\eta$ decrease to zero. This also establishes the energy
bound needed in the next step.

\subsection{Minimizers and exact vertical charts}
The direct method again gives exact minimizers. Nonnegative scalar
deficits and Section~\ref{sec:slab-upper} imply that every limiting
pair is a radius-$R$ ball inside a width-$2R$ coordinate slab.
Inclusion centers the ball normally between the two faces. A common
translation and signed coordinate permutation put the limit in the
standard position. The slab branch applies even above half volume:
$2R\in(1/\pi,2/3)\subset(1/\pi,1-1/\pi)$.

Section~\ref{sec:regularity} gives global $C^1$-small graphs.
Center the sphere to remove its degree-one moments, and center the
slab to give each face height mean zero. A common translation fixes
the sphere transversely and the slab axially. Retain the relative axial
center $h_j$. With $\ell_j=|\log\eps_j|$, the expansions and gaps give
\begin{equation}\label{eq:scale-slab}
 \|\psi_j\|_{H^1}+\|V_j^+\|_{H^1}+\|V_j^-\|_{H^1}
 =O(\eps_j/\sqrt{\ell_j}).
\end{equation}
The sphere mean is $O(\eps_j^2/\ell_j)$.

For $|x|<2\delta$ at either pole, let $W_j^\pm(x)$ be the actual
outward vertical height of the sphere relative to its reference
sphere. The upper boundary is
$h_j+\sqrt{R^2-|x|^2}+W_j^+(x)$, and the lower boundary is
$h_j-\sqrt{R^2-|x|^2}-W_j^-(x)$. Actual inclusion gives
\begin{equation}\label{eq:vertical-obstacle}
 V_j^\pm(x)-W_j^\pm(x)
 \ge\eps_j\pm h_j-[R-\sqrt{R^2-|x|^2}].
\end{equation}
Let $\Theta_j$ map $x$ to its direction in the actual radial graph, and put
$p_j(x)=\psi_j(\Theta_j(x))$. The following identity is exact:
\begin{equation}\label{eq:chart}
 W_j(x)=A_j(x)p_j(x),\qquad
 A_j(x)=\frac{2R+p_j(x)}{\sqrt{(R+p_j(x))^2-|x|^2}
 +\sqrt{R^2-|x|^2}}.
\end{equation}
The chart metric has distortion $1+O_R(\delta^2)+o_j(1)$,
$A_j=1+O_R(\delta^2)+o_j(1)$, and
$|\nabla A_j|\le C_R\delta+o_j(1)$.
These follow by differentiating \eqref{eq:chart} and
$x=(R+\psi_j)\nu_{\rm tr}$, using unscaled $C^1$ smallness.
We do not replace \eqref{eq:chart} by an uncontrolled pointwise
first-order expansion at scale $\eps_j$.

\subsection{Localization and annular capacity}
Fix a spherical spectral cutoff $N$. Decompose
$\psi_j=b_j+l_j+k_j$ into its constant part, degrees $2$ through $N$,
and degrees above $N$. There is no degree-one part.
Equation~\eqref{eq:scale-slab} gives
\[
 |b_j|=O(\eps_j^2/\ell_j),\qquad
 \|l_j\|_{C^1}=O_N(\eps_j/\sqrt{\ell_j})=o_N(\eps_j).
\]
Use the actual maps and coefficients in \eqref{eq:chart} to define
$W_{j,\mathrm{hi}}^\pm=A_j^\pm k_j\circ\Theta_j^\pm$.
Their difference from $W_j^\pm$ is $o_N(\eps_j)$ uniformly.
Differentiation, change of variables, and Young's inequality give,
for every fixed $\xi>0$,
\begin{align}\label{eq:chart-energy}
 \sum_\pm\int_{|x|<2\delta}|\nabla W_{j,\mathrm{hi}}^\pm|^2
 &\le(1+C_R\delta^2+\xi+o_j(1))\int_{\partial B_R}|\nabla k_j|^2
 +C_{R,\delta,\xi}\int_{\partial B_R}k_j^2.
\end{align}
The two spherical patches are disjoint. Spectral separation gives
\[
 \int k_j^2\le\frac{R^2}{(N+1)(N+2)}\int|\nabla k_j|^2,\qquad
 Q_B(k_j)\ge\left(1-\frac2{(N+1)(N+2)}\right)\int|\nabla k_j|^2.
\]
The low-degree Jacobi energy is nonnegative, and the constant part has
energy $o(\eps_j^2/\ell_j)$. Given $\zeta>0$, choose $\delta$ small,
then $\xi$ small, then $N$ large, all fixed before $j\to\infty$.
Equations~\eqref{eq:sphere-area}, \eqref{eq:slab-area}, and
\eqref{eq:chart-energy} then give, with
$Z_j^\pm=V_j^\pm-W_{j,\mathrm{hi}}^\pm$,
\begin{equation}\label{eq:localized}
 L_R(\eps_j)\ge\frac{1-\zeta}{4}
 \sum_\pm\int_{|x|<2\delta}|\nabla Z_j^\pm|^2
 +o(\eps_j^2/\ell_j).
\end{equation}
The factor $1/4$ includes
$|\nabla(V-W)|^2\le2(|\nabla V|^2+|\nabla W|^2)$.
Also $\|Z_j^\pm\|_{H^1}=O_\delta(\eps_j/\sqrt{\ell_j})$.

For a circle average
$m(\rho)=(2\pi)^{-1}\int_0^{2\pi}Z(\rho,\theta)\,d\theta$,
Jensen and weighted Cauchy--Schwarz imply
\begin{equation}\label{eq:annulus}
 \int_{a<|x|<b}|\nabla Z|^2
 \ge2\pi\int_a^b\rho|m'(\rho)|^2\,d\rho
 \ge\frac{2\pi[m(a)-m(b)]^2}{\log(b/a)}.
\end{equation}
Polar integration of the $L^2$ bound supplies, separately for the two
signs, $b_j\in[\delta,2\delta]$ with $|m_j(b_j)|=o(\eps_j)$.
Fix small $c>0$ and put $a_j=c\sqrt{\eps_j}$.
Equation~\eqref{eq:vertical-obstacle} and the discarded low-mode bound
yield
\[
 m_j^\pm(a_j)\ge[1-c^2/(2R)+o_j(1)]\eps_j\pm h_j.
\]
Since $\log(b_j/a_j)=\ell_j/2+O(1)$,
\begin{equation}\label{eq:annular-lower}
 \int_{|x|<2\delta}|\nabla Z_j^\pm|^2
 \ge\frac{4\pi+o_j(1)}{\ell_j}
 \bigl([1-c^2/(2R)+o_j(1)]\eps_j\pm h_j\bigr)_+^2.
\end{equation}
Decrease the common error to cover both signs. For every $s\ge0$
and $h\in\R$, $(s+h)_+^2+(s-h)_+^2\ge2s^2$.
Thus no bound on $h_j/\eps_j$ is needed. Combining
\eqref{eq:localized} and \eqref{eq:annular-lower} gives
\[
 \liminf_j\frac{\ell_j L_R(\eps_j)}{\eps_j^2}
 \ge2\pi(1-\zeta)(1-c^2/(2R))^2.
\]
Let $c,\zeta$ decrease to zero. The upper bound already proved matches,
which proves Theorem~\ref{thm:slab}.
