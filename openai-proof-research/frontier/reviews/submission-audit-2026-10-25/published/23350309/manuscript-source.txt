\begin{abstract}
Under the sharp contraction and joining results specified below, we prove
linear stability of binary-information contraction on every compact
amplitude and correlation rectangle, uniformly over finite cube dimensions.
For information $\psi(a)$ tending to zero, the constant grows at most as
$a^{-4}$, and this exponent is necessary. The statements apply to arbitrary
$[-1,1]$-valued kernels, without a mean restriction or a pointwise interior
margin. We also strengthen the inequality $D(u)\ge F(\cI(u))$ by an
absolute positive multiple of $(\E u)^2\cI(u)^2$. The proof combines strict Gaussian
contraction, a compactness argument for joining laws, and estimates that
retain the entropy of rare endpoint values.
\end{abstract}
\maketitle

\section{Statements and dependencies}
On the uniform cube $\Omega_n=\{-1,1\}^n$, let $T_r$ be the noise
operator with Walsh multipliers $r^{|S|}$. Write
\[
 L=\log2,\quad \psi(t)=\frac{(1+t)\log(1+t)+(1-t)\log(1-t)}2,
 \quad H(t)=L-\psi(t),\quad V(t)=\atanh t,
\]
with continuous endpoint values for $\psi,H$. Define
\[
 \cI(u)=\E\psi(u)-\psi(\E u),\qquad
 d_a(u)=\min_{i,\,s\in\{-1,1\}}\E(u-saX_i)^2,
 \quad \Delta_r(u;a)=\psi(ra)-\cI(T_ru).
\]
All logarithms are natural. The inverse of $\psi$ is its nonnegative
branch. When $\cI(u)=\psi(a)$ we abbreviate the deficit to $\Delta_r(u)$.

The sharp soft-channel inequality and its local joining construction in
\cite{OpenAI} are inputs to this paper. Our conclusions are conditional on
those results at the fixed revision specified in the bibliography. Their
precise forms, including the scalar and canonical-optimizer inputs, are
listed in Section~\ref{sec:inputs}. Proof-internal identities needed for
strictness are derived there rather than included in that assumption.

\begin{theorem}[Compact amplitudes]\label{thm:compact}
Assume the inputs in Section~\ref{sec:inputs}. For
$0<a_0\le a_1<1$ and $0<r_0\le r_1<1$ there is a finite constant
$C=C(a_0,a_1,r_0,r_1)$ such that, for every finite $n\ge1$, every
$u:\Omega_n\to[-1,1]$, $a\in[a_0,a_1]$, and $r\in[r_0,r_1]$,
\[
 \cI(u)=\psi(a)\quad\Longrightarrow\quad d_a(u)\le C\Delta_r(u).
\]
The power one of the deficit cannot be replaced by a larger power.
\end{theorem}

\begin{theorem}[Small information]\label{thm:endpoint}
Under the same assumptions, for $a_1<1$ and
$0<r_0\le r_1<1$ there is $C=C(a_1,r_0,r_1)<\infty$ such that
\[
 d_a(u)\le C a^{-4}\Delta_r(u)
 \qquad(0<a\le a_1,\ r_0\le r\le r_1,\ \cI(u)=\psi(a)).
\]
The assertion is uniform in finite dimension and in the mean of $u$.
No exponent smaller than four can replace four.
\end{theorem}

For an interior-valued field define
\[
 D(u)=\frac14\sum_{i=1}^n
 \E[(u-u^{(i)})(V(u)-V(u^{(i)}))],\qquad
 F(\psi(t))=tV(t),\quad 0\le t<1.
\]
Here $u^{(i)}$ means that coordinate $i$ is flipped.

\begin{theorem}[Biased production]\label{thm:production}
Under the same inputs there is an absolute $c>0$ such that
\[
 D(u)\ge F(\cI(u))+c(\E u)^2\cI(u)^2
\]
for every finite-dimensional interior-valued field. An explicit choice
of $c$ is given in \eqref{eq:c}. For boundary-valued fields the
positive-time limit gives the corresponding extended inequality.
\end{theorem}

The stability constants are existential: the proof uses compactness and
a contradiction argument, and does not give effective values for them.
The production constant is specified by scalar functions. Theorems
\ref{thm:compact} and \ref{thm:endpoint} concern soft kernels at a fixed
information budget. Boolean stability, such as \cite{KT}, has a different
normalization and does not supply the small-information assertion here.
The joining method and sharp comparison curve are credited to
\cite{OpenAI}. The differential-equation and dimension-induction approach
appears in \cite{CGN}. An entropy-production inequality for arbitrary
arrays with a mean-dependent profile is proved in
\cite[Section~9, Lemma~9.2]{CGJLMNW}. That profile reduces to $F(\cI)$
on an interval of small information for some fixed nonzero means.
Theorem~\ref{thm:production} adds a positive multiple of
$(\E u)^2\cI(u)^2$ uniformly in that regime.
The second-variation computation along translations in
Section~\ref{sec:gauss} follows the method of
\cite[Lemmas~5.1 and~5.3]{HT}.
We also use classical hypercontractivity, Borell's two-set inequality,
and the low-influence replacement method of \cite{MOO}.

\section{Local inputs and strict margins}\label{sec:inputs}
Put $S(J)=F(J)-2J$ for $0\le J<L$, extended by zero to $J<0$, and
\[
 \cB_m(J)=2J+(1-m^2)S\left(L-\frac{H(m)-J}{1-m^2}\right),
 \qquad \Phi_m(J)=\max\{F(J),\cB_m(J)\}.
\]
The physical range is $|m|<1$, $0\le J<H(m)$.
The cost
\[
 c(x,y)=\frac{x-y}{4}(V(x)-V(y))
\]
is extended by zero at $(1,1),(-1,-1)$ and by infinity at every other
boundary pair. It is nonnegative and lower semicontinuous.
For $a=m+d>b=m-d$, define
\[
 \bar h(m,d)=\frac{H(m+d)+H(m-d)}2.
\]
\[
 \quad \cK(m,d,e)=\inf\left\{\E c(A,B):
 \E A=a,\ \E B=b,\ \frac{\E H(A)+\E H(B)}2\le e\right\}.
\]

\begin{assumption}\label{ass:inputs}
We use the following results of \cite{OpenAI}, with their stated ranges.
\begin{enumerate}
\item Theorem \texttt{main:soft}: for every finite cube and bounded
kernel, $\cI(T_ru)\le f_{|r|}(\cI(u))$, where
$f_r(J)=\psi(r\psi^{-1}(J))$. Theorem \texttt{upgrade:production}:
$D(u)\ge\Phi_{\E u}(\cI(u))$ for interior fields.
\item Lemmas \texttt{scalar:entropy-bounds},
\texttt{scalar:profile-convexity},\\ \texttt{scalar:curvature-growth}, and
\texttt{scalar:scaled-curvature}: $t^2/2\le\psi(t)\le Lt^2$,
$F(0)=0$, $F'(0)=2$, $F''\ge4/3$,
$e^{-2J}F''(J)$ strictly increases, and
\[
 \frac12\le (L-J)^2F''(J)\le M:=4L^2/3.
\]
The zero-extended $S$ is convex, nondecreasing, and $C^1$.
\item Lemma \texttt{capacity:onebit}: for $|m|+|d|<1$, if
$J=H(m)-\bar h(m,d)$ and $\alpha=1-m^2$, then
$J<\alpha L$ and $c(m+d,m-d)\ge\alpha F(J/\alpha)$.
\item Lemma \texttt{canonical:priced} and Proposition
\texttt{canonical:coverage}: for $-1<b<a<1$ and
$0<e\le\bar h(m,d)$ the cost minimum is attained with entropy exactly
$e$ by a unique priced optimizer
$k\delta_{(x,y)}+\pi_+\delta_{(1,1)}+\pi_-\delta_{(-1,-1)}$,
where $x>y$ are interior. Its core minimizes the strictly convex symmetric
priced objective in the logarithmic variables
$z=\log((1+x)/(1+y))$, $w=\log((1-y)/(1-x))$, under
$z\ge z_0=\log((1+a)/(1+b))$ and
$w\ge w_0=\log((1-b)/(1-a))$.
\item Theorems \texttt{reserve:joining} and \texttt{upgrade:joining}:
if $e_a=\E H(A)>0$, $e_b=\E H(B)>0$, $a=\E A$, $b=\E B$,
$m=(a+b)/2$, $e=(e_a+e_b)/2$, $J=H(m)-e$, and
$J_a=H(a)-e_a$, $J_b=H(b)-e_b$, then
\[
 \E c(A,B)\ge P_m(J)-\tfrac12(P_a(J_a)+P_b(J_b))
 \quad\text{for }P=\cB\text{ and }P=\Phi.
\]
\item Lemma \texttt{reserve:geometry}: for $|m|+d<1$,
\[
 (1-m)^2\log\frac{1-m^2}{1-m^2-d^2}
 \le3\kappa,\qquad
 \kappa=-\tfrac12\log(1-d^2/(1+m)^2).
\]
Lemma \texttt{thinning:window}: mix an interior core of nonnegative
center $x_c$ and half-difference $d_c>0$ with $(1,1)$, retaining mass
$\lambda\in(0,1]$. Put $x=1-\lambda(1-x_c)$, $d=\lambda d_c$,
$e=\lambda\bar h(x_c,d_c)$, $I=H(x)-e$,
$\nu=\bar h(x,d)-e$, $q=L-I$, $s=L-\nu$, $b_*=\log(1+x)$.
Then $\int_q^s(h-b_*)_+h^{-2}\,dh\le3\kappa$.
\end{enumerate}
\end{assumption}
These are the external mathematical inputs, not additional stability
hypotheses. We next derive the strict margins used later from them.

\subsection{Canonical geometries}
After reflection let $m\ge0$, so $z_0\le w_0$. Symmetry and uniqueness
of the priced minimum give $z\le w$. If $w>w_0$ and $z<w$, strict
convexity and symmetry give $(F_p)_z<(F_p)_w=0$, contradicting
minimality. Thus either $z=w$ or $w=w_0$. In the first case the core
is $(t,-t)$ and
\begin{equation}\label{eq:reflected}
 d=kt,\quad e=kH(t),\quad \cK=dV(t),\quad 0\le m\le1-k.
\end{equation}
In the second case $\pi_-=0$ and the core has center $x_c\ge0$;
$1-m\le k\le1$, $x=1-(1-a)/k$, $y=1-(1-b)/k$.
The two descriptions meet at $k=1-m$. The price-zero case is the
deterministic pair. This reproduces the classification in the proof of
\cite[Section 3]{OpenAI} and uses no stability assertion about that minimum.

\subsection{Two retained losses}
Let $\delta_0=2-3M=2-4L^2>0$. On the one-positive-corner path in
Assumption~\ref{ass:inputs}(6), put $A_* =\log(2/(1+x))$ and
$T(\lambda)=F(I)-F(\nu)$. Differentiating binary entropy gives
$\lambda I'=I-A_*$ and $\lambda\nu'=\nu-A_*-\kappa$.
Consequently
\[
 \lambda T'-T=\int_\nu^I(u-A_*)F''(u)\,du+F'(\nu)\kappa
 \ge2\kappa-M\int_q^s\frac{(h-b_*)_+}{h^2}\,dh
 \ge\delta_0\kappa.
\]
Since $\kappa\ge\lambda^2d_c^2/8$, integration of $(T/\lambda)'$
from $\lambda$ to one gives
\begin{equation}\label{eq:thinloss}
 \lambda F(J_c)-[F(I)-F(\nu)]
 \ge\frac{\delta_0}{8}\lambda(1-\lambda)d_c^2,
 \qquad J_c=H(x_c)-\bar h(x_c,d_c).
\end{equation}
The differentiation is along the full path; $A_*$ is held fixed only
inside the displayed auxiliary integral.

For either-corner addition define
\[
 R(v,e)=vS(L-e/v),\quad \alpha=1-m^2,\quad\beta=\alpha-d^2,
 \]
\[
 W(m,d,e)=2[H(m)-\bar h(m,d)]+R(\alpha,e)-R(\beta,e).
\]
Convex perspective pooling gives
$\cB_m(I)-\tfrac12(\cB_a(I_a)+\cB_b(I_b))\le W$.
On the positive-corner ray set $A=1-m$, holding $d/A,e/A$ fixed,
$\eta_*=1+d^2/A^2$, and
\[
 \gamma(h)=S(L-h)+hS'(L-h)
 =\int_{\min(h,L)}^L uF''(L-u)\,du.
\]
Differentiating $R/A$ and binary entropy gives
\[
 A W_A-W=2\kappa+A^2\{\eta_*\gamma(e/\beta)-\gamma(e/\alpha)\}.
\]
The scaled-curvature bound implies
$0\le\gamma(e/\alpha)-\gamma(e/\beta)\le M\log(\alpha/\beta)$.
The geometric input therefore gives $AW_A-W\ge\delta_0\kappa$.
These identities remain valid across the zero-extension seam because
$S$ is $C^1$. Integration, and reflection for the negative corner, yield
\begin{equation}\label{eq:reserveloss}
 \tau W(m,d,e)-W(\pm(1-\tau)+\tau m,\tau d,\tau e)
 \ge\frac{\delta_0}{8}\tau(1-\tau)d^2.
\end{equation}
This is the quantitative form of the source's corner argument.

\subsection{Scalar margins below and above half information}
Set $k_0=L-1/2$ and $\Xi(J)=(J-k_0)S'(J)-S(J)$.
The curvature-growth input and
$\int_0^{L/2}(s-k_0)e^{2s}\,ds=0$ show
\[
 \Xi(L/2)=\int_0^{L/2}(s-k_0)F''(s)\,ds>0.
\]
Also $\Xi'(J)>0$ for $J\ge L/2$.
If $c_m=H(m)-(1-m^2)L$, then $0\le c_m\le m^2k_0$.
The tangent inequality for $S$ at $J$ gives
\begin{equation}\label{eq:high}
 \cB_m(J)-F(J)\ge m^2\Xi(J)\qquad(J\ge L/2).
\end{equation}
Below half information put
\begin{equation}\label{eq:eta}
 \eta=2-\max_{0\le J\le L/2}F''(J)/F'(J)>0.
\end{equation}
To check positivity, set $h=L-J\ge L/2$. The curvature bounds give
$F'(J)\ge2+(h^{-1}-L^{-1})/2$, so
$2h^2F'(J)\ge(4-L^{-1})h^2+h\ge L^2+L/4>M\ge h^2F''(J)$.
For fixed $m,d>0$, $m+d<1$, set
$J_*=H(m)-\bar h(m,d)$ and
$A_0=(V(m+d)+V(m-d))/2$. Then
\begin{equation}\label{eq:entropyratio}
 A_0/V(m)\ge e^{2J_*}.
\end{equation}
Indeed, with $v_\pm=V(m\pm d)$ the derivative in $d$ of
$\log(A_0/V(m))-2J_*$ is
$[G(v_+)-G(v_-)]/(v_++v_-)>0$, where
$G(v)=\cosh^2v-v^2$ is even and strictly increasing for $v>0$.
At $d=0$ the expression vanishes. This also proves that
$F(H(m)-e)-F(\bar h(m,d)-e)$ strictly increases in $m$ whenever
$H(m)-e\le L/2$, by comparing \eqref{eq:eta} and
\eqref{eq:entropyratio}.

\section{Cube estimates at a fixed positive amplitude}\label{sec:compact}
Throughout this section $a\in[a_0,a_1]$ and $r\in[r_0,r_1]$, with
all four endpoints fixed as in Theorem~\ref{thm:compact}.
Constants depend only on those endpoints unless another dependence is stated.

\subsection{Elementary norm and entropy estimates}
For $|x|\le\gamma<1$ and $y\in[-1,1]$, let
$B_\psi(y,x)=\psi(y)-\psi(x)-V(x)(y-x)$. Then
\begin{equation}\label{eq:bregman}
 \tfrac12(y-x)^2\le B_\psi(y,x)\le U_\gamma(y-x)^2,
 \quad |B_\psi(x+z,x)-\tfrac12\psi''(x)z^2|
 \le C_{\gamma,p}|z|^p\quad(2<p\le3).
\end{equation}
The lower bound is $\psi''\ge1$. The other two follow from interior
Taylor expansion near $z=0$ and continuity on the remaining compact
set, including $y=\pm1$. In particular no endpoint derivative is used.
Put
\[
 p=\min\{3,1+r_1^{-2}\}>2,\qquad
 \theta=(1+r_1^2)^{-1}-\tfrac12>0.
\]
Real hypercontractivity \cite{Bonami,Beckner} gives
\begin{equation}\label{eq:hyper}
 \|T_rz\|_p\le\|z\|_2,\qquad
 \|T_rz\|_2\le P^\theta\|z\|_2
 \quad\text{if }\Pr(z\ne0)\le P.
\end{equation}
For clarity this applies to real functions, not only indicators.
A two-point proof starts with $M(t)=\E|A+tBX|^q$, $1<q\le2$:
the second derivative of $M^{2/q}$ is a nonnegative square plus
$2(q-1)B^2 M^{2/q-1}\E|A+tBX|^{q-2}\ge2(q-1)B^2$.
Jensen proves the last inequality; continuity across its integrable
zeros permits integration. Tensorization and duality give
\eqref{eq:hyper}.
Also, if $z$ has support mass $P$,
\begin{equation}\label{eq:lowproj}
 \|\Pi_{\le1}z\|_2\le C P^{1/4}\|z\|_2.
\end{equation}
Indeed degree-at-most-one polynomials of norm one have uniformly
bounded fourth moments. Duality and H\"older prove the assertion.

\subsection{The semigroup budget and return to the original kernel}
Let $P_t=T_{e^{-t}}$, $t=-\log r$, and set
\[
 y(s)=\cI(P_su),\quad w(s)=\psi(ae^{-s}),\quad z(s)=w(s)-y(s),
 \quad R(s)=D(P_su)-F(y(s)).
\]
For $s>0$, a nonconstant finite-cube kernel becomes strictly interior.
The flow identity gives $z'=R-A(s)z$, where
$2\le A(s)\le K:=\max_{[0,\psi(a_1)]}F'$ is the secant slope of $F$.
Here $R\ge0$, $z\ge0$ by the assumed inequalities. Integration from
positive time and then continuity at zero give
\begin{equation}\label{eq:budget}
 \Delta_r(u)=\int_0^t e^{-\int_s^t A(v)\,dv}R(s)\,ds,
 \quad \int_0^tR(s)\,ds\le e^{Kt}\Delta_r(u),
 \quad z(s)\le e^{Kt}\Delta_r(u).
\end{equation}
No finite production at the original endpoint-valued kernel is required.

For a coordinate oriented so that $b=\E uX_i\ge0$, strong convexity
of $\psi$ at $bX_i$ gives
\[
 \|u-bX_i\|_2^2\le2\{\psi(a)-\psi(b)+\psi(m)\},\qquad m=\E u.
\]
If $\|T_\lambda u-\lambda aX_i\|_2^2=q$, then $m^2\le q$ and
$|b-a|\le\sqrt q/\lambda$. Thus, for $\lambda\ge r_0$ and small $q$,
\begin{equation}\label{eq:return}
 \|u-aX_i\|_2^2
 \le 2V((1+a_1)/2)\sqrt q/r_0+2Lq+q/r_0^2.
\end{equation}
We used the orthogonal identity
$\|u-aX_i\|^2=\|u-bX_i\|^2+(a-b)^2$.
This is an entropy estimate, not an inverse-continuity assertion for noise.

\begin{lemma}[Local stability]\label{lem:local}
There are $\varepsilon_*,c_*>0$ such that
$\|u-saX_i\|_2\le\varepsilon_*$ and $\cI(u)=\psi(a)$ imply
$\Delta_r(u)\ge c_*\|u-saX_i\|_2^2$.
\end{lemma}
\begin{proof}
Write $v=aX_1$, $h=u-v$, $m=\E h$, and $k=rV(ra)/V(a)$.
The exact Bregman expression for
$G=\Delta_r(u)+k(\cI(u)-\psi(a))$ is
\[
 G=k\E B_\psi(v+h,v)-\E B_\psi(rv+T_rh,rv)+(1-k)\psi(m).
\]
Its quadratic form is
\[
 Q(h)=\frac{k\|h\|_2^2}{1-a^2}
 -\frac{\|T_rh\|_2^2}{1-r^2a^2}+(1-k)m^2.
\]
Every nonconstant Walsh coefficient has coefficient at least
$q=k/(1-a^2)-r^2/(1-r^2a^2)$; the constant coefficient is $a^2q$.
The function $(1-t^2)V(t)/t$ has derivative
$[t-(1+t^2)V(t)]/t^2<0$, proving $q>0$. Moreover $0<k<r^2<1$.
Consequently $Q\ge c_H\|h\|^2$, where $c_H=a_0^2\min q>0$.

Choose $\eta_0\le(1-a_1)/2$ with $D_3\eta_0/6\le c_H/8$,
where $D_3$ bounds $|\psi'''|$ on the resulting interior interval.
Split $h=g+b$ on the sets $|h|\le\eta_0$ and $|h|>\eta_0$.
The latter has mass $P\le\|h\|^2/\eta_0^2$.
Taylor expansion on the good set and strong convexity on the other give
\[
 k\E B_\psi(v+h,v)\ge
 \frac{k\|g\|^2}{2(1-a^2)}-\frac{c_H}8\|g\|^2
 +\frac{k_{\min}}2\|b\|^2.
\]
The output remainder is at most $C\|h\|^p$ by
\eqref{eq:bregman}--\eqref{eq:hyper}. Use $\psi(m)\ge m^2/2$.
The good quadratic terms equal $Q(g)/2$. With
$B_*=(1-r_1^2a_1^2)^{-1}$, the bad square costs at most
$B_*P^{2\theta}\|b\|^2/2$ and the cross terms at most
$(B_*P^\theta+\sqrt P)\|g\|\|b\|$.
Young's inequality bounds the latter by
$c_H\|g\|^2/8+2(B_*P^\theta+\sqrt P)^2\|b\|^2/c_H$.
Choose $P_*>0$ so that these two bad coefficients sum to at most
$k_{\min}/4$. Choose $\varepsilon_*$ so that
$\varepsilon_*^2/\eta_0^2\le P_*$ and
$C\varepsilon_*^{p-2}\le\min(c_H,k_{\min})/8$.
The remaining positive coefficients prove the claim. The input values
on the bad set have never been expanded.
\end{proof}

\subsection{Equality and compactness for a joining law}
For the notation in Assumption~\ref{ass:inputs}(5), write
\[
 \mathcal J(\mu)=\E c(A,B)
 +\tfrac12\{\Phi_a(J_a)+\Phi_b(J_b)\}-\Phi_m(J).
\]

\begin{lemma}\label{lem:joining}
For a law $\mu$ of finite cost with $(a-b)/2>0$, $\mathcal J(\mu)=0$
if and only if $\mu$ is a point mass at $(t,-t)$, $0<t<1$.
On the set of laws with $(a-b)/2\ge\beta>0$, $j_0\le J\le j_1<L$,
and $\E c\le C_0$, a joining surplus tending to zero forces
\[
 \tfrac12\E[(A-t)^2+(B+t)^2]\longrightarrow0,
 \qquad t=\psi^{-1}(J),
\]
uniformly over that set.
\end{lemma}
\begin{proof}
Finite cost and positive separation exclude zero child entropy: if
$H(A)=0$ almost surely, then $A$ is an endpoint, and finite cost forces
$B=A$ almost surely, contradicting separation. The same proves positive
entropy for $B$ and excludes endpoint means.
If the parent maximum is its reserve, including a tie, perspective
pooling gives $\mathcal J\ge\E c-W\ge\cK-W$.
Let its optimizer have core mass $k$ and half-difference $d_c$.
Compose \eqref{eq:reserveloss} with retentions
$\tau_1=k/(k+\pi_+)$ and $\tau_2=k+\pi_+$. The loss is at least
\[
 \frac{\delta_0d_c^2}{8}
 \frac{k\pi_++k^2\pi_-}{k+\pi_+}
 \ge\frac{\delta_0}{8}(1-k)d^2.
\]
The deterministic core is paid by the capacity inequality:
$W_c\le(1-m_c^2)F(J_c/(1-m_c^2))\le c_c$.
If $m_c\ne0$, then $Lm_c^2-\psi(m_c)>0$ by the positive entropy
series, and the first inequality is strict: the reserve argument is
strictly below $J_c/(1-m_c^2)>0$, where $S$ strictly increases.
Equality therefore forces $k=1$ and $m_c=0$.
It also forces $\E c=\cK$. Uniqueness of the priced optimizer, at
its exact entropy budget, then forces the actual law to be this point mass.

For a strict $F$ parent, $m\ne0$ and $J<L/2$ by
\eqref{eq:high}. Reflect to $m>0$.
Convexity of $F$ at the original children gives
$\mathcal J\ge\cK-[F(J)-F(j)]$, $j=(J_a+J_b)/2$.
For a one-positive-corner optimizer, \eqref{eq:thinloss} is strict
if $k<1$. If $k=1$, strict convexity of $F$ and the capacity inequality
give $c_c>F(J_c)$ because $m>0$. For a reflected optimizer,
\eqref{eq:reflected} implies $k<1$. Increase $m$ to $1-k$, keeping
$d,e,k,t$ fixed. Its information decreases, stays below $L/2$, and
the demand increases by \eqref{eq:entropyratio}. At that wall the
strict one-corner payment applies. Equality is impossible.
Conversely a point mass $(t,-t)$ has zero child information and cost
$F(\psi(t))$, so its surplus is zero.

Finally, probability laws on the fixed compact square are weakly
compact. The cost sublevel is closed by lower semicontinuity; means
and entropy moments are continuous. The constraints in the lemma thus
define a compact set. The preceding endpoint exclusion applies at every
point of it, so compactness supplies a common positive child entropy
and interior mean margin. All profile terms are consequently continuous,
even across ties and the $S$ seam; $\mathcal J$ is lower semicontinuous.
The squared distance in the statement is continuous. Its closed subset
at distance at least $\varepsilon$ has strictly positive minimum surplus,
by the equality classification. This proves the uniform assertion.
\end{proof}

\begin{lemma}[A marked coordinate]\label{lem:marked}
For each $\beta>0$, kernels with
$\max_i|\widehat u(\{i\})|\ge\beta$ satisfy
$d_a(u)\le C_\beta\Delta_r(u)$.
\end{lemma}
\begin{proof}
Split an interior field $g$ at the marked coordinate. The exact identity
$D(g)=(D(g_+)+D(g_-))/2+\E c(g_+,g_-)$ and the face production
bound give $D(g)-F(\cI(g))\ge\mathcal J$.
Also $\E c\le D(g)$. On a compact positive information interval,
small production surplus and a singleton lower bound therefore imply
arbitrarily small coordinate distance, by Lemma~\ref{lem:joining}.
Apply \eqref{eq:budget} on $[0,t]$. Some time in its middle third
has $R(s)\le3e^{Kt_{\max}}\Delta/t_{\min}$, where
$t_{\min}=-\log r_1$, $t_{\max}=-\log r_0$.
For small $\Delta$, $y(s)\in[\psi(r_0a_0)/2,\psi(a_1)]$,
and the marked singleton is at least $r_0\beta$.
The preceding compactness lemma puts $P_su$ close to the coordinate
of amplitude $\psi^{-1}(y(s))$. Equation~\eqref{eq:budget} and
Lipschitz continuity of this inverse on the positive compact interval
put that amplitude within $O(\Delta)$ of $ae^{-s}$.
Choose the distance tolerance first to enter the neighborhood in
Lemma~\ref{lem:local} via \eqref{eq:return}, then choose the production
and deficit thresholds. The local lemma gives the desired linear estimate.
Above the chosen deficit threshold the bound $d_a\le4$ suffices.
\end{proof}

\subsection{Coordinate sorting}
Sorting the two values on each coordinate fiber into increasing order
preserves the distribution of $u$, hence its mean and information.
It preserves every other singleton, and it preserves monotonicity in
coordinates already sorted. After one sweep the resulting field $g$
is increasing. Sorting does not decrease $\cI(T_ru)$: write the fiber
as $b+X_id$, and after noise in the other coordinates write
$B=T_rb$, $D=T_rd$, $D_+=T_r|d|$. Then $D_+\ge|D|$ and
\[
 f_B(q)=\tfrac12\{\psi(B+rq)+\psi(B-rq)\},\qquad f_B''(q)\ge r^2.
\]
The gain is at least $r^2\E(D_+^2-D^2)/2$.

Let $M_2=\E u^2$ and choose a coordinate with maximal final singleton
$c=\widehat g(\{i\})$. At its sorting step the initial singleton is
$b_i=\widehat u(\{i\})$ and $\E|d|=c$, because all other sorting
steps preserve that singleton. Noise spectral bounds give
\[
 \E(D_+^2-D^2)\ge(1-r^2)(c^2-b_i^2)-r^2(M_2-c^2).
\]
If $\ell=\cI(T_rg)-\cI(T_ru)$, this step's gain is at most $\ell$.
Also $d_a(g)=M_2+a^2-2ac=(M_2-c^2)+(a-c)^2$.
When $d_a(g)\le a^2/4$, $c\ge a/2$, and
\[
 d_a(u)-d_a(g)\le2a(c-|b_i|)\le4(c^2-b_i^2).
\]
Here $|b_i|\le c$, and use of a nonmaximal original singleton only
weakens the first bound. Otherwise $d_a(u)\le4\le16d_a(g)/a_0^2$.
It follows that
\begin{equation}\label{eq:compression}
 d_a(u)\le C\{d_a(g)+\ell\},\qquad
 \Delta_r(u)=\Delta_r(g)+\ell.
\end{equation}
Thus it remains to exclude increasing near-extremizers with all
singletons small. The next section supplies that exclusion and completes
Theorem~\ref{thm:compact}.
\section{Strict Gaussian contraction and the diffuse case}\label{sec:gauss}
Let $\gamma_d$ be standard Gaussian measure and
$Q_rv(x)=\E v(rx+\sqrt{1-r^2}Z)$. Define $\cI_\gamma$ by the same
entropy formula as on the cube.

\begin{lemma}\label{lem:gaussstrict}
For every measurable nonconstant $v:\R\to[-1,1]$ with
$0<\cI_\gamma(v)<L$ and every $0<r<1$,
\[
 \cI_\gamma(Q_rv)<f_r(\cI_\gamma(v)).
\]
\end{lemma}
\begin{proof}
First the non-strict inequality transfers from the assumed cube theorem.
For smooth bounded $v$ with bounded third derivative apply it to the
normalized sum $S_N=N^{-1/2}\sum X_i$. Conditional on the whole noisy
output $Y=y$, the centered input summands have variance $(1-r^2)/N$
and third absolute moments $O(N^{-3/2})$. Successive Gaussian replacement,
with second-order Taylor cancellation, gives
\[
 \sup_y|T_r(v\circ S_N)(y)-Q_rv(S_N(y))|
 \le C_r\|v'''\|_\infty N^{-1/2}.
\]
The usual central limit theorem and uniform continuity of $\psi$ now
transfer both input and output information. Gaussian smoothing approximates
any bounded measurable kernel in $L^2$, and $Q_r$ is a contraction,
so the non-strict bound follows for every such kernel. Independent blocks
give the same statement in each fixed Gaussian dimension.

We need the following scalar inequality. If $0<a,r<1$,
$k=rV(ra)/V(a)$, and $b=e^{\psi(a)}-1$, then
\begin{equation}\label{eq:scalarstrict}
 0<k<r^2<1,\qquad k>\frac{r^2(1-b)}{1-r^2b}.
\end{equation}
Here are details of the strict comparison. Put
$M_0(a)=[a/(1-a^2)-V(a)]/[a/(1-a^2)+V(a)]$ and
\[
 K_0(a)=\psi(a)+\log[a+(1-a^2)V(a)]-\log(2a).
\]
The latter has zero limits at both endpoints. Its derivative has the sign
of $1-t\coth t+t^2\operatorname{sech}^2t$, where $a=\tanh t$.
The partial fraction for $\coth$ \cite[4.36.3]{DLMF} gives
\[
 \frac{t\coth t-1}{t^2\operatorname{sech}^2t}
 =2\sum_{j\ge1}\frac{\cosh^2t}{t^2+\pi^2j^2}.
\]
Each summand strictly increases, since
$\tanh t>t/(1+t^2)>t/(t^2+\pi^2j^2)$; the first inequality follows
by differentiating $(1+t^2)\sinh t-t\cosh t$.
The sum increases from $1/3$ to infinity. Thus $K_0$ first increases
and then decreases, so $K_0(a)>0$, equivalently $b>M_0(a)$.
Moreover $M_0(z)/z^2$ increases: its numerator and denominator series
have coefficients $(2j+2)/(2j+3)$ and $(2j+2)/(2j+1)$, whose ratios
strictly increase. Pairing the quotient-rule numerator over $j>i$
proves that sign directly. If $A(x)=V(a\sqrt x)/(a\sqrt x)$, then
\[
 \frac{A'(x)}{A(x)+xA'(x)}=\frac{M_0(a\sqrt x)}x\le M_0(a)<b.
\]
Hence $(1-bx)A(x)$ decreases; comparison at $r^2$ and one proves
\eqref{eq:scalarstrict}. Also $f_r'>0$ and $f_r''<0$, because
$aV'(a)/V(a)=\sinh(2t)/(2t)$ strictly increases.

Suppose first that $w=Q_\tau v$, $0<\tau<1$, is nonconstant and
attains equality. Write $h=Q_rw$, $m=\E w$, $\cI_\gamma(w)=\psi(a)$.
The likelihood ratio of a binary output of conditional mean $w$ relative
to its product marginals is $(1+sw)/(1+sm)$, $s=\pm1$.
Its product second moment is $1+q$, where
$q=\Var(w)/(1-m^2)$. Jensen under the joint measure gives
\begin{equation}\label{eq:likelihood}
 \cI_\gamma(w)\le\log(1+q),\qquad q\ge b.
\end{equation}

We record the regularity needed for the variation. Both $1\pm v$ have
positive integrals on some finite interval. Their Gaussian convolutions
therefore satisfy $1\pm w(x)\ge c\exp(-Cx^2)$; hence
$|V(w(x))|\le C(1+x^2)$. All ordinary derivatives of $w$ are bounded
by differentiating the convolution kernel. With $A_s=1+sw$ and
$g_s=1+sv$, its score formula is
$A_s'=(\tau/\sqrt{1-\tau^2})\E[Zg_s(\tau x+\sqrt{1-\tau^2}Z)]$.
Weighted Cauchy--Schwarz, averaged over $s=\pm1$, gives
\begin{equation}\label{eq:fisher}
 \frac{w'^2}{1-w^2}\le\frac{\tau^2}{1-\tau^2}.
\end{equation}
For the tilted density $g_s/A_s$ its entropy is at most
$\log(2/A_s)$. The Gaussian entropy variational inequality
$\lambda\E_{\rm tilt}Z-\lambda^2/2\le\operatorname{Ent}(g_s/A_s)$,
which follows by comparison with the exponential density, bounds the
tilted mean by $\sqrt{2\log(2/A_s)}$.
Consequently $|(V\circ w)'(x)|\le C(1+|x|)$.
The same bounds hold for $h$.

The nonnegative functional
$\mathcal G(v)=f_r(\cI_\gamma(v))-\cI_\gamma(Q_rv)$ vanishes at $w$.
A smooth compactly supported perturbation $\phi$ obeys
$|\phi|\le C(1\pm w)$; positivity of $Q_r$ gives
$|Q_r\phi|\le C(1\pm h)$. Thus perturbations of both signs are
admissible and dominated differentiation of entropy is justified by the
quadratic growth above. The first variation and self-adjointness give
\begin{equation}\label{eq:euler}
 Q_rV(h)=kV(w)+(1-k)V(m).
\end{equation}
The terms are continuous, so this identity is pointwise.
Differentiation, the commutation $(Q_rf)'=rQ_rf'$, and multiplication
by $w'$ give
\begin{equation}\label{eq:energygauss}
 r^2\E\frac{(Q_rw')^2}{1-h^2}=k\mathcal D,
 \qquad \mathcal D=\E\frac{w'^2}{1-w^2}>0.
\end{equation}
All integrals and the self-adjoint interchange are legitimate: the
composed derivative of $V(h)$ has linear growth, $w'$ is bounded,
and \eqref{eq:fisher} controls the Fisher integrals.

Translate $w_c(x)=w(x+c)$. Its noisy output is $h(x+c/r)$.
The same bounds, locally uniformly in $c$, allow two entropy derivatives.
Put $A=\E[(V(w)-V(m))w']$. The linear terms involving $w''$
cancel by \eqref{eq:euler}, leaving
\[
 \frac{d^2}{dc^2}\mathcal G(w_c)\bigg|_{c=0}
 =k\mathcal D+f_r''(\psi(a))A^2
 -\E\frac{(Q_rw')^2}{1-h^2}
 +(1-k)\frac{(\E w')^2}{1-m^2}.
\]
Since $(\E w')^2\le\mathcal D(1-m^2)(1-q)$,
\eqref{eq:energygauss}, \eqref{eq:likelihood}, and strict concavity give
\[
 \mathcal G''(0)\le\mathcal D
 \{k(1-r^{-2})+(1-k)(1-b)\}<0
\]
by \eqref{eq:scalarstrict}. This contradicts minimality.
For an arbitrary equality kernel choose $r<\tau<1$, $s=r/\tau$.
The non-strict inequality and the identity $f_s\circ f_\tau=f_r$ force
$Q_\tau v$ to attain equality at $s$, with positive information below
$L$. This is impossible by the smoothed case.
\end{proof}

\begin{corollary}\label{cor:gaussgap}
For every compact rectangle $0<b_0\le b\le b_1<1$,
$0<\sigma_0\le\sigma\le\sigma_1<1$, the deficit
$\psi(\sigma b)-\cI_\gamma(Q_\sigma v)$ has a strictly positive
lower bound over all finite Gaussian dimensions and bounded kernels
with $\cI_\gamma(v)=\psi(b)$.
\end{corollary}
\begin{proof}
Put $f=(1+v)/2$ and let $f^*$ be its increasing scalar Gaussian quantile.
For an output set $A$ of measure $p$, layer cake and Borell's two-set
inequality \cite[Theorem 1.1]{MN} give
\[
 \int_A Q_\sigma f\,d\gamma
 \le\int_{A^*}Q_\sigma f^*\,d\gamma,
\]
where $A^*$ is the upper ray of measure $p$. The level sets of the same
$f^*$ are nested rays, so all comparisons are simultaneous.
Taking largest-set integrals and then the supremum over $p$ of their
value minus $cp$ gives the stop-loss inequalities for every $c$.
Equal means and convex piecewise-linear approximation give convex order.
Apply it to $t\mapsto\psi(2t-1)$: the rearrangement preserves input
information and does not decrease output information.
The class of bounded nondecreasing scalar kernels is compact in
$L^2(\gamma)$ by Helly selection and dominated convergence.
Information is continuous there. Joint continuity of $Q_\sigma$ in
kernel and correlation follows from contraction and the Hermite
multipliers for the fixed limit kernel. The constrained rectangle is
compact, and its continuous deficit is positive by
Lemma~\ref{lem:gaussstrict}. Its minimum proves the claim.
\end{proof}

\subsection{Transfer from increasing cube kernels}
Suppose increasing kernels $g_j$, possibly in changing dimensions,
satisfy $\Delta_{r_j}(g_j)\to0$ while all singletons are bounded by
$\alpha_j\to0$. Fix $r_1<\lambda<1$, put $h_j=T_\lambda g_j$,
and $\sigma_j=r_j/\lambda$. Equation~\eqref{eq:budget} gives
\[
 \cI(h_j)=\psi(\lambda a_j)+o(1),\qquad
 \cI(T_{\sigma_j}h_j)=\psi(r_ja_j)+o(1).
\]
For a fixed degree $d$, let $p_j=\Pi_{\le d}h_j$ and use the same
multilinear polynomial on Gaussian variables. Its cube tail has norm
at most $\lambda^{d+1}$.
Monotonicity gives $0\le\partial_i h_j\le1$, so
\[
 \operatorname{Inf}_i(p_j)\le\E(\partial_i h_j)^2
 \le\E\partial_i h_j=\lambda\widehat g_j(\{i\})\le\alpha_j,
 \qquad \sum_i\operatorname{Inf}_i(p_j)\le d.
\]
For a $C^4$ test function with bounded fourth derivative, replacing one
sign at a time by a Gaussian cancels its first three moments. The
fourth-order error sums to
$C_d\|\varphi^{(4)}\|_\infty\sum_i\operatorname{Inf}_i(p_j)^2
\le C_d d\alpha_j\|\varphi^{(4)}\|_\infty$.
Here the needed mixed-product fourth moment is
$\E B^4\le9^{d-1}(\E B^2)^2$ for the derivative polynomial $B$;
it follows by tensorizing the two-point/Gaussian inequality
$\E(A+XB/\sqrt3)^4\le(A^2+B^2)^2$.
This is the fixed-degree replacement argument of \cite{MOO}, applied
also to $q_j=Q_{\sigma_j}p_j$, whose coefficients are multiplied by
$\sigma_j^{|S|}$.

The bounded tests $\clip(t)$ and $\psi(\clip(t))$, where clipping is
to $[-1,1]$, are handled by uniform smooth approximation.
For the unbounded test $D(t)=\dist(t,[-1,1])^2$, mollify at radius
$\tau$. Its fourth derivative is bounded for each $\tau>0$, while
$|D_\tau(t)-D(t)|\le C\tau(1+|t|)+C\tau^2$.
The second moments of the polynomials equal their coefficient norms
and are at most one under both measures. Thus first $j\to\infty$,
then $\tau\downarrow0$, transfers this test as well.
Under cube measure $\E D(p_j)\le\lambda^{2d+2}$, hence the same
Gaussian expectation is at most $\lambda^{2d+2}+o(1)$.
Set $v_j=\clip(p_j)$ in Gaussian space. Since $Q_{\sigma_j}v_j$ is
bounded, contraction and the metric projection property give
\[
 \|v_j-p_j\|_2\le\lambda^{d+1}+o(1),\qquad
 \|Q_{\sigma_j}v_j-\clip(q_j)\|_2\le\|v_j-p_j\|_2.
\]
Together with the bounded test comparisons and uniform entropy
continuity, these transfer both information budgets with an error
that tends to zero as $d\to\infty$, uniformly after $j\to\infty$.
Choose the Gaussian gap in Corollary~\ref{cor:gaussgap} on a slightly
larger compact rectangle around $\lambda[a_0,a_1]$ and
$[r_0/\lambda,r_1/\lambda]$ first. Then choose $d$ so all tail errors
are smaller than a fixed fraction of that gap, and finally choose $j$.
At the exact amplitude $b_j=\psi^{-1}(\cI_\gamma(v_j))$, the transferred
deficit contradicts that gap.

Therefore there are $\beta,\varepsilon>0$ such that every increasing
kernel with deficit below $\varepsilon$ has a singleton at least $\beta$.
Lemma~\ref{lem:marked} handles those kernels; above the threshold use
$d_a\le4$. Sorting and \eqref{eq:compression} prove
Theorem~\ref{thm:compact}.

\subsection{Sharpness of the deficit power}
Fix $a,r\in(0,1)$ and use the two-dimensional kernel
$u_\epsilon=c(\epsilon)X_1+\epsilon X_2$, normalized by
$\cI(u_\epsilon)=\psi(a)$. The implicit function theorem gives
\[
 c(\epsilon)=a-\frac{\epsilon^2}{2(1-a^2)V(a)}+O(\epsilon^4).
\]
It is interior for small $\epsilon$, has distance
$d_a=\epsilon^2+O(\epsilon^4)$, and
\[
 \Delta_r(u_\epsilon)=\frac12
 \left\{\frac{rV(ra)}{(1-a^2)V(a)}
             -\frac{r^2}{1-r^2a^2}\right\}\epsilon^2+O(\epsilon^4).
\]
The coefficient is positive by Lemma~\ref{lem:local}.
Thus no larger power of the deficit can give a uniform stability bound.
\section{Small information with an interior mean}\label{sec:small}
We now fix only $a_1<1$ and the correlation interval $[r_0,r_1]$.
Write the orthogonal decomposition
\[
 u=m+\ell+W,\qquad \ell=\sum_i c_iX_i,\quad
 s^2=\|\ell\|_2^2,\quad E_{\ge2}=\|W\|_2^2,
 \quad b=\max_i|c_i|.
\]
For a maximizing coordinate let
$B_0=\|u-bX_i\|_2^2=m^2+s^2-b^2+E_{\ge2}$.
Conditional Jensen and the convexity in $m$ of
$[\psi(m+b)+\psi(m-b)]/2-\psi(m)$ show $b\le a$.
The linear Bregman term at $bX_i$ cancels, so
\begin{equation}\label{eq:modal}
 B_0\le d_a=B_0+(a-b)^2\le(1+2U_{a_1})B_0,
 \qquad s^2+E_{\ge2}=\Var(u)\le2\psi(a)\le C a^2.
\end{equation}
Indeed $(a-b)^2\le2(\psi(a)-\psi(b))$ and
$\psi(a)-\psi(b)\le\E B_\psi(u,bX_i)\le U_{a_1}B_0$.

\begin{proposition}\label{prop:interiormean}
For every $m_0<1$ there exist $a_*,c_*>0$ such that
$|m|\le m_0$, $0<a\le a_*$, and $\cI(u)=\psi(a)$ imply
\[
 \Delta_r(u)\ge c_* a^4 B_0.
\]
The constants do not depend on dimension or an interior margin for $u$.
\end{proposition}
We prove it in three steps. All errors and constants in those steps
are uniform in dimension, $|m|\le m_0$, and the fixed correlation range.

\subsection{Control of the original higher modes}
For a first-chaos polynomial of norm $s$, the inequality
$\E e^{t\ell}\le e^{t^2s^2/2}$ follows from $\cosh x\le e^{x^2/2}$.
It gives $\|\ell\|_j\le C_js$ and, for fixed $\eta_0>0$,
\begin{equation}\label{eq:cliptail}
 \|\ell-C_{\eta_0}(\ell)\|_j\le C_j e^{-c_j/s^2},
\end{equation}
where $C_\eta$ clips to $[-\eta,\eta]$. At $s=0$ this error is zero.
Choose $\eta_0$ small relative to $1-m_0$ and put
$v=m+C_{\eta_0}(\ell)$, $t=\ell-C_{\eta_0}(\ell)$,
$e=u-v=W+t$. Symmetry gives $\E v=m$, $\E e=0$.
Both $v,T_rv$ lie in a fixed interior interval.
Let $E_a$ denote an error bounded by $Ce^{-c/a^2}$, with constants
allowed to change between occurrences.

The curve $f_r$ is concave by the scalar calculation above. Its tangent
at $\psi(a)$, with $k=rV(ra)/V(a)$, gives for any bounded comparison
field $v$ of amplitude $\beta=\psi^{-1}(\cI(v))$
\begin{equation}\label{eq:tangent}
 \Delta_r(u)\ge\Delta_r(v;\beta)
 +k[\cI(u)-\cI(v)]-[\cI(T_ru)-\cI(T_rv)].
\end{equation}
There is no ordering condition on the two input informations.
For the clipped comparison above the linear Bregman contribution is
\[
 C_{\rm lin}=k\E[V(v)e]-\E[V(T_rv)T_re],\qquad
 |C_{\rm lin}|\le Ca^2\sqrt{E_{\ge2}}+E_a.
\]
To see the bound, expand $V(m+z)=V(m)+z/(1-m^2)+O(z^2)$.
The constant terms vanish, and the first-chaos terms against $W$
vanish by orthogonality and self-adjointness. Moment bounds control the
quadratic remainder by $Ca^2\|W\|$, and \eqref{eq:cliptail} handles $t$.

We claim that for small $a$
\begin{equation}\label{eq:entrancepayment}
 k\E B_\psi(u,v)-\E B_\psi(T_ru,T_rv)
 \ge cE_{\ge2}-E_a.
\end{equation}
Here are details retaining large input residuals. Split $e=G+Z$ at a
small fixed threshold $\eta_1$. The bad support mass satisfies
$P\le Ca^2/\eta_1^2+E_a$. On the good set the input quadratic
coefficient can be made arbitrarily close to $1/[2(1-m^2)]$ by choosing
$\eta_0,\eta_1$ small. On the bad set keep the global payment
$k\|Z\|^2/2$. The output has the same approximate quadratic coefficient
and a remainder at most $C\|e\|^p$ by \eqref{eq:hyper}.
Since $k=r^2+O(a^2)$, the constant-coefficient quadratic form has a
uniform positive gap on degrees at least two:
$(r^2-r^4)/[2(1-m^2)]$.
Its low-degree block has bounded norm. By \eqref{eq:lowproj},
$\|\Pi_{\le1}G\|^2\le CP^{1/2}\|Z\|^2+E_a$, because
$\Pi_{\le1}e=\Pi_{\le1}t$.
The bad output square is bounded by $CP^{2\theta}\|Z\|^2$,
and its cross term by $CP^\theta\|G\|\|Z\|$.
Young's inequality absorbs this into a fixed fraction of the good gap
and $CP^{2\theta}\|Z\|^2$.
Finally $\|e\|^p\le Ca^{p-2}(\|G\|^2+\|Z\|^2)+E_a$.
As $a\to0$, every bad coefficient tends to zero and can be absorbed
into the retained input payment. This proves \eqref{eq:entrancepayment}.
Use $\Delta_r(v;\beta)\ge0$ in \eqref{eq:tangent}. We obtain
\begin{equation}\label{eq:entrance}
 \Delta_r(u)\ge cE_{\ge2}-Ca^2\sqrt{E_{\ge2}}-E_a,
 \qquad E_{\ge2}\le C(\Delta_r(u)+a^4).
\end{equation}

\subsection{A quartic expansion retaining bad entropy}
Assume now $E_{\ge2}\le K_0a^4$, with fixed $K_0$.
Retain the preceding $v,e,t$ and split $e=G+Z$ at the shrinking threshold
$\tau=a^{1/2}$. Its bad mass is $P\le C_{K_0}a^3+E_a$.
Set $h=1-m^2$ and
\[
 A_{\rm bad}=\E[B_\psi(u,v)\mathbf1_{\{|e|>\tau\}}].
\]
Then
\[
 \tfrac12\|Z\|^2\le A_{\rm bad}\le U_\gamma\|Z\|^2,
 \quad \|T_rZ\|=o(a^2),\quad \|\Pi_{\le1}G\|=o(a^2),
 \quad |\E\ell^2Z|\le Ca^2P^{1/4}\|Z\|=o(a^4).
\]
These follow from \eqref{eq:bregman}, \eqref{eq:hyper},
\eqref{eq:lowproj}, and fourth/eighth moment bounds for $\ell$.
Taylor expansion only on the good residual set gives
$\E[B_\psi(u,v)\mathbf1_{\rm good}]=\|G\|^2/(2h)+o(a^4)$.
For its multiplier, first restrict to $|\ell|\le a^{1/2}$; the variation
is $O(a^{1/2})$. The complement has mass exponentially small in $1/a$,
and $|G|\le\tau$ controls its contribution. The cubic residual cost
is $O(\tau\|G\|^2)$.
At the output the remainder is $O(a^{2p})=o(a^4)$, while H\"older
bounds multiplier variation by
$C\|T_rv-m\|_{p/(p-2)}\|T_re\|_p^2=O(a^5)+E_a$.
The first-chaos linear terms cancel for the full $e=W+t$ before the
bad part is removed.

Writing $C_4=(1+3m^2)/(12h^3)$, these estimates give
\begin{align}
 \psi(a)&=\frac{s^2}{2h}+C_4\E\ell^4+\frac m{h^2}\E\ell^2G
       +\frac{\|G\|^2}{2h}+A_{\rm bad}+o(a^4),\label{eq:quarticin}\\
 \cI(T_ru)&=\frac{r^2s^2}{2h}+C_4r^4\E\ell^4
       +\frac{mr^2}{h^2}\E[\ell^2T_rG]
       +\frac{\|T_rG\|^2}{2h}+o(a^4).\label{eq:quarticout}
\end{align}
For example, expanding $V(m+\ell)$ through order two leaves an
$L^2$ remainder $O(a^3)$ paired with an $O(a^2)$ residual.
Equation~\eqref{eq:cliptail} pays all clipping terms, and the preceding
bound on $\E\ell^2Z$ removes the bad cubic coupling. This proves
uniformity of these errors without a pointwise assumption on $W$.

Equation~\eqref{eq:quarticin} gives $s^2=ha^2+O(a^4)$.
For $\kappa=\sum_i c_i^4/s^4$ and $G_j=\Pi_jG$, eliminate $s^2/(2h)$
and complete the level-two square to obtain
\begin{align}
 \Delta_r(u)={}&r^2(1-r^2)a^4
       \left\{\frac{\kappa m^2}{3h}+\frac{1-\kappa}{6}\right\}
   +\frac{r^2(1-r^2)}{2h}
       \left\|G_2+\frac m h(\ell^2-s^2)\right\|^2\notag\\
 &+\sum_{j\ge3}\frac{r^2-r^{2j}}{2h}\|G_j\|^2
       +r^2A_{\rm bad}+o(a^4).\label{eq:normal}
\end{align}
Indeed $\E\ell^4=s^4+2(s^4-\sum c_i^4)$, and square completion
subtracts $r^2(1-r^2)m^2\,2(s^4-\sum c_i^4)/(2h^3)$.
Substitute $s^4=h^2a^4+O(a^6)$ and
$\psi(ra)-r^2\psi(a)=-r^2(1-r^2)a^4/12+O(a^6)$ to verify the
bracket. It is at least $(m^2+1-\kappa)/12$.
Hence $\Delta=o(a^4)$ forces
\begin{equation}\label{eq:coneentrance}
 m\to0,\quad\kappa\to1,\quad E_{\ge2}=o(a^4),\quad
 s^2-b^2=o(a^2),\quad b^2/(ha^2)\to1.
\end{equation}
For the higher energy use $A_{\rm bad}\ge\|Z\|^2/2$, the negligible
low projection of $G$, and
$\|(m/h)(\ell^2-s^2)\|=o(a^2)$ in \eqref{eq:normal}.
Also $\kappa\le b^2/s^2$. The absolute error $o(a^4)$ is used only
for this entrance statement.

\subsection{Relative coercivity near a coordinate}
There are small fixed $\mu,\varepsilon,\delta,a_c>0$ such that, if
\begin{equation}\label{eq:cone}
 |m|\le\mu,\quad \ell_0=\sum_{i\ne1}c_iX_i,\quad
 l_0=\|\ell_0\|\le\varepsilon a,\quad
 |b^2/((1-m^2)a^2)-1|\le\delta,\quad a\le a_c,
\end{equation}
with $c_1=b\ge0$, then
\begin{equation}\label{eq:relative}
 \Delta_r(u)\ge c(a^4m^2+a^2l_0^2+E_{\ge2}).
\end{equation}
To prove it, take the exact conditional field $v=m+bX_1$ and
$A=v+C_\eta(\ell_0)$ for fixed small $\eta$.
Put $t=\ell_0-C_\eta(\ell_0)$, $e=u-A=W+t$.
Choose the constants so that $|A|,|T_rA|\le\gamma$ with
$\gamma^2<1-r_1^2$.
All fixed-norm clipping errors are bounded by
$E_l=Ce^{-c/l_0^2}$, interpreted as zero when $l_0=0$.
Symmetry and independence of $X_1$ cancel the linear $v$-to-$A$ terms.
Equation~\eqref{eq:tangent} becomes
\begin{align}
 \Delta_r(u)\ge{}&\Delta_r(v;\beta)
 +k\E B_\psi(A,v)-\E B_\psi(T_rA,T_rv)\notag\\
 &+k\E[V(A)e]-\E[V(T_rA)T_re]\notag\\
 &+k\E B_\psi(u,A)-\E B_\psi(T_ru,T_rA),\label{eq:conedecomp}
\end{align}
where $\beta=\psi^{-1}(\cI(v))$ and $k=f_r'(\psi(a))$.

For this genuine one-bit field, capacity and convexity of
$F(J)-(2/3)J^2$ give
$D(v)-F(\cI(v))\ge(2/3)m^2\cI(v)^2$.
Along its noise flow, \eqref{eq:budget} applies.
If the terminal gap is at least half $\psi(r\beta)$ it bounds
$c m^2\beta^4$ immediately. Otherwise its information at every earlier
time is at least its terminal value, at least $c\beta^2$;
integrating the production surplus gives the same bound.
From \eqref{eq:cone}, $\beta\asymp a$. Thus
\begin{equation}\label{eq:onebitpay}
 \Delta_r(v;\beta)\ge c m^2a^4.
\end{equation}

The cubic terms in the next line of \eqref{eq:conedecomp}'s first row
average to zero, since the clipped first chaos is symmetric and independent
of $X_1$. Its fourth moments are $O(l_0^4)$. Consequently this contribution
is $(Q_{11}/2)l_0^2+O(l_0^4)+O(E_l)$, where
\[
 Q_{11}=k\E(1-v^2)^{-1}-r^2\E(1-(T_rv)^2)^{-1}
 =r^2(1-r^2)\left\{\frac{(1+3m^2)b^2}{(1-m^2)^3}
                 -\frac{a^2}{3(1-m^2)}\right\}+O(a^4).
\]
Here $k=r^2-r^2(1-r^2)a^2/3+O(a^4)$; all expansions are analytic
on the fixed interior interval. At $m=0$, $b^2=a^2$, the leading
coefficient is $2r^2(1-r^2)/3$. Choose $\mu,\delta,a_c$ small and
then $\varepsilon$ small to obtain
\begin{equation}\label{eq:otherpay}
 k\E B_\psi(A,v)-\E B_\psi(T_rA,T_rv)
 \ge ca^2l_0^2-CE_l.
\end{equation}

Strong convexity at the input, \eqref{eq:bregman} at the output,
and the degree-at-least-two spectral gap give for the last row
\begin{align}
 k\E B_\psi(u,A)-\E B_\psi(T_ru,T_rA)
 &\ge\left\{\frac k2-\frac{r^4}{2(1-\gamma^2)}\right\}E_{\ge2}
       -C E_{\ge2}^{p/2}-CE_l\notag\\
 &\ge cE_{\ge2}-CE_l.\label{eq:higherpay}
\end{align}
In the first line a fixed arbitrarily small fraction of the positive
coefficient may be spent on cross terms involving $t$; its exponentially
small norm is included in $CE_l$. The coefficient remains positive by
our choice of $\gamma$, and $E_{\ge2}\le Ca^2$ absorbs the remainder.

Finally expand $V(v+\ell_0)$ through order two in $\ell_0$ while
keeping its full dependence on $v$. Conditional orthogonality removes
$V(v)W$ and the even-in-$X_1$ part of $\psi''(v)\ell_0 W$.
The odd coefficient is at most $C|m|b$, and
$|\psi'''(v)|\le C(|m|+b)$. The cubic remainder has norm $O(l_0^3)$.
The identical calculation after noise, using self-adjointness, gives
\begin{equation}\label{eq:coupling}
 |k\E[V(A)e]-\E[V(T_rA)T_re]|
 \le C\{|m|a l_0+(|m|+a)l_0^2+l_0^3\}\sqrt{E_{\ge2}}+CE_l.
\end{equation}
Young's inequality spends half of \eqref{eq:higherpay}; its other cost
is bounded by
$C(\mu^2+\varepsilon^2+\mu^2\varepsilon^2+\varepsilon^4)a^2l_0^2$,
which is absorbed in \eqref{eq:otherpay} by decreasing $\mu,\varepsilon$.
For $l_0\le\varepsilon a$,
$E_l/(a^2l_0^2)\le C\varepsilon^2e^{-c/l_0^2}/l_0^4\to0$
uniformly as $a\to0$; the error is exactly zero when $l_0=0$.
Thus the clipping costs are relative even if $B_0$ is exceptionally
small. Combining \eqref{eq:conedecomp}--\eqref{eq:coupling} proves
\eqref{eq:relative}.

\begin{proof}[Proof of Proposition~\ref{prop:interiormean}]
Failure would give $a_j\to0$ and $B_{0,j}>0$ with
$\Delta_j/(a_j^4B_{0,j})\to0$. Since $B_{0,j}\le m_0^2+Ca_j^2$,
we have $\Delta_j=o(a_j^4)$. Equation~\eqref{eq:entrance} gives
$E_{\ge2}=O(a_j^4)$, and \eqref{eq:normal} implies
\eqref{eq:coneentrance}. Choose the largest singleton and orient it.
The fixed conditions \eqref{eq:cone} eventually hold. Then
\eqref{eq:relative} gives
$\Delta_j\ge c(a_j^4m_j^2+a_j^2l_{0,j}^2+E_{\ge2,j})
\ge ca_j^4B_{0,j}$, a contradiction. If $B_0=0$, \eqref{eq:modal}
already gives $d_a=0$.
\end{proof}
\section{Uniform biased production and completion of the endpoint bound}
\label{sec:production}
We strengthen the joining potential rather than applying a one-bit
inequality to a general field. With $\eta$ and $\Xi$ from
Section~\ref{sec:inputs}, put
\begin{equation}\label{eq:c}
 \begin{gathered}
\mu=4/3,\quad A_1=\mu/2,\quad B_1=\delta_0/8,\quad
 C_1=4L^2+3L/2,\\
 c=\tfrac12\min\{\mu/8,B_1/C_1,\eta,\Xi(L/2)/L^2\}>0.
\end{gathered}
\end{equation}
This explicit scalar formula suffices; no numerical optimization is used.
Define $G(m,J)=F(J)+cm^2J^2$ and
$\cP_m(J)=\max\{\cB_m(J),G(m,J)\}$.
The Hessian of $G$, in variables $(m,J)$, has entries
$2cJ^2$, $4cmJ$, $F''(J)+2cm^2$, and determinant
$2cJ^2(F''(J)-6cm^2)\ge0$. Thus $G$ is jointly convex on the
full rectangle $(-1,1)\times[0,L)$.
By \eqref{eq:high}, $\cB_m\ge G(m,\cdot)$ above half information;
at $m=0$ they agree. A strict $G$ parent therefore has
$m\ne0$ and $J<L/2$.

\subsection{The one-positive-corner payment}
Take the core and path of \eqref{eq:thinloss}. Write $x_c\ge0$,
$\sigma_c=1-x_c$, $\alpha_c=1-x_c^2$, and let $J_c$ be its information.
The capacity inequality and convexity of $F(J)-\mu J^2/2$ give
\begin{equation}\label{eq:corebias}
 c_c-F(J_c)\ge A_1(\alpha_c^{-1}-1)J_c^2
 \ge A_1x_c^2J_c^2.
\end{equation}
The first inequality follows by applying the perspective inequality to
that convex function, which vanishes at zero.

To bound the added demand, interpret the mixture using a fair input
$X$ and an independent flag of probability $\lambda$ choosing the core.
If $U$ is its binary output, then $I=I(U;X,\mathrm{flag})$,
$\nu=I(U;\mathrm{flag}\mid X)$, and $J=I-\nu=I(U;X)$.
The chain rule and data processing give
\[
 I=\lambda J_c+E_\lambda,\quad J\le\lambda J_c,
 \quad E_\lambda=h_{\rm bin}(\lambda\sigma_c/2)
                       -\lambda h_{\rm bin}(\sigma_c/2).
\]
For $g(t)=-(1-t)\log(1-t)$, $-2\le g''\le-1$ on $[0,1/2]$;
convexity of $g(t)+t^2$ implies
$0\le g(\lambda p)-\lambda g(p)\le\lambda(1-\lambda)p^2$.
Separating the term $-t\log t$ gives
\[
 0\le E_\lambda\le\tfrac34\lambda\sigma_c\log(1/\lambda).
\]
The core posterior identity, obtained by expanding binary information,
is
\[
 J_c=\frac{1+x_c}{2}\psi\!\left(\frac{d_c}{1+x_c}\right)
     +\frac{1-x_c}{2}\psi\!\left(\frac{d_c}{1-x_c}\right).
\]
Hence $J_c\le Ld_c^2/[\sigma_c(1+x_c)]$,
$\sigma_cJ_c\le Ld_c^2$, and $J_c^2\le L^2d_c^2$.
Since $x=x_c+(1-\lambda)\sigma_c$,
\begin{align}
 x^2(I^2-\nu^2)&=x^2J(2I-J)\notag\\
 &\le2\lambda^2x^2J_c^2
       +\tfrac32\lambda^2x^2\sigma_cJ_c\log(1/\lambda)\notag\\
 &\le4\lambda x_c^2J_c^2+C_1\lambda(1-\lambda)d_c^2.\label{eq:added}
\end{align}
For the last line use $x^2\le2x_c^2+2(1-\lambda)^2$,
$\lambda\log(1/\lambda)\le1-\lambda$, and $x^2\le1$ in the logarithmic
term. Combining \eqref{eq:thinloss}, \eqref{eq:corebias}, and
\eqref{eq:added}, with $4c\le A_1$ and $cC_1\le B_1$, proves
\begin{equation}\label{eq:newcorner}
 \lambda c_c\ge G(x,I)-G(x,\nu)
\end{equation}
for every $0<\lambda\le1$.

\subsection{Reflected cores and original child budgets}
Hold $d,e$ fixed and suppose $m>0$,
$I=H(m)-e\le L/2$, $j=\bar h(m,d)-e\ge0$, $J=I-j>0$.
Put $W_0=V(m)$ and $A_0=(V(m+d)+V(m-d))/2$.
Equations~\eqref{eq:eta}--\eqref{eq:entropyratio} imply
\[
 A_0F'(j)-W_0F'(I)\ge2\eta W_0J.
\]
Indeed $F'(I)/F'(j)\le e^{(2-\eta)J}$ and $F'(j)\ge2$.
Differentiate the upgraded demand to get
\[
 \partial_m[G(m,I)-G(m,j)]
 =A_0F'(j)-W_0F'(I)+2cm(I^2-j^2)+2cm^2(jA_0-IW_0).
\]
The middle term is nonnegative. Since $A_0\ge W_0$,
$jA_0-IW_0\ge-JW_0$, so the derivative is at least
$2(\eta-c)W_0J>0$.
For the reflected optimizer \eqref{eq:reflected}, increase $m$ to
$1-k$, preserving its cost, entropy, and separation. All states remain
physical and $I$ decreases. The demand increases up to the
one-positive-corner wall, where \eqref{eq:newcorner} pays it.
Thus that inequality, with the exact canonical cost on the left,
holds for both optimizer geometries whenever the $G$ parent is strict.

Now take arbitrary original child states with positive entropies as in
Assumption~\ref{ass:inputs}(5). If the parent $\cP$ is its reserve,
the assumed reserve joining inequality pays it, because the children
$\cP$ dominate their reserves. Otherwise reflect to $m>0$ and put
$d=|a-b|/2$, $j=(J_a+J_b)/2$. If $d=0$, $I=j$ and joint convexity
of $G$ suffices. If $d>0$, the actual pair law is feasible for
$\cK(m,d,e)$, so the two preceding payments give
\[
 \E c(A,B)\ge\cK(m,d,e)\ge G(m,I)-G(m,j).
\]
Joint convexity at the original individual child budgets gives
\[
 \tfrac12\{\cP_a(J_a)+\cP_b(J_b)\}
 \ge\tfrac12\{G(a,J_a)+G(b,J_b)\}\ge G(m,j).
\]
This proves the full joining inequality for $\cP$ without replacing
any individual child entropy by a canonical one.
At dimension zero, $\cP_m(0)=0$, since $H(m)\ge L(1-m^2)$.
Induction using the exact production split therefore yields
$D(u)\ge\cP_m(\cI(u))\ge G(m,\cI(u))$ for every interior field.
This proves Theorem~\ref{thm:production}. For nonconstant boundary-valued
fields, apply it at positive noise time and let time decrease to zero;
the finite edge costs converge to their extended values. Constant endpoint
fields have zero information and production.

\subsection{All means and sharpness at zero information}
Let $K=\max_{[0,\psi(a_1)]}F'$. For any exactly normalized original
kernel, the semigroup budget \eqref{eq:budget} and the new production
bound give $R(s)\ge cm^2y(s)^2$ at positive time.
If $\Delta\ge\psi(ra)/2$, then $\Delta\ge r_0^2a^2/4$.
Otherwise monotonicity of information gives
$y(s)\ge y(t)>\psi(ra)/2\ge r_0^2a^2/4$ throughout $[0,t]$.
Integration yields
\begin{equation}\label{eq:uniformbias}
 \Delta_r(u)\ge\frac{c m^2r_0^4a^4}{16K}(1-r_1^K).
\end{equation}
Thus $|m|\ge1/2$ implies $\Delta\ge c_ba^4$ in either case.
Since $d_a=\E u^2+a^2-2ab\le2$, this proves the endpoint theorem
uniformly for these means, including limits $|m|\to1$.
For $|m|\le1/2$ and small $a$, use
Proposition~\ref{prop:interiormean} and \eqref{eq:modal}.
For the remaining amplitudes bounded away from zero, use
Theorem~\ref{thm:compact}. Taking the maximum of the fixed constants
proves Theorem~\ref{thm:endpoint}.

Finally fix $0<|m|<1$ and choose the one-bit field
$u_a=m+b(a)X_1$ with exact information $\psi(a)$.
The analytic inverse at zero gives, with $h=1-m^2$,
\[
 b(a)^2=ha^2-\tfrac23m^2a^4+O(a^6),\qquad
 \Delta_r(u_a)=\frac{r^2(1-r^2)m^2}{3h}a^4+O(a^6).
\]
These follow by substituting
$\cI(m+bX)=b^2/(2h)+(1+3m^2)b^4/(12h^3)+O(b^6)$
at $b$ and $rb$. The kernel is interior for small $a$ and
$d_a(u_a)=m^2+(a-b(a))^2\to m^2>0$.
An $a^{-p}$ bound with $p<4$ would have right side tending to zero,
which is impossible.
