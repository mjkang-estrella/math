\begin{abstract}
Let $T$ be a regular tetrahedron and let $A_{\min}(T)$ be the least area
of an orthogonal projection of $T$ onto a plane. We prove that every
finite cover of $T$ by cylinders with measurable perpendicular bases
$B_i$ satisfies $\sum_i|B_i|\ge (\sqrt{15}/9)A_{\min}(T)$.
The proof computes the largest line integral of a radial quadratic
density on $T$. We also prove that $\sqrt{15}/9$ is the best constant
obtainable from any nonnegative polynomial density of degree at most
two by dividing its mass by a uniform bound on its line integrals.
This optimality concerns the density method within that class, rather
than the cylinder-covering infimum.
\end{abstract}

\section{Introduction}
A cylinder in $\R^3$ is a set $B+\R u$, where $u$ is a unit vector
and $B\subset u^\perp$ is Lebesgue measurable. Its cost is the planar
area $|B|$. For a convex body $K$, put
\[
 A_{\min}(K)=\min_{|u|=1}|\operatorname{proj}_{u^\perp}K|.
\]
The directions of different cylinders in a cover need not agree.

Bezdek and Litvak proved that every finite cylinder cover of a
three-dimensional convex body satisfies
\[
 \sum_i\frac{|B_i|}{|\operatorname{proj}_{u_i^\perp}K|}\ge\frac13,
\]
and that the right side can be replaced by $1$ for ellipsoids
\cite[Theorem~3.1]{BL09}. In particular, the total base area is at least
$A_{\min}(K)/3$. Their later work treats multiple covers as well as
packings \cite{BL16}. For a regular tetrahedron of edge length $e$,
the inscribed ball has radius $e/\sqrt{24}$. The ellipsoid result
therefore gives the stronger consequence
\begin{equation}\label{eq:old}
 \sum_i|B_i|\ge\frac{\pi e^2}{24}
   =\frac{\pi}{6\sqrt2}A_{\min}(T).
\end{equation}
The next theorem improves this consequence by the factor
$2\sqrt{30}/(3\pi)$, approximately $1.162$.

\begin{theorem}\label{thm:cover}
Let $T\subset\R^3$ be a regular tetrahedron of edge length $e>0$.
Suppose
\[
 T\subset\bigcup_{i=1}^N(B_i+\R u_i),
\]
where $N<\infty$, each $u_i$ is a unit vector, and each
$B_i\subset u_i^\perp$ is Lebesgue measurable with finite area. Then
\begin{equation}\label{eq:main}
 \sum_{i=1}^N|B_i|\ge\frac{\sqrt{15}}9 A_{\min}(T)
             =\frac{e^2\sqrt{30}}{36}.
\end{equation}
\end{theorem}

The bases may be unbounded, disconnected, or nonconvex. The proof
also applies when the cylinders cover $T$ up to a volume-null set.

For a nonzero polynomial $q$ of degree at most two that is nonnegative
on $T$, define
\begin{equation}\label{eq:certificate}
 \mathcal C_T(q)=
 \frac{\displaystyle\int_Tq(x)\,dx}
 {\displaystyle A_{\min}(T)\sup_{\ell}
       \int_{\ell\cap T}q(x)\,ds},
\end{equation}
where the supremum is over all affine lines in $\R^3$ and $ds$ is
arclength. The denominator is positive and finite.

\begin{theorem}\label{thm:optimal}
For a regular tetrahedron $T$,
\[
 \max_{\substack{q\not\equiv0,\ \deg q\le2\\q\ge0\text{ on }T}}
           \mathcal C_T(q)=\frac{\sqrt{15}}9.
\]
For the edge-two tetrahedron with centroid at the origin, a maximizing
density is $q(x)=19/10-|x|^2$.
\end{theorem}

The density argument itself is classical; it is used in the ellipsoid
proof of \cite{BL09}. Our calculation identifies the exact maximum
line integral for a family of quadratic densities and then optimizes
over all quadratic polynomials. The theorem does not determine the
covering infimum or the best constant from arbitrary densities.
It concerns total base area normalized by the minimum projection area;
it does not assert the same improvement for the direction-dependent
ratios in the Bezdek--Litvak theorem.

\section{Tetrahedral geometry and weighted chords}
Normalize the edge length to $2$ and the centroid to $0$. Write the
vertices as $v_1,\ldots,v_4$. Then
\begin{equation}\label{eq:gram}
 |v_i|^2=\frac32,\qquad v_i\cdot v_j=-\frac12\quad(i\ne j),
 \qquad\sum_i v_i=0.
\end{equation}
For barycentric coordinates $\lambda_i\ge0$ with $\sum_i\lambda_i=1$,
\begin{equation}\label{eq:bary}
 \left|\sum_i\lambda_i v_i\right|^2
       =2\sum_i\lambda_i^2-\frac12.
\end{equation}

\begin{lemma}\label{lem:data}
In this normalization,
\[
 \vol(T)=\frac{2\sqrt2}{3},\qquad
 A_{\min}(T)=\sqrt2,\qquad
 \frac1{\vol(T)}\int_T|x|^2\,dx=\frac3{10}.
\]
\end{lemma}
\begin{proof}
The Gram matrix of the three edges from a vertex has diagonal entries
$4$ and off-diagonal entries $2$. Its determinant is $32$, giving
the volume $\sqrt{32}/6=2\sqrt2/3$.

To compute a projection area, let $d_i=D_u\lambda_i$ for a unit vector
$u$. Then $\sum_i d_i=0$ and $u=\sum_i d_i v_i$. If $F_i$ is the face
opposite $v_i$, its outward unit normal is
$-\nabla\lambda_i/|\nabla\lambda_i|$, and
$|F_i|=3\vol(T)|\nabla\lambda_i|$. Summing the projected areas of
the entering faces gives
\[
 |\operatorname{proj}_{u^\perp}T|
       =3\vol(T)\sum_{d_i>0}d_i.
\]
Put $s=\sum_{d_i>0}d_i$. The positive and negative coefficients express
$u=s(p-q)$ for points $p,q\in T$. Since the diameter is $2$, we have
$s\ge1/2$. Equality holds for $u=(v_i-v_j)/2$, proving the projection
formula.

Under uniform volume on $T$, each $\lambda_i$ has density
$3(1-t)^2$ on $[0,1]$. Thus its second moment is $1/10$.
Equation~\eqref{eq:bary} now gives the last identity.
\end{proof}

For $a\ge3/2$, the function $a-|x|^2$ is nonnegative on $T$. Put
\[
 M(a)=\sup_\ell\int_{\ell\cap T}(a-|x|^2)\,ds.
\]

\begin{proposition}\label{prop:chord}
For every $a\ge3/2$,
\begin{equation}\label{eq:Ma}
 M(a)=
 \begin{cases}
 \dfrac23(a+\tfrac12)^{3/2},&\dfrac32\le a\le\dfrac72,\\[3pt]
 2(a-\tfrac56),&a\ge\dfrac72.
 \end{cases}
\end{equation}
\end{proposition}
\begin{proof}
Let a nondegenerate chord have endpoints $p,q$, midpoint $m$, and
length $l$. Integration along the segment gives
\begin{equation}\label{eq:integral}
 \int_{[p,q]}(a-|x|^2)\,ds
       =l\left(a-|m|^2-\frac{l^2}{12}\right).
\end{equation}
This identity does not require $m$ to be perpendicular to the chord
direction. Write $p=\sum_i r_iv_i$ and $q=\sum_i s_iv_i$ in
barycentric coordinates. By \eqref{eq:gram},
\begin{equation}\label{eq:midpoint}
 |m|^2-\frac{l^2}{4}=p\cdot q
              =2\sum_i r_i s_i-\frac12\ge-\frac12.
\end{equation}
Together with $|m|^2\ge0$ and $l\le2$, this bounds
\eqref{eq:integral} by
\begin{equation}\label{eq:envelope}
 \begin{cases}
 l(a-l^2/12),&0\le l\le\sqrt2,\\
 l(a+1/2-l^2/3),&\sqrt2\le l\le2.
 \end{cases}
\end{equation}
The first expression is increasing on its interval. The derivative of
the second is $a+1/2-l^2$, so the largest value in
\eqref{eq:envelope} is the right side of \eqref{eq:Ma}.

To attain this bound, take endpoints on opposite edges with
barycentric coordinates
\[
 (z,1-z,0,0),\qquad(0,0,z,1-z),\qquad\frac12\le z\le1.
\]
Their supports are disjoint, so equality holds in
\eqref{eq:midpoint}, and
\[
 l^2=2+8(z-\tfrac12)^2.
\]
These chords realize every length in $[\sqrt2,2]$. Each segment is
the full intersection of its line with $T$, since extending past an
endpoint makes a barycentric coordinate negative. Thus a maximizing
length is always realized, including the edge case $l=2$.
Empty intersections and singletons have zero line integral.
\end{proof}

\section{The covering inequality and quadratic optimality}
\begin{proof}[Proof of Theorem~\ref{thm:cover}]
Continue with the edge-two normalization and extend
$f(x)=(19/10-|x|^2)\mathbf1_T(x)$ by zero outside $T$.
Lemma~\ref{lem:data} and Proposition~\ref{prop:chord} give
\[
 \int_{\R^3}f(x)\,dx=\frac{16\sqrt2}{15},\qquad
 \sup_\ell\int_\ell f\,ds=\frac{16\sqrt{15}}{25}.
\]
For any cylinder $B+\R u$, Tonelli's theorem in orthogonal
coordinates yields
\[
 \int_{B+\R u}f(x)\,dx
   =\int_B\int_\R f(b+tu)\,dt\,db
   \le\frac{16\sqrt{15}}{25}|B|.
\]
Lebesgue-measurable bases cause no completion issue: replacing a base
by a Borel representative changes its cylinder only on a countable
union of products of a planar null set with bounded intervals.
Nonnegativity and the covering hypothesis imply
\[
 \frac{16\sqrt2}{15}
   \le\sum_i\int_{B_i+\R u_i}f(x)\,dx
   \le\frac{16\sqrt{15}}{25}\sum_i|B_i|.
\]
This gives $\sum_i|B_i|\ge\sqrt{30}/9$.
A similarity of ratio $e/2$ multiplies all areas by $e^2/4$,
which proves \eqref{eq:main}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:optimal}]
The ratio \eqref{eq:certificate} is invariant under similarities of
$T$, with the density transported by the similarity. We again use the
edge-two normalization. If $q$ is admissible, it has positive mass:
otherwise its continuity and nonnegativity would make it zero on the
interior of $T$, hence identically zero. Its supremum line integral
is positive and at most $2\max_Tq$.

Average $q$ over the finite orthogonal symmetry group $G$ of $T$:
\[
 \overline q(x)=\frac1{|G|}\sum_{g\in G}q(gx).
\]
This preserves mass, nonnegativity, and degree. Every line integral
of $\overline q$ is an average of line integrals of $q$, so
$\mathcal C_T(q)\le\mathcal C_T(\overline q)$.
In coordinates with vertices
\[
 (1,1,1)/\sqrt2,\quad(1,-1,-1)/\sqrt2,\quad
 (-1,1,-1)/\sqrt2,\quad(-1,-1,1)/\sqrt2,
\]
the symmetries include coordinate permutations and changes of sign
of any two coordinates. Invariance eliminates linear and mixed
quadratic terms and equates the diagonal coefficients. Consequently
$\overline q(x)=\alpha+\beta|x|^2$.

If $\beta\ge0$, nonnegativity at $0$ gives $\alpha\ge0$.
An edge chord has length $2$ and mean squared radius $5/6$.
It follows that
\[
 \mathcal C_T(\overline q)
 \le\frac{\alpha+(3/10)\beta}{3\alpha+(5/2)\beta}\le\frac13.
\]
If $\beta<0$, positive rescaling reduces to
$\overline q(x)=a-|x|^2$, where $a\ge3/2$.
Lemma~\ref{lem:data} and \eqref{eq:Ma} give
\[
 \mathcal C_T(a-|x|^2)=
 \begin{cases}
 (a-3/10)/(a+1/2)^{3/2},&3/2\le a\le7/2,\\
 (a-3/10)/(3a-5/2),&a\ge7/2.
 \end{cases}
\]
The first derivative is
$(19/20-a/2)/(a+1/2)^{5/2}$; the second is
$(-8/5)/(3a-5/2)^2$. Thus the maximum occurs at $a=19/10$,
where its value is $\sqrt{15}/9>1/3$. This proves both the upper
bound and its attainment. No uniqueness claim for nonsymmetric
maximizers is needed.
\end{proof}
