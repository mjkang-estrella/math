\title{Fourth-moment comparison of zero-field SK\texorpdfstring{\\}{ }free-energy fluctuations}
\author{Myeongjin Kang}
\date{October 17, 2026}
\begin{document}
\maketitle
\begin{abstract}
We compare zero-field Sherrington--Kirkpatrick log partition functions with independent, possibly nonidentically distributed, centered variance-one couplings under a uniform fourth-moment bound. The mean error is bounded independently of system size, the variance error is bounded by a constant times one plus the Gaussian standard deviation, and bounded smooth tests at scale $a\ge1$ have error $O(a^{-1})$. Third moments need not vanish. The cubic term in coordinate replacement is controlled by a second replacement over a reflected vertex star. An expansion of the square and a finite family of hybrid laws give the variance estimate without fifth moments. Assuming the explicitly identified Gaussian low-temperature theorem, these comparisons transfer its exact variance prefactor and its limiting law with each model's own mean and variance. The Gaussian theorem is an input, not a result proved here.
\end{abstract}

\section{Statement, inputs and antecedents}
For $n\ge2$, put
\[
 E_n=\{\{i,j\}:1\le i<j\le n\},\qquad m=|E_n|=\frac{n(n-1)}2,
 \qquad q=\frac\beta{\sqrt n}.
\]
For $x\in\R^{E_n}$ define the counting-prior, zero-field log partition function
\begin{equation}\label{eq:model}
 F(x)=\log\sum_{\sigma\in\{-1,1\}^n}
       \exp\!\left(q\sum_{e=\{i,j\}\in E_n}x_e\sigma_i\sigma_j\right).
\end{equation}
Let $g=(g_e)$ have independent standard Gaussian coordinates and write
$c_G=\E F(g)$ and $V_G=\Var F(g)$. Throughout, the temperature is fixed.

\begin{theorem}[Finite-size comparison]\label{thm:comparison}
For every $\beta>0$ and $K\ge1$ there is $C=C(\beta,K)<\infty$ such that the following holds for every $n\ge2$. Let $(J_e)_{e\in E_n}$ be independent real variables with
\begin{equation}\label{eq:moments}
 \E J_e=0,\qquad \E J_e^2=1,\qquad \E|J_e|^4\le K.
\end{equation}
Then
\begin{align}
 |\E F(J)-c_G|&\le C,\label{eq:mean}\\
 |\Var F(J)-V_G|&\le C(1+\sqrt{V_G}).\label{eq:variance}
\end{align}
For every $a\ge1$, $c\in\R$, and real $\phi\in C^6(\R)$ whose derivatives of orders zero through six are bounded,
\begin{equation}\label{eq:test}
 \left|\E\phi\!\left(\frac{F(J)-c}{a}\right)
          -\E\phi\!\left(\frac{F(g)-c}{a}\right)\right|
 \le\frac Ca\sum_{r=1}^6\norm{\phi^{(r)}}_\infty.
\end{equation}
The same constant is independent of $n$, the individual coordinate laws, $a$, $c$ and $\phi$. No symmetry, identical-distribution, zero-third-moment, fifth-moment or tail-shape assumption is required.
\end{theorem}

The finite-size theorem is proved below without an upstream probabilistic theorem. Its low-temperature consequence uses a separate Gaussian input. We state that input explicitly rather than deriving it from the fluctuation exponent alone.

\begin{proposition}[Assumed Gaussian input]\label{prop:gaussian}
Accept the limiting-law theorem and moment theorem of the pinned manuscript \cite{GaussianLaw}, at the revision specified in the bibliography. For every fixed $\beta>1$, they supply $a_2(\beta)>0$ and a probability law $\nu_\beta$ such that
\begin{equation}\label{eq:gaussian}
 \frac{V_G}{n^{1/3}}\longrightarrow a_2(\beta),\qquad
 \frac{F(g)-c_G}{\sqrt{V_G}}\Longrightarrow\nu_\beta
\end{equation}
as $n\to\infty$ through all integers. The law has mean zero and variance one.
\end{proposition}

The model in \cite[\texttt{eq:model}]{GaussianLaw} is exactly \eqref{eq:model}. That source's moment theorem concerns
\[
 F_n^c=F(g)-n\log2+\zeta_n,\qquad
 Y_n=n^{-1/6}(F_n^c-\E F_n^c),
\]
where $\zeta_n$ is independent centered Gaussian with variance $\beta^2/2$. Its second-moment limit $a_2>0$ implies
\[
 n^{-1/3}V_G=\Var Y_n-\frac{\beta^2}{2n^{1/3}}\longrightarrow a_2.
\]
Its limiting-law theorem uses the uncompleted model's exact mean and variance; the completion-removal argument is also given in \cite[\texttt{red:prop-moments}]{GaussianReduction}. The companion \cite{GaussianScale} supplies only the exponent $n^{1/3+o(1)}$, which would not by itself identify the prefactor or limiting law. These upstream assertions are stipulated inputs here; their proofs are not audited or reproduced.

\begin{corollary}[Conditional low-temperature transfer]\label{cor:limit}
Assume Proposition~\ref{prop:gaussian}. For every fixed $\beta>1$ and every triangular array of independent edge families satisfying \eqref{eq:moments} with one finite uniform $K$,
\begin{equation}\label{eq:limit}
 \frac{\Var F(J)}{n^{1/3}}\longrightarrow a_2(\beta),\qquad
 \frac{F(J)-\E F(J)}{\sqrt{\Var F(J)}}\Longrightarrow\nu_\beta.
\end{equation}
The denominator is positive for all sufficiently large $n$, and convergence is through all integers. No coupling between different sizes is required.
\end{corollary}

\paragraph{Contribution and antecedents.}
Carmona--Hu \cite{CH} use approximate integration by parts and interpolation to compare free energies; their Section~4.2 obtains an absolute $O(1)$ comparison with a vanishing third moment and a finite fourth moment. Coordinate replacement and bounded-spin log-partition derivative estimates are classical; see Chatterjee \cite{Chatterjee}. Cheng--Liu--Shao--Xu \cite[Proposition~3.2]{CLSX} give a particularly close scaled smooth-test comparison under matching of the first three moments.

There is an earlier conceptual precedent for controlling a mismatched cubic coefficient by an extra cancellation in its expectation. In edge Green-function comparison, Erd\H{o}s--Yau--Yin \cite[end of Section~6.1 and Lemmas~6.5--6.6]{EYY} expand after deleting a matrix row and column; local-law estimates isolate a centered linear term independent of the minor. That principle is credited here. Our implementation for SK uses deterministic gauge oddness and comparison with a reflected product law, together with uniform Gibbs cumulant bounds. It does not import a resolvent expansion or local-law estimate.

The contribution considered here is the integrated comparison \eqref{eq:mean}--\eqref{eq:test} with unrestricted third moments. Gauge symmetry and ordinary matched-moment replacement are antecedents. The square-remainder identity and finite-maximum bootstrap are a technical part of the argument, not a separate contribution. The law and prefactor in Corollary~\ref{cor:limit} are inherited from the Gaussian input. This is a bounded comparison with the primary operations cited above, not a worldwide priority assertion. No external-field, varying-temperature, critical-window or ground-state extension is claimed.

\section{Derivatives and the vertex gauge}
Write $s_e(\sigma)=\sigma_i\sigma_j$ for $e=\{i,j\}$. For an ordered list $\boldsymbol e=(e_1,\ldots,e_k)$, repetitions allowed, let $D_{\boldsymbol e}$ denote the corresponding partial derivative.

\begin{lemma}[Uniform mixed derivatives]\label{lem:derivatives}
For each $k\ge1$ there is $B_k<\infty$, depending only on $k$, with
\begin{equation}\label{eq:derivative}
 |D_{\boldsymbol e}F(x)|\le B_kq^k\qquad(x\in\R^{E_n}).
\end{equation}
In particular $|\partial_e F|\le q$. For $U(x)=\phi((F(x)-c)/a)$ as in Theorem~\ref{thm:comparison}, put $A_\phi=\sum_{r=1}^6\norm{\phi^{(r)}}_\infty$. Then
\begin{equation}\label{eq:chain}
 |D_{\boldsymbol e}U(x)|\le C_kq^k a^{-1}A_\phi,
 \qquad 1\le k\le6.
\end{equation}
\end{lemma}
\begin{proof}
Under the finite Gibbs law at $x$, $D_{\boldsymbol e}F$ is $q^k$ times the joint cumulant of $s_{e_1},\ldots,s_{e_k}$. Its exact partition formula is
\[
 \kappa(s_1,\ldots,s_k)
 =\sum_{\pi\in\mathcal P([k])}(|\pi|-1)!(-1)^{|\pi|-1}
       \prod_{A\in\pi}\E_x\prod_{r\in A}s_r.
\]
Every spin product has absolute value one, so take
$B_k=\sum_{\pi\in\mathcal P([k])}(|\pi|-1)!$.
No distinctness of the indices is used. The first derivative is the Gibbs mean of $qs_e$, giving its sharper bound $q$.

For the composition, the multivariate chain rule gives
\[
 D_{\boldsymbol e}U
 =\sum_{\pi\in\mathcal P([k])}a^{-|\pi|}
   \phi^{(|\pi|)}((F-c)/a)
   \prod_{A\in\pi}D_{(e_r:r\in A)}F.
\]
The product has total derivative order $k$. Since $a\ge1$ and $|\pi|\ge1$, $a^{-|\pi|}\le a^{-1}$, proving \eqref{eq:chain} with no dependence on $c$.
\end{proof}

The finite sum also gives the elementary growth bound
\begin{equation}\label{eq:growth}
 |F(x)|\le n\log2+q\sum_{e\in E_n}|x_e|.
\end{equation}
Thus every product law used below has finite first and second free-energy moments.

Fix a vertex $i$ and let $T_i$ negate all incident edge coordinates. Negating $\sigma_i$ in the spin sum gives
\begin{equation}\label{eq:gauge}
 F(T_i x)=F(x).
\end{equation}
This is a deterministic identity, not invariance of an asymmetric disorder law. It holds also for $U=\Psi(F)$. For $e=\{i,j\}$, let $S$ be the other incident edges, so $|S|=n-2$. Differentiate \eqref{eq:gauge} three times in $x_e$ before setting $x_e=0$. For
\[
 h(x_{E_n\setminus\{e\}})=\left.\partial_e^3U(x)\right|_{x_e=0}
\]
we obtain
\begin{equation}\label{eq:odd}
 h(-x_S,x_{E_n\setminus(S\cup\{e\})})
 =-h(x_S,x_{E_n\setminus(S\cup\{e\})}).
\end{equation}
For $n=2$ the star is empty, and this identity states $h=0$.

\section{Two replacements: means and bounded tests}
Put $K_0=\max(K,3)$. Consider product laws that select independently at each edge one of
\begin{equation}\label{eq:hybrid-laws}
 \operatorname{law}(J_e),\quad\operatorname{law}(-J_e),
 \quad N(0,1).
\end{equation}
Their coordinate fourth moments are at most $K_0$, third absolute moments at most $K_0^{3/4}$, and first absolute moments at most one. All means are zero and variances one.

In this section let $U=F$ with $L=1$, or let $U=\phi((F-c)/a)$ with $L=a^{-1}A_\phi$. The derivative bound in either case is $C_kq^kL$ through order six.

\begin{lemma}[Reflected-star comparison]\label{lem:star}
Under every product law from \eqref{eq:hybrid-laws}, with $e$ deleted,
\begin{equation}\label{eq:star}
 \left|\E\left.\partial_e^3U\right|_{x_e=0}\right|
 \le C n q^6 L,
\end{equation}
where $C$ depends only on $K$ and absolute derivative constants.
\end{lemma}
\begin{proof}
Keep the exterior coordinates fixed. Compare the star product law $P$ with the product law $P^-$ in which each star coordinate is reflected, replacing the coordinates of $S$ one at a time. For a replaced coordinate $f$, Taylor-expand $h$ at $x_f=0$ through degree two. Its remainder is at most
\[
 Cq^6L|x_f|^3,
\]
because $\partial_f^3h$ is a mixed sixth derivative. The first two moments of a law and its reflection agree, so the polynomial terms cancel. Each replacement costs at most $Cq^6LK_0^{3/4}$ after expectation. Averaging the exterior does not change this bound. The endpoints have opposite expectations by \eqref{eq:odd}; dividing their difference by two proves \eqref{eq:star}. Every intermediate law is a product law. In particular, no shared random sign is used to create dependent coordinates.
\end{proof}

\begin{proof}[Proof of \eqref{eq:mean} and \eqref{eq:test}]
Enumerate the edges and replace any product law in \eqref{eq:hybrid-laws} by the Gaussian law one coordinate at a time. At edge $e$, condition on the remaining coordinates and write $z=x_e$. Taylor expansion gives
\[
 U(z)=U(0)+z\partial_eU(0)+\frac{z^2}{2}\partial_e^2U(0)
       +\frac{z^3}{6}\partial_e^3U(0)+R(z),
 \qquad |R(z)|\le Cq^4L|z|^4.
\]
The constant, linear and quadratic terms cancel. The cubic difference is the third-moment difference times the expectation of the deleted-edge cubic derivative, divided by six: the replaced variable is independent of that coefficient. Lemma~\ref{lem:star} and the moment bounds show that one replacement costs at most
$CL(q^4+nq^6)$. Exactly,
\begin{equation}\label{eq:counts}
 mq^4=\frac{\beta^4(n-1)}{2n}\le\frac{\beta^4}{2},
 \qquad mnq^6=\frac{\beta^6(n-1)}{2n}\le\frac{\beta^6}{2}.
\end{equation}
Summing proves a bound $C(\beta,K)L$. For $U=F$, integrability follows from \eqref{eq:growth}. The mean comparison is proved uniformly for every hybrid law, which will be used below. For the test observable it is exactly \eqref{eq:test}. The sixth derivative in the inner comparison is integrated against a third power of its own coordinate; it is not a sixth-moment assumption.
\end{proof}

\section{Variance with only fourth moments}
\subsection{A finite maximum and deleted coordinates}
Let $\Hyb$ be the finite class of all product laws in \eqref{eq:hybrid-laws}, with at most $3^m$ members. It is closed under every outer replacement and star reflection used above. Define
\begin{equation}\label{eq:M}
 M=\max_{P\in\Hyb}\E_P(F-c_G)^2.
\end{equation}
This maximum is finite before any variance estimate: \eqref{eq:growth} and finite second moments suffice. No bound depending on $3^m$ will be used.

If $r\le2$ coordinates of a hybrid are set to zero, restore them independently with compatible laws from $\Hyb$. The restored array has a law in $\Hyb$, so coordinate Lipschitz continuity and Cauchy--Schwarz give
\begin{equation}\label{eq:restore}
 \E|F_{\mathrm{deleted}}-c_G|\le\sqrt M+r q.
\end{equation}
Indeed, pointwise the left-hand absolute value is at most the restored absolute value plus $q$ times the sum of the restored coordinate magnitudes. This estimate is unconditional, after the exterior coordinates are averaged. We do not claim a conditional bound by $\sqrt M$ at each fixed exterior realization.

\subsection{The square remainder}
Fix an edge $e$ and its exterior coordinates. Write
\[
 H(z)=F(x_{e}=z)-c_G=H_0+u(z),\qquad u(0)=0,
 \qquad b_j=u^{(j)}(0).
\]
All derivatives of positive order satisfy $|u^{(k)}|\le B_kq^k$, and $|b_1|\le q$. Taylor's theorem gives
\begin{equation}\label{eq:u-remainder}
 \left|u(z)-b_1z-\frac{b_2z^2}{2}-\frac{b_3z^3}{6}\right|
 \le Cq^4|z|^4.
\end{equation}
The following exact identity avoids a fifth moment:
\begin{align}\label{eq:square}
 u(z)^2-b_1^2z^2-b_1b_2z^3
 &=2b_1z\left(u(z)-b_1z-\frac{b_2z^2}{2}\right)
   +(u(z)-b_1z)^2.
\end{align}
The bracket in the first term is $O(q^3|z|^3)$, and the last bracket is $O(q^2|z|^2)$. Thus the absolute value of \eqref{eq:square} is at most $Cq^4|z|^4$. Expanding $H^2=H_0^2+2H_0u+u^2$ gives its degree-three Taylor polynomial with remainder
\begin{equation}\label{eq:square-rem}
 |R_{\mathrm{sq}}(z)|\le Cq^4|z|^4(|H_0|+1).
\end{equation}
Under a product law, $H_0$ is independent of $z$. By \eqref{eq:restore} with one deletion,
\begin{equation}\label{eq:square-rem-mean}
 \E|R_{\mathrm{sq}}(z)|\le Cq^4K_0(\sqrt M+q+1).
\end{equation}
This argument does not bound a fourth derivative of $H^2$ at a random interpolation point by $|H_0|+q|z|$ and then multiply it by $|z|^4$; that route would unnecessarily introduce a fifth moment.

\subsection{The cubic square coefficient}
With $e$ set to zero, put
\begin{equation}\label{eq:B}
 B=\partial_e^3[(F-c_G)^2]
   =2(F-c_G)\partial_e^3F+6(\partial_eF)(\partial_e^2F).
\end{equation}
Equation~\eqref{eq:odd} applies to this observable too. To control its star replacement, set $H=F-c_G$ before restriction. For any other incident edge $f$ the full product formula is
\begin{equation}\label{eq:mixed-square}
 \partial_e^3\partial_f^3(H^2)
 =\sum_{a,b=0}^3\binom3a\binom3b
   (\partial_e^a\partial_f^bH)
   (\partial_e^{3-a}\partial_f^{3-b}H).
\end{equation}
Only the pairs $(a,b)=(0,0),(3,3)$ retain an undifferentiated $H$; they contribute $2H\partial_e^3\partial_f^3F$. Every other term is a product of positive-order derivatives with total order six. Lemma~\ref{lem:derivatives} therefore gives
\begin{equation}\label{eq:B-derivative}
 |\partial_f^3B|\le Cq^6(|F-c_G|+1),
\end{equation}
including the repeated occurrences of $e$ and $f$.

For the inner Taylor variable $z=x_f$ with $e$ still zero, Lipschitz continuity gives
$|F(z)-c_G|\le|F(0)-c_G|+q|z|$. The order-two integral Taylor remainder for $B$ is consequently at most
\begin{equation}\label{eq:B-rem}
 Cq^6\bigl[(|F(0)-c_G|+1)|z|^3+q|z|^4\bigr].
\end{equation}
Here $F(0)$ has exactly $e$ and $f$ deleted and is independent of $z$. Use \eqref{eq:restore} with two deletions, then the third and fourth moment bounds. A comparison of the law of $z$ with its reflection cancels the polynomial through degree two and costs at most
\begin{align*}
 Cq^6\bigl[(\sqrt M+2q+1)K_0^{3/4}+qK_0\bigr]
 \le C_{\beta,K}q^6(\sqrt M+1),
\end{align*}
since $q\le\beta/\sqrt2$. At each step the remaining coordinates have independent hybrid laws. The coefficients of the canceled polynomial have at most linear growth in the twice-deleted $|F(0)-c_G|$, so their expectations are finite. Formula~\eqref{eq:B-rem} is integrable as well; conditional cancellation may therefore be averaged legitimately.

The two endpoint expectations of $B$ are opposite by gauge oddness. Telescoping the $n-2$ star coordinates gives
\begin{equation}\label{eq:B-mean}
 |\E B|\le C_{\beta,K}nq^6(\sqrt M+1).
\end{equation}
Only a third and a fourth moment have been used in this inner replacement.

\subsection{Closing the bootstrap}
\begin{proof}[Proof of \eqref{eq:variance}]
Telescope $(F-c_G)^2$ from an arbitrary $P\in\Hyb$ to the Gaussian law. Its degree-three expansion has cubic coefficient $B/6$. The degree-zero, one and two terms cancel in expectation. Independence of the replaced coordinate and its deleted base permits multiplication of its third-moment difference by $\E B$. Bounds \eqref{eq:square-rem-mean} and \eqref{eq:B-mean} show that one edge costs at most
$C_{\beta,K}(q^4+nq^6)(\sqrt M+1)$. The polynomial terms are integrable: the constant term is integrable because $H_0$ has finite second moment, and other coefficients have at most linear growth in a deleted free energy, independently of the replacing variable. Summing with \eqref{eq:counts} yields, for every $P\in\Hyb$,
\begin{equation}\label{eq:bootstrap-start}
 \left|\E_P(F-c_G)^2-V_G\right|\le C(\sqrt M+1).
\end{equation}
Taking the finite maximum gives $M\le V_G+C\sqrt M+C$. For $y=\sqrt M$, the positive quadratic root gives
\begin{equation}\label{eq:bootstrap}
 y\le\frac C2+\sqrt{V_G+C+\frac{C^2}{4}}
   \le\sqrt{V_G}+C_1.
\end{equation}
The maximum was finite independently of this inequality, so no desired hybrid variance bound has been assumed. Substituting \eqref{eq:bootstrap} into \eqref{eq:bootstrap-start} bounds the square comparison by $C_2(1+\sqrt{V_G})$. Finally,
\[
 \Var_P F=\E_P(F-c_G)^2-(\E_PF-c_G)^2.
\]
The uniform hybrid mean comparison already proved bounds the subtracted term by a constant. This proves \eqref{eq:variance} and completes Theorem~\ref{thm:comparison}.
\end{proof}

\section{Exact standardization and scope boundaries}
\begin{proof}[Proof of Corollary~\ref{cor:limit}]
At fixed $\beta>1$, Proposition~\ref{prop:gaussian} gives $V_G\sim a_2n^{1/3}\to\infty$. By \eqref{eq:variance}, uniformly over the permitted arrays,
\[
 \frac{\Var F(J)}{V_G}=1+O(V_G^{-1/2}+V_G^{-1})\longrightarrow1.
\]
This proves the prefactor assertion and eventual positivity. In \eqref{eq:test} take $a=\sqrt{V_G}$, which is at least one eventually, and $c=c_G$. For each fixed $t\in\R$, apply it to $\cos(tx)$ and $\sin(tx)$. All derivatives through order six are bounded. The characteristic functions of $(F(J)-c_G)/\sqrt{V_G}$ and $(F(g)-c_G)/\sqrt{V_G}$ differ by $O_t(V_G^{-1/2})$. L\'evy's continuity theorem and \eqref{eq:gaussian} therefore give convergence of the former to $\nu_\beta$. Equation~\eqref{eq:mean} implies $(\E F(J)-c_G)/\sqrt{V_G}\to0$. Slutsky's theorem with the variance ratio gives \eqref{eq:limit}. Each estimate holds at every size and uniformly in the coordinate laws; no joint probability space across sizes or subsequence is needed.
\end{proof}

For $n=2$ and symmetric signs $J_{12}\in\{-1,1\}$, \eqref{eq:model} is $2\log2+\log\cosh(q)$, a constant. Thus eventual positivity is the correct statement; no finite-size standardized variable is asserted when its variance vanishes. A nonzero external field would break \eqref{eq:gauge}, and is excluded. At $a=1$ the smooth-test estimate is merely bounded, not a claim of convergence.

Under only a fourth moment, the pressure need not have all higher moments or exponential moments. The properties of the already defined limiting measure $\nu_\beta$ remain those supplied by the Gaussian input, but the proof does not transfer convergence of every finite-size moment. The assumptions permit asymmetric heavy tails with no fifth moment: for example, center and normalize a Pareto variable with tail $t^{-9/2}$ for $t\ge1$. The proof uses none of the absent moments.
